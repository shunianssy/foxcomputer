# 4.1 函数定义

函数是组织好的、可重复使用的代码块，用于实现特定功能。

## 函数基础

### 定义函数

```python
# 使用 def 关键字定义函数
def greet():
    """打印问候语"""
    print("Hello, World!")

# 调用函数
greet()  # 输出: Hello, World!
```

### 函数参数

```python
# 位置参数
def greet(name):
    """向指定的人打招呼"""
    print(f"Hello, {name}!")

greet("张三")  # 输出: Hello, 张三!

# 多个参数
def add(a, b):
    """返回两个数的和"""
    return a + b

result = add(3, 5)
print(result)  # 输出: 8
```

### 默认参数

```python
# 默认参数值
def greet(name, greeting="Hello"):
    """使用指定问候语打招呼"""
    print(f"{greeting}, {name}!")

greet("张三")              # 输出: Hello, 张三!
greet("李四", "Hi")        # 输出: Hi, 李四!
greet("王五", greeting="你好")  # 输出: 你好, 王五!
```

### 关键字参数

```python
# 使用关键字参数
def describe_person(name, age, city):
    """描述一个人"""
    print(f"{name}, {age}岁, 来自{city}")

# 位置参数
describe_person("张三", 25, "北京")

# 关键字参数（顺序可以改变）
describe_person(age=25, name="张三", city="北京")

# 混合使用（位置参数必须在关键字参数前面）
describe_person("张三", city="上海", age=30)
```

## 可变参数

### *args：可变位置参数

```python
# *args 接收任意数量的位置参数
def sum_all(*args):
    """计算所有参数的和"""
    total = 0
    for num in args:
        total += num
    return total

print(sum_all(1, 2, 3))        # 输出: 6
print(sum_all(1, 2, 3, 4, 5))  # 输出: 15

# args 是一个元组
def show_args(*args):
    print(f"类型: {type(args)}")
    print(f"内容: {args}")

show_args(1, 2, 3)
# 输出:
# 类型: <class 'tuple'>
# 内容: (1, 2, 3)
```

### **kwargs：可变关键字参数

```python
# **kwargs 接收任意数量的关键字参数
def print_info(**kwargs):
    """打印所有关键字参数"""
    for key, value in kwargs.items():
        print(f"{key}: {value}")

print_info(name="张三", age=25, city="北京")
# 输出:
# name: 张三
# age: 25
# city: 北京

# kwargs 是一个字典
def show_kwargs(**kwargs):
    print(f"类型: {type(kwargs)}")
    print(f"内容: {kwargs}")

show_kwargs(a=1, b=2)
# 输出:
# 类型: <class 'dict'>
# 内容: {'a': 1, 'b': 2}
```

### 混合使用

```python
# 参数顺序：位置参数 -> *args -> 默认参数 -> **kwargs
def func(a, b, *args, default=10, **kwargs):
    print(f"a={a}, b={b}")
    print(f"args={args}")
    print(f"default={default}")
    print(f"kwargs={kwargs}")

func(1, 2, 3, 4, 5, default=20, x=100, y=200)
# 输出:
# a=1, b=2
# args=(3, 4, 5)
# default=20
# kwargs={'x': 100, 'y': 200}
```

## 仅位置参数和仅关键字参数

```python
# 仅位置参数（Python 3.8+）
# / 之前的参数只能通过位置传递
def greet(name, /, greeting="Hello"):
    print(f"{greeting}, {name}!")

greet("张三")           # 正确
greet("张三", "Hi")     # 正确
# greet(name="张三")    # 错误：name 是仅位置参数

# 仅关键字参数
# * 之后的参数只能通过关键字传递
def create_user(name, *, age, city):
    print(f"{name}, {age}岁, {city}")

create_user("张三", age=25, city="北京")  # 正确
# create_user("张三", 25, "北京")        # 错误

# 组合使用
def func(a, b, /, c, d, *, e, f):
    print(f"a={a}, b={b}, c={c}, d={d}, e={e}, f={f}")

func(1, 2, 3, d=4, e=5, f=6)
func(1, 2, c=3, d=4, e=5, f=6)
```

## 解包参数

```python
# 解包列表/元组作为位置参数
numbers = [1, 2, 3, 4, 5]
print(*numbers)  # 等价于 print(1, 2, 3, 4, 5)

def add(a, b, c):
    return a + b + c

values = [1, 2, 3]
print(add(*values))  # 输出: 6

# 解包字典作为关键字参数
def greet(name, age, city):
    print(f"{name}, {age}岁, 来自{city}")

person = {"name": "张三", "age": 25, "city": "北京"}
greet(**person)
```

## 文档字符串

```python
def calculate_bmi(weight, height):
    """
    计算身体质量指数（BMI）。
    
    BMI = 体重(kg) / 身高(m)^2
    
    Args:
        weight: 体重，单位千克
        height: 身高，单位米
    
    Returns:
        float: BMI 值
    
    Raises:
        ValueError: 如果身高或体重为负数
    
    Examples:
        >>> calculate_bmi(70, 1.75)
        22.857142857142858
    """
    if weight <= 0 or height <= 0:
        raise ValueError("体重和身高必须为正数")
    
    return weight / (height ** 2)

# 查看文档字符串
print(calculate_bmi.__doc__)
help(calculate_bmi)
```

## 类型注解

```python
# 使用类型注解提高代码可读性
def add(a: int, b: int) -> int:
    """返回两个整数的和"""
    return a + b

def greet(name: str) -> str:
    """返回问候语"""
    return f"Hello, {name}!"

# 复杂类型注解
from typing import List, Dict, Optional, Union

def process_items(items: List[int]) -> int:
    """处理整数列表，返回总和"""
    return sum(items)

def get_user(user_id: int) -> Optional[Dict[str, str]]:
    """获取用户信息，可能返回 None"""
    users = {1: {"name": "张三"}}
    return users.get(user_id)

def process(value: Union[int, str]) -> str:
    """处理整数或字符串"""
    return str(value)
```

## 实战练习

### 练习一：计算器函数

```python
def calculator(a: float, b: float, operator: str) -> float:
    """
    简单计算器函数。
    
    Args:
        a: 第一个数字
        b: 第二个数字
        operator: 运算符（+、-、*、/）
    
    Returns:
        计算结果
    """
    if operator == "+":
        return a + b
    elif operator == "-":
        return a - b
    elif operator == "*":
        return a * b
    elif operator == "/":
        if b == 0:
            raise ValueError("除数不能为零")
        return a / b
    else:
        raise ValueError(f"不支持的运算符: {operator}")

# 使用示例
print(calculator(10, 5, "+"))  # 输出: 15
print(calculator(10, 5, "*"))  # 输出: 50
```

### 练习二：可变参数求平均值

```python
def average(*numbers: float) -> float:
    """
    计算任意数量数字的平均值。
    
    Args:
        *numbers: 任意数量的数字
    
    Returns:
        平均值
    """
    if not numbers:
        return 0.0
    
    return sum(numbers) / len(numbers)

print(average(1, 2, 3, 4, 5))  # 输出: 3.0
print(average(10, 20, 30))     # 输出: 20.0
```

### 练习三：构建 URL

```python
def build_url(base: str, **params: str) -> str:
    """
    构建 URL 查询字符串。
    
    Args:
        base: 基础 URL
        **params: 查询参数
    
    Returns:
        完整的 URL
    
    Examples:
        >>> build_url("https://api.example.com", key="value", page="1")
        'https://api.example.com?key=value&page=1'
    """
    if not params:
        return base
    
    query_string = "&".join(f"{k}={v}" for k, v in params.items())
    return f"{base}?{query_string}"

url = build_url(
    "https://api.example.com/search",
    q="python",
    page="1",
    limit="10"
)
print(url)
# 输出: https://api.example.com/search?q=python&page=1&limit=10
```

## 小结

本章要点：

1. **函数定义**：使用 def 关键字
2. **参数类型**：位置参数、默认参数、关键字参数
3. **可变参数**：*args 接收位置参数，**kwargs 接收关键字参数
4. **参数限制**：/ 之前为仅位置参数，* 之后为仅关键字参数
5. **文档字符串**：使用三引号编写函数说明
6. **类型注解**：提高代码可读性和 IDE 支持

> **提示**：好的函数应该只做一件事，并且做好
