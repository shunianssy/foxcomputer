# 4.2 返回值与作用域

函数可以返回值，变量有不同的作用域范围。

## 返回值

### return 语句

```python
# 返回单个值
def add(a, b):
    return a + b

result = add(3, 5)
print(result)  # 输出: 8

# 没有返回值的函数（隐式返回 None）
def greet(name):
    print(f"Hello, {name}!")

result = greet("张三")
print(result)  # 输出: None

# 提前返回
def check_positive(num):
    if num <= 0:
        return "非正数"
    return "正数"

print(check_positive(5))   # 输出: 正数
print(check_positive(-1))  # 输出: 非正数
```

### 返回多个值

```python
# 返回多个值（实际是元组）
def get_min_max(numbers):
    return min(numbers), max(numbers)

minimum, maximum = get_min_max([1, 5, 3, 9, 2])
print(f"最小值: {minimum}, 最大值: {maximum}")

# 接收为元组
result = get_min_max([1, 5, 3, 9, 2])
print(result)  # 输出: (1, 9)

# 返回列表或字典
def get_stats(numbers):
    return {
        "min": min(numbers),
        "max": max(numbers),
        "avg": sum(numbers) / len(numbers)
    }

stats = get_stats([1, 2, 3, 4, 5])
print(stats)  # 输出: {'min': 1, 'max': 5, 'avg': 3.0}
```

### 返回函数

```python
# 函数可以作为返回值
def get_multiplier(factor):
    """返回一个乘法函数"""
    def multiplier(x):
        return x * factor
    return multiplier

double = get_multiplier(2)
triple = get_multiplier(3)

print(double(5))  # 输出: 10
print(triple(5))  # 输出: 15

# 使用 lambda
def get_multiplier(factor):
    return lambda x: x * factor
```

## 变量作用域

### 局部变量

```python
def my_function():
    # 局部变量：只在函数内部有效
    local_var = "我是局部变量"
    print(local_var)

my_function()
# print(local_var)  # 错误：局部变量无法在函数外访问
```

### 全局变量

```python
# 全局变量：在模块级别定义
global_var = "我是全局变量"

def my_function():
    print(global_var)  # 可以访问全局变量

my_function()  # 输出: 我是全局变量
print(global_var)  # 输出: 我是全局变量
```

### 变量遮蔽

```python
# 局部变量遮蔽全局变量
name = "全局名称"

def show_name():
    name = "局部名称"  # 创建了一个新的局部变量
    print(name)

show_name()    # 输出: 局部名称
print(name)    # 输出: 全局名称
```

### global 关键字

```python
counter = 0

def increment():
    global counter  # 声明使用全局变量
    counter += 1

increment()
increment()
print(counter)  # 输出: 2

# 不使用 global 会创建局部变量
def wrong_increment():
    counter = 0  # 局部变量
    counter += 1

wrong_increment()
print(counter)  # 输出: 2（全局变量未改变）
```

### nonlocal 关键字

```python
# nonlocal 用于嵌套函数中引用外层变量
def outer():
    count = 0
    
    def inner():
        nonlocal count  # 引用外层函数的变量
        count += 1
        return count
    
    return inner

counter = outer()
print(counter())  # 输出: 1
print(counter())  # 输出: 2
print(counter())  # 输出: 3
```

## 作用域规则（LEGB）

Python 按照以下顺序查找变量：

1. **L**ocal（局部）：函数内部
2. **E**nclosing（嵌套）：外层函数
3. **G**lobal（全局）：模块级别
4. **B**uilt-in（内置）：Python 内置

```python
# LEGB 规则示例
x = "global"

def outer():
    x = "enclosing"
    
    def inner():
        x = "local"
        print(x)  # 输出: local
    
    inner()
    print(x)  # 输出: enclosing

outer()
print(x)  # 输出: global

# 内置作用域
def show_builtin():
    print(len)  # len 是内置函数
    # len = "重新定义"  # 不要这样做！
```

## 闭包

闭包是指函数记住并访问其外部作用域的变量。

```python
# 闭包示例
def make_counter():
    count = 0
    
    def counter():
        nonlocal count
        count += 1
        return count
    
    return counter

# 每个闭包维护自己的状态
counter1 = make_counter()
counter2 = make_counter()

print(counter1())  # 输出: 1
print(counter1())  # 输出: 2
print(counter2())  # 输出: 1

# 闭包的实际应用：配置函数
def make_url_fetcher(base_url):
    """创建一个 URL 获取函数"""
    def fetch(endpoint):
        return f"{base_url}{endpoint}"
    return fetch

api_fetcher = make_url_fetcher("https://api.example.com")
print(api_fetcher("/users"))     # 输出: https://api.example.com/users
print(api_fetcher("/products"))  # 输出: https://api.example.com/products
```

## 实战练习

### 练习一：计数器工厂

```python
def create_counter(start=0, step=1):
    """
    创建一个计数器。
    
    Args:
        start: 起始值
        step: 每次增加的步长
    
    Returns:
        计数器函数
    """
    count = start
    
    def counter():
        nonlocal count
        current = count
        count += step
        return current
    
    return counter

# 使用示例
counter = create_counter(10, 2)
print(counter())  # 输出: 10
print(counter())  # 输出: 12
print(counter())  # 输出: 14
```

### 练习二：缓存装饰器

```python
def memoize(func):
    """简单的缓存装饰器"""
    cache = {}
    
    def wrapper(*args):
        if args in cache:
            print(f"从缓存获取: {args}")
            return cache[args]
        
        result = func(*args)
        cache[args] = result
        return result
    
    return wrapper

@memoize
def fibonacci(n):
    """计算斐波那契数"""
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

print(fibonacci(10))  # 第一次计算
print(fibonacci(10))  # 从缓存获取
```

### 练习三：配置管理

```python
def create_config_manager():
    """创建配置管理器"""
    config = {}
    
    def get(key, default=None):
        return config.get(key, default)
    
    def set(key, value):
        config[key] = value
    
    def get_all():
        return config.copy()
    
    return {"get": get, "set": set, "get_all": get_all}

# 使用示例
manager = create_config_manager()
manager["set"]("host", "localhost")
manager["set"]("port", 8080)

print(manager["get"]("host"))      # 输出: localhost
print(manager["get"]("port"))      # 输出: 8080
print(manager["get"]("debug", False))  # 输出: False
print(manager["get_all"]())        # 输出: {'host': 'localhost', 'port': 8080}
```

## 小结

本章要点：

1. **返回值**：使用 return 返回结果，可以返回多个值
2. **局部变量**：函数内部定义的变量
3. **全局变量**：模块级别定义的变量
4. **global**：在函数内修改全局变量
5. **nonlocal**：在嵌套函数中修改外层变量
6. **LEGB 规则**：变量查找顺序
7. **闭包**：函数记住外部作用域的变量

> **提示**：尽量避免使用全局变量
