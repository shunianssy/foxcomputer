本章将综合运用前面学到的知识，开发一个完整的打砖块游戏。

## 游戏设计

### 游戏规则
- 玩家控制底部的挡板，反弹小球
- 小球击碎上方的砖块获得分数
- 所有砖块消除后获胜
- 小球落到底部则游戏结束

### 游戏元素
- 挡板（玩家控制）
- 小球（自动移动）
- 砖块（被击中后消失）
- 分数系统

## 完整代码

```python
"""
打砖块游戏 - Pygame 完整示例

功能:
- 玩家控制挡板反弹小球
- 击碎砖块获得分数
- 多关卡设计
- 音效系统
- 游戏状态管理

作者: Python 教程
"""

import pygame
import sys
import math
import random
import array

# 初始化 Pygame
pygame.init()
pygame.mixer.init()

# 游戏常量
SCREEN_WIDTH = 800
SCREEN_HEIGHT = 600
FPS = 60

# 颜色定义
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)
GRAY = (128, 128, 128)
RED = (255, 0, 0)
GREEN = (0, 255, 0)
BLUE = (0, 0, 255)
YELLOW = (255, 255, 0)
CYAN = (0, 255, 255)
MAGENTA = (255, 0, 255)
ORANGE = (255, 165, 0)

# 砖块颜色
BRICK_COLORS = [RED, ORANGE, YELLOW, GREEN, CYAN, BLUE, MAGENTA]


class SoundManager:
    """音效管理器"""
    
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._initialized = False
        return cls._instance
    
    def __init__(self):
        if self._initialized:
            return
        
        self._initialized = True
        self.sounds = {}
        self.enabled = True
        self._generate_sounds()
    
    def _generate_sounds(self):
        """生成游戏音效"""
        # 碰撞音效
        self.sounds['paddle'] = self._create_beep(440, 0.05)
        self.sounds['brick'] = self._create_beep(660, 0.05)
        self.sounds['wall'] = self._create_beep(330, 0.03)
        self.sounds['lose'] = self._create_beep(150, 0.3)
        self.sounds['win'] = self._create_beep(880, 0.2)
    
    def _create_beep(self, frequency, duration):
        """生成蜂鸣音效"""
        sample_rate = 44100
        n_samples = int(sample_rate * duration)
        samples = array.array('h')
        
        for i in range(n_samples):
            t = i / sample_rate
            envelope = min(1, min(i / 500, (n_samples - i) / 500))
            value = int(20000 * envelope * math.sin(2 * math.pi * frequency * t))
            samples.append(value)
            samples.append(value)
        
        return pygame.mixer.Sound(buffer=samples)
    
    def play(self, name):
        """播放音效"""
        if self.enabled and name in self.sounds:
            self.sounds[name].play()


class Paddle(pygame.sprite.Sprite):
    """挡板类 - 玩家控制"""
    
    def __init__(self):
        super().__init__()
        
        self.width = 100
        self.height = 15
        self.speed = 8
        
        # 创建挡板图像
        self.image = pygame.Surface((self.width, self.height), pygame.SRCALPHA)
        pygame.draw.rect(self.image, WHITE, (0, 0, self.width, self.height), 
                         border_radius=5)
        pygame.draw.rect(self.image, GRAY, (0, 0, self.width, self.height), 
                         2, border_radius=5)
        
        self.rect = self.image.get_rect()
        self.rect.centerx = SCREEN_WIDTH // 2
        self.rect.bottom = SCREEN_HEIGHT - 30
    
    def update(self):
        """更新挡板位置"""
        keys = pygame.key.get_pressed()
        
        # 键盘控制
        if keys[pygame.K_LEFT] or keys[pygame.K_a]:
            self.rect.x -= self.speed
        if keys[pygame.K_RIGHT] or keys[pygame.K_d]:
            self.rect.x += self.speed
        
        # 边界限制
        self.rect.clamp_ip(pygame.Rect(0, 0, SCREEN_WIDTH, SCREEN_HEIGHT))
    
    def draw(self, screen):
        """绘制挡板"""
        screen.blit(self.image, self.rect)


class Ball(pygame.sprite.Sprite):
    """小球类"""
    
    def __init__(self, paddle):
        super().__init__()
        
        self.radius = 8
        self.speed = 6
        self.paddle = paddle
        
        # 创建小球图像
        self.image = pygame.Surface((self.radius * 2, self.radius * 2), 
                                     pygame.SRCALPHA)
        pygame.draw.circle(self.image, WHITE, (self.radius, self.radius), 
                           self.radius)
        # 高光效果
        pygame.draw.circle(self.image, (200, 200, 200), 
                           (self.radius - 2, self.radius - 2), 3)
        
        self.rect = self.image.get_rect()
        self.reset()
    
    def reset(self):
        """重置小球位置"""
        self.rect.centerx = self.paddle.rect.centerx
        self.rect.bottom = self.paddle.rect.top
        
        # 随机初始方向
        angle = random.uniform(-60, 60)
        self.dx = self.speed * math.sin(math.radians(angle))
        self.dy = -self.speed * math.cos(math.radians(angle))
        
        self.active = False  # 是否已发射
    
    def launch(self):
        """发射小球"""
        if not self.active:
            self.active = True
    
    def update(self):
        """更新小球位置"""
        if not self.active:
            # 跟随挡板
            self.rect.centerx = self.paddle.rect.centerx
            self.rect.bottom = self.paddle.rect.top
            return
        
        # 移动
        self.rect.x += self.dx
        self.rect.y += self.dy
        
        # 墙壁碰撞
        if self.rect.left <= 0:
            self.rect.left = 0
            self.dx = abs(self.dx)
            SoundManager().play('wall')
        
        if self.rect.right >= SCREEN_WIDTH:
            self.rect.right = SCREEN_WIDTH
            self.dx = -abs(self.dx)
            SoundManager().play('wall')
        
        if self.rect.top <= 0:
            self.rect.top = 0
            self.dy = abs(self.dy)
            SoundManager().play('wall')
    
    def bounce_paddle(self, paddle):
        """与挡板碰撞"""
        if self.rect.colliderect(paddle.rect):
            # 计算反弹角度（根据碰撞位置）
            hit_pos = (self.rect.centerx - paddle.rect.centerx) / (paddle.width / 2)
            angle = hit_pos * 60  # 最大 60 度
            
            self.dx = self.speed * math.sin(math.radians(angle))
            self.dy = -abs(self.speed * math.cos(math.radians(angle)))
            
            self.rect.bottom = paddle.rect.top
            SoundManager().play('paddle')
    
    def bounce_brick(self, brick):
        """与砖块碰撞"""
        if self.rect.colliderect(brick.rect):
            # 判断碰撞方向
            dx = self.rect.centerx - brick.rect.centerx
            dy = self.rect.centery - brick.rect.centery
            
            if abs(dx) / brick.rect.width > abs(dy) / brick.rect.height:
                self.dx = -self.dx
            else:
                self.dy = -self.dy
            
            SoundManager().play('brick')
            return True
        return False
    
    def is_out(self):
        """检查是否掉落"""
        return self.rect.top > SCREEN_HEIGHT


class Brick(pygame.sprite.Sprite):
    """砖块类"""
    
    def __init__(self, x, y, color, points=10):
        super().__init__()
        
        self.width = 70
        self.height = 25
        self.points = points
        self.color = color
        
        # 创建砖块图像
        self.image = pygame.Surface((self.width, self.height), pygame.SRCALPHA)
        pygame.draw.rect(self.image, color, (0, 0, self.width, self.height), 
                         border_radius=3)
        
        # 高光效果
        highlight = tuple(min(255, c + 50) for c in color)
        pygame.draw.line(self.image, highlight, (2, 2), (self.width - 2, 2), 2)
        
        # 边框
        border = tuple(max(0, c - 50) for c in color)
        pygame.draw.rect(self.image, border, (0, 0, self.width, self.height), 
                         2, border_radius=3)
        
        self.rect = self.image.get_rect()
        self.rect.x = x
        self.rect.y = y


class Game:
    """游戏主类"""
    
    def __init__(self):
        """初始化游戏"""
        self.screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT))
        pygame.display.set_caption("打砖块 - Pygame 教程示例")
        
        self.clock = pygame.time.Clock()
        self.font = pygame.font.Font(None, 36)
        self.big_font = pygame.font.Font(None, 72)
        
        self.sound_manager = SoundManager()
        
        self.reset()
    
    def reset(self):
        """重置游戏状态"""
        self.score = 0
        self.lives = 3
        self.level = 1
        self.game_over = False
        self.game_won = False
        
        # 创建游戏对象
        self.paddle = Paddle()
        self.ball = Ball(self.paddle)
        self.bricks = pygame.sprite.Group()
        
        self._create_bricks()
    
    def _create_bricks(self):
        """创建砖块"""
        self.bricks.empty()
        
        rows = 5 + self.level  # 随关卡增加行数
        cols = 10
        
        brick_width = 70
        brick_height = 25
        padding = 5
        offset_x = (SCREEN_WIDTH - cols * (brick_width + padding)) // 2
        offset_y = 60
        
        for row in range(rows):
            color = BRICK_COLORS[row % len(BRICK_COLORS)]
            points = (rows - row) * 10  # 越高分越多
            
            for col in range(cols):
                x = offset_x + col * (brick_width + padding)
                y = offset_y + row * (brick_height + padding)
                brick = Brick(x, y, color, points)
                self.bricks.add(brick)
    
    def next_level(self):
        """进入下一关"""
        self.level += 1
        self.ball.reset()
        self._create_bricks()
        self.ball.speed = min(12, 6 + self.level * 0.5)  # 增加难度
    
    def handle_events(self):
        """处理事件"""
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                return False
            
            if event.type == pygame.KEYDOWN:
                if event.key == pygame.K_ESCAPE:
                    return False
                
                if event.key == pygame.K_SPACE:
                    if self.game_over or self.game_won:
                        self.reset()
                    else:
                        self.ball.launch()
        
        return True
    
    def update(self):
        """更新游戏状态"""
        if self.game_over or self.game_won:
            return
        
        # 更新游戏对象
        self.paddle.update()
        self.ball.update()
        
        # 挡板碰撞
        self.ball.bounce_paddle(self.paddle)
        
        # 砖块碰撞
        for brick in self.bricks:
            if self.ball.bounce_brick(brick):
                self.score += brick.points
                brick.kill()
                break
        
        # 检查胜利
        if len(self.bricks) == 0:
            if self.level < 3:  # 最多 3 关
                self.next_level()
            else:
                self.game_won = True
                self.sound_manager.play('win')
        
        # 检查失败
        if self.ball.is_out():
            self.lives -= 1
            if self.lives <= 0:
                self.game_over = True
                self.sound_manager.play('lose')
            else:
                self.ball.reset()
    
    def draw(self):
        """绘制游戏画面"""
        # 背景
        self.screen.fill((20, 20, 40))
        
        # 绘制边框
        pygame.draw.rect(self.screen, GRAY, (0, 0, SCREEN_WIDTH, SCREEN_HEIGHT), 3)
        
        # 绘制游戏对象
        self.bricks.draw(self.screen)
        self.paddle.draw(self.screen)
        self.screen.blit(self.ball.image, self.ball.rect)
        
        # 绘制 UI
        self._draw_ui()
        
        # 游戏结束画面
        if self.game_over:
            self._draw_game_over()
        elif self.game_won:
            self._draw_victory()
        
        pygame.display.flip()
    
    def _draw_ui(self):
        """绘制用户界面"""
        # 分数
        score_text = self.font.render(f"分数: {self.score}", True, WHITE)
        self.screen.blit(score_text, (20, 15))
        
        # 生命
        lives_text = self.font.render(f"生命: {self.lives}", True, WHITE)
        self.screen.blit(lives_text, (SCREEN_WIDTH - 120, 15))
        
        # 关卡
        level_text = self.font.render(f"关卡: {self.level}", True, WHITE)
        self.screen.blit(level_text, (SCREEN_WIDTH // 2 - 40, 15))
        
        # 提示
        if not self.ball.active and not self.game_over and not self.game_won:
            hint = self.font.render("按空格发射", True, YELLOW)
            hint_rect = hint.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT - 80))
            self.screen.blit(hint, hint_rect)
    
    def _draw_game_over(self):
        """绘制游戏结束画面"""
        # 半透明遮罩
        overlay = pygame.Surface((SCREEN_WIDTH, SCREEN_HEIGHT), pygame.SRCALPHA)
        overlay.fill((0, 0, 0, 180))
        self.screen.blit(overlay, (0, 0))
        
        # 游戏结束文字
        text = self.big_font.render("游戏结束", True, RED)
        text_rect = text.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2 - 50))
        self.screen.blit(text, text_rect)
        
        # 最终分数
        score = self.font.render(f"最终分数: {self.score}", True, WHITE)
        score_rect = score.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2 + 20))
        self.screen.blit(score, score_rect)
        
        # 重新开始提示
        hint = self.font.render("按空格重新开始", True, YELLOW)
        hint_rect = hint.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2 + 80))
        self.screen.blit(hint, hint_rect)
    
    def _draw_victory(self):
        """绘制胜利画面"""
        # 半透明遮罩
        overlay = pygame.Surface((SCREEN_WIDTH, SCREEN_HEIGHT), pygame.SRCALPHA)
        overlay.fill((0, 0, 0, 180))
        self.screen.blit(overlay, (0, 0))
        
        # 胜利文字
        text = self.big_font.render("恭喜通关!", True, GREEN)
        text_rect = text.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2 - 50))
        self.screen.blit(text, text_rect)
        
        # 最终分数
        score = self.font.render(f"最终分数: {self.score}", True, WHITE)
        score_rect = score.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2 + 20))
        self.screen.blit(score, score_rect)
        
        # 重新开始提示
        hint = self.font.render("按空格重新开始", True, YELLOW)
        hint_rect = hint.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2 + 80))
        self.screen.blit(hint, hint_rect)
    
    def run(self):
        """游戏主循环"""
        running = True
        
        while running:
            running = self.handle_events()
            self.update()
            self.draw()
            self.clock.tick(FPS)
        
        pygame.quit()
        sys.exit()


def main():
    """程序入口"""
    game = Game()
    game.run()


if __name__ == "__main__":
    main()
```

## 代码结构解析

### 1. 游戏架构

```
Game (游戏主类)
├── Paddle (挡板) - 玩家控制
├── Ball (小球) - 物理运动
├── Brick (砖块) - 目标对象
└── SoundManager (音效管理器)
```

### 2. 核心类说明

| 类名 | 职责 |
|------|------|
| `Game` | 游戏主循环、状态管理、渲染 |
| `Paddle` | 玩家控制的挡板 |
| `Ball` | 小球运动和碰撞逻辑 |
| `Brick` | 砖块显示和分数 |
| `SoundManager` | 音效管理（单例模式）|

### 3. 游戏循环

```python
while running:

    update()         # 更新状态
```
