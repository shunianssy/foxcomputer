# 11.3 PyPDF2 PDF 文件处理

PyPDF2 是 Python 中用于处理 PDF 文件的库，支持读取、合并、拆分、旋转和加密 PDF 文件。

## 安装

```bash
# 使用 pip 安装
pip install PyPDF2

# 验证安装
python -c "import PyPDF2; print(PyPDF2.__version__)"
```

## 基本概念

- **PdfReader（读取器）**：用于读取 PDF 文件
- **PdfWriter（写入器）**：用于创建和修改 PDF 文件
- **PageObject（页面对象）**：代表 PDF 中的单个页面
- **PdfMerger（合并器）**：用于合并多个 PDF 文件

## 读取 PDF

### 打开 PDF 文件

```python
from PyPDF2 import PdfReader

# 创建读取器
reader = PdfReader("example.pdf")

# 获取 PDF 信息
print(f"页数: {len(reader.pages)}")

# 获取元数据
metadata = reader.metadata
if metadata:
    print(f"标题: {metadata.title}")
    print(f"作者: {metadata.author}")
    print(f"创建者: {metadata.creator}")
    print(f"创建日期: {metadata.creation_date}")
```

### 读取页面内容

```python

reader = PdfReader("example.pdf")
```

### 提取所有文本

```python
from PyPDF2 import PdfReader
from typing import Optional

def extract_all_text(pdf_path: str) -> str:
    """
    提取 PDF 文件中的所有文本
    
    Args:
        pdf_path: PDF 文件路径
        
    Returns:
        提取的文本内容
    """
    reader = PdfReader(pdf_path)
    all_text = []
    
    for i, page in enumerate(reader.pages):
        text = page.extract_text()
        if text:
            all_text.append(f"--- 第 {i+1} 页 ---\n{text}")
    
    return "\n\n".join(all_text)
```

### 使用 PdfWriter 合并

```python
from PyPDF2 import PdfReader, PdfWriter
from typing import List

def merge_pdfs(file_paths: List[str], output_path: str) -> None:
    """
    合并多个 PDF 文件
    
    Args:
        file_paths: PDF 文件路径列表
        output_path: 输出文件路径
    """
    writer = PdfWriter()
    
    for file_path in file_paths:
        reader = PdfReader(file_path)
        for page in reader.pages:
            writer.add_page(page)
    
    with open(output_path, "wb") as f:
        writer.write(f)
    
    print(f"已合并 {len(file_paths)} 个文件到 {output_path}")


# 使用示例
# merge_pdfs(["file1.pdf", "file2.pdf", "file3.pdf"], "merged.pdf")
```

## 拆分 PDF

### 拆分为单页

```python
from PyPDF2 import PdfReader, PdfWriter
import os

def split_pdf_to_pages(pdf_path: str, output_dir: str) -> None:
    """
    将 PDF 拆分为单页文件
    
    Args:
        pdf_path: PDF 文件路径
        output_dir: 输出目录
    """
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)
    
    reader = PdfReader(pdf_path)
    base_name = os.path.splitext(os.path.basename(pdf_path))[0]
    
    for i, page in enumerate(reader.pages):
        writer = PdfWriter()
        writer.add_page(page)
        
        output_path = os.path.join(output_dir, f"{base_name}_page_{i+1}.pdf")
        with open(output_path, "wb") as f:
            writer.write(f)
    
    print(f"已拆分为 {len(reader.pages)} 个单页文件")


# 使用示例
# split_pdf_to_pages("document.pdf", "split_pages")
```

### 按范围拆分

```python
from PyPDF2 import PdfReader, PdfWriter

def split_pdf_by_range(
    pdf_path: str,
    start_page: int,
    end_page: int,
    output_path: str
) -> None:
    """
    按页面范围拆分 PDF
    
    Args:
        pdf_path: PDF 文件路径
        start_page: 起始页（从0开始）
        end_page: 结束页（不包含）
        output_path: 输出文件路径
    """
    reader = PdfReader(pdf_path)
    writer = PdfWriter()
    
    for i in range(start_page, min(end_page, len(reader.pages))):
        writer.add_page(reader.pages[i])
    
    with open(output_path, "wb") as f:
        writer.write(f)
    
    print(f"已提取第 {start_page+1} 到第 {end_page} 页")


# 使用示例
# split_pdf_by_range("document.pdf", 0, 5, "first_5_pages.pdf")
```

### 批量旋转

```python
from PyPDF2 import PdfReader, PdfWriter
from typing import List

def rotate_pages(
    pdf_path: str,
    output_path: str,
    page_indices: List[int],
    rotation: int = 90
) -> None:
    """
    旋转指定页面
    
    Args:
        pdf_path: PDF 文件路径
        output_path: 输出文件路径
        page_indices: 要旋转的页面索引列表（从0开始）
        rotation: 旋转角度（90, 180, 270）
    """
    reader = PdfReader(pdf_path)
    writer = PdfWriter()
    
    for i, page in enumerate(reader.pages):
        if i in page_indices:
            page.rotate(rotation)
        writer.add_page(page)
    
    with open(output_path, "wb") as f:
        writer.write(f)


# 使用示例
# rotate_pages("document.pdf", "rotated.pdf", [0, 1, 2], 90)
```

## 删除页面

```python
from PyPDF2 import PdfReader, PdfWriter
from typing import List

def delete_pages(
    pdf_path: str,
    output_path: str,
    pages_to_delete: List[int]
) -> None:
    """
    删除指定页面
    
    Args:
        pdf_path: PDF 文件路径
        output_path: 输出文件路径
        pages_to_delete: 要删除的页面索引列表（从0开始）
    """
    reader = PdfReader(pdf_path)
    writer = PdfWriter()
    
    for i, page in enumerate(reader.pages):
        if i not in pages_to_delete:
            writer.add_page(page)
    
    with open(output_path, "wb") as f:
        writer.write(f)
    
    print(f"已删除第 {[p+1 for p in pages_to_delete]} 页")


# 使用示例
# delete_pages("document.pdf", "deleted.pdf", [0, 2, 4])  # 删除第1、3、5页
```

## 提取图片

### 从 PDF 提取图片

```python
from PyPDF2 import PdfReader
import os
from typing import List

def extract_images_from_pdf(pdf_path: str, output_dir: str) -> List[str]:
    """
    从 PDF 中提取图片
    
    Args:
        pdf_path: PDF 文件路径
        output_dir: 输出目录
        
    Returns:
        提取的图片路径列表
    """
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)
    
    reader = PdfReader(pdf_path)
    extracted_files = []
    
    for page_num, page in enumerate(reader.pages):
        if '/XObject' in page['/Resources']:
            x_objects = page['/Resources']['/XObject'].get_object()
            
            for obj_name in x_objects:
                obj = x_objects[obj_name]
                
                # 检查是否为图片
                if obj['/Subtype'] == '/Image':
                    # 获取图片数据
                    data = obj.get_data()
                    
                    # 确定图片格式
                    if '/Filter' in obj:
                        filter_type = obj['/Filter']
                        if filter_type == '/DCTDecode':
                            ext = 'jpg'
                        elif filter_type == '/FlateDecode':
                            ext = 'png'
                        else:
                            ext = 'bin'
                    else:
                        ext = 'raw'
                    
                    # 保存图片
                    output_name = f"page{page_num+1}_{obj_name[1:]}.{ext}"
                    output_path = os.path.join(output_dir, output_name)
                    
                    with open(output_path, 'wb') as f:
                        f.write(data)
                    
                    extracted_files.append(output_path)
    
    return extracted_files


# 使用示例
# images = extract_images_from_pdf("document.pdf", "extracted_images")
# print(f"提取了 {len(images)} 张图片")
```

## 添加水印

```python
from PyPDF2 import PdfReader, PdfWriter
from typing import Tuple

def add_watermark(
    pdf_path: str,
    watermark_path: str,
    output_path: str,
    position: Tuple[int, int] = (0, 0)
) -> None:
    """
    为 PDF 添加水印
    
    Args:
        pdf_path: 原 PDF 文件路径
        watermark_path: 水印 PDF 文件路径
        output_path: 输出文件路径
        position: 水印位置 (x, y)
    """
    pdf_reader = PdfReader(pdf_path)
    watermark_reader = PdfReader(watermark_path)
    writer = PdfWriter()
    
    # 获取水印页面
    watermark_page = watermark_reader.pages[0]
    
    for page in pdf_reader.pages:
        # 合并水印到页面
        page.merge_page(watermark_page)
        writer.add_page(page)
    
    with open(output_path, "wb") as f:
        writer.write(f)
    
    print("水印添加完成")


# 使用示例（需要先创建一个水印 PDF）
# add_watermark("document.pdf", "watermark.pdf", "watermarked.pdf")
```

## 添加页码

```python
from PyPDF2 import PdfReader, PdfWriter
from reportlab.pdfgen import canvas
from reportlab.lib.pagesizes import letter
import io

def add_page_numbers(pdf_path: str, output_path: str) -> None:
    """
    为 PDF 添加页码
    
    Args:
        pdf_path: PDF 文件路径
        output_path: 输出文件路径
    """
    reader = PdfReader(pdf_path)
    writer = PdfWriter()
    total_pages = len(reader.pages)
    
    for i, page in enumerate(reader.pages):
        # 创建页码 PDF
        packet = io.BytesIO()
        c = canvas.Canvas(packet, pagesize=letter)
        
        # 设置页码文本
        page_num_text = f"第 {i+1} 页 / 共 {total_pages} 页"
        c.setFont("Helvetica", 10)
        c.drawCentredString(letter[0] / 2, 30, page_num_text)
        c.save()
        
        # 合并页码到原页面
        packet.seek(0)
        page_num_reader = PdfReader(packet)
        page.merge_page(page_num_reader.pages[0])
        
        writer.add_page(page)
    
    with open(output_path, "wb") as f:
        writer.write(f)
    
    print("页码添加完成")


# 注意：此功能需要安装 reportlab
# pip install reportlab
```

### PDF 信息查看器

```python
from PyPDF2 import PdfReader
from typing import Dict, Any
import os

def get_pdf_info(pdf_path: str) -> Dict[str, Any]:
    """
    获取 PDF 文件详细信息
    
    Args:
        pdf_path: PDF 文件路径
        
    Returns:
        PDF 信息字典
    """
    reader = PdfReader(pdf_path)
    
    info = {
        'file_name': os.path.basename(pdf_path),
        'file_size': os.path.getsize(pdf_path),
        'page_count': len(reader.pages),
        'is_encrypted': reader.is_encrypted,
    }
    
    # 元数据
    metadata = reader.metadata
    if metadata:
        info['metadata'] = {
            'title': metadata.title,
            'author': metadata.author,
            'subject': metadata.subject,
            'creator': metadata.creator,
            'producer': metadata.producer,
            'creation_date': str(metadata.creation_date) if metadata.creation_date else None,
            'modification_date': str(metadata.modification_date) if metadata.modification_date else None,
        }
    
    # 页面尺寸
    if reader.pages:
        first_page = reader.pages[0]
        info['page_size'] = {
            'width': first_page.mediabox.width,
            'height': first_page.mediabox.height,
        }
    
    return info
```

### PDF 文本搜索

```python
from PyPDF2 import PdfReader
from typing import List, Dict

def search_in_pdf(pdf_path: str, keyword: str) -> List[Dict]:
    """
    在 PDF 中搜索关键词
    
    Args:
        pdf_path: PDF 文件路径
        keyword: 搜索关键词
        
    Returns:
        匹配结果列表
    """
    reader = PdfReader(pdf_path)
    results = []
    
    for i, page in enumerate(reader.pages):
        text = page.extract_text()
        if text and keyword.lower() in text.lower():
            # 找到关键词位置
            lower_text = text.lower()
            start = lower_text.find(keyword.lower())
            
            # 获取上下文
            context_start = max(0, start - 50)
            context_end = min(len(text), start + len(keyword) + 50)
            context = text[context_start:context_end]
            
            results.append({
                'page': i + 1,
                'context': context,
            })
    
    return results
```

### 批量处理 PDF

```python
from PyPDF2 import PdfReader, PdfWriter
import os
from typing import List, Callable

def batch_process_pdfs(
    input_dir: str,
    output_dir: str,
    process_func: Callable,
    file_pattern: str = "*.pdf"
) -> List[str]:
    """
    批量处理 PDF 文件
    
    Args:
        input_dir: 输入目录
        output_dir: 输出目录
        process_func: 处理函数，接收 reader 返回 writer
        file_pattern: 文件匹配模式
        
    Returns:
        处理后的文件路径列表
    """
    import glob
    
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)
    
    processed_files = []
    pdf_files = glob.glob(os.path.join(input_dir, file_pattern))
    
    for pdf_path in pdf_files:
        try:
            reader = PdfReader(pdf_path)
            writer = process_func(reader)
            
            if writer:
                output_name = os.path.basename(pdf_path)
                output_path = os.path.join(output_dir, output_name)
                
                with open(output_path, "wb") as f:
                    writer.write(f)
                
                processed_files.append(output_path)
                print(f"已处理: {output_name}")
        
        except Exception as e:
            print(f"处理失败 {pdf_path}: {e}")
    
    return processed_files
```

### PDF 转文本文件

```python
from PyPDF2 import PdfReader
import os
from typing import Optional

def pdf_to_text_file(
    pdf_path: str,
    output_path: Optional[str] = None,
    encoding: str = 'utf-8'
) -> str:
    """
    将 PDF 转换为文本文件
    
    Args:
        pdf_path: PDF 文件路径
        output_path: 输出文本文件路径，默认为同名 .txt 文件
        encoding: 文件编码
        
    Returns:
        输出文件路径
    """
    if output_path is None:
        output_path = os.path.splitext(pdf_path)[0] + '.txt'
    
    reader = PdfReader(pdf_path)
    
    with open(output_path, 'w', encoding=encoding) as f:
        for i, page in enumerate(reader.pages):
            text = page.extract_text()
            if text:
                f.write(f"=== 第 {i+1} 页 ===\n")
                f.write(text)
                f.write("\n\n")
    
    return output_path
```
