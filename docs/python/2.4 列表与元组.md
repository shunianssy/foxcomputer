# 2.4 列表与元组

列表和元组是 Python 中最常用的序列类型，用于存储有序的数据集合。

## 列表（List）

列表是可变的有序集合，可以存储任意类型的元素。

### 创建列表

```python
# 创建空列表
empty_list = []
empty_list2 = list()

# 创建有元素的列表
numbers = [1, 2, 3, 4, 5]
strings = ["apple", "banana", "orange"]
mixed = [1, "hello", 3.14, True, [1, 2]]

# 使用 list() 函数
chars = list("Python")  # ['P', 'y', 't', 'h', 'o', 'n']
range_list = list(range(5))  # [0, 1, 2, 3, 4]
```

### 列表索引与切片

```python
fruits = ["apple", "banana", "orange", "grape", "mango"]

# 索引访问
print(fruits[0])   # 输出: apple
print(fruits[-1])  # 输出: mango

# 切片
print(fruits[1:3])   # 输出: ['banana', 'orange']
print(fruits[:3])    # 输出: ['apple', 'banana', 'orange']
print(fruits[2:])    # 输出: ['orange', 'grape', 'mango']
print(fruits[::2])   # 输出: ['apple', 'orange', 'mango']
print(fruits[::-1])  # 输出: ['mango', 'grape', 'orange', 'banana', 'apple']
```

### 修改列表

```python
numbers = [1, 2, 3, 4, 5]

# 修改单个元素
numbers[0] = 10
print(numbers)  # 输出: [10, 2, 3, 4, 5]

# 修改多个元素（切片赋值）
numbers[1:3] = [20, 30]
print(numbers)  # 输出: [10, 20, 30, 4, 5]
```

### 添加元素

```python
fruits = ["apple", "banana"]

# append：在末尾添加一个元素
fruits.append("orange")
print(fruits)  # 输出: ['apple', 'banana', 'orange']

# extend：在末尾添加多个元素
fruits.extend(["grape", "mango"])
print(fruits)  # 输出: ['apple', 'banana', 'orange', 'grape', 'mango']

# insert：在指定位置插入元素
fruits.insert(1, "kiwi")
print(fruits)  # 输出: ['apple', 'kiwi', 'banana', 'orange', 'grape', 'mango']

# 使用 + 号
fruits = fruits + ["peach"]
print(fruits)
```

### 删除元素

```python
numbers = [1, 2, 3, 4, 5, 3, 6]

# remove：删除第一个匹配的元素
numbers.remove(3)
print(numbers)  # 输出: [1, 2, 4, 5, 3, 6]

# pop：删除指定索引的元素（默认最后一个）
last = numbers.pop()
print(last)     # 输出: 6
print(numbers)  # 输出: [1, 2, 4, 5, 3]

second = numbers.pop(1)
print(second)   # 输出: 2

# del：删除指定索引或切片
del numbers[0]
print(numbers)  # 输出: [4, 5, 3]

# clear：清空列表
numbers.clear()
print(numbers)  # 输出: []
```

### 列表方法

```python
numbers = [3, 1, 4, 1, 5, 9, 2, 6]

# 排序
numbers.sort()
print(numbers)  # 输出: [1, 1, 2, 3, 4, 5, 6, 9]

# 降序排序
numbers.sort(reverse=True)
print(numbers)  # 输出: [9, 6, 5, 4, 3, 2, 1, 1]

# 反转
numbers.reverse()
print(numbers)  # 输出: [1, 1, 2, 3, 4, 5, 6, 9]

# 查找元素位置
print(numbers.index(5))  # 输出: 5

# 统计元素出现次数
print(numbers.count(1))  # 输出: 2

# 复制列表
copy_list = numbers.copy()
```

### 列表推导式

```python
# 基本语法：[expression for item in iterable]

# 生成平方数列表
squares = [x ** 2 for x in range(1, 6)]
print(squares)  # 输出: [1, 4, 9, 16, 25]

# 带条件的推导式
evens = [x for x in range(10) if x % 2 == 0]
print(evens)  # 输出: [0, 2, 4, 6, 8]

# 带表达式的推导式
words = ["hello", "world", "python"]
upper_words = [word.upper() for word in words]
print(upper_words)  # 输出: ['HELLO', 'WORLD', 'PYTHON']

# 嵌套循环
matrix = [[i * j for j in range(1, 4)] for i in range(1, 4)]
print(matrix)  # 输出: [[1, 2, 3], [2, 4, 6], [3, 6, 9]]

# 展平嵌套列表
flat = [x for row in matrix for x in row]
print(flat)  # 输出: [1, 2, 3, 2, 4, 6, 3, 6, 9]
```

## 元组（Tuple）

元组是不可变的有序集合，一旦创建就不能修改。

### 创建元组

```python
# 创建空元组
empty_tuple = ()
empty_tuple2 = tuple()

# 创建有元素的元组
point = (3, 4)
colors = ("red", "green", "blue")

# 单元素元组（注意逗号）
single = (42,)  # 必须有逗号
not_tuple = (42)  # 这只是数字 42

# 省略括号
coordinates = 10, 20, 30

# 使用 tuple() 函数
from_list = tuple([1, 2, 3])
from_string = tuple("Python")  # ('P', 'y', 't', 'h', 'o', 'n')
```

### 元组操作

```python
point = (3, 4)

# 索引和切片
print(point[0])    # 输出: 3
print(point[-1])   # 输出: 4

# 元组解包
x, y = point
print(f"x={x}, y={y}")  # 输出: x=3, y=4

# 交换变量
a, b = 1, 2
a, b = b, a

# 扩展解包
first, *rest = (1, 2, 3, 4, 5)
print(first)  # 输出: 1
print(rest)   # 输出: [2, 3, 4, 5]

# 连接元组
combined = (1, 2) + (3, 4)
print(combined)  # 输出: (1, 2, 3, 4)

# 重复元组
repeated = (1, 2) * 3
print(repeated)  # 输出: (1, 2, 1, 2, 1, 2)
```

### 元组方法

```python
numbers = (1, 2, 3, 2, 4, 2)

# 计数
print(numbers.count(2))  # 输出: 3

# 查找索引
print(numbers.index(3))  # 输出: 2

# 长度
print(len(numbers))  # 输出: 6

# 成员判断
print(2 in numbers)  # True
```

### 命名元组

```python
from collections import namedtuple

# 定义命名元组
Point = namedtuple('Point', ['x', 'y'])
Person = namedtuple('Person', 'name age city')

# 创建实例
p = Point(3, 4)
print(p.x, p.y)  # 输出: 3 4
print(p[0], p[1])  # 输出: 3 4

person = Person(name="张三", age=25, city="北京")
print(person.name)  # 输出: 张三
print(person.age)   # 输出: 25

# 转换为字典
print(person._asdict())
# 输出: {'name': '张三', 'age': 25, 'city': '北京'}
```

## 列表 vs 元组

| 特性 | 列表 | 元组 |
|------|------|------|
| 可变性 | 可变 | 不可变 |
| 语法 | [] | () |
| 方法 | 多（append, remove 等） | 少（count, index） |
| 性能 | 较慢 | 较快 |
| 内存 | 较大 | 较小 |
| 用途 | 需要修改的数据 | 固定数据、字典键 |

```python
# 元组可以作为字典的键
locations = {
    (0, 0): "原点",
    (1, 0): "x轴上",
    (0, 1): "y轴上"
}
print(locations[(0, 0)])  # 输出: 原点

# 列表不能作为字典的键
# locations = {[0, 0]: "原点"}  # TypeError
```

## 实战练习

### 练习一：列表操作

```python
# 管理学生成绩
scores = []

# 添加成绩
while True:
    score = input("请输入成绩（输入 q 结束）：")
    if score.lower() == 'q':
        break
    scores.append(float(score))

# 计算统计信息
if scores:
    print(f"最高分：{max(scores)}")
    print(f"最低分：{min(scores)}")
    print(f"平均分：{sum(scores) / len(scores):.2f}")
    print(f"成绩列表：{sorted(scores, reverse=True)}")
```

### 练习二：列表去重

```python
# 列表去重的几种方法
numbers = [1, 2, 3, 2, 4, 3, 5, 1]

# 方法一：使用 set（不保持顺序）
unique1 = list(set(numbers))
print(unique1)  # 输出: [1, 2, 3, 4, 5]（顺序可能不同）

# 方法二：使用 dict（保持顺序，Python 3.7+）
unique2 = list(dict.fromkeys(numbers))
print(unique2)  # 输出: [1, 2, 3, 4, 5]

# 方法三：循环判断（保持顺序）
unique3 = []
for num in numbers:
    if num not in unique3:
        unique3.append(num)
print(unique3)  # 输出: [1, 2, 3, 4, 5]
```

### 练习三：矩阵操作

```python
# 矩阵转置
matrix = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 9]
]

# 方法一：嵌套循环
transposed = []
for i in range(len(matrix[0])):
    row = []
    for j in range(len(matrix)):
        row.append(matrix[j][i])
    transposed.append(row)
print(transposed)

# 方法二：列表推导式
transposed = [[matrix[j][i] for j in range(len(matrix))] for i in range(len(matrix[0]))]
print(transposed)

# 方法三：使用 zip
transposed = list(map(list, zip(*matrix)))
print(transposed)
```

### 练习四：购物车

```python
# 简单购物车
cart = []

def add_item(name, price, quantity=1):
    """添加商品到购物车"""
    cart.append({"name": name, "price": price, "quantity": quantity})

def remove_item(name):
    """从购物车移除商品"""
    for item in cart:
        if item["name"] == name:
            cart.remove(item)
            return True
    return False

def get_total():
    """计算总价"""
    return sum(item["price"] * item["quantity"] for item in cart)

def show_cart():
    """显示购物车内容"""
    print("\n购物车内容：")
    print("-" * 40)
    for item in cart:
        print(f"{item['name']} x {item['quantity']} = ¥{item['price'] * item['quantity']:.2f}")
    print("-" * 40)
    print(f"总计：¥{get_total():.2f}\n")

# 使用示例
add_item("苹果", 5.5, 3)
add_item("香蕉", 3.0, 2)
add_item("牛奶", 8.5, 1)

show_cart()
```

## 小结

本章要点：

1. **列表**：可变有序集合，支持增删改查
2. **元组**：不可变有序集合，性能更好
3. **列表方法**：append、extend、insert、remove、pop、sort 等
4. **列表推导式**：简洁地创建列表
5. **命名元组**：具有命名字段的元组

> **提示**：如果数据不需要修改，优先使用元组
