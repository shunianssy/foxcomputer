# 10.5 Playwright 现代浏览器自动化

Playwright 是新一代浏览器自动化工具，支持 Chromium、Firefox 和 WebKit，具有更快的速度和更好的 API 设计。

## 安装

```bash
# 安装 Playwright
pip install playwright

# 安装浏览器
playwright install

# 只安装特定浏览器
playwright install chromium
playwright install firefox
playwright install webkit

# 验证安装
python -c "from playwright.sync_api import sync_playwright; print('安装成功')"
```

## 快速开始

### 同步 API

```python
from playwright.sync_api import sync_playwright


def main():
    """同步方式使用 Playwright"""
    with sync_playwright() as p:
        # 启动浏览器
        browser = p.chromium.launch(headless=False)
        
        # 创建页面
        page = browser.new_page()
        
        # 访问网页
        page.goto("https://www.example.com")
        
        # 获取页面标题
        print(page.title())
        
        # 截图
        page.screenshot(path="screenshot.png")
        
        # 关闭浏览器
        browser.close()


if __name__ == "__main__":
    main()
```

### 异步 API

```python
import asyncio
from playwright.async_api import async_playwright


async def main():
    """异步方式使用 Playwright"""
    async with async_playwright() as p:
        # 启动浏览器
        browser = await p.chromium.launch(headless=False)
        
        # 创建页面
        page = await browser.new_page()
        
        # 访问网页
        await page.goto("https://www.example.com")
        
        # 获取页面标题
        print(await page.title())
        
        # 截图
        await page.screenshot(path="screenshot.png")
        
        # 关闭浏览器
        await browser.close()


if __name__ == "__main__":
    asyncio.run(main())
```

```python
from playwright.sync_api import sync_playwright


def main():
```

```python
        browser = p.chromium.launch(
            headless=False,  # 是否无头模式
            slow_mo=100,  # 操作延迟（毫秒）

                "--start-maximized",  # 最大化窗口

            viewport={"width": 1920, "height": 1080},  # 视口大小

            locale="zh-CN",  # 语言

            java_script_enabled=True,  # 启用 JavaScript

            is_mobile=True,  # 移动设备
            color_scheme="dark",  # 深色模式
            device_scale_factor=2,  # 设备像素比
        )
```

### 显式等待

```python
from playwright.sync_api import sync_playwright, expect


def main():
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=False)
        page = browser.new_page()
        page.goto("https://example.com")
        
        # 等待元素出现
        page.wait_for_selector("#content")
        
        # 等待元素可见
        page.wait_for_selector("#content", state="visible")
        
        # 等待元素隐藏
        page.wait_for_selector("#loading", state="hidden")
        
        # 等待元素附加到 DOM
        page.wait_for_selector("#content", state="attached")
        
        # 等待元素消失
        page.wait_for_selector("#loading", state="detached")
        
        # 等待导航完成
        page.click("#link")
        page.wait_for_load_state("load")  # 等待 load 事件
        page.wait_for_load_state("domcontentloaded")  # 等待 DOM 加载
        page.wait_for_load_state("networkidle")  # 等待网络空闲
        
        # 等待 URL 变化
        page.wait_for_url("**/dashboard")
        page.wait_for_url(re.compile(r".*/dashboard"))
        
        # 等待请求
        with page.expect_request("**/api/data") as req:
            page.click("#fetch-data")
        request = req.value
        print(request.url)
        
        # 等待响应
        with page.expect_response("**/api/data") as resp:
            page.click("#fetch-data")
        response = resp.value
        print(response.json())
        
        # 等待弹窗
        with page.expect_popup() as popup:
            page.click("#open-window")
        new_page = popup.value
        
        # 等待文件下载
        with page.expect_download() as download:
            page.click("#download-btn")
        download_info = download.value
        print(download_info.path())  # 临时文件路径
        download_info.save_as("saved_file.pdf")  # 保存到指定路径
        
        # 等待函数
        page.wait_for_function("() => document.querySelector('.status').textContent === '完成'")
        
        # 等待超时
        page.wait_for_timeout(1000)  # 等待 1 秒（不推荐）
        
        browser.close()


if __name__ == "__main__":
    main()
```

### 网络拦截

```python
from playwright.sync_api import sync_playwright, Route


def main():
    with sync_playwright() as p:
        browser = p.chromium.launch()
        page = browser.new_page()
        
        # 拦截请求
        def handle_route(route: Route):
            # 阻止加载图片
            if route.request.resource_type == "image":
                route.abort()
            else:
                route.continue_()
        
        page.route("**/*", handle_route)
        
        # 拦截特定请求
        page.route("**/api/data", lambda route: route.fulfill(
            status=200,
            content_type="application/json",
            body='{"message": "Mocked"}'
        ))
        
        # 修改请求头
        page.route("**/*", lambda route: route.continue_(
            headers={**route.request.headers, "X-Custom": "value"}
        ))
        
        # 修改响应
        def modify_response(route: Route):
            response = route.fetch()
            body = response.text()
            body = body.replace("旧文本", "新文本")
            route.fulfill(response=response, body=body)
        
        page.route("**/*.html", modify_response)
        
        page.goto("https://example.com")
        
        browser.close()


if __name__ == "__main__":
    main()
```

```python
from playwright.sync_api import sync_playwright


def main():
    with sync_playwright() as p:
        browser = p.chromium.launch()
        page = browser.new_page()
        
        # 监听请求
        page.on("request", lambda req: print(f">> {req.method} {req.url}"))
        
        # 监听响应
        page.on("response", lambda res: print(f"<< {res.status} {res.url}"))
        
        # 监听请求失败
        page.on("requestfailed", lambda req: print(f"Failed: {req.url}"))
        
        # 获取所有请求
        requests = []
        page.on("request", lambda req: requests.append(req))
        
        page.goto("https://example.com")
        
        # 分析请求
        api_requests = [r for r in requests if "/api/" in r.url]
        print(f"API 请求数: {len(api_requests)}")
        
        browser.close()


if __name__ == "__main__":
    main()
```

### JavaScript 执行

```python
from playwright.sync_api import sync_playwright


def main():
    with sync_playwright() as p:
        browser = p.chromium.launch()
        page = browser.new_page()
        page.goto("https://example.com")
        
        # 执行 JavaScript

        )

        result = element.evaluate("el => el.textContent")

                document.querySelector('#button').click();
            }
```

```python
    """
    新闻网站爬虫
    
    使用 Playwright 爬取动态加载的新闻文章
    """
    
    def __init__(self, headless: bool = True):
        """
        初始化爬虫
        
        Args:
            headless: 是否使用无头模式
        """
        self.headless = headless
        self.browser = None
        self.context = None
        self.page = None
    
    def start(self):
        """启动浏览器"""
        self.playwright = sync_playwright().start()
        self.browser = self.playwright.chromium.launch(
            headless=self.headless,
            args=["--disable-blink-features=AutomationControlled"]
        )
        
        self.context = self.browser.new_context(
            viewport={"width": 1920, "height": 1080},
            user_agent="Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
        )
        
        # 拦截图片请求（提高速度）
        self.context.route("**/*.{png,jpg,jpeg,gif,webp,svg}", 
                          lambda route: route.abort())
        
        self.page = self.context.new_page()
        logger.info("浏览器启动成功")
    
    def stop(self):
        """关闭浏览器"""
        if self.context:
            self.context.close()
        if self.browser:
            self.browser.close()
        if self.playwright:
            self.playwright.stop()
        logger.info("浏览器已关闭")
    
    def scrape_articles(self, url: str, max_articles: int = 10) -> List[ArticleInfo]:
        """
        爬取文章列表
        
        Args:
            url: 网站 URL
            max_articles: 最大文章数量
            
        Returns:
            文章列表
        """
        articles = []
        
        try:
            # 访问网站
            logger.info(f"访问: {url}")
            self.page.goto(url, wait_until="networkidle")
            
            # 等待文章列表加载
            self.page.wait_for_selector(".article-item")
            
            # 获取文章链接
            article_links = self.page.locator(".article-item a.title").all()[:max_articles]
            urls = [link.get_attribute("href") for link in article_links]
            
            # 爬取每篇文章
            for article_url in urls:
                if article_url:
                    article = self._scrape_article(article_url)
                    if article:
                        articles.append(article)
            
            logger.info(f"共爬取 {len(articles)} 篇文章")
            return articles
            
        except Exception as e:
            logger.error(f"爬取失败: {e}")
            return articles
    
    def _scrape_article(self, url: str) -> ArticleInfo | None:
        """
        爬取单篇文章
        
        Args:
            url: 文章 URL
            
        Returns:
            文章信息或 None
        """
        try:
            # 打开新标签页
            page = self.context.new_page()
            
            try:
                logger.info(f"爬取文章: {url}")
                page.goto(url, wait_until="networkidle")
                
                # 提取文章信息
                title = page.locator("h1.title").text_content(timeout=5000)
                author = page.locator(".author-name").text_content(timeout=5000)
                date = page.locator(".publish-date").text_content(timeout=5000)
                content = page.locator(".article-content").inner_text(timeout=5000)
                
                # 清理数据
                title = title.strip() if title else ""
                author = author.strip() if author else "未知"
                date = date.strip() if date else ""
                content = content.strip() if content else ""
                
                return ArticleInfo(
                    title=title,
                    author=author,
                    date=date,
                    content=content,
                    url=url
                )
                
            finally:
                page.close()
                
        except Exception as e:
            logger.warning(f"爬取文章失败 {url}: {e}")
            return None
    
    def __enter__(self):
        self.start()
        return self
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        self.stop()


# 使用示例
if __name__ == "__main__":
    with NewsScraper(headless=True) as scraper:
        articles = scraper.scrape_articles(
            "https://news.example.com",
            max_articles=5
        )
        
        for article in articles:
            print(f"标题: {article.title}")
            print(f"作者: {article.author}")
            print(f"日期: {article.date}")
            print(f"链接: {article.url}")
            print("-" * 50)
```
