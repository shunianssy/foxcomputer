# 9.2 图形绘制

Pygame 提供了丰富的图形绘制函数，可以绘制各种基本形状。

## 绘制基本图形

### 1. 矩形

```python
import pygame
import sys

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("绘制矩形")

# 颜色定义
RED = (255, 0, 0)
GREEN = (0, 255, 0)
BLUE = (0, 0, 255)
WHITE = (255, 255, 255)

clock = pygame.time.Clock()
running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
    
    screen.fill((30, 30, 30))  # 深灰色背景
    
    # 绘制填充矩形
    # 参数：目标 Surface, 颜色, (x, y, 宽度, 高度)
    pygame.draw.rect(screen, RED, (50, 50, 150, 100))
    
    # 绘制空心矩形（边框）
    # 参数：目标 Surface, 颜色, (x, y, 宽度, 高度), 边框宽度
    pygame.draw.rect(screen, GREEN, (250, 50, 150, 100), 3)
    
    # 使用 Rect 对象
    rect = pygame.Rect(450, 50, 150, 100)
    pygame.draw.rect(screen, BLUE, rect)
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

### 4. 多边形

```python
import pygame
import sys
import math

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("绘制多边形")

clock = pygame.time.Clock()
running = True

def get_polygon_points(center, radius, sides):
    """
    计算正多边形的顶点坐标
    
    参数:
        center: 中心点坐标 (x, y)
        radius: 外接圆半径
        sides: 边数
    
    返回:
        顶点坐标列表
    """
    points = []
    for i in range(sides):
        angle = 2 * math.pi * i / sides - math.pi / 2
        x = center[0] + radius * math.cos(angle)
        y = center[1] + radius * math.sin(angle)
        points.append((x, y))
    return points

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
    
    screen.fill((30, 30, 30))
    
    # 绘制三角形（填充）
    triangle = [(150, 150), (100, 250), (200, 250)]
    pygame.draw.polygon(screen, (255, 0, 0), triangle)
    
    # 绘制正方形（空心）
    square = [(300, 150), (400, 150), (400, 250), (300, 250)]
    pygame.draw.polygon(screen, (0, 255, 0), square, 3)
    
    # 绘制五边形
    pentagon = get_polygon_points((600, 200), 60, 5)
    pygame.draw.polygon(screen, (0, 0, 255), pentagon)
    
    # 绘制六边形
    hexagon = get_polygon_points((150, 400), 60, 6)
    pygame.draw.polygon(screen, (255, 255, 0), hexagon)
    
    # 绘制八边形
    octagon = get_polygon_points((350, 400), 60, 8)
    pygame.draw.polygon(screen, (255, 0, 255), octagon)
    
    # 绘制星形
    star_points = []
    for i in range(10):
        angle = 2 * math.pi * i / 10 - math.pi / 2
        radius = 60 if i % 2 == 0 else 30
        x = 600 + radius * math.cos(angle)
        y = 400 + radius * math.sin(angle)
        star_points.append((x, y))
    pygame.draw.polygon(screen, (0, 255, 255), star_points)
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

### 5. 椭圆与弧线

```python
import pygame
import sys

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("绘制椭圆与弧线")

clock = pygame.time.Clock()
running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
    
    screen.fill((30, 30, 30))
    
    # 绘制填充椭圆
    # 参数：目标 Surface, 颜色, (x, y, 宽度, 高度)
    pygame.draw.ellipse(screen, (255, 0, 0), (50, 150, 200, 100))
    
    # 绘制空心椭圆
    pygame.draw.ellipse(screen, (0, 255, 0), (300, 150, 150, 150), 3)
    
    # 绘制圆形（特殊的椭圆）
    pygame.draw.ellipse(screen, (0, 0, 255), (500, 150, 120, 120))
    
    # 绘制弧线
    # 参数：目标 Surface, 颜色, 边界矩形, 起始角度, 结束角度, 线宽
    # 角度使用弧度制，0 为右侧，顺时针方向
    
    # 绘制半圆弧
    pygame.draw.arc(screen, (255, 255, 0), (50, 350, 150, 150), 
                    0, math.pi, 5)
    
    # 绘制四分之一圆弧
    pygame.draw.arc(screen, (255, 0, 255), (250, 350, 150, 150), 
                    0, math.pi / 2, 5)
    
    # 绘制笑脸
    pygame.draw.circle(screen, (255, 255, 0), (550, 425), 75)  # 脸
    pygame.draw.circle(screen, (0, 0, 0), (520, 400), 10)       # 左眼
    pygame.draw.circle(screen, (0, 0, 0), (580, 400), 10)       # 右眼
    pygame.draw.arc(screen, (0, 0, 0), (500, 420, 100, 50), 
                    math.pi, 2 * math.pi, 3)  # 微笑
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

## Rect 对象详解

```python
import pygame

# 创建 Rect 对象的多种方式
rect1 = pygame.Rect(10, 20, 100, 50)  # (x, y, width, height)
rect2 = pygame.Rect((10, 20), (100, 50))  # ((x, y), (width, height))

# Rect 属性
print(f"位置: ({rect1.x}, {rect1.y})")      # 左上角坐标
print(f"大小: {rect1.width} x {rect1.height}")
print(f"左上角: {rect1.topleft}")
print(f"右下角: {rect1.bottomright}")
print(f"中心: {rect1.center}")
print(f"左边: {rect1.left}, 右边: {rect1.right}")
print(f"上边: {rect1.top}, 下边: {rect1.bottom}")

# Rect 方法
rect1.move(10, 20)      # 移动（返回新 Rect）
rect1.move_ip(10, 20)   # 原地移动（in-place）
rect1.inflate(10, 10)   # 放大（返回新 Rect）
rect1.clamp(another_rect)  # 将 rect 移动到另一个 rect 内部
rect1.colliderect(another_rect)  # 检测两个矩形是否碰撞
rect1.collidepoint((x, y))       # 检测点是否在矩形内
```

## 小结

本章要点：

1. **pygame.draw.rect()**：绘制矩形
2. **pygame.draw.circle()**：绘制圆形
3. **pygame.draw.line()**：绘制直线
4. **pygame.draw.polygon()**：绘制多边形
5. **pygame.draw.ellipse()**：绘制椭圆
6. **pygame.draw.arc()**：绘制弧线
7. **pygame.Rect**：矩形对象，用于碰撞检测

> 💡 **提示**：绘制函数的最后一个参数通常是线宽，设为 0 或省略表示填充。
