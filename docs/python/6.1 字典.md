# 6.1 字典

字典是 Python 中最重要的数据结构之一，用于存储键值对。

## 字典基础

### 创建字典

```python
# 创建空字典
empty_dict = {}
empty_dict2 = dict()

# 创建有元素的字典
person = {
    "name": "张三",
    "age": 25,
    "city": "北京"
}

# 使用 dict() 函数
person2 = dict(name="李四", age=30, city="上海")

# 从键值对列表创建
items = [("a", 1), ("b", 2), ("c", 3)]
d = dict(items)

# 使用 dict.fromkeys()
keys = ["a", "b", "c"]
d = dict.fromkeys(keys)  # {'a': None, 'b': None, 'c': None}
d = dict.fromkeys(keys, 0)  # {'a': 0, 'b': 0, 'c': 0}
```

### 访问元素

```python
person = {"name": "张三", "age": 25, "city": "北京"}

# 通过键访问
print(person["name"])  # 输出: 张三

# 使用 get() 方法（推荐）
print(person.get("name"))  # 输出: 张三
print(person.get("job", "未知"))  # 输出: 未知（键不存在时返回默认值）

# 键不存在时抛出异常
# print(person["job"])  # KeyError
```

### 修改元素

```python
person = {"name": "张三", "age": 25}

# 修改值
person["age"] = 26
print(person)  # 输出: {'name': '张三', 'age': 26}

# 添加新键值对
person["city"] = "北京"
print(person)  # 输出: {'name': '张三', 'age': 26, 'city': '北京'}

# 使用 update() 方法
person.update({"job": "工程师", "salary": 10000})
print(person)
```

### 删除元素

```python
person = {"name": "张三", "age": 25, "city": "北京"}

# del 删除
del person["city"]
print(person)  # 输出: {'name': '张三', 'age': 25}

# pop() 删除并返回值
age = person.pop("age")
print(age)  # 输出: 25
print(person)  # 输出: {'name': '张三'}

# popitem() 删除并返回最后一个键值对
person["age"] = 25
item = person.popitem()
print(item)  # 输出: ('age', 25)

# clear() 清空字典
person.clear()
print(person)  # 输出: {}
```

## 字典方法

### 获取键、值、键值对

```python
person = {"name": "张三", "age": 25, "city": "北京"}

# 获取所有键
keys = person.keys()
print(list(keys))  # 输出: ['name', 'age', 'city']

# 获取所有值
values = person.values()
print(list(values))  # 输出: ['张三', 25, '北京']

# 获取所有键值对
items = person.items()
print(list(items))  # 输出: [('name', '张三'), ('age', 25), ('city', '北京')]
```

### 成员判断

```python
person = {"name": "张三", "age": 25}

# 判断键是否存在
print("name" in person)  # 输出: True
print("job" in person)   # 输出: False

# 判断值是否存在
print("张三" in person.values())  # 输出: True
```

### setdefault() 方法

```python
# setdefault()：如果键不存在，则添加并设置默认值
person = {"name": "张三"}

# 键存在，返回已有值
name = person.setdefault("name", "李四")
print(name)  # 输出: 张三

# 键不存在，添加并返回默认值
job = person.setdefault("job", "工程师")
print(job)  # 输出: 工程师
print(person)  # 输出: {'name': '张三', 'job': '工程师'}
```

## 遍历字典

```python
person = {"name": "张三", "age": 25, "city": "北京"}

# 遍历键
for key in person:
    print(key)

for key in person.keys():
    print(key)

# 遍历值
for value in person.values():
    print(value)

# 遍历键值对（推荐）
for key, value in person.items():
    print(f"{key}: {value}")
```

## 字典推导式

```python
# 基本语法：{key_expr: value_expr for item in iterable}

# 创建平方数字典
squares = {x: x ** 2 for x in range(1, 6)}
print(squares)  # 输出: {1: 1, 2: 4, 3: 9, 4: 16, 5: 25}

# 带条件
even_squares = {x: x ** 2 for x in range(1, 11) if x % 2 == 0}
print(even_squares)  # 输出: {2: 4, 4: 16, 6: 36, 8: 64, 10: 100}

# 键值互换
original = {"a": 1, "b": 2, "c": 3}
swapped = {v: k for k, v in original.items()}
print(swapped)  # 输出: {1: 'a', 2: 'b', 3: 'c'}

# 从两个列表创建字典
keys = ["name", "age", "city"]
values = ["张三", 25, "北京"]
person = {k: v for k, v in zip(keys, values)}
print(person)  # 输出: {'name': '张三', 'age': 25, 'city': '北京'}
```

## 字典合并

```python
# Python 3.9+ 使用 | 运算符
dict1 = {"a": 1, "b": 2}
dict2 = {"c": 3, "d": 4}
merged = dict1 | dict2
print(merged)  # 输出: {'a': 1, 'b': 2, 'c': 3, 'd': 4}

# 使用 |= 更新
dict1 |= dict2
print(dict1)  # 输出: {'a': 1, 'b': 2, 'c': 3, 'd': 4}

# 使用 update() 方法
dict1 = {"a": 1, "b": 2}
dict2 = {"c": 3, "d": 4}
dict1.update(dict2)
print(dict1)  # 输出: {'a': 1, 'b': 2, 'c': 3, 'd': 4}

# 使用 ** 解包
dict1 = {"a": 1, "b": 2}
dict2 = {"c": 3, "d": 4}
merged = {**dict1, **dict2}
print(merged)  # 输出: {'a': 1, 'b': 2, 'c': 3, 'd': 4}

# 键冲突时，后面的覆盖前面的
dict1 = {"a": 1, "b": 2}
dict2 = {"b": 3, "c": 4}
merged = {**dict1, **dict2}
print(merged)  # 输出: {'a': 1, 'b': 3, 'c': 4}
```

## collections 模块

### defaultdict

```python
from collections import defaultdict

# 普通 dict 需要检查键是否存在
word_count = {}
for word in ["apple", "banana", "apple", "orange", "banana", "apple"]:
    if word in word_count:
        word_count[word] += 1
    else:
        word_count[word] = 1

# 使用 defaultdict 更简洁
word_count = defaultdict(int)  # 默认值为 0
for word in ["apple", "banana", "apple", "orange", "banana", "apple"]:
    word_count[word] += 1

print(dict(word_count))  # 输出: {'apple': 3, 'banana': 2, 'orange': 1}

# 其他默认值类型
d = defaultdict(list)   # 默认值为 []
d = defaultdict(set)    # 默认值为 set()
d = defaultdict(str)    # 默认值为 ""
```

### OrderedDict

```python
from collections import OrderedDict

# 有序字典（Python 3.7+ 普通 dict 已有序）
od = OrderedDict()
od["a"] = 1
od["b"] = 2
od["c"] = 3
print(od)  # 输出: OrderedDict([('a', 1), ('b', 2), ('c', 3)])

# 移动到末尾
od.move_to_end("a")
print(od)  # 输出: OrderedDict([('b', 2), ('c', 3), ('a', 1)])

# 按插入顺序排序
d = {"c": 3, "a": 1, "b": 2}
sorted_d = OrderedDict(sorted(d.items()))
print(sorted_d)  # 输出: OrderedDict([('a', 1), ('b', 2), ('c', 3)])
```

### Counter

```python
from collections import Counter

# 计数器
text = "hello world"
counter = Counter(text)
print(counter)  # 输出: Counter({'l': 3, 'o': 2, 'h': 1, 'e': 1, ' ': 1, 'w': 1, 'r': 1, 'd': 1})

# 统计列表
words = ["apple", "banana", "apple", "orange", "banana", "apple"]
word_count = Counter(words)
print(word_count)  # 输出: Counter({'apple': 3, 'banana': 2, 'orange': 1})

# 最常见的元素
print(word_count.most_common(2))  # 输出: [('apple', 3), ('banana', 2)]

# 计数器运算
c1 = Counter("aabbcc")
c2 = Counter("aabbdd")
print(c1 + c2)  # 输出: Counter({'a': 4, 'b': 4, 'c': 2, 'd': 2})
print(c1 - c2)  # 输出: Counter({'c': 2})
print(c1 & c2)  # 交集: Counter({'a': 2, 'b': 2})
print(c1 | c2)  # 并集: Counter({'a': 2, 'b': 2, 'c': 2, 'd': 2})
```

## 实战练习

### 练习一：词频统计

```python
def word_frequency(text):
    """统计文本中单词的出现频率"""
    # 分割单词并转小写
    words = text.lower().split()
    
    # 统计频率
    freq = {}
    for word in words:
        freq[word] = freq.get(word, 0) + 1
    
    # 按频率排序
    sorted_freq = sorted(freq.items(), key=lambda x: x[1], reverse=True)
    return sorted_freq

text = "hello world hello python world python python"
result = word_frequency(text)
print(result)  # 输出: [('python', 3), ('hello', 2), ('world', 2)]
```

### 练习二：学生成绩管理

```python
class GradeManager:
    """学生成绩管理类"""
    
    def __init__(self):
        self.grades = {}
    
    def add_student(self, name):
        """添加学生"""
        if name not in self.grades:
            self.grades[name] = []
    
    def add_grade(self, name, grade):
        """添加成绩"""
        if name not in self.grades:
            self.grades[name] = []
        self.grades[name].append(grade)
    
    def get_average(self, name):
        """获取平均分"""
        if name in self.grades and self.grades[name]:
            return sum(self.grades[name]) / len(self.grades[name])
        return 0
    
    def get_all_averages(self):
        """获取所有学生的平均分"""
        return {name: self.get_average(name) for name in self.grades}
    
    def get_top_student(self):
        """获取成绩最好的学生"""
        averages = self.get_all_averages()
        if averages:
            return max(averages, key=averages.get)
        return None

# 使用示例
manager = GradeManager()
manager.add_grade("张三", 85)
manager.add_grade("张三", 90)
manager.add_grade("李四", 92)
manager.add_grade("李四", 88)
manager.add_grade("王五", 78)

print(f"张三平均分: {manager.get_average('张三'):.1f}")
print(f"所有学生平均分: {manager.get_all_averages()}")
print(f"成绩最好的学生: {manager.get_top_student()}")
```

### 练习三：分组数据

```python
def group_by(items, key_func):
    """按指定键函数分组"""
    groups = {}
    for item in items:
        key = key_func(item)
        if key not in groups:
            groups[key] = []
        groups[key].append(item)
    return groups

# 示例：按首字母分组
words = ["apple", "banana", "apricot", "blueberry", "cherry", "avocado"]
grouped = group_by(words, lambda x: x[0])
print(grouped)
# 输出: {'a': ['apple', 'apricot', 'avocado'], 'b': ['banana', 'blueberry'], 'c': ['cherry']}

# 示例：按成绩等级分组
students = [
    {"name": "张三", "score": 85},
    {"name": "李四", "score": 92},
    {"name": "王五", "score": 78},
    {"name": "赵六", "score": 95},
    {"name": "钱七", "score": 60}
]

def get_grade(student):
    score = student["score"]
    if score >= 90:
        return "A"
    elif score >= 80:
        return "B"
    elif score >= 70:
        return "C"
    else:
        return "D"

grouped = group_by(students, get_grade)
for grade, students_list in grouped.items():
    names = [s["name"] for s in students_list]
    print(f"{grade} 级: {names}")
```

### 练习四：缓存装饰器

```python
def memoize(func):
    """使用字典实现缓存装饰器"""
    cache = {}
    
    def wrapper(*args):
        # 将参数转换为可哈希的键
        key = str(args)
        if key not in cache:
            cache[key] = func(*args)
            print(f"计算并缓存: {args}")
        else:
            print(f"从缓存获取: {args}")
        return cache[key]
    
    return wrapper

@memoize
def fibonacci(n):
    """计算斐波那契数"""
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

print(fibonacci(10))
```

本章要点：
