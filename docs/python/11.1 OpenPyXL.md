# 11.1 OpenPyXL Excel 文件处理

OpenPyXL 是 Python 中用于读写 Excel 2010+ xlsx/xlsm 文件的库，功能强大且易于使用。

## 安装

```bash
# 使用 pip 安装
pip install openpyxl

# 验证安装
python -c "import openpyxl; print(openpyxl.__version__)"
```

## 基本概念

- **Workbook（工作簿）**：整个 Excel 文件
- **Worksheet（工作表）**：工作簿中的单个表格
- **Cell（单元格）**：工作表中的单个格子
- **Row（行）**：水平方向的一行单元格
- **Column（列）**：垂直方向的一列单元格

## 创建工作簿

### 新建 Excel 文件

```python
from openpyxl import Workbook

# 创建新的工作簿
wb = Workbook()

# 获取默认活动工作表
ws = wb.active
ws.title = "Sheet1"  # 重命名工作表

# 保存文件
wb.save("example.xlsx")
```

### 创建多个工作表

```python
from openpyxl import Workbook

wb = Workbook()

# 获取活动工作表并重命名
ws1 = wb.active
ws1.title = "数据表"

# 创建新的工作表
ws2 = wb.create_sheet("分析表")  # 在末尾添加
ws3 = wb.create_sheet("汇总表", 0)  # 在指定位置添加（索引0表示第一个）

# 获取所有工作表名称
print(wb.sheetnames)  # 输出: ['汇总表', '数据表', '分析表']

# 删除工作表
wb.remove(ws3)

wb.save("multi_sheets.xlsx")
```

### 遍历数据

```python
from openpyxl import load_workbook

wb = load_workbook("batch_data.xlsx")
ws = wb.active

# 遍历所有行
print("=== 遍历所有行 ===")
for row in ws.iter_rows():
    for cell in row:
        print(cell.value, end="\t")
    print()

# 遍历指定范围
print("\n=== 遍历指定范围 ===")
for row in ws.iter_rows(min_row=2, max_row=5, min_col=1, max_col=3):
    for cell in row:
        print(cell.value, end="\t")
    print()

# 只获取值（不获取单元格对象）
print("\n=== 只获取值 ===")
for row in ws.iter_rows(values_only=True):
    print(row)

# 遍历列
print("\n=== 遍历列 ===")
for col in ws.iter_cols(min_col=1, max_col=2):
    for cell in col:
        print(cell.value, end="\t")
    print()
```

## 实用示例

### 读取 Excel 并转换为字典列表

```python
from openpyxl import load_workbook
from typing import List, Dict, Any

def excel_to_dict_list(file_path: str, sheet_name: str = None) -> List[Dict[str, Any]]:
    """
    将 Excel 表格数据转换为字典列表
    
    Args:
        file_path: Excel 文件路径
        sheet_name: 工作表名称，默认为活动工作表
        
    Returns:
        字典列表，每个字典代表一行数据
    """
    wb = load_workbook(file_path)
    ws = wb[sheet_name] if sheet_name else wb.active
    
    # 获取表头
    headers = [cell.value for cell in ws[1]]
    
    # 转换数据
    result = []
    for row in ws.iter_rows(min_row=2, values_only=True):
        row_dict = {}
        for i, value in enumerate(row):
            if i < len(headers) and headers[i]:
                row_dict[headers[i]] = value
        result.append(row_dict)
    
    wb.close()
    return result


# 使用示例
data = excel_to_dict_list("batch_data.xlsx")
for item in data:
    print(item)
```

### 导出数据到 Excel

```python
from openpyxl import Workbook
from openpyxl.styles import Font, Alignment, PatternFill, Border, Side
from typing import List, Dict, Any

def export_to_excel(
    data: List[Dict[str, Any]],
    file_path: str,
    sheet_name: str = "Sheet1"
) -> None:
    """
    将字典列表导出为 Excel 文件
    
    Args:
        data: 字典列表数据
        file_path: 输出文件路径
        sheet_name: 工作表名称
    """
    if not data:
        raise ValueError("数据不能为空")
    
    wb = Workbook()
    ws = wb.active
    ws.title = sheet_name
    
    # 获取所有字段名
    headers = list(data[0].keys())
    
    # 写入表头
    header_font = Font(bold=True, color="FFFFFF")
    header_fill = PatternFill(start_color="4472C4", end_color="4472C4", fill_type="solid")
    header_align = Alignment(horizontal="center", vertical="center")
    thin_border = Border(
        left=Side(style='thin'),
        right=Side(style='thin'),
        top=Side(style='thin'),
        bottom=Side(style='thin')
    )
    
    for col, header in enumerate(headers, 1):
        cell = ws.cell(row=1, column=col, value=header)
        cell.font = header_font
        cell.fill = header_fill
        cell.alignment = header_align
        cell.border = thin_border
    
    # 写入数据
    for row_idx, row_data in enumerate(data, 2):
        for col_idx, header in enumerate(headers, 1):
            cell = ws.cell(row=row_idx, column=col_idx, value=row_data.get(header))
            cell.border = thin_border
            cell.alignment = Alignment(horizontal="center")
    
    # 自动调整列宽
    for col in ws.columns:
        max_length = 0
        column = col[0].column_letter
        for cell in col:
            try:
                if cell.value:
                    max_length = max(max_length, len(str(cell.value)))
            except:
                pass
        ws.column_dimensions[column].width = max_length + 2
    
    wb.save(file_path)


# 使用示例
sample_data = [
    {"ID": 1, "姓名": "张三", "部门": "技术部", "工资": 15000},
    {"ID": 2, "姓名": "李四", "部门": "销售部", "工资": 12000},
    {"ID": 3, "姓名": "王五", "部门": "人事部", "工资": 10000},
]

export_to_excel(sample_data, "employees.xlsx", "员工信息")
```

### 批量处理多个工作表

```python
from openpyxl import load_workbook, Workbook
from typing import Dict, List, Any
import os

def merge_excel_files(
    file_paths: List[str],
    output_path: str,
    merge_mode: str = "sheets"
) -> None:
    """
    合并多个 Excel 文件
    
    Args:
        file_paths: 要合并的文件路径列表
        output_path: 输出文件路径
        merge_mode: 合并模式
            - "sheets": 每个文件作为一个工作表
            - "rows": 所有数据合并到一个工作表
    """
    output_wb = Workbook()
    output_ws = output_wb.active
    
    if merge_mode == "sheets":
        # 每个文件作为一个工作表
        for i, file_path in enumerate(file_paths):
            file_name = os.path.splitext(os.path.basename(file_path))[0]
            
            if i == 0:
                ws = output_ws
                ws.title = file_name[:31]  # 工作表名最长31字符
            else:
                ws = output_wb.create_sheet(file_name[:31])
            
            # 复制数据
            source_wb = load_workbook(file_path)
            source_ws = source_wb.active
            
            for row in source_ws.iter_rows(values_only=True):
                ws.append(row)
            
            source_wb.close()
    
    elif merge_mode == "rows":
        # 所有数据合并到一个工作表
        header_written = False
        
        for file_path in file_paths:
            source_wb = load_workbook(file_path)
            source_ws = source_wb.active
            
            rows = list(source_ws.iter_rows(values_only=True))
            
            if not header_written and rows:
                output_ws.append(rows[0])  # 写入表头
                header_written = True
                rows = rows[1:]  # 跳过表头
            
            for row in rows:
                output_ws.append(row)
            
            source_wb.close()
    
    output_wb.save(output_path)


# 使用示例
# merge_excel_files(["file1.xlsx", "file2.xlsx"], "merged.xlsx", "sheets")
```

## 小结

本章要点：

1. **基本操作**：创建、读取、保存工作簿和工作表
2. **数据读写**：单元格操作、批量写入、遍历数据
