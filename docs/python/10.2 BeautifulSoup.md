# 10.2 BeautifulSoup HTML 解析

BeautifulSoup 是一个强大的 HTML/XML 解析库，能够方便地从网页中提取数据。

## 安装

```bash
# 安装 beautifulsoup4 和解析器
pip install beautifulsoup4 lxml

# 验证安装
python -c "from bs4 import BeautifulSoup; print('安装成功')"
```

## 基本使用

### 创建解析对象

```python
from bs4 import BeautifulSoup
import requests

# 从字符串创建
html = """
<html>
<head><title>测试页面</title></head>
<body>
    <h1>欢迎</h1>
    <p class="intro">这是一个测试段落</p>
    <div id="content">
        <ul>
            <li>项目 1</li>
            <li>项目 2</li>
            <li>项目 3</li>
        </ul>
    </div>
</body>
</html>
"""

# 创建 BeautifulSoup 对象
soup = BeautifulSoup(html, "lxml")  # 使用 lxml 解析器

# 从网页获取并解析
response = requests.get("https://example.com")
soup = BeautifulSoup(response.text, "lxml")

# 从文件解析
with open("page.html", "r", encoding="utf-8") as f:
    soup = BeautifulSoup(f, "lxml")
```

```python
from bs4 import BeautifulSoup
```

## 标签选择

### 基本选择

```python
from bs4 import BeautifulSoup

html = """
<html>
<body>
    <h1>标题</h1>
    <p>段落 1</p>

        <a href="https://example.com">链接</a>

soup = BeautifulSoup(html, "lxml")

# 选择第一个匹配的标签
print(soup.h1)  # <h1>标题</h1>
print(soup.p)  # <p>段落 1</p>（只返回第一个）

# 获取标签属性

print(soup.a.get("href"))  # https://example.com
print(soup.a.attrs)  # {'href': 'https://example.com'}
```

### find 方法

```python
from bs4 import BeautifulSoup

html = """
<div>
    <p class="intro" id="p1">介绍段落</p>
    <p class="content">内容段落</p>
    <p class="intro">另一个介绍</p>
    <a href="link1">链接1</a>
    <a href="link2">链接2</a>
</div>
"""

soup = BeautifulSoup(html, "lxml")

# find() - 查找第一个匹配的标签
p = soup.find("p")
print(p)  # <p class="intro" id="p1">介绍段落</p>

# 按类名查找
p = soup.find("p", class_="intro")  # class 是关键字，使用 class_
print(p)

# 按 id 查找
p = soup.find("p", id="p1")
print(p)

# 按属性查找
a = soup.find("a", href="link1")
print(a)

# 多条件查找
p = soup.find("p", class_="intro", id="p1")
print(p)

# 使用 attrs 字典
p = soup.find("p", attrs={"class": "intro", "id": "p1"})
print(p)

# 查找包含特定文本的标签
p = soup.find("p", string="内容段落")
print(p)

# 正则匹配
import re
p = soup.find("p", class_=re.compile(r"intro|content"))
print(p)
```

### find_all 方法

```python
from bs4 import BeautifulSoup

html = """
<ul>
    <li class="item">项目 1</li>
    <li class="item">项目 2</li>
    <li class="special">项目 3</li>
    <li class="item">项目 4</li>
</ul>
"""

soup = BeautifulSoup(html, "lxml")

# find_all() - 查找所有匹配的标签
items = soup.find_all("li")
print(len(items))  # 4

# 限制返回数量
items = soup.find_all("li", limit=2)
print(len(items))  # 2

# 按类名查找
items = soup.find_all("li", class_="item")
print(len(items))  # 3

# 多类名匹配（同时包含）
items = soup.find_all("li", class_=["item", "special"])
print(len(items))  # 4

# 递归查找（默认 True）
items = soup.find_all("li", recursive=False)  # 只查找直接子元素

# 查找多种标签
tags = soup.find_all(["li", "ul"])
print(len(tags))  # 5

# 使用函数过滤
def has_special_class(tag):
    """判断标签是否有 special 类"""
    return tag.has_attr("class") and "special" in tag["class"]

special = soup.find_all(has_special_class)
print(special)
```

### CSS 选择器

```python
from bs4 import BeautifulSoup

html = """
<div id="container">
    <h1 class="title main">主标题</h1>

items = soup.select("li")

items = soup.select(".item")

container = soup.select_one("#container")

# 后代选择器
items = soup.select("ul li")

# 子元素选择器
items = soup.select("ul > li")
print(len(items))  # 3

# 多类选择器（同时包含）
title = soup.select_one(".title.main")
print(title.string)  # 主标题

# 属性选择器
link = soup.select_one('a[href^="https"]')  # href 以 https 开头
print(link.string)  # 链接

# 组合选择器
elements = soup.select("h1, p")
print(len(elements))  # 2

# 伪类选择器
first_item = soup.select_one("li:first-child")
print(first_item.string)  # 项目 1

last_item = soup.select_one("li:last-child")
print(last_item.string)  # 项目 3

# nth-child
second_item = soup.select_one("li:nth-child(2)")
print(second_item.string)  # 项目 2
```

## 遍历文档树

### 子节点遍历

```python
from bs4 import BeautifulSoup

html = """
<div id="parent">
    <p>段落 1</p>
    <p>段落 2</p>
    <span>文本</span>
</div>
"""

soup = BeautifulSoup(html, "lxml")
div = soup.find("div")

# 获取所有子节点（包括换行符等）
print(div.contents)  # 列表形式

# 遍历子节点
for child in div.children:
    print(child)

# 获取所有子标签（不含文本节点）
print(div.find_all(recursive=False))

# 获取直接子标签
for tag in div.find_all(True, recursive=False):
    print(tag.name)

# 获取第一个子标签
print(div.find(True))  # <p>段落 1</p>
```

### 父节点遍历

```python
from bs4 import BeautifulSoup

html = """
<body>
    <div>
        <p>段落</p>
    </div>
</body>
"""

soup = BeautifulSoup(html, "lxml")
p = soup.find("p")

# 直接父节点
print(p.parent.name)  # div

# 所有父节点
for parent in p.parents:
    print(parent.name)
# 输出: div -> body -> html -> [document]

# 查找父级中最近的指定标签
div = p.find_parent("div")
print(div.name)  # div

# 查找所有父级中的指定标签
parents = p.find_parents("div")
print(len(parents))  # 1
```

### 兄弟节点遍历

```python
from bs4 import BeautifulSoup

html = """
<ul>
    <li id="item1">项目 1</li>
    <li id="item2">项目 2</li>
    <li id="item3">项目 3</li>
    <li id="item4">项目 4</li>
</ul>
"""

soup = BeautifulSoup(html, "lxml")
item2 = soup.find("li", id="item2")

# 下一个兄弟节点（包括文本节点）
print(item2.next_sibling)  # 换行符

# 下一个兄弟标签
print(item2.next_sibling.next_sibling)  # <li id="item3">项目 3</li>

# 上一个兄弟节点
print(item2.previous_sibling)  # 换行符

# 上一个兄弟标签
print(item2.previous_sibling.previous_sibling)  # <li id="item1">项目 1</li>

# 遍历后面的兄弟节点
for sibling in item2.next_siblings:
    if sibling.name:  # 忽略文本节点
        print(sibling.get("id"))

# 遍历前面的兄弟节点
for sibling in item2.previous_siblings:
    if sibling.name:
        print(sibling.get("id"))

# 查找下一个匹配的兄弟标签
next_item = item2.find_next_sibling("li")
print(next_item.get("id"))  # item3

# 查找所有后面的兄弟标签
next_items = item2.find_next_siblings("li")
print(len(next_items))  # 2
```

## 文本处理

```python
from bs4 import BeautifulSoup

html = """
<div>
    <p>段落 <b>加粗</b> 文本</p>
    <p>另一个段落</p>
    <!-- 这是注释 -->
</div>
"""

soup = BeautifulSoup(html, "lxml")
div = soup.find("div")

# 获取所有文本（包含子标签文本）
print(div.text)  # 合并所有文本
print(div.get_text())  # 同上

# 指定分隔符
print(div.get_text(separator="|"))

# 去除空白
print(div.get_text(strip=True))

# 获取纯文本列表
print(div.stripped_strings)  # 生成器
print(list(div.stripped_strings))

# 遍历文本
for string in div.strings:
    print(repr(string))

# 遍历非空文本
for string in div.stripped_strings:
    print(string)

# NavigableString 对象
p = soup.find("p")
print(type(p.string))  # <class 'bs4.element.NavigableString'>

# 替换文本
p.string.replace_with("新文本")
print(p.string)

# 注释处理
comment = soup.find(string=lambda text: isinstance(text, Comment))
print(comment)  # 这是注释
```

## 修改文档

```python
from bs4 import BeautifulSoup

html = """
<div>
    <p>原段落</p>
</div>
"""

soup = BeautifulSoup(html, "lxml")
div = soup.find("div")
p = soup.find("p")

# 修改标签内容
p.string = "修改后的段落"
print(p)  # <p>修改后的段落</p>

# 修改标签属性
p["class"] = "new-class"
p["id"] = "new-id"
print(p)  # <p class="new-class" id="new-id">修改后的段落</p>

# 添加属性
p["data-value"] = "123"

# 删除属性
del p["class"]

# 创建新标签
new_tag = soup.new_tag("span")
new_tag.string = "新标签"
print(new_tag)  # <span>新标签</span>

# 在末尾添加
div.append(new_tag)
print(div)

# 在开头添加
another_tag = soup.new_tag("a", href="https://example.com")
another_tag.string = "链接"
div.insert(0, another_tag)
print(div)

# 在指定位置插入
p.insert_after(soup.new_tag("br"))
print(div)

# 清空内容
p.clear()
print(p)  # <p id="new-id"></p>

# 删除标签
p.decompose()
print(div)  # p 标签已被删除

# 提取标签（返回被提取的标签）
extracted = div.extract()
print(extracted)
```

## 实战示例

### 爬取新闻标题

```python
import requests
from bs4 import BeautifulSoup
import logging
from typing import List, Dict
from dataclasses import dataclass

# 配置日志
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


@dataclass
class NewsItem:
    """新闻条目数据类"""
    title: str
    link: str
    source: str


def fetch_news(url: str, headers: dict = None) -> List[NewsItem]:
    """
    爬取新闻标题
    
    Args:
        url: 新闻网站 URL
        headers: 请求头
        
    Returns:
        新闻列表
    """
    default_headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
    }
    headers = headers or default_headers
    
    try:
        # 发送请求
        response = requests.get(url, headers=headers, timeout=10)
        response.raise_for_status()
        response.encoding = response.apparent_encoding
        
        # 解析 HTML
        soup = BeautifulSoup(response.text, "lxml")
        news_list = []
        
        # 示例：提取新闻标题和链接（需要根据实际网站结构调整）
        # 假设新闻在 <div class="news-item"> 中
        news_items = soup.select(".news-item")
        
        for item in news_items:
            # 提取标题
            title_tag = item.select_one("a")
            if title_tag:
                title = title_tag.get_text(strip=True)
                link = title_tag.get("href", "")
                
                # 处理相对链接
                if link.startswith("/"):
                    link = url.rstrip("/") + link
                
                news_list.append(NewsItem(
                    title=title,
                    link=link,
                    source=url
                ))
        
        logger.info(f"成功获取 {len(news_list)} 条新闻")
        return news_list
        
    except requests.RequestException as e:
        logger.error(f"请求失败: {e}")
        return []
    except Exception as e:
        logger.error(f"解析失败: {e}")
        return []


# 使用示例
if __name__ == "__main__":
    # 示例 URL（需要替换为实际网站）
    news = fetch_news("https://news.example.com")
    for item in news[:5]:
        print(f"标题: {item.title}")
        print(f"链接: {item.link}")
        print("-" * 50)
```

### 爬取表格数据

```python
import requests
from bs4 import BeautifulSoup
import pandas as pd
from typing import List, Dict
import logging

logger = logging.getLogger(__name__)


def parse_html_table(url: str, table_index: int = 0) -> pd.DataFrame:
    """
    解析 HTML 表格为 DataFrame
    
    Args:
        url: 网页 URL
        table_index: 表格索引（页面上第几个表格）
        
    Returns:
        pandas DataFrame
    """
    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
    }
    
    try:
        response = requests.get(url, headers=headers, timeout=10)
        response.raise_for_status()
        
        soup = BeautifulSoup(response.text, "lxml")
        
        # 找到指定表格
        tables = soup.find_all("table")
        if table_index >= len(tables):
            raise ValueError(f"表格索引 {table_index} 超出范围")
        
        table = tables[table_index]
        
        # 提取表头
        headers = []
        header_row = table.find("thead")
        if header_row:
            headers = [th.get_text(strip=True) for th in header_row.find_all("th")]
        else:
            # 尝试从第一行获取表头
            first_row = table.find("tr")
            if first_row:
                headers = [th.get_text(strip=True) for th in first_row.find_all("th")]
        
        # 提取数据行
        rows = []
        tbody = table.find("tbody") or table
        for tr in tbody.find_all("tr"):
            # 跳过表头行
            if tr.find("th"):
                continue
            
            cells = tr.find_all(["td", "th"])
            row = [cell.get_text(strip=True) for cell in cells]
            if row:  # 跳过空行
                rows.append(row)
        
        # 创建 DataFrame
        if headers and len(headers) == len(rows[0]) if rows else True:
            df = pd.DataFrame(rows, columns=headers)
        else:
            df = pd.DataFrame(rows)
        
        logger.info(f"成功解析表格，共 {len(df)} 行数据")
        return df
        
    except Exception as e:
        logger.error(f"解析表格失败: {e}")
        return pd.DataFrame()


# 使用示例
if __name__ == "__main__":
    # 示例：爬取维基百科表格
    df = parse_html_table("https://example.com/data-table")
    print(df.head())
    
    # 保存为 CSV
    df.to_csv("table_data.csv", index=False, encoding="utf-8-sig")
```

### 分页爬取

```python
import requests
from bs4 import BeautifulSoup
import time
import logging
from typing import List, Generator

logger = logging.getLogger(__name__)


def crawl_paginated_site(
    base_url: str,
    total_pages: int,
    delay: float = 1.0
) -> Generator[List[str], None, None]:
    """
    分页爬取网站内容
    
    Args:
        base_url: 基础 URL（包含 {page} 占位符）
        total_pages: 总页数
        delay: 请求间隔（秒）
        
    Yields:
        每页的内容列表
    """
    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
    }
    
    for page in range(1, total_pages + 1):
        url = base_url.format(page=page)
        logger.info(f"正在爬取第 {page} 页: {url}")
        
        try:
            response = requests.get(url, headers=headers, timeout=10)
            response.raise_for_status()
            
            soup = BeautifulSoup(response.text, "lxml")
            
            # 提取内容（根据实际网站结构调整选择器）
            items = []
            for item in soup.select(".content-item"):
                title = item.select_one(".title")
                if title:
                    items.append(title.get_text(strip=True))
            
            yield items
            
            # 延迟，避免请求过快
            time.sleep(delay)
            
        except requests.RequestException as e:
            logger.error(f"第 {page} 页请求失败: {e}")
            yield []


# 使用示例
if __name__ == "__main__":
    # 示例 URL（需要替换为实际网站）
    base_url = "https://example.com/list?page={page}"
    
    all_items = []
    for page_items in crawl_paginated_site(base_url, total_pages=5, delay=1.0):
        all_items.extend(page_items)
        print(f"本页获取 {len(page_items)} 条")
    
    print(f"总共获取 {len(all_items)} 条数据")
```

## 小结
