# 2.3 字符串

字符串是 Python 中最常用的数据类型之一，用于表示文本信息。

## 字符串定义

### 基本定义

```python
# 单引号
s1 = 'Hello'

# 双引号
s2 = "World"

# 三引号（可跨行）
s3 = '''这是
多行
字符串'''

s4 = """这也是
多行字符串"""

print(type(s1))  # <class 'str'>
```

### 引号嵌套

```python
# 单引号内包含双引号
s1 = '他说："你好"'

# 双引号内包含单引号
s2 = "It's a book"

# 使用转义字符
s3 = '他说：\'你好\''
s4 = "It\"s a book"
```

### 原始字符串

```python
# 原始字符串（不转义）
path = r"C:\Users\name\Documents"
print(path)  # 输出: C:\Users\name\Documents

# 对比普通字符串
path2 = "C:\\Users\\name\\Documents"
print(path2)  # 输出: C:\Users\name\Documents
```

## 转义字符

| 转义字符 | 说明 |
|----------|------|
| `\n` | 换行 |
| `\t` | 制表符 |
| `\\` | 反斜杠 |
| `\'` | 单引号 |
| `\"` | 双引号 |
| `\r` | 回车 |
| `\b` | 退格 |

```python
print("Hello\nWorld")     # 换行
print("Hello\tWorld")     # 制表符
print("Hello\\World")     # 反斜杠
print("Hello\"World")     # 双引号
```

## 字符串索引与切片

### 索引

```python
s = "Python"

# 正向索引（从 0 开始）
print(s[0])  # 输出: P
print(s[1])  # 输出: y
print(s[5])  # 输出: n

# 负向索引（从 -1 开始）
print(s[-1])  # 输出: n
print(s[-2])  # 输出: o
print(s[-6])  # 输出: P
```

### 切片

```python
s = "Python"

# 切片语法：s[start:end:step]
# start: 起始索引（包含）
# end: 结束索引（不包含）
# step: 步长

# 基本切片
print(s[0:3])   # 输出: Pyt
print(s[2:5])   # 输出: tho
print(s[:3])    # 输出: Pyt（从头开始）
print(s[3:])    # 输出: hon（到末尾）

# 使用步长
print(s[::2])   # 输出: Pto（每隔一个字符）
print(s[1::2])  # 输出: yhn

# 反转字符串
print(s[::-1])  # 输出: nohtyP

# 负索引切片
print(s[-3:])   # 输出: hon
print(s[:-3])   # 输出: Pyt
```

## 字符串操作

### 拼接

```python
# 使用 + 号
s1 = "Hello"
s2 = "World"
result = s1 + " " + s2
print(result)  # 输出: Hello World

# 使用 * 号重复
print("=" * 20)  # 输出: ====================
print("Hi" * 3)  # 输出: HiHiHi

# 使用 join 方法
words = ["Hello", "World", "Python"]
result = " ".join(words)
print(result)  # 输出: Hello World Python
```

### 长度

```python
s = "Hello, World!"

# 获取字符串长度
print(len(s))  # 输出: 13

# 统计字符出现次数
print(s.count("l"))  # 输出: 3
print(s.count("o"))  # 输出: 2
```

### 成员判断

```python
s = "Hello, World!"

# in 和 not in
print("Hello" in s)      # True
print("Python" in s)     # False
print("Python" not in s) # True
```

## 字符串方法

### 大小写转换

```python
s = "Hello World"

# 转换为大写
print(s.upper())      # 输出: HELLO WORLD

# 转换为小写
print(s.lower())      # 输出: hello world

# 首字母大写
print(s.capitalize()) # 输出: Hello world

# 每个单词首字母大写
print(s.title())      # 输出: Hello World

# 大小写互换
print(s.swapcase())   # 输出: hELLO wORLD
```

### 查找与替换

```python
s = "Hello, World! World is beautiful."

# 查找子串位置
print(s.find("World"))     # 输出: 7（第一次出现的位置）
print(s.find("Python"))    # 输出: -1（未找到）
print(s.rfind("World"))    # 输出: 14（从右边查找）

# 查找（未找到会报错）
print(s.index("World"))    # 输出: 7
# print(s.index("Python")) # ValueError

# 替换
print(s.replace("World", "Python"))     # 替换所有
print(s.replace("World", "Python", 1))  # 只替换第一个
```

### 分割与连接

```python
# 分割字符串
s = "apple,banana,orange"
fruits = s.split(",")
print(fruits)  # 输出: ['apple', 'banana', 'orange']

# 分割行
text = "第一行\n第二行\n第三行"
lines = text.splitlines()
print(lines)  # 输出: ['第一行', '第二行', '第三行']

# 连接字符串
words = ["Hello", "World"]
result = " ".join(words)
print(result)  # 输出: Hello World
```

### 去除空白

```python
s = "  Hello World  "

# 去除两端空白
print(s.strip())   # 输出: "Hello World"

# 去除左边空白
print(s.lstrip())  # 输出: "Hello World  "

# 去除右边空白
print(s.rstrip())  # 输出: "  Hello World"

# 去除指定字符
s2 = "xxHello Worldxx"
print(s2.strip("x"))  # 输出: "Hello World"
```

### 判断方法

```python
# 判断开头和结尾
s = "Hello World"
print(s.startswith("Hello"))  # True
print(s.endswith("World"))     # True

# 判断字符类型
print("123".isdigit())     # True（只包含数字）
print("abc".isalpha())     # True（只包含字母）
print("abc123".isalnum())  # True（只包含字母和数字）
print("   ".isspace())     # True（只包含空白字符）
print("Hello".islower())   # False（是否全小写）
print("HELLO".isupper())   # True（是否全大写）
print("Hello World".istitle())  # True（是否标题格式）
```

### 填充与对齐

```python
s = "Hello"

# 居中对齐
print(s.center(20, "-"))  # 输出: -------Hello--------

# 左对齐
print(s.ljust(20, "-"))   # 输出: Hello---------------

# 右对齐
print(s.rjust(20, "-"))   # 输出: ---------------Hello

# 补零
num = "42"
print(num.zfill(5))  # 输出: 00042
```

## 字符串格式化

### f-string（推荐）

```python
name = "张三"
age = 25
score = 95.5

# 基本格式化
print(f"姓名：{name}，年龄：{age}")

# 表达式
print(f"明年 {age + 1} 岁")

# 格式化数字
print(f"成绩：{score:.1f}")
print(f"百分比：{score / 100:.1%}")

# 宽度和对齐
print(f"|{name:>10}|")  # 右对齐，宽度10
print(f"|{name:<10}|")  # 左对齐
print(f"|{name:^10}|")  # 居中对齐
```

### format 方法

```python
# 位置参数
print("姓名：{}，年龄：{}".format("张三", 25))

# 索引参数
print("{0} 喜欢 {1}，{0} 也喜欢 {2}".format("张三", "编程", "阅读"))

# 关键字参数
print("姓名：{name}，年龄：{age}".format(name="张三", age=25))

# 格式化
print("{:.2f}".format(3.14159))  # 输出: 3.14
print("{:010d}".format(42))      # 输出: 0000000042
```

### 百分号格式化

```python
name = "张三"
age = 25

# 基本格式化
print("姓名：%s，年龄：%d" % (name, age))

# 格式化符号
# %s: 字符串
# %d: 整数
# %f: 浮点数
# %x: 十六进制
# %o: 八进制
# %e: 科学计数法

print("圆周率：%.2f" % 3.14159)  # 输出: 圆周率：3.14
```

## 实战练习

### 练习一：字符串统计

```python
# 统计字符串中的字母、数字、空格和其他字符
s = input("请输入一个字符串：")

letters = 0
digits = 0
spaces = 0
others = 0

for char in s:
    if char.isalpha():
        letters += 1
    elif char.isdigit():
        digits += 1
    elif char.isspace():
        spaces += 1
    else:
        others += 1

print(f"字母：{letters} 个")
print(f"数字：{digits} 个")
print(f"空格：{spaces} 个")
print(f"其他：{others} 个")
```

### 练习二：回文判断

```python
# 判断是否为回文字符串
s = input("请输入一个字符串：")

# 方法一：反转比较
if s == s[::-1]:
    print(f"'{s}' 是回文字符串")
else:
    print(f"'{s}' 不是回文字符串")

# 方法二：双指针
def is_palindrome(s):
    left, right = 0, len(s) - 1
    while left < right:
        if s[left] != s[right]:
            return False
        left += 1
        right -= 1
    return True

print(is_palindrome(s))
```

### 练习三：字符串加密

```python
# 凯撒密码加密
def caesar_encrypt(text, shift):
    """凯撒密码加密"""
    result = ""
    for char in text:
        if char.isupper():
            result += chr((ord(char) - ord('A') + shift) % 26 + ord('A'))
        elif char.islower():
            result += chr((ord(char) - ord('a') + shift) % 26 + ord('a'))
        else:
            result += char
    return result

def caesar_decrypt(text, shift):
    """凯撒密码解密"""
    return caesar_encrypt(text, -shift)

# 使用示例
original = "Hello World"
encrypted = caesar_encrypt(original, 3)
decrypted = caesar_decrypt(encrypted, 3)

print(f"原文：{original}")
print(f"加密：{encrypted}")
print(f"解密：{decrypted}")
```
