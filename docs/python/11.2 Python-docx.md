# 11.2 Python-docx Word 文档处理

Python-docx 是 Python 中用于创建和更新 Microsoft Word (.docx) 文件的库。

## 安装

```bash
# 使用 pip 安装
pip install python-docx

# 验证安装
python -c "import docx; print(docx.__version__)"
```

## 基本概念

- **Document（文档）**：整个 Word 文档
- **Paragraph（段落）**：文档中的段落，包含文本和格式
- **Run（文本运行）**：段落中具有相同格式的连续文本
- **Section（节）**：文档中的节，可以有不同的页面设置
- **Table（表格）**：文档中的表格
- **Style（样式）**：预定义的格式集合

## 创建文档

### 新建文档

```python
from docx import Document

# 创建新文档
doc = Document()

# 添加标题
doc.add_heading('文档标题', level=0)

# 添加段落
doc.add_paragraph('这是第一个段落。')

# 保存文档
doc.save('basic_document.docx')
```

### 打开现有文档

```python
from docx import Document

# 打开现有文档
doc = Document('basic_document.docx')

# 修改内容
for paragraph in doc.paragraphs:
    if '第一个' in paragraph.text:
        paragraph.text = paragraph.text.replace('第一个', '修改后的第一个')

# 另存为新文件
doc.save('modified_document.docx')
```

### 段落格式

```python
from docx import Document
from docx.shared import Pt, Inches
from docx.enum.text import WD_ALIGN_PARAGRAPH

doc = Document()

# 添加段落并设置格式
p = doc.add_paragraph('这是一个格式化的段落。')

# 设置对齐方式
p.alignment = WD_ALIGN_PARAGRAPH.CENTER  # 居中对齐

# 设置段落格式
p_format = p.paragraph_format
p_format.left_indent = Inches(0.5)  # 左缩进
p_format.right_indent = Inches(0.5)  # 右缩进
p_format.first_line_indent = Inches(0.3)  # 首行缩进
p_format.space_before = Pt(12)  # 段前间距
p_format.space_after = Pt(12)  # 段后间距
p_format.line_spacing = 1.5  # 行距

doc.save('paragraph_format.docx')
```

## 表格

### 创建表格

```python
from docx import Document

doc = Document()

# 创建 3行4列 的表格
table = doc.add_table(rows=3, cols=4, style='Table Grid')

# 填充表头
header_cells = table.rows[0].cells
header_cells[0].text = '姓名'
header_cells[1].text = '年龄'
header_cells[2].text = '城市'
header_cells[3].text = '职业'

# 填充数据
data = [
    ['张三', '25', '北京', '工程师'],
    ['李四', '30', '上海', '设计师'],
]

for i, row_data in enumerate(data, 1):
    row_cells = table.rows[i].cells
    for j, cell_data in enumerate(row_data):
        row_cells[j].text = cell_data

doc.save('table_basic.docx')
```

### 表格样式

```python
from docx import Document
from docx.shared import Pt, Inches
from docx.enum.table import WD_TABLE_ALIGNMENT
from docx.oxml.ns import qn
from docx.oxml import OxmlElement

doc = Document()

# 创建表格
table = doc.add_table(rows=4, cols=3, style='Table Grid')

# 设置表格宽度
table.autofit = False
table.allow_autofit = False

# 填充数据
headers = ['产品', '单价', '数量']
products = [
    ['苹果', '5.00', '100'],
    ['香蕉', '3.00', '200'],
    ['橙子', '4.00', '150'],
]

# 设置表头
for i, header in enumerate(headers):
    cell = table.rows[0].cells[i]
    cell.text = header
    # 设置表头样式
    for paragraph in cell.paragraphs:
        paragraph.alignment = WD_TABLE_ALIGNMENT.CENTER
        for run in paragraph.runs:
            run.bold = True
            run.font.size = Pt(12)

# 填充数据行
for i, product in enumerate(products, 1):
    for j, value in enumerate(product):
        cell = table.rows[i].cells[j]
        cell.text = value
        for paragraph in cell.paragraphs:
            paragraph.alignment = WD_TABLE_ALIGNMENT.CENTER

# 设置列宽
for col in table.columns:
    col.width = Inches(1.5)

doc.save('table_style.docx')
```

## 实用示例

### 生成报告文档

```python
from docx import Document
from docx.shared import Pt, Inches, RGBColor
from docx.enum.text import WD_ALIGN_PARAGRAPH
from docx.enum.table import WD_TABLE_ALIGNMENT
from typing import List, Dict, Any
from datetime import datetime

def generate_report(
    title: str,
    content: Dict[str, Any],
    output_path: str
) -> None:
    """
    生成格式化报告文档
    
    Args:
        title: 报告标题
        content: 报告内容字典
        output_path: 输出文件路径
    """
    doc = Document()
    
    # 设置页边距
    section = doc.sections[0]
    section.top_margin = Inches(1)
    section.bottom_margin = Inches(1)
    section.left_margin = Inches(1.25)
    section.right_margin = Inches(1.25)
    
    # 添加标题
    heading = doc.add_heading(title, level=0)
    heading.alignment = WD_ALIGN_PARAGRAPH.CENTER
    
    # 添加日期
    date_para = doc.add_paragraph()
    date_para.alignment = WD_ALIGN_PARAGRAPH.CENTER
    date_run = date_para.add_run(f"生成日期：{datetime.now().strftime('%Y-%m-%d')}")
    date_run.font.size = Pt(10)
    date_run.font.color.rgb = RGBColor(128, 128, 128)
    
    doc.add_paragraph()  # 空行
    
    # 添加摘要
    if 'summary' in content:
        doc.add_heading('摘要', level=1)
        doc.add_paragraph(content['summary'])
    
    # 添加正文
    if 'sections' in content:
        for section_data in content['sections']:
            doc.add_heading(section_data['title'], level=2)
            doc.add_paragraph(section_data['content'])
    
    # 添加表格
    if 'table' in content:
        doc.add_heading('数据表格', level=1)
        table_data = content['table']
        
        table = doc.add_table(
            rows=len(table_data),
            cols=len(table_data[0]),
            style='Table Grid'
        )
        
        for i, row_data in enumerate(table_data):
            for j, cell_data in enumerate(row_data):
                cell = table.rows[i].cells[j]
                cell.text = str(cell_data)
                
                # 表头加粗
                if i == 0:
                    for paragraph in cell.paragraphs:
                        paragraph.alignment = WD_TABLE_ALIGNMENT.CENTER
                        for run in paragraph.runs:
                            run.bold = True
    
    # 添加结论
    if 'conclusion' in content:
        doc.add_heading('结论', level=1)
        doc.add_paragraph(content['conclusion'])
    
    doc.save(output_path)


# 使用示例
report_content = {
    'summary': '本报告总结了2024年第一季度的销售情况，包括各产品线的销售数据和趋势分析。',
    'sections': [
        {
            'title': '销售概况',
            'content': '第一季度总销售额达到500万元，同比增长15%。其中线上渠道占比60%，线下渠道占比40%。'

        ['A系列', '300', '+10%'],
        ['B系列', '100', '+50%'],
```

### 批量生成合同文档

```python
from docx import Document
from docx.shared import Pt
from typing import List, Dict, Any
import os

def generate_contracts(
    template_path: str,
    data_list: List[Dict[str, str]],
    output_dir: str
) -> List[str]:
    """
    根据模板批量生成合同文档
    
    Args:
        template_path: 模板文件路径
        data_list: 数据字典列表
        output_dir: 输出目录
        
    Returns:
        生成的文件路径列表
    """
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)
    
    generated_files = []
    
    for data in data_list:
        # 加载模板
        doc = Document(template_path)
        
        # 替换占位符
        for paragraph in doc.paragraphs:
            for key, value in data.items():
                placeholder = f'{{{{{key}}}}}'
                if placeholder in paragraph.text:
                    paragraph.text = paragraph.text.replace(placeholder, value)
        
        # 替换表格中的占位符
        for table in doc.tables:
            for row in table.rows:
                for cell in row.cells:
                    for key, value in data.items():
                        placeholder = f'{{{{{key}}}}}'
                        if placeholder in cell.text:
                            cell.text = cell.text.replace(placeholder, value)
        
        # 保存文档
        output_name = f"合同_{data.get('name', 'unknown')}.docx"
        output_path = os.path.join(output_dir, output_name)
        doc.save(output_path)
        generated_files.append(output_path)
    
    return generated_files
```

### 提取文档内容

```python
from docx import Document
from typing import Dict, List, Any

def extract_document_content(file_path: str) -> Dict[str, Any]:
    """
    提取 Word 文档的内容
    
    Args:
        file_path: 文档路径
        
    Returns:
        包含文档内容的字典
    """
    doc = Document(file_path)
    
    result = {
        'paragraphs': [],
        'tables': [],
        'headings': [],
    }
    
    # 提取段落
    for paragraph in doc.paragraphs:
        para_data = {
            'text': paragraph.text,
            'style': paragraph.style.name,
        }
        result['paragraphs'].append(para_data)
        
        # 记录标题
        if paragraph.style.name.startswith('Heading'):
            result['headings'].append({
                'level': paragraph.style.name,
                'text': paragraph.text,
            })
    
    # 提取表格
    for table in doc.tables:
        table_data = []
        for row in table.rows:
            row_data = [cell.text for cell in row.cells]
            table_data.append(row_data)
        result['tables'].append(table_data)
    
    return result
```
