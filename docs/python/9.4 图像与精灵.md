# 9.4 图像与精灵

Pygame 提供了图像加载和精灵系统，用于处理游戏中的图形对象。

## 加载与显示图像

### 1. 基本图像操作

```python
import pygame
import sys
import os

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("图像加载演示")

clock = pygame.time.Clock()

# 加载图像
# 支持：PNG, JPG, GIF, BMP 等格式
try:
    # 加载图像并自动转换格式（提高绘制性能）
    image = pygame.image.load("player.png").convert_alpha()
except FileNotFoundError:
    # 如果文件不存在，创建一个示例图像
    image = pygame.Surface((64, 64), pygame.SRCALPHA)
    pygame.draw.circle(image, (0, 255, 0), (32, 32), 30)

# 获取图像尺寸
image_width, image_height = image.get_size()
print(f"图像尺寸: {image_width} x {image_height}")

# 缩放图像
scaled_image = pygame.transform.scale(image, (128, 128))

# 平滑缩放（质量更好，速度较慢）
smooth_scaled = pygame.transform.smoothscale(image, (200, 200))

# 旋转图像
rotated_image = pygame.transform.rotate(image, 45)

# 水平翻转
flipped_h = pygame.transform.flip(image, True, False)

# 垂直翻转
flipped_v = pygame.transform.flip(image, False, True)

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
    
    screen.fill((50, 50, 50))
    
    # 绘制原始图像
    screen.blit(image, (50, 50))
    
    # 绘制缩放后的图像
    screen.blit(scaled_image, (150, 50))
    
    # 绘制旋转后的图像
    screen.blit(rotated_image, (300, 50))
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

### 2. 图像变换

```python
import pygame
import sys
import math

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("图像变换演示")

clock = pygame.time.Clock()

# 创建一个示例图像
def create_sample_image():
    surface = pygame.Surface((100, 100), pygame.SRCALPHA)
    # 绘制一个箭头形状
    pygame.draw.polygon(surface, (255, 100, 0), [
        (50, 10), (90, 50), (70, 50), (70, 90), (30, 90), (30, 50), (10, 50)
    ])
    return surface

original = create_sample_image()
angle = 0
scale = 1.0

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        elif event.type == pygame.KEYDOWN:
            if event.key == pygame.K_ESCAPE:
                running = False
    
    screen.fill((30, 30, 30))
    
    # 动态旋转
    angle = (angle + 2) % 360
    rotated = pygame.transform.rotate(original, angle)
    
    # 动态缩放
    scale = 1.0 + 0.3 * math.sin(pygame.time.get_ticks() / 500)
    scaled = pygame.transform.scale(original, 
                                    (int(100 * scale), int(100 * scale)))
    
    # 绘制原始图像
    screen.blit(original, (100, 250))
    
    # 绘制旋转图像（居中显示）
    rot_rect = rotated.get_rect(center=(350, 300))
    screen.blit(rotated, rot_rect)
    
    # 绘制缩放图像
    scaled_rect = scaled.get_rect(center=(600, 300))
    screen.blit(scaled, scaled_rect)
    
    # 显示标签
    font = pygame.font.Font(None, 24)
    screen.blit(font.render("原始", True, (255, 255, 255)), (120, 380))
    screen.blit(font.render("旋转", True, (255, 255, 255)), (330, 380))
    screen.blit(font.render("缩放", True, (255, 255, 255)), (580, 380))
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

## 精灵系统

### 1. Sprite 基础

```python
import pygame
import sys
import random

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("精灵系统演示")

clock = pygame.time.Clock()

class Player(pygame.sprite.Sprite):
    """
    玩家精灵类
    
    继承自 pygame.sprite.Sprite，需要实现：
    - __init__: 初始化精灵
    - update: 更新精灵状态
    """
    
    def __init__(self, x, y):
        super().__init__()
        
        # 创建图像
        self.image = pygame.Surface((50, 50), pygame.SRCALPHA)
        pygame.draw.rect(self.image, (0, 255, 0), (0, 0, 50, 50))
        pygame.draw.polygon(self.image, (0, 200, 0), [
            (25, 5), (45, 45), (25, 35), (5, 45)
        ])
        
        # 获取矩形对象（用于位置和碰撞检测）
        self.rect = self.image.get_rect()
        self.rect.center = (x, y)
        
        # 速度
        self.speed = 5
    
    def update(self):
        """更新玩家位置"""
        keys = pygame.key.get_pressed()
        
        if keys[pygame.K_LEFT] or keys[pygame.K_a]:
            self.rect.x -= self.speed
        if keys[pygame.K_RIGHT] or keys[pygame.K_d]:
            self.rect.x += self.speed
        if keys[pygame.K_UP] or keys[pygame.K_w]:
            self.rect.y -= self.speed
        if keys[pygame.K_DOWN] or keys[pygame.K_s]:
            self.rect.y += self.speed
        
        # 边界限制
        self.rect.clamp_ip(screen.get_rect())


class Enemy(pygame.sprite.Sprite):
    """敌人精灵类"""
    
    def __init__(self, x, y):
        super().__init__()
        
        self.image = pygame.Surface((40, 40), pygame.SRCALPHA)
        pygame.draw.circle(self.image, (255, 0, 0), (20, 20), 20)
        pygame.draw.circle(self.image, (200, 0, 0), (20, 20), 15)
        
        self.rect = self.image.get_rect()
        self.rect.center = (x, y)
        
        self.speed = random.randint(1, 3)
    
    def update(self):
        """更新敌人位置"""
        self.rect.y += self.speed
        
        # 超出屏幕则移除
        if self.rect.top > 600:
            self.kill()


class Bullet(pygame.sprite.Sprite):
    """子弹精灵类"""
    
    def __init__(self, x, y):
        super().__init__()
        
        self.image = pygame.Surface((10, 20), pygame.SRCALPHA)
        pygame.draw.ellipse(self.image, (255, 255, 0), (0, 0, 10, 20))
        
        self.rect = self.image.get_rect()
        self.rect.center = (x, y)
        
        self.speed = 10
    
    def update(self):
        """更新子弹位置"""
        self.rect.y -= self.speed
        
        if self.rect.bottom < 0:
            self.kill()


# 创建精灵组
all_sprites = pygame.sprite.Group()
enemies = pygame.sprite.Group()
bullets = pygame.sprite.Group()

# 创建玩家
player = Player(400, 500)
all_sprites.add(player)

# 敌人生成计时器
ENEMY_SPAWN = pygame.USEREVENT + 1
pygame.time.set_timer(ENEMY_SPAWN, 1000)

running = True
score = 0
font = pygame.font.Font(None, 36)

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        elif event.type == pygame.KEYDOWN:
            if event.key == pygame.K_ESCAPE:
                running = False
            elif event.key == pygame.K_SPACE:
                # 发射子弹
                bullet = Bullet(player.rect.centerx, player.rect.top)
                all_sprites.add(bullet)
                bullets.add(bullet)
        elif event.type == ENEMY_SPAWN:
            # 生成敌人
            enemy = Enemy(random.randint(50, 750), -20)
            all_sprites.add(enemy)
            enemies.add(enemy)
    
    # 更新所有精灵
    all_sprites.update()
    
    # 碰撞检测：子弹与敌人
    hits = pygame.sprite.groupcollide(bullets, enemies, True, True)
    for hit in hits:
        score += 10
    
    # 碰撞检测：玩家与敌人
    hits = pygame.sprite.spritecollide(player, enemies, True)
    if hits:
        score -= 50
    
    # 绘制
    screen.fill((30, 30, 30))
    all_sprites.draw(screen)
    
    # 显示分数
    score_text = font.render(f"分数: {score}", True, (255, 255, 255))
    screen.blit(score_text, (10, 10))
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

### 3. 动画精灵

```python
import pygame
import sys

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("动画精灵演示")

clock = pygame.time.Clock()

class AnimatedSprite(pygame.sprite.Sprite):
    """
    动画精灵类
    
    通过切换多帧图像实现动画效果
    """
    
    def __init__(self, x, y):
        super().__init__()
        
        # 创建动画帧
        self.frames = []
        self.frame_index = 0
        self.animation_speed = 10  # 每10帧切换一次
        self.frame_counter = 0
        
        # 创建简单的动画帧（一个跳动的球）
        for i in range(8):
            frame = pygame.Surface((64, 64), pygame.SRCALPHA)
            # 球的大小随帧变化
            radius = 25 + 5 * abs(4 - i)
            color = (
                int(255 * (i / 7)),  # R
                int(100 + 100 * (i / 7)),  # G
                int(255 * (1 - i / 7))  # B
            )
            pygame.draw.circle(frame, color, (32, 32), radius)
            self.frames.append(frame)
        
        # 设置初始图像
        self.image = self.frames[0]
        self.rect = self.image.get_rect()
        self.rect.center = (x, y)
        
        # 移动速度
        self.speed_x = 3
        self.speed_y = 2
    
    def update(self):
        """更新动画和位置"""
        # 更新动画帧
        self.frame_counter += 1
        if self.frame_counter >= self.animation_speed:
            self.frame_counter = 0
            self.frame_index = (self.frame_index + 1) % len(self.frames)
            self.image = self.frames[self.frame_index]
        
        # 更新位置
        self.rect.x += self.speed_x
        self.rect.y += self.speed_y
        
        # 边界反弹
        if self.rect.left <= 0 or self.rect.right >= 800:
            self.speed_x = -self.speed_x
        if self.rect.top <= 0 or self.rect.bottom >= 600:
            self.speed_y = -self.speed_y


# 创建精灵组
sprites = pygame.sprite.Group()

# 创建多个动画精灵
for i in range(5):
    sprite = AnimatedSprite(100 + i * 150, 300)
    sprites.add(sprite)

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        elif event.type == pygame.KEYDOWN:
            if event.key == pygame.K_ESCAPE:
                running = False
    
    # 更新
    sprites.update()
    
    # 绘制
    screen.fill((30, 30, 30))
    sprites.draw(screen)
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

## 小结

本章要点：

1. **pygame.image.load()**：加载图像文件
2. **convert() / convert_alpha()**：转换图像格式以提高性能
3. **pygame.transform**：图像变换（缩放、旋转、翻转）
4. **pygame.sprite.Sprite**：精灵基类
5. **pygame.sprite.Group**：精灵组，批量管理精灵
6. **pygame.sprite.spritecollide()**：精灵碰撞检测
7. **pygame.sprite.groupcollide()**：精灵组碰撞检测

> 💡 **提示**：使用精灵系统可以更好地组织游戏对象，简化碰撞检测和批量渲染。
