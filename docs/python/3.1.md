# 3.1 条件语句

条件语句用于根据条件执行不同的代码块。Python 使用 if、elif、else 关键字实现条件判断。

## if 语句

### 基本语法

```python
# 基本 if 语句
age = 18

if age >= 18:
    print("你已成年")

# if-else 语句
score = 75

if score >= 60:
    print("及格")
else:
    print("不及格")

# if-elif-else 语句
score = 85

if score >= 90:
    print("优秀")
elif score >= 80:
    print("良好")
elif score >= 60:
    print("及格")
else:
    print("不及格")
```

### 嵌套条件

```python
# 嵌套 if 语句
age = 25
has_license = True

if age >= 18:
    if has_license:
        print("可以驾驶")
    else:
        print("需要先考取驾照")
else:
    print("年龄不够，不能驾驶")

# 更好的写法：使用逻辑运算符
if age >= 18 and has_license:
    print("可以驾驶")
elif age >= 18 and not has_license:
    print("需要先考取驾照")
else:
    print("年龄不够，不能驾驶")
```

## 比较运算符

| 运算符 | 说明 | 示例 |
|--------|------|------|
| `==` | 等于 | `a == b` |
| `!=` | 不等于 | `a != b` |
| `>` | 大于 | `a > b` |
| `<` | 小于 | `a < b` |
| `>=` | 大于等于 | `a >= b` |
| `<=` | 小于等于 | `a <= b` |

```python
a, b = 10, 20

print(a == b)  # False
print(a != b)  # True
print(a > b)   # False
print(a < b)   # True
print(a >= b)  # False
print(a <= b)  # True

# 比较字符串
print("apple" < "banana")  # True（按字典序比较）
print("Apple" < "apple")   # True（大写字母 ASCII 码更小）
```

## 逻辑运算符

| 运算符 | 说明 | 示例 |
|--------|------|------|
| `and` | 与（都为真才为真） | `a and b` |
| `or` | 或（有一个为真就为真） | `a or b` |
| `not` | 非（取反） | `not a` |

```python
# and 运算符
age = 25
has_license = True

if age >= 18 and has_license:
    print("可以驾驶")

# or 运算符
day = "周六"

if day == "周六" or day == "周日":
    print("周末")

# not 运算符
is_raining = False

if not is_raining:
    print("可以出门")

# 组合使用
age = 25
is_student = True

if age >= 18 and not is_student:
    print("成年工作者")
elif age >= 18 and is_student:
    print("成年学生")
```

### 短路求值

```python
# and 短路：如果第一个为 False，不计算第二个
a = False
b = True
result = a and b  # a 为 False，不会计算 b

# or 短路：如果第一个为 True，不计算第二个
a = True
b = False
result = a or b  # a 为 True，不会计算 b

# 实际应用：避免错误
x = None
if x is not None and x > 0:
    print("x 是正数")
```

## 成员运算符

```python
# in：判断元素是否在序列中
fruits = ["apple", "banana", "orange"]

if "apple" in fruits:
    print("有苹果")

# not in：判断元素是否不在序列中
if "grape" not in fruits:
    print("没有葡萄")

# 用于字符串
text = "Hello, World!"

if "World" in text:
    print("包含 World")

# 用于字典（判断键）
person = {"name": "张三", "age": 25}

if "name" in person:
    print(f"姓名：{person['name']}")
```

## 身份运算符

```python
# is：判断两个变量是否引用同一个对象
a = [1, 2, 3]
b = [1, 2, 3]
c = a

print(a is b)  # False（不同对象）
print(a is c)  # True（同一对象）
print(a == b)  # True（值相等）

# is not：判断是否不是同一对象
print(a is not b)  # True

# 常用：判断 None
value = None

if value is None:
    print("值为 None")

if value is not None:
    print("值不为 None")
```

## 三元表达式

```python
# 语法：value_if_true if condition else value_if_false
age = 20

# 传统写法
if age >= 18:
    status = "成年"
else:
    status = "未成年"

# 三元表达式
status = "成年" if age >= 18 else "未成年"
print(status)

# 实际应用
score = 75
result = "及格" if score >= 60 else "不及格"
print(result)

# 嵌套三元（不推荐，可读性差）
score = 85
grade = "优秀" if score >= 90 else "良好" if score >= 80 else "及格" if score >= 60 else "不及格"
print(grade)
```

## 条件表达式

### 链式比较

```python
# 链式比较
x = 15

# 传统写法
if x >= 10 and x <= 20:
    print("x 在 10 到 20 之间")

# 链式比较
if 10 <= x <= 20:
    print("x 在 10 到 20 之间")

# 多个比较
a, b, c = 1, 2, 3
if a < b < c:
    print("a < b < c")
```

### 真值判断

```python
# 以下值在布尔上下文中为 False：
# None, False, 0, 0.0, '', [], (), {}, set()

# 检查列表是否为空
items = []

# 不推荐
if len(items) == 0:
    print("列表为空")

# 推荐
if not items:
    print("列表为空")

# 检查列表是否有元素
if items:
    print("列表不为空")
```

## 模式匹配（Python 3.10+）

```python
# match-case 语句（Python 3.10+）
def http_status(status):
    match status:
        case 200:
            return "OK"
        case 201:
            return "Created"
        case 400:
            return "Bad Request"
        case 404:
            return "Not Found"
        case 500:
            return "Internal Server Error"
        case _:  # 默认情况
            return "Unknown Status"

print(http_status(200))  # 输出: OK
print(http_status(404))  # 输出: Not Found

# 带条件的模式匹配
def describe_point(point):
    match point:
        case (0, 0):
            return "原点"
        case (x, 0):
            return f"在 x 轴上，x={x}"
        case (0, y):
            return f"在 y 轴上，y={y}"
        case (x, y) if x == y:
            return f"在对角线上，x=y={x}"
        case (x, y):
            return f"坐标: ({x}, {y})"

print(describe_point((0, 0)))   # 输出: 原点
print(describe_point((3, 0)))   # 输出: 在 x 轴上，x=3
print(describe_point((5, 5)))   # 输出: 在对角线上，x=y=5
```

score = float(input("请输入成绩："))

    print("等级：D")

    print("等级：E（不及格）")

### 练习二：闰年判断

```python
# 判断是否为闰年
year = int(input("请输入年份："))

# 闰年规则：
# 1. 能被 4 整除但不能被 100 整除
# 2. 能被 400 整除

if (year % 4 == 0 and year % 100 != 0) or (year % 400 == 0):
    print(f"{year} 年是闰年")
else:
    print(f"{year} 年不是闰年")
```

### 练习三：简易计算器

```python
# 简易计算器
num1 = float(input("请输入第一个数字："))
operator = input("请输入运算符（+、-、*、/）：")
num2 = float(input("请输入第二个数字："))

if operator == "+":
    result = num1 + num2
elif operator == "-":
    result = num1 - num2
elif operator == "*":
    result = num1 * num2
elif operator == "/":
    if num2 != 0:
        result = num1 / num2
    else:
        result = "错误：除数不能为零"
else:
    result = "错误：无效的运算符"

print(f"结果：{result}")
```

### 练习四：登录验证

```python
# 简单的登录验证
users = {
    "admin": "123456",
    "user1": "password",
    "user2": "abc123"
}

username = input("请输入用户名：")
password = input("请输入密码：")

if username in users:
    if users[username] == password:
        print(f"欢迎，{username}！登录成功")
    else:
        print("密码错误")
else:
    print("用户名不存在")
```

## 小结

本章要点：

1. **if-elif-else**：条件判断的基本结构
2. **比较运算符**：==、!=、>、<、>=、<=
3. **逻辑运算符**：and、or、not
4. **成员运算符**：in、not in
5. **身份运算符**：is、is not
6. **三元表达式**：简洁的条件赋值

> 💡 **提示**：条件语句要注意代码缩进，Python 使用缩进来表示代码块。建议每个 if-elif-else 分支都保持一致的缩进层级。
