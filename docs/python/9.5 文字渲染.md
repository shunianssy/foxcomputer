# 9.5 文字渲染

Pygame 提供了字体模块，可以在游戏中显示文字。

## 基本文字渲染

### 1. 使用系统字体

```python
import pygame
import sys

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("文字渲染演示")

clock = pygame.time.Clock()

# 初始化字体模块
pygame.font.init()

# 使用系统默认字体
# 参数：字体名称（None 为默认）, 字号
font_default = pygame.font.Font(None, 36)

# 使用系统字体
# pygame.font.get_fonts() 返回所有可用字体列表
available_fonts = pygame.font.get_fonts()
print(f"可用字体数量: {len(available_fonts)}")

# 创建系统字体对象
font_arial = pygame.font.SysFont("arial", 48)
font_times = pygame.font.SysFont("timesnewroman", 32)

# 支持中文字体
font_chinese = pygame.font.SysFont("microsoftyahei", 36)  # 微软雅黑
font_simhei = pygame.font.SysFont("simhei", 36)           # 黑体

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
    
    screen.fill((30, 30, 30))
    
    # 渲染文字
    # 参数：文字内容, 是否抗锯齿, 颜色
    text1 = font_default.render("Hello Pygame!", True, (255, 255, 255))
    text2 = font_arial.render("Arial Font", True, (255, 100, 100))
    text3 = font_times.render("Times New Roman", True, (100, 255, 100))
    
    # 中文文字
    text4 = font_chinese.render("你好，Pygame！", True, (100, 100, 255))
    
    # 绘制文字
    screen.blit(text1, (50, 50))
    screen.blit(text2, (50, 100))
    screen.blit(text3, (50, 150))
    screen.blit(text4, (50, 200))
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

## 文字样式

### 1. 粗体、斜体、下划线

```python
import pygame
import sys

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("文字样式演示")

clock = pygame.time.Clock()

# 创建字体
font = pygame.font.SysFont("arial", 36)
font_bold = pygame.font.SysFont("arial", 36, bold=True)
font_italic = pygame.font.SysFont("arial", 36, italic=True)
font_both = pygame.font.SysFont("arial", 36, bold=True, italic=True)

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
    
    screen.fill((30, 30, 30))

    text2 = font_bold.render("Bold Text", True, (255, 100, 100))
    screen.blit(text2, (50, 160))
    
    # 斜体
    text3 = font_italic.render("Italic Text", True, (100, 255, 100))

    # 粗体+斜体
```

### 2. 文字对齐与换行

```python
import pygame
import sys

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("文字对齐演示")

clock = pygame.time.Clock()
font = pygame.font.SysFont("arial", 32)

def render_text_center(text, font, color, y):
    """居中渲染文字"""
    text_surface = font.render(text, True, color)
    text_rect = text_surface.get_rect(center=(400, y))
    screen.blit(text_surface, text_rect)

def render_text_right(text, font, color, y):
    """右对齐渲染文字"""
    text_surface = font.render(text, True, color)
    text_rect = text_surface.get_rect(right=750, y=y)
    screen.blit(text_surface, text_rect)

def render_multiline(text, font, color, x, y, line_height=40):
    """渲染多行文字"""
    lines = text.split('\n')
    for i, line in enumerate(lines):
        text_surface = font.render(line, True, color)
        screen.blit(text_surface, (x, y + i * line_height))

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
    
    screen.fill((30, 30, 30))
    
    # 左对齐（默认）
    text1 = font.render("Left Aligned Text", True, (255, 255, 255))
    screen.blit(text1, (50, 50))
    
    # 居中对齐
    render_text_center("Centered Text", font, (255, 100, 100), 120)
    
    # 右对齐
    render_text_right("Right Aligned Text", font, (100, 255, 100), 190)
    
    # 多行文字
    multiline_text = """这是第一行文字
这是第二行文字
这是第三行文字"""
    render_multiline(multiline_text, font, (100, 100, 255), 50, 280)
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

## 文字特效

### 1. 文字阴影与描边

```python
import pygame
import sys

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("文字特效演示")

clock = pygame.time.Clock()
font = pygame.font.SysFont("arial", 64)

def render_text_with_shadow(text, font, color, shadow_color, pos, offset=3):
    """
    渲染带阴影的文字
    
    参数:
        text: 文字内容
        font: 字体对象
        color: 文字颜色
        shadow_color: 阴影颜色
        pos: 文字位置
        offset: 阴影偏移量
    """
    # 先绘制阴影
    shadow = font.render(text, True, shadow_color)
    screen.blit(shadow, (pos[0] + offset, pos[1] + offset))
    
    # 再绘制文字
    text_surface = font.render(text, True, color)
    screen.blit(text_surface, pos)

def render_text_with_outline(text, font, color, outline_color, pos, outline_width=2):
    """
    渲染带描边的文字
    
    参数:
        text: 文字内容
        font: 字体对象
        color: 文字颜色
        outline_color: 描边颜色
        pos: 文字位置
        outline_width: 描边宽度
    """
    # 绘制描边（在各个方向绘制文字）
    for dx in range(-outline_width, outline_width + 1):
        for dy in range(-outline_width, outline_width + 1):
            if dx != 0 or dy != 0:
                outline = font.render(text, True, outline_color)
                screen.blit(outline, (pos[0] + dx, pos[1] + dy))
    
    # 绘制文字
    text_surface = font.render(text, True, color)
    screen.blit(text_surface, pos)

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
    
    screen.fill((30, 30, 30))
    
    # 阴影效果
    render_text_with_shadow("Shadow Effect", font, 
                            (255, 255, 255), (100, 100, 100), (50, 100))
    
    # 描边效果
    render_text_with_outline("Outline Effect", font, 
                             (255, 255, 255), (255, 100, 0), (50, 250))
    
    # 发光效果（多层阴影）
    text = "Glow Effect"
    for i in range(5, 0, -1):
        glow = font.render(text, True, (0, i * 30, i * 50))
        glow_rect = glow.get_rect(center=(400, 420))
        screen.blit(glow, glow_rect)
    
    text_surface = font.render(text, True, (100, 200, 255))
    text_rect = text_surface.get_rect(center=(400, 420))
    screen.blit(text_surface, text_rect)
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

### 2. 动态文字效果

```python
import pygame
import sys
import math

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("动态文字效果")

clock = pygame.time.Clock()
font = pygame.font.SysFont("arial", 48)

def render_wave_text(text, font, color, base_y, time, amplitude=10, frequency=0.1):
    """波浪文字效果"""
    for i, char in enumerate(text):
        # 计算每个字符的 Y 偏移
        offset = amplitude * math.sin(frequency * time + i * 0.5)
        
        # 渲染单个字符
        char_surface = font.render(char, True, color)
        
        # 计算位置
        x = 100 + i * 30
        y = base_y + offset
        
        screen.blit(char_surface, (x, y))

def render_typing_text(text, font, color, pos, current_char):
    """打字机效果"""
    visible_text = text[:current_char]
    text_surface = font.render(visible_text, True, color)
    screen.blit(text_surface, pos)
    
    # 绘制光标
    if current_char < len(text):
        cursor_x = pos[0] + text_surface.get_width()
        cursor = font.render("|", True, color)
        screen.blit(cursor, (cursor_x, pos[1]))

running = True
time_counter = 0
typing_index = 0
typing_timer = 0

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
    
    # 更新时间
    time_counter += 1
    
    # 更新打字机效果
    typing_timer += 1
    if typing_timer >= 10:  # 每10帧显示一个字符
        typing_timer = 0
        typing_index = (typing_index + 1) % 30  # 循环显示
    
    screen.fill((30, 30, 30))
    
    # 波浪效果
    render_wave_text("WAVE TEXT!", font, (255, 100, 100), 150, time_counter)
    
    # 打字机效果
    render_typing_text("Typing Effect Demo...", font, (100, 255, 100), 
                       (100, 300), typing_index)
    
    # 闪烁效果
    if (time_counter // 30) % 2 == 0:
        blink_text = font.render("Blinking Text", True, (100, 100, 255))
        screen.blit(blink_text, (100, 400))
    
    # 缩放效果
    scale = 1.0 + 0.2 * math.sin(time_counter * 0.05)
    text_surface = font.render("Scaling Text", True, (255, 255, 0))
    scaled = pygame.transform.scale(text_surface, 
                                    (int(text_surface.get_width() * scale),
                                     int(text_surface.get_height() * scale)))
    scaled_rect = scaled.get_rect(center=(400, 500))
    screen.blit(scaled, scaled_rect)
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

## 文本输入框

```python
import pygame
import sys

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("文本输入框演示")

clock = pygame.time.Clock()
font = pygame.font.SysFont("arial", 32)

class TextInput:
    """
    文本输入框类
    
    支持文本输入、删除、光标显示等功能
    """
    
    def __init__(self, x, y, width, height, font, 
                 bg_color=(50, 50, 50), text_color=(255, 255, 255),
                 border_color=(100, 100, 100), active_color=(0, 150, 255)):
        self.rect = pygame.Rect(x, y, width, height)
        self.font = font
        self.bg_color = bg_color
        self.text_color = text_color
        self.border_color = border_color
        self.active_color = active_color
        
        self.text = ""
        self.active = False
        self.cursor_visible = True
        self.cursor_timer = 0
    
    def handle_event(self, event):
        """处理输入事件"""
        if event.type == pygame.MOUSEBUTTONDOWN:
            # 点击激活/取消激活
            self.active = self.rect.collidepoint(event.pos)
        
        elif event.type == pygame.KEYDOWN and self.active:
            if event.key == pygame.K_BACKSPACE:
                self.text = self.text[:-1]
            elif event.key == pygame.K_RETURN:
                print(f"输入内容: {self.text}")
                self.text = ""  # 清空
            elif event.unicode.isprintable():
                self.text += event.unicode
    
    def update(self):
        """更新光标闪烁"""
        self.cursor_timer += 1
        if self.cursor_timer >= 30:  # 每30帧切换
            self.cursor_timer = 0
            self.cursor_visible = not self.cursor_visible
    
    def draw(self, surface):
        """绘制输入框"""
        # 绘制背景
        pygame.draw.rect(surface, self.bg_color, self.rect)
        
        # 绘制边框
        border_color = self.active_color if self.active else self.border_color
        pygame.draw.rect(surface, border_color, self.rect, 2)
        
        # 绘制文字
        text_surface = self.font.render(self.text, True, self.text_color)
        text_rect = text_surface.get_rect(midleft=(self.rect.x + 10, 
                                                    self.rect.centery))
        surface.blit(text_surface, text_rect)
        
        # 绘制光标
        if self.active and self.cursor_visible:
            cursor_x = text_rect.right + 2
            cursor_y = self.rect.y + 10
            cursor_height = self.rect.height - 20
            pygame.draw.line(surface, self.text_color, 
                           (cursor_x, cursor_y), 
                           (cursor_x, cursor_y + cursor_height), 2)

# 创建输入框
input_box = TextInput(200, 250, 400, 50, font)
input_box2 = TextInput(200, 350, 400, 50, font)

# 标签字体
label_font = pygame.font.SysFont("arial", 24)

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        
        input_box.handle_event(event)
        input_box2.handle_event(event)
    
    input_box.update()
    input_box2.update()
    
    screen.fill((30, 30, 30))
    
    # 标题
    title = font.render("点击输入框开始输入", True, (255, 255, 255))
    title_rect = title.get_rect(center=(400, 100))
    screen.blit(title, title_rect)
    
    # 标签
    label1 = label_font.render("用户名:", True, (200, 200, 200))
    screen.blit(label1, (100, 265))
    
    label2 = label_font.render("密码:", True, (200, 200, 200))
    screen.blit(label2, (100, 365))
    
    # 绘制输入框
    input_box.draw(screen)
    input_box2.draw(screen)
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

## 小结

本章要点：

1. **pygame.font.Font()**：创建字体对象
2. **pygame.font.SysFont()**：使用系统字体
3. **font.render()**：渲染文字为 Surface
4. **font.set_bold() / set_italic() / set_underline()**：设置文字样式
5. **text.get_rect()**：获取文字矩形，用于对齐
6. **pygame.font.get_fonts()**：获取可用字体列表

> **提示**：渲染中文时，确保使用支持中文的字体
