# 7.3 上下文管理器

上下文管理器用于管理资源，确保资源正确获取和释放。Python 使用 with 语句来使用上下文管理器。

## with 语句基础

### 为什么使用 with

```python
# 传统方式：需要手动关闭文件
file = open("example.txt", "r")
try:
    content = file.read()
finally:
    file.close()  # 必须记得关闭

# 使用 with 语句：自动关闭文件
with open("example.txt", "r", encoding="utf-8") as file:
    content = file.read()
# 文件在此自动关闭，即使发生异常
```

### with 语句的工作原理

```python
# with 语句等价于以下代码
manager = open("example.txt", "r")
file = manager.__enter__()
try:
    content = file.read()
finally:
    manager.__exit__(None, None, None)
```

## 文件操作中的 with

```python
# 读取文件
with open("example.txt", "r", encoding="utf-8") as file:
    content = file.read()
    print(content)

# 写入文件
with open("output.txt", "w", encoding="utf-8") as file:
    file.write("Hello, World!")

# 同时打开多个文件
with open("input.txt", "r", encoding="utf-8") as infile, \
     open("output.txt", "w", encoding="utf-8") as outfile:
    content = infile.read()
    outfile.write(content.upper())

# Python 3.10+ 可以使用括号
with (
    open("input.txt", "r", encoding="utf-8") as infile,
    open("output.txt", "w", encoding="utf-8") as outfile
):
    content = infile.read()
    outfile.write(content.upper())
```

## 自定义上下文管理器

### 使用类实现

```python
class Timer:
    """计时器上下文管理器"""
    
    def __init__(self, name="Timer"):
        self.name = name
        self.start_time = None
        self.end_time = None
    
    def __enter__(self):
        """进入上下文时调用"""
        import time
        self.start_time = time.time()
        print(f"{self.name} 开始...")
        return self  # 返回 self，可以通过 as 获取
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        """退出上下文时调用"""
        import time
        self.end_time = time.time()
        elapsed = self.end_time - self.start_time
        print(f"{self.name} 结束，耗时: {elapsed:.4f} 秒")
        return False  # 返回 True 会抑制异常

# 使用示例
with Timer("数据处理"):
    # 模拟耗时操作
    sum(range(10000000))
# 输出:
# 数据处理 开始...
# 数据处理 结束，耗时: 0.1234 秒
```

### __exit__ 方法详解

```python
class SafeDivision:
    """安全除法上下文管理器"""
    
    def __enter__(self):
        print("进入安全除法模式")
        return self
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        """
        参数说明：
        - exc_type: 异常类型（无异常时为 None）
        - exc_val: 异常值（无异常时为 None）
        - exc_tb: 异常追踪信息（无异常时为 None）
        
        返回值：
        - True: 抑制异常
        - False: 继续传播异常
        """
        if exc_type is ZeroDivisionError:
            print(f"捕获到除零错误: {exc_val}")
            return True  # 抑制异常
        return False  # 其他异常继续传播

# 使用示例
with SafeDivision():
    result = 10 / 0  # 不会抛出异常
    print(result)
# 输出:
# 进入安全除法模式
# 捕获到除零错误: division by zero
```

### 使用 contextlib 模块

```python
from contextlib import contextmanager

@contextmanager
def timer(name="Timer"):
    """使用装饰器创建上下文管理器"""
    import time
    start = time.time()
    print(f"{name} 开始...")
    try:
        yield  # yield 之前的代码相当于 __enter__
    finally:
        # yield 之后的代码相当于 __exit__
        elapsed = time.time() - start
        print(f"{name} 结束，耗时: {elapsed:.4f} 秒")

# 使用示例
with timer("数据处理"):
    sum(range(10000000))
```

### 带 yield 值的上下文管理器

```python
from contextlib import contextmanager

@contextmanager
def open_file(filename, mode="r"):
    """自定义文件打开上下文管理器"""
    file = open(filename, mode, encoding="utf-8")
    try:
        yield file  # yield 的值可以通过 as 获取
    finally:
        file.close()

# 使用示例
with open_file("example.txt", "r") as f:
    content = f.read()
    print(content)
```

## contextlib 模块工具

### contextlib.suppress

```python
from contextlib import suppress

# 抑制指定异常
with suppress(FileNotFoundError):
    os.remove("不存在的文件.txt")  # 文件不存在也不会报错

# 等价于
try:
    os.remove("不存在的文件.txt")
except FileNotFoundError:
    pass

# 抑制多种异常
with suppress(FileNotFoundError, PermissionError):
    os.remove("some_file.txt")
```

### contextlib.ExitStack

```python
from contextlib import ExitStack

# 动态管理多个上下文
with ExitStack() as stack:
    files = [
        stack.enter_context(open(f"file{i}.txt", "w"))
        for i in range(3)
    ]
    for i, f in enumerate(files):
        f.write(f"这是文件 {i} 的内容")

# 所有文件都会正确关闭

# 条件性进入上下文
def process_files(filenames, mode="r"):
    with ExitStack() as stack:
        files = [stack.enter_context(open(f, mode)) for f in filenames]
        # 处理文件...
        return [f.read() for f in files]
```

### contextlib.nullcontext

```python
from contextlib import nullcontext

# 当可能需要上下文管理器，也可能不需要时使用
def process_data(data, lock=None):
    # 如果 lock 为 None，使用 nullcontext
    with lock or nullcontext():
        # 处理数据
        return data.upper()

# 使用示例
process_data("hello")  # 不需要锁
process_data("hello", threading.Lock())  # 需要锁
```

## 实际应用场景

### 数据库连接管理

```python
from contextlib import contextmanager

@contextmanager
def database_connection(host, port, database):
    """数据库连接上下文管理器"""
    import sqlite3
    conn = sqlite3.connect(f"{host}:{port}/{database}")
    try:
        yield conn
    finally:
        conn.close()

# 使用示例
with database_connection("localhost", 5432, "mydb") as conn:
    cursor = conn.cursor()
    cursor.execute("SELECT * FROM users")
    results = cursor.fetchall()
```

### 计时器

import time
from contextlib import contextmanager

@contextmanager
def timer(name="操作"):
    """计时上下文管理器"""
    start = time.perf_counter()
    try:
        yield
    finally:
        elapsed = time.perf_counter() - start

### 临时目录

```python
import os
import tempfile
from contextlib import contextmanager

@contextmanager
def temp_directory():
    """临时目录上下文管理器"""
    temp_dir = tempfile.mkdtemp()
    try:
        yield temp_dir
    finally:
        # 清理临时目录
        import shutil
        shutil.rmtree(temp_dir)

# 使用示例
with temp_directory() as temp_dir:
    # 在临时目录中工作
    temp_file = os.path.join(temp_dir, "temp.txt")
    with open(temp_file, "w") as f:
        f.write("临时内容")
    print(f"临时文件路径: {temp_file}")
# 临时目录自动删除
```

### 改变工作目录

```python
import os
from contextlib import contextmanager

@contextmanager
def change_directory(path):
    """临时改变工作目录"""
    old_dir = os.getcwd()
    try:
        os.chdir(path)
        yield
    finally:
        os.chdir(old_dir)

# 使用示例
with change_directory("/tmp"):
    # 在 /tmp 目录下工作
    print(f"当前目录: {os.getcwd()}")
# 自动恢复原目录
```

### 锁管理

```python
import threading
from contextlib import contextmanager

class LockManager:
    """线程锁管理器"""
    
    def __init__(self):
        self.lock = threading.Lock()
    
    @contextmanager
    def acquire(self):
        """获取锁的上下文管理器"""
        self.lock.acquire()
        try:
            yield
        finally:
            self.lock.release()

# 使用示例
lock_manager = LockManager()

def safe_increment(counter):
    with lock_manager.acquire():
        counter["value"] += 1

counter = {"value": 0}
threads = [
    threading.Thread(target=safe_increment, args=(counter,))
    for _ in range(100)
]
for t in threads:
    t.start()
for t in threads:
    t.join()
print(counter["value"])  # 输出: 100
```

## 嵌套上下文管理器

```python
from contextlib import contextmanager

@contextmanager
def step(name):
    """步骤上下文管理器"""
    print(f"开始: {name}")
    try:
        yield
    finally:
        print(f"结束: {name}")

# 嵌套使用
with step("步骤1"):
    print("执行步骤1")
    with step("步骤2"):
        print("执行步骤2")
        with step("步骤3"):
            print("执行步骤3")

# 输出:
# 开始: 步骤1
# 执行步骤1
# 开始: 步骤2
# 执行步骤2
# 开始: 步骤3
# 执行步骤3
# 结束: 步骤3
# 结束: 步骤2
# 结束: 步骤1
```

### 练习二：事务管理器

```python
from contextlib import contextmanager

class Transaction:
    """数据库事务模拟"""
    
    def __init__(self):
        self.committed = False
        self.data = {}
    
    def set(self, key, value):
        self.data[key] = value
    
    def commit(self):
        self.committed = True
        print("事务已提交")
    
    def rollback(self):
        self.data.clear()
        print("事务已回滚")

@contextmanager
def transaction():
    """事务上下文管理器"""
    tx = Transaction()
    try:
        yield tx
        tx.commit()  # 正常结束时提交
    except Exception as e:
        tx.rollback()  # 异常时回滚
        raise

# 使用示例
try:
    with transaction() as tx:
        tx.set("name", "张三")
        tx.set("age", 25)
        # 模拟错误
        # raise ValueError("测试错误")
except ValueError:
    print("事务失败")
```

### 练习三：配置临时修改

```python
from contextlib import contextmanager

@contextmanager
def temporary_config(obj, **kwargs):
    """临时修改对象属性"""
    # 保存原值
    original = {k: getattr(obj, k) for k in kwargs}
    
    try:
        # 设置新值
        for k, v in kwargs.items():
            setattr(obj, k, v)
        yield
    finally:
        # 恢复原值
        for k, v in original.items():
            setattr(obj, k, v)

# 使用示例
class Config:
    debug = False
    timeout = 30

config = Config()
print(f"原始: debug={config.debug}, timeout={config.timeout}")

with temporary_config(config, debug=True, timeout=60):
    print(f"临时: debug={config.debug}, timeout={config.timeout}")

print(f"恢复: debug={config.debug}, timeout={config.timeout}")
```

## 小结

本章要点：

1. **with 语句**：自动管理资源，确保正确释放
2. **__enter__ 和 __exit__**：上下文管理器协议
3. **contextmanager 装饰器**：简化上下文管理器创建
4. **contextlib 模块**：suppress、redirect_stdout、ExitStack 等工具
5. **实际应用**：文件操作、数据库连接、锁管理、计时器等

> **提示**：上下文管理器是 Python 资源管理的最佳实践
