# 9.5 声音与音乐

Pygame 提供了强大的音频处理功能，可以播放音效和背景音乐。

## 音频系统初始化

```python
import pygame

# 初始化音频模块
pygame.mixer.init()

# 初始化参数
pygame.mixer.init(
    frequency=44100,    # 采样率，默认 44100 Hz
    size=-16,           # 采样大小，-16 表示有符号 16 位
    channels=2,         # 声道数，2 表示立体声
    buffer=512          # 缓冲区大小
)

# 检查音频系统
print(f"音频初始化: {pygame.mixer.get_init()}")
print(f"声道数: {pygame.mixer.get_num_channels()}")
```

## 播放音效

### 1. 加载与播放音效

```python
import pygame
import sys
import os

pygame.init()
pygame.mixer.init()

screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("音效播放演示")

clock = pygame.time.Clock()
font = pygame.font.Font(None, 36)

# 创建音效（如果没有音频文件，使用程序生成）
def create_beep_sound(frequency=440, duration=0.2):
    """
    生成简单的蜂鸣音效
    
    参数:
        frequency: 频率 (Hz)
        duration: 持续时间 (秒)
    
    返回:
        Sound 对象
    """
    import math
    
    sample_rate = 44100
    n_samples = int(sample_rate * duration)
    
    # 创建音频缓冲区
    buf = bytes(n_samples * 2)  # 16 位立体声
    
    # 生成正弦波
    import array
    samples = array.array('h')  # 有符号 16 位整数
    
    for i in range(n_samples):
        t = i / sample_rate
        # 正弦波 + 淡入淡出
        envelope = min(1, min(i / 1000, (n_samples - i) / 1000))
        value = int(32767 * envelope * math.sin(2 * math.pi * frequency * t))
        samples.append(value)  # 左声道
        samples.append(value)  # 右声道
    
    # 创建 Sound 对象
    sound = pygame.mixer.Sound(buffer=samples)
    return sound

# 创建不同音调的音效
sound_c = create_beep_sound(262, 0.15)  # C 音
sound_e = create_beep_sound(330, 0.15)  # E 音
sound_g = create_beep_sound(392, 0.15)  # G 音
sound_high_c = create_beep_sound(523, 0.3)  # 高音 C

# 设置音量 (0.0 - 1.0)
sound_c.set_volume(0.5)
sound_e.set_volume(0.5)
sound_g.set_volume(0.5)
sound_high_c.set_volume(0.5)

# 按键映射
key_sounds = {
    pygame.K_a: sound_c,
    pygame.K_s: sound_e,
    pygame.K_d: sound_g,
    pygame.K_f: sound_high_c,
}

key_names = {
    pygame.K_a: "C",
    pygame.K_s: "E",
    pygame.K_d: "G",
    pygame.K_f: "高音 C",
}

current_note = "按 A/S/D/F 演奏音符"

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        elif event.type == pygame.KEYDOWN:
            if event.key == pygame.K_ESCAPE:
                running = False
            elif event.key in key_sounds:
                key_sounds[event.key].play()
                current_note = f"正在演奏: {key_names[event.key]}"
    
    screen.fill((30, 30, 30))
    
    # 显示提示
    title = font.render("简易钢琴 - 按 A/S/D/F 演奏", True, (255, 255, 255))
    screen.blit(title, (250, 200))
    
    note_text = font.render(current_note, True, (255, 200, 0))
    screen.blit(note_text, (280, 300))
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

## 播放背景音乐

### 1. 音乐播放基础

```python
import pygame
import sys

pygame.init()
pygame.mixer.init()

screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("音乐播放器")

clock = pygame.time.Clock()
font = pygame.font.Font(None, 36)

# 音乐状态
music_playing = False
music_volume = 0.5

# 注意：音乐文件需要自己准备
# 支持格式：MP3, OGG, WAV, MIDI
# pygame.mixer.music.load("background.mp3")

# 使用程序生成简单的背景音乐效果
def play_tone_sequence():
    """播放音调序列作为背景音乐"""
    import math
    import array
    
    sample_rate = 44100
    duration = 2.0  # 2秒循环
    n_samples = int(sample_rate * duration)
    
    samples = array.array('h')
    
    # 生成简单的旋律
    notes = [262, 294, 330, 349, 392, 440, 494, 523]  # C 大调音阶
    
    for i in range(n_samples):
        t = i / sample_rate
        note_idx = int(t * 4) % len(notes)  # 每秒4个音符
        freq = notes[note_idx]
        
        # 淡入淡出
        note_time = (t * 4) % 1
        envelope = min(note_time * 10, (1 - note_time) * 10, 1)
        
        value = int(16000 * envelope * math.sin(2 * math.pi * freq * t))
        samples.append(value)
        samples.append(value)
    
    # 保存为临时文件
    import tempfile
    import wave
    
    with tempfile.NamedTemporaryFile(suffix='.wav', delete=False) as f:
        temp_path = f.name
    
    with wave.open(temp_path, 'w') as wav:
        wav.setnchannels(2)
        wav.setsampwidth(2)
        wav.setframerate(sample_rate)
        wav.writeframes(samples.tobytes())
    
    return temp_path

# 生成音乐文件
try:
    music_file = play_tone_sequence()
    pygame.mixer.music.load(music_file)
    pygame.mixer.music.set_volume(music_volume)
    has_music = True
except Exception as e:
    print(f"无法创建音乐: {e}")
    has_music = False

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        elif event.type == pygame.KEYDOWN:
            if event.key == pygame.K_ESCAPE:
                running = False
            elif event.key == pygame.K_SPACE and has_music:
                if music_playing:
                    pygame.mixer.music.pause()
                    music_playing = False
                else:
                    if pygame.mixer.music.get_pos() == -1:
                        pygame.mixer.music.play(-1)  # 无限循环
                    else:
                        pygame.mixer.music.unpause()
                    music_playing = True
            elif event.key == pygame.K_UP:
                music_volume = min(1.0, music_volume + 0.1)
                pygame.mixer.music.set_volume(music_volume)
            elif event.key == pygame.K_DOWN:
                music_volume = max(0.0, music_volume - 0.1)
                pygame.mixer.music.set_volume(music_volume)
    
    screen.fill((30, 30, 30))
    
    # 显示状态
    status = "播放中" if music_playing else "已暂停"
    status_text = font.render(f"音乐状态: {status}", True, (255, 255, 255))
    screen.blit(status_text, (250, 200))
    
    volume_text = font.render(f"音量: {int(music_volume * 100)}%", True, (255, 255, 255))
    screen.blit(volume_text, (250, 250))
    
    help_text = font.render("空格: 播放/暂停 | 上下键: 音量", True, (150, 150, 150))
    screen.blit(help_text, (200, 350))
    
    pygame.display.flip()
    clock.tick(60)

# 清理
if has_music:
    pygame.mixer.music.stop()
    import os
    try:
        os.unlink(music_file)
    except:
        pass

pygame.quit()
sys.exit()
```

## 音效管理器

```python
import pygame
import sys
import math
import array

pygame.init()
pygame.mixer.init()

screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("音效管理器演示")

clock = pygame.time.Clock()
font = pygame.font.Font(None, 36)


class SoundManager:
    """
    音效管理器
    
    集中管理游戏中的所有音效和音乐
    """
    
    _instance = None
    
    def __new__(cls):
        """单例模式"""
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._initialized = False
        return cls._instance
    
    def __init__(self):
        if self._initialized:
            return
        
        self._initialized = True
        self.sounds = {}       # 音效字典
        self.music_volume = 0.7
        self.sfx_volume = 0.8
        self.muted = False
    
    def generate_tone(self, frequency, duration, wave_type='sine'):
        """
        生成音调
        
        参数:
            frequency: 频率 (Hz)
            duration: 持续时间 (秒)
            wave_type: 波形类型 ('sine', 'square', 'sawtooth')
        
        返回:
            Sound 对象
        """
        sample_rate = 44100
        n_samples = int(sample_rate * duration)
        samples = array.array('h')
        
        for i in range(n_samples):
            t = i / sample_rate
            
            # 淡入淡出
            attack = min(1, i / (sample_rate * 0.01))
            decay = min(1, (n_samples - i) / (sample_rate * 0.01))
            envelope = attack * decay
            
            # 波形生成
            if wave_type == 'sine':
                value = math.sin(2 * math.pi * frequency * t)
            elif wave_type == 'square':
                value = 1 if math.sin(2 * math.pi * frequency * t) > 0 else -1
            elif wave_type == 'sawtooth':
                value = 2 * (t * frequency % 1) - 1
            else:
                value = math.sin(2 * math.pi * frequency * t)
            
            sample = int(32767 * envelope * value)
            samples.append(sample)
            samples.append(sample)
        
        return pygame.mixer.Sound(buffer=samples)
    
    def load_sound(self, name, file_path=None, **kwargs):
        """
        加载或生成音效
        
        参数:
            name: 音效名称
            file_path: 音频文件路径（可选）
            **kwargs: 生成音效的参数
        """
        if file_path:
            try:
                self.sounds[name] = pygame.mixer.Sound(file_path)
            except FileNotFoundError:
                self.sounds[name] = self.generate_tone(**kwargs) if kwargs else None
        elif kwargs:
            self.sounds[name] = self.generate_tone(**kwargs)
    
    def play(self, name, volume=None):
        """播放音效"""
        if self.muted or name not in self.sounds:
            return None
        
        sound = self.sounds[name]
        if sound:
            vol = volume if volume is not None else self.sfx_volume
            sound.set_volume(vol)
            return sound.play()
        return None
    
    def play_music(self, file_path, loops=-1):
        """播放背景音乐"""
        if not self.muted:
            try:
                pygame.mixer.music.load(file_path)
                pygame.mixer.music.set_volume(self.music_volume)
                pygame.mixer.music.play(loops)
            except:
                pass
    
    def stop_music(self):
        """停止背景音乐"""
        pygame.mixer.music.stop()
    
    def set_music_volume(self, volume):
        """设置音乐音量"""
        self.music_volume = max(0, min(1, volume))
        pygame.mixer.music.set_volume(self.music_volume)
    
    def set_sfx_volume(self, volume):
        """设置音效音量"""
        self.sfx_volume = max(0, min(1, volume))
    
    def toggle_mute(self):
        """切换静音"""
        self.muted = not self.muted
        if self.muted:
            pygame.mixer.music.set_volume(0)
        else:
            pygame.mixer.music.set_volume(self.music_volume)
        return self.muted


# 使用音效管理器
sound_manager = SoundManager()

# 生成游戏音效
sound_manager.load_sound('jump', frequency=400, duration=0.1, wave_type='square')
sound_manager.load_sound('coin', frequency=800, duration=0.15)
sound_manager.load_sound('hit', frequency=150, duration=0.2, wave_type='sawtooth')
sound_manager.load_sound('win', frequency=600, duration=0.3)

# 游戏状态
score = 0
messages = []

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        elif event.type == pygame.KEYDOWN:
            if event.key == pygame.K_ESCAPE:
                running = False
            elif event.key == pygame.K_SPACE:
                sound_manager.play('jump')
                messages.append(("跳跃!", 60))
            elif event.key == pygame.K_c:
                sound_manager.play('coin')
                score += 10
                messages.append(("收集金币 +10", 60))
            elif event.key == pygame.K_h:
                sound_manager.play('hit')
                score = max(0, score - 5)
                messages.append(("受到伤害 -5", 60))
            elif event.key == pygame.K_w:
                sound_manager.play('win')
                messages.append(("胜利!", 60))
            elif event.key == pygame.K_m:
                muted = sound_manager.toggle_mute()
                messages.append(("静音" if muted else "取消静音", 60))
            elif event.key == pygame.K_UP:
                sound_manager.set_sfx_volume(sound_manager.sfx_volume + 0.1)
            elif event.key == pygame.K_DOWN:
                sound_manager.set_sfx_volume(sound_manager.sfx_volume - 0.1)
    
    # 更新消息
    messages = [(msg, timer - 1) for msg, timer in messages if timer > 0]
    
    screen.fill((30, 30, 30))
    
    # 显示分数
    score_text = font.render(f"分数: {score}", True, (255, 255, 255))
    screen.blit(score_text, (50, 50))
    
    # 显示音量
    vol_text = font.render(f"音效音量: {int(sound_manager.sfx_volume * 100)}%", 
                           True, (255, 255, 255))
    screen.blit(vol_text, (50, 100))
    
    # 显示操作提示
    hints = [
        "空格: 跳跃音效",
        "C: 金币音效",
        "H: 受伤音效",
        "W: 胜利音效",
        "M: 静音切换",
        "上下键: 调节音量",
    ]
    for i, hint in enumerate(hints):
        hint_text = font.render(hint, True, (150, 150, 150))
        screen.blit(hint_text, (50, 200 + i * 35))
    
    # 显示消息
    for i, (msg, timer) in enumerate(messages[-5:]):
        msg_text = font.render(msg, True, (255, 200, 0))
        screen.blit(msg_text, (400, 200 + i * 40))
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

## 小结

本章要点：

1. **pygame.mixer.init()**：初始化音频系统
2. **pygame.mixer.Sound**：音效对象，适合短音效
3. **pygame.mixer.music**：音乐模块，适合背景音乐
4. **Sound.play()**：播放音效
5. **music.play()**：播放音乐
6. **set_volume()**：设置音量
7. **Channel**：通道管理，控制同时播放的音效数量

> **提示**：音效使用 Sound 对象，可以同时播放多个
