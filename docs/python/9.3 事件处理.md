# 9.3 事件处理

Pygame 通过事件系统处理用户输入，包括键盘、鼠标、游戏手柄等。

## 事件循环

```python
import pygame
import sys

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("事件处理演示")

clock = pygame.time.Clock()
running = True

# 字体用于显示文本
font = pygame.font.Font(None, 36)
event_log = []  # 存储最近的事件

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        
        # 记录事件（最多保留10条）
        event_text = f"事件: {pygame.event.event_name(event.type)}"
        event_log.append(event_text)
        if len(event_log) > 10:
            event_log.pop(0)
    
    screen.fill((30, 30, 30))
    
    # 显示事件日志
    y = 50
    for text in event_log[-10:]:
        surface = font.render(text, True, (255, 255, 255))
        screen.blit(surface, (50, y))
        y += 30
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

## 键盘事件

### 1. 键盘按下与释放

```python
import pygame
import sys

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("键盘事件")

clock = pygame.time.Clock()
font = pygame.font.Font(None, 48)

# 按键状态
pressed_keys = set()
key_names = {}

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        
        # 键盘按下事件
        elif event.type == pygame.KEYDOWN:
            pressed_keys.add(event.key)
            key_names[event.key] = pygame.key.name(event.key)
            print(f"按下: {pygame.key.name(event.key)}")
            
            # ESC 键退出
            if event.key == pygame.K_ESCAPE:
                running = False
        
        # 键盘释放事件
        elif event.type == pygame.KEYUP:
            pressed_keys.discard(event.key)
            print(f"释放: {pygame.key.name(event.key)}")
    
    screen.fill((30, 30, 30))
    
    # 显示当前按下的键
    text = "按下的键: " + ", ".join(key_names.get(k, "") for k in pressed_keys)
    if not pressed_keys:
        text = "请按下键盘上的任意键..."
    
    surface = font.render(text, True, (255, 255, 255))
    rect = surface.get_rect(center=(400, 300))
    screen.blit(surface, rect)
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

### 2. 持续按键检测

```python
import pygame
import sys

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("持续按键检测 - 移动方块")

clock = pygame.time.Clock()

# 玩家位置和速度
player_x = 400
player_y = 300
player_speed = 5
player_size = 50

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        elif event.type == pygame.KEYDOWN:
            if event.key == pygame.K_ESCAPE:
                running = False
    
    # 使用 get_pressed() 检测持续按键
    keys = pygame.key.get_pressed()
    
    # 移动玩家
    if keys[pygame.K_LEFT] or keys[pygame.K_a]:
        player_x -= player_speed
    if keys[pygame.K_RIGHT] or keys[pygame.K_d]:
        player_x += player_speed
    if keys[pygame.K_UP] or keys[pygame.K_w]:
        player_y -= player_speed
    if keys[pygame.K_DOWN] or keys[pygame.K_s]:
        player_y += player_speed
    
    # 边界检测
    player_x = max(0, min(player_x, 800 - player_size))
    player_y = max(0, min(player_y, 600 - player_size))
    
    screen.fill((30, 30, 30))
    
    # 绘制玩家
    pygame.draw.rect(screen, (0, 255, 0), 
                     (player_x, player_y, player_size, player_size))
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

## 鼠标事件

### 1. 鼠标点击

```python
import pygame
import sys

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("鼠标点击演示")

clock = pygame.time.Clock()
font = pygame.font.Font(None, 36)

# 存储点击位置
clicks = []
mouse_pos = (0, 0)

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        
        # 鼠标移动事件
        elif event.type == pygame.MOUSEMOTION:
            mouse_pos = event.pos
        
        # 鼠标按下事件
        elif event.type == pygame.MOUSEBUTTONDOWN:
            # event.button: 1=左键, 2=中键, 3=右键, 4=滚轮上, 5=滚轮下
            if event.button == 1:  # 左键
                clicks.append(('left', event.pos))
            elif event.button == 3:  # 右键
                clicks.append(('right', event.pos))
        
        # 鼠标释放事件
        elif event.type == pygame.MOUSEBUTTONUP:
            pass  # 处理释放事件
    
    screen.fill((30, 30, 30))
    
    # 显示鼠标位置
    pos_text = font.render(f"鼠标位置: {mouse_pos}", True, (255, 255, 255))
    screen.blit(pos_text, (50, 50))
    
    # 绘制点击位置
    for button, pos in clicks[-20:]:  # 只显示最近20个
        color = (255, 0, 0) if button == 'left' else (0, 0, 255)
        pygame.draw.circle(screen, color, pos, 10)
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

### 2. 鼠标拖拽绘制

```python
import pygame
import sys

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("鼠标绘图 - 按住左键绘制")

clock = pygame.time.Clock()

# 绘图表面（用于保存绘制内容）
drawing_surface = pygame.Surface((800, 600))
drawing_surface.fill((30, 30, 30))

# 画笔设置
brush_color = (255, 255, 255)
brush_size = 5
drawing = False
last_pos = None

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        
        elif event.type == pygame.MOUSEBUTTONDOWN:
            if event.button == 1:  # 左键开始绘制
                drawing = True
                last_pos = event.pos
            elif event.button == 4:  # 滚轮上 - 增大画笔
                brush_size = min(50, brush_size + 2)
            elif event.button == 5:  # 滚轮下 - 减小画笔
                brush_size = max(1, brush_size - 2)
        
        elif event.type == pygame.MOUSEBUTTONUP:
            if event.button == 1:
                drawing = False
                last_pos = None
        
        elif event.type == pygame.MOUSEMOTION:
            if drawing:
                # 绘制线条（连接上一个点和当前点）
                if last_pos:
                    pygame.draw.line(drawing_surface, brush_color, 
                                     last_pos, event.pos, brush_size)
                last_pos = event.pos
        
        elif event.type == pygame.KEYDOWN:
            # 按键切换颜色
            if event.key == pygame.K_r:
                brush_color = (255, 0, 0)
            elif event.key == pygame.K_g:
                brush_color = (0, 255, 0)
            elif event.key == pygame.K_b:
                brush_color = (0, 0, 255)
            elif event.key == pygame.K_w:
                brush_color = (255, 255, 255)
            elif event.key == pygame.K_c:
                drawing_surface.fill((30, 30, 30))  # 清屏
    
    # 绘制
    screen.blit(drawing_surface, (0, 0))
    
    # 显示画笔大小
    font = pygame.font.Font(None, 24)
    info = font.render(f"画笔大小: {brush_size} | R/G/B/W 切换颜色 | C 清屏", 
                       True, (200, 200, 200))
    screen.blit(info, (10, 10))
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

## 自定义事件

```python
import pygame
import sys
import random

pygame.init()
screen = pygame.display.set_mode((800, 600))
pygame.display.set_caption("自定义事件")

clock = pygame.time.Clock()
font = pygame.font.Font(None, 36)

# 创建自定义事件
# 使用 pygame.USEREVENT 作为基础值
SPAWN_ENEMY = pygame.USEREVENT + 1
POWER_UP = pygame.USEREVENT + 2

# 设置定时器
# 参数：事件类型，间隔时间（毫秒）
pygame.time.set_timer(SPAWN_ENEMY, 1000)  # 每秒触发一次
pygame.time.set_timer(POWER_UP, 3000)      # 每3秒触发一次

enemies = []
power_ups = []

running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        
        # 处理自定义事件
        elif event.type == SPAWN_ENEMY:
            # 随机生成敌人
            x = random.randint(50, 750)
            enemies.append({'x': x, 'y': 0, 'speed': random.randint(2, 5)})
        
        elif event.type == POWER_UP:
            # 随机生成道具
            x = random.randint(50, 750)
            power_ups.append({'x': x, 'y': 0})
    
    # 更新敌人位置
    for enemy in enemies[:]:
        enemy['y'] += enemy['speed']
        if enemy['y'] > 600:
            enemies.remove(enemy)
    
    # 更新道具位置
    for power in power_ups[:]:
        power['y'] += 2
        if power['y'] > 600:
            power_ups.remove(power)
    
    screen.fill((30, 30, 30))
    
    # 绘制敌人（红色方块）
    for enemy in enemies:
        pygame.draw.rect(screen, (255, 0, 0), 
                         (enemy['x'] - 15, enemy['y'] - 15, 30, 30))
    
    # 绘制道具（黄色圆形）
    for power in power_ups:
        pygame.draw.circle(screen, (255, 255, 0), 
                           (power['x'], int(power['y'])), 15)
    
    # 显示信息
    info = font.render(f"敌人: {len(enemies)} | 道具: {len(power_ups)}", 
                       True, (255, 255, 255))
    screen.blit(info, (50, 50))
    
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

## 小结

本章要点：

1. **pygame.event.get()**：获取事件队列
2. **pygame.KEYDOWN / KEYUP**：键盘按下/释放事件
3. **pygame.key.get_pressed()**：获取持续按键状态
4. **pygame.MOUSEBUTTONDOWN / UP**：鼠标按下/释放事件
5. **pygame.MOUSEMOTION**：鼠标移动事件
6. **pygame.mouse.get_pos()**：获取鼠标位置
7. **pygame.USEREVENT**：自定义事件基类
8. **pygame.time.set_timer()**：设置定时事件

> **提示**：对于需要持续响应的按键，使用 `get_pressed()` 比事件检测更合适
