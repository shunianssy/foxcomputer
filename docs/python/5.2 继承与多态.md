# 5.2 继承与多态

继承允许创建基于现有类的新类，多态允许不同类的对象对同一方法做出不同响应。

## 继承基础

### 基本继承

```python
# 父类（基类）
class Animal:
    """动物类"""
    
    def __init__(self, name):
        self.name = name
    
    def speak(self):
        """发声"""
        pass
    
    def info(self):
        return f"我是{self.name}"


# 子类（派生类）
class Dog(Animal):
    """狗类"""
    
    def speak(self):
        return "汪汪汪！"


class Cat(Animal):
    """猫类"""
    
    def speak(self):
        return "喵喵喵！"


# 使用示例
dog = Dog("小黄")
cat = Cat("小白")

print(dog.info())      # 输出: 我是小黄
print(dog.speak())     # 输出: 汪汪汪！
print(cat.speak())     # 输出: 喵喵喵！

# 检查继承关系
print(isinstance(dog, Dog))     # True
print(isinstance(dog, Animal))  # True
print(issubclass(Dog, Animal))  # True
```

### super() 函数

```python
class Person:
    """人员类"""
    
    def __init__(self, name, age):
        self.name = name
        self.age = age
    
    def info(self):
        return f"姓名: {self.name}, 年龄: {self.age}"


class Student(Person):
    """学生类"""
    
    def __init__(self, name, age, grade):
        # 调用父类的构造方法
        super().__init__(name, age)
        self.grade = grade
    
    def info(self):
        # 调用父类的方法
        base_info = super().info()
        return f"{base_info}, 年级: {self.grade}"


# 使用示例
student = Student("张三", 18, "高三")
print(student.info())  # 输出: 姓名: 张三, 年龄: 18, 年级: 高三
```

## 多重继承

```python
class Flyable:
    """可飞行的"""
    
    def fly(self):
        return "正在飞行"


class Swimmable:
    """可游泳的"""
    
    def swim(self):
        return "正在游泳"


class Duck(Animal, Flyable, Swimmable):
    """鸭子类"""
    
    def speak(self):
        return "嘎嘎嘎！"


# 使用示例
duck = Duck("唐老鸭")
print(duck.speak())  # 输出: 嘎嘎嘎！
print(duck.fly())    # 输出: 正在飞行
print(duck.swim())   # 输出: 正在游泳

# 方法解析顺序（MRO）
print(Duck.__mro__)
# 输出: (<class '__main__.Duck'>, <class '__main__.Animal'>, 
#       <class '__main__.Flyable'>, <class '__main__.Swimmable'>, <class 'object'>)
```

## 方法重写

```python
class Shape:
    """形状基类"""
    
    def __init__(self, name):
        self.name = name
    
    def area(self):
        """计算面积"""
        raise NotImplementedError("子类必须实现此方法")
    
    def perimeter(self):
        """计算周长"""
        raise NotImplementedError("子类必须实现此方法")
    
    def __str__(self):
        return f"{self.name}: 面积={self.area():.2f}, 周长={self.perimeter():.2f}"


class Rectangle(Shape):
    """矩形"""
    
    def __init__(self, width, height):
        super().__init__("矩形")
        self.width = width
        self.height = height
    
    def area(self):
        return self.width * self.height
    
    def perimeter(self):
        return 2 * (self.width + self.height)


class Circle(Shape):
    """圆形"""
    
    def __init__(self, radius):
        super().__init__("圆形")
        self.radius = radius
    
    def area(self):
        return 3.14159 * self.radius ** 2
    
    def perimeter(self):
        return 2 * 3.14159 * self.radius


# 使用示例
shapes = [
    Rectangle(5, 3),
    Circle(4)
]

for shape in shapes:
    print(shape)
# 输出:
# 矩形: 面积=15.00, 周长=16.00
# 圆形: 面积=50.27, 周长=25.13
```

## 多态

```python
# 多态：不同对象对同一方法做出不同响应
def make_sound(animal):
    """让动物发声"""
    print(f"{animal.name}说: {animal.speak()}")


# 使用示例
animals = [Dog("小黄"), Cat("小白"), Duck("唐老鸭")]

for animal in animals:
    make_sound(animal)
# 输出:
# 小黄说: 汪汪汪！
# 小白说: 喵喵喵！
# 唐老鸭说: 嘎嘎嘎！
```

## 抽象类

```python
from abc import ABC, abstractmethod


class Vehicle(ABC):
    """交通工具抽象类"""
    
    def __init__(self, brand):
        self.brand = brand
    
    @abstractmethod
    def start(self):
        """启动"""
        pass
    
    @abstractmethod
    def stop(self):
        """停止"""
        pass
    
    def info(self):
        return f"品牌: {self.brand}"


class Car(Vehicle):
    """汽车"""
    
    def start(self):
        return "启动引擎"
    
    def stop(self):
        return "熄火"


class Bicycle(Vehicle):
    """自行车"""
    
    def start(self):
        return "开始蹬踏板"
    
    def stop(self):
        return "刹车停止"


# 使用示例
car = Car("丰田")
bike = Bicycle("捷安特")

print(f"{car.brand}: {car.start()}")  # 输出: 丰田: 启动引擎
print(f"{bike.brand}: {bike.start()}")  # 输出: 捷安特: 开始蹬踏板

# 不能实例化抽象类
# vehicle = Vehicle("未知")  # TypeError
```

## 实战练习

### 练习一：员工管理系统

```python
class Employee:
    """员工基类"""
    
    def __init__(self, name, base_salary):
        self.name = name
        self.base_salary = base_salary
    
    def calculate_salary(self):
        """计算工资"""
        return self.base_salary
    
    def __str__(self):
        return f"{self.name}: 工资 {self.calculate_salary():.2f}"


class Manager(Employee):
    """经理"""
    
    def __init__(self, name, base_salary, bonus):
        super().__init__(name, base_salary)
        self.bonus = bonus
    
    def calculate_salary(self):
        return self.base_salary + self.bonus


class Salesperson(Employee):
    """销售"""
    
    def __init__(self, name, base_salary, sales, commission_rate):
        super().__init__(name, base_salary)
        self.sales = sales
        self.commission_rate = commission_rate
    
    def calculate_salary(self):
        return self.base_salary + self.sales * self.commission_rate


# 使用示例
employees = [
    Manager("张经理", 10000, 5000),
    Salesperson("李销售", 5000, 100000, 0.05),
    Employee("王员工", 8000)
]

for emp in employees:
    print(emp)
# 输出:
# 张经理: 工资 15000.00
# 李销售: 工资 10000.00
# 王员工: 工资 8000.00
```

### 练习二：图形计算器

```python
from abc import ABC, abstractmethod
import math


class Shape(ABC):
    """图形抽象类"""
    
    @abstractmethod
    def area(self):
        pass
    
    @abstractmethod
    def perimeter(self):
        pass


class Rectangle(Shape):
    def __init__(self, width, height):
        self.width = width
        self.height = height
    
    def area(self):
        return self.width * self.height
    
    def perimeter(self):
        return 2 * (self.width + self.height)


class Circle(Shape):
    def __init__(self, radius):
        self.radius = radius
    
    def area(self):
        return math.pi * self.radius ** 2
    
    def perimeter(self):
        return 2 * math.pi * self.radius


class Triangle(Shape):
    def __init__(self, a, b, c):
        self.a = a
        self.b = b
        self.c = c
    
    def area(self):
        # 海伦公式
        s = (self.a + self.b + self.c) / 2
        return math.sqrt(s * (s - self.a) * (s - self.b) * (s - self.c))
    
    def perimeter(self):
        return self.a + self.b + self.c


# 计算总面积
def total_area(shapes):
    return sum(shape.area() for shape in shapes)


shapes = [
    Rectangle(5, 3),
    Circle(4),
    Triangle(3, 4, 5)
]

print(f"总面积: {total_area(shapes):.2f}")
```

## 小结

本章要点：

1. **继承**：子类继承父类的属性和方法
2. **super()**：调用父类的方法
3. **多重继承**：一个类可以继承多个父类
4. **方法重写**：子类重新定义父类的方法
5. **多态**：不同对象对同一方法做出不同响应
6. **抽象类**：使用 ABC 和 @abstractmethod 定义

> **提示**：继承要遵循"is-a"关系
