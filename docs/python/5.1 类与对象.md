# 5.1 类与对象

面向对象编程（OOP）是一种编程范式，它使用"对象"来设计程序。Python 是一门面向对象的语言。

## 类的基本概念

### 定义类

```python
# 使用 class 关键字定义类
class Person:
    """人员类"""
    
    # 类属性（所有实例共享）
    species = "人类"
    
    # 构造方法
    def __init__(self, name, age):
        # 实例属性
        self.name = name
        self.age = age
    
    # 实例方法
    def greet(self):
        """打招呼"""
        return f"你好，我是{self.name}"
    
    def get_info(self):
        """获取信息"""
        return f"{self.name}, {self.age}岁"


# 创建对象（实例化）
person1 = Person("张三", 25)
person2 = Person("李四", 30)

# 访问属性
print(person1.name)           # 输出: 张三
print(person2.age)            # 输出: 30
print(person1.species)        # 输出: 人类

# 调用方法
print(person1.greet())        # 输出: 你好，我是张三
print(person2.get_info())     # 输出: 李四, 30岁
```

### self 参数

```python
# self 指向当前实例对象
class Counter:
    def __init__(self):
        self.count = 0
    
    def increment(self):
        self.count += 1
        return self.count
    
    def reset(self):
        self.count = 0

c1 = Counter()
c2 = Counter()

print(c1.increment())  # 输出: 1
print(c1.increment())  # 输出: 2
print(c2.increment())  # 输出: 1（c2 有自己的 count）
```

## 类属性与实例属性

```python
class Student:
    # 类属性
    school = "清华大学"
    total_count = 0
    
    def __init__(self, name, score):
        # 实例属性
        self.name = name
        self.score = score
        # 修改类属性
        Student.total_count += 1
    
    def get_info(self):
        return f"{self.name} - {self.school} - {self.score}分"


# 创建实例
s1 = Student("张三", 85)
s2 = Student("李四", 92)

print(s1.get_info())  # 输出: 张三 - 清华大学 - 85分
print(f"总人数: {Student.total_count}")  # 输出: 总人数: 2

# 修改类属性
Student.school = "北京大学"
print(s1.get_info())  # 输出: 张三 - 北京大学 - 85分

# 实例属性遮蔽类属性
s1.school = "复旦大学"  # 创建了实例属性
print(s1.school)        # 输出: 复旦大学（实例属性）
print(s2.school)        # 输出: 北京大学（类属性）
```

## 类方法与静态方法

```python
class Date:
    """日期类"""
    
    def __init__(self, year, month, day):
        self.year = year
        self.month = month
        self.day = day
    
    # 实例方法
    def display(self):
        return f"{self.year}-{self.month:02d}-{self.day:02d}"
    
    # 类方法（使用 @classmethod 装饰器）
    @classmethod
    def from_string(cls, date_str):
        """从字符串创建日期"""
        year, month, day = map(int, date_str.split('-'))
        return cls(year, month, day)
    
    # 静态方法（使用 @staticmethod 装饰器）
    @staticmethod
    def is_leap_year(year):
        """判断是否为闰年"""
        return (year % 4 == 0 and year % 100 != 0) or (year % 400 == 0)


# 使用实例方法
d1 = Date(2024, 1, 15)
print(d1.display())  # 输出: 2024-01-15

# 使用类方法创建实例
d2 = Date.from_string("2024-12-25")
print(d2.display())  # 输出: 2024-12-25

# 使用静态方法
print(Date.is_leap_year(2024))  # 输出: True
print(Date.is_leap_year(2023))  # 输出: False
```

## 属性装饰器

```python
class Circle:
    """圆类"""
    
    def __init__(self, radius):
        self._radius = radius
    
    # getter
    @property
    def radius(self):
        """获取半径"""
        return self._radius
    
    # setter
    @radius.setter
    def radius(self, value):
        """设置半径"""
        if value <= 0:
            raise ValueError("半径必须为正数")
        self._radius = value
    
    # 只读属性
    @property
    def area(self):
        """计算面积（只读）"""
        return 3.14159 * self._radius ** 2
    
    @property
    def circumference(self):
        """计算周长（只读）"""
        return 2 * 3.14159 * self._radius


# 使用示例
circle = Circle(5)
print(f"半径: {circle.radius}")           # 输出: 5
print(f"面积: {circle.area:.2f}")         # 输出: 78.54
print(f"周长: {circle.circumference:.2f}") # 输出: 31.42

# 修改半径
circle.radius = 10
print(f"新面积: {circle.area:.2f}")       # 输出: 314.16

# 尝试设置无效值
try:
    circle.radius = -5
except ValueError as e:
    print(e)  # 输出: 半径必须为正数
```

## 私有属性

```python
class BankAccount:
    """银行账户类"""
    
    def __init__(self, owner, balance):
        self.owner = owner
        self.__balance = balance  # 私有属性（名称改编）
    
    def deposit(self, amount):
        """存款"""
        if amount > 0:
            self.__balance += amount
            return True
        return False
    
    def withdraw(self, amount):
        """取款"""
        if 0 < amount <= self.__balance:
            self.__balance -= amount
            return True
        return False
    
    def get_balance(self):
        """获取余额"""
        return self.__balance


# 使用示例
account = BankAccount("张三", 1000)
print(account.get_balance())  # 输出: 1000

account.deposit(500)
print(account.get_balance())  # 输出: 1500

# 无法直接访问私有属性
# print(account.__balance)  # AttributeError

# 可以通过名称改编访问（不推荐）
print(account._BankAccount__balance)  # 输出: 1500
```

## 实战练习

### 练习一：学生管理

```python
class StudentManager:
    """学生管理类"""
    
    def __init__(self):
        self.__students = []
    
    def add_student(self, name, score):
        """添加学生"""
        student = {"name": name, "score": score}
        self.__students.append(student)
        print(f"已添加学生: {name}")
    
    def remove_student(self, name):
        """移除学生"""
        for i, student in enumerate(self.__students):
            if student["name"] == name:
                self.__students.pop(i)
                print(f"已移除学生: {name}")
                return True
        print(f"未找到学生: {name}")
        return False
    
    def get_average(self):
        """获取平均分"""
        if not self.__students:
            return 0
        return sum(s["score"] for s in self.__students) / len(self.__students)
    
    def get_all(self):
        """获取所有学生"""
        return self.__students.copy()


# 使用示例
manager = StudentManager()
manager.add_student("张三", 85)
manager.add_student("李四", 92)
manager.add_student("王五", 78)

print(f"平均分: {manager.get_average():.2f}")
print("所有学生:", manager.get_all())
```

### 练习二：矩形类

```python
class Rectangle:
    """矩形类"""
    
    def __init__(self, width, height):
        self.width = width
        self.height = height
    
    @property
    def area(self):
        """面积"""
        return self.width * self.height
    
    @property
    def perimeter(self):
        """周长"""
        return 2 * (self.width + self.height)
    
    def __str__(self):
        return f"矩形({self.width}x{self.height})"
    
    def __repr__(self):
        return f"Rectangle({self.width}, {self.height})"


# 使用示例
rect = Rectangle(5, 3)
print(rect)              # 输出: 矩形(5x3)
print(f"面积: {rect.area}")          # 输出: 15
print(f"周长: {rect.perimeter}")     # 输出: 16
```

## 小结

本章要点：

1. **类定义**：使用 class 关键字
2. **实例属性**：在 __init__ 中定义
3. **类属性**：所有实例共享
4. **实例方法**：第一个参数是 self
5. **类方法**：使用 @classmethod，第一个参数是 cls
6. **静态方法**：使用 @staticmethod，不需要 self 或 cls
7. **属性装饰器**：@property 实现 getter/setter
8. **私有属性**：使用 __ 前缀

> **提示**：类是创建对象的模板，对象是类的实例
