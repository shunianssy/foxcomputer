# 7.4 JSON 与 CSV

JSON 和 CSV 是两种最常见的数据交换格式，Python 提供了内置模块来处理它们。

## JSON 基础

### 什么是 JSON

JSON（JavaScript Object Notation）是一种轻量级的数据交换格式，易于阅读和编写。

```python
# JSON 数据类型
# - 对象：{"key": "value"}
# - 数组：[1, 2, 3]
# - 字符串："hello"
# - 数字：123, 3.14
# - 布尔：true, false
# - 空值：null
```

### 导入 json 模块

```python
import json
```

## JSON 编码（Python 对象转 JSON）

### json.dumps() - 转换为字符串

```python
import json

# 字典转 JSON 字符串
data = {"name": "张三", "age": 25, "city": "北京"}
json_str = json.dumps(data)
print(json_str)  # 输出: {"name": "\u5f20\u4e09", "age": 25, "city": "\u5317\u4eac"}

# ensure_ascii=False 保留中文
json_str = json.dumps(data, ensure_ascii=False)
print(json_str)  # 输出: {"name": "张三", "age": 25, "city": "北京"}

# 格式化输出
json_str = json.dumps(data, ensure_ascii=False, indent=2)
print(json_str)
# 输出:
# {
#   "name": "张三",
#   "age": 25,
#   "city": "北京"
# }

# 排序键
json_str = json.dumps(data, ensure_ascii=False, sort_keys=True)
print(json_str)  # 输出: {"age": 25, "city": "北京", "name": "张三"}
```

## JSON 解码（JSON 转 Python 对象）

### json.loads() - 从字符串解析

```python
import json

# JSON 字符串转 Python 对象
json_str = '{"name": "张三", "age": 25, "city": "北京"}'
data = json.loads(json_str)
print(data)  # 输出: {'name': '张三', 'age': 25, 'city': '北京'}
print(type(data))  # 输出: <class 'dict'>

# 解析数组
json_arr = '[1, 2, 3, 4, 5]'
arr = json.loads(json_arr)
print(arr)  # 输出: [1, 2, 3, 4, 5]
```

## 自定义编码和解码

### 自定义编码器

```python
import json
from datetime import datetime

class DateTimeEncoder(json.JSONEncoder):
    """自定义 JSON 编码器，处理 datetime 对象"""
    
    def default(self, obj):
        if isinstance(obj, datetime):
            return obj.isoformat()
        return super().default(obj)

# 使用示例
data = {
    "name": "张三",
    "created_at": datetime.now()
}

json_str = json.dumps(data, cls=DateTimeEncoder, ensure_ascii=False)
print(json_str)
```

### 自定义解码器

```python
import json
from datetime import datetime

def datetime_hook(obj):
    """自定义解码钩子"""
    for key, value in obj.items():
        if key.endswith('_at') and isinstance(value, str):
            try:
                obj[key] = datetime.fromisoformat(value)
            except ValueError:
                pass
    return obj

json_str = '{"name": "张三", "created_at": "2024-01-15T10:30:00"}'
data = json.loads(json_str, object_hook=datetime_hook)
print(data)
print(type(data["created_at"]))  # 输出: <class 'datetime.datetime'>
```

### 使用 default 参数

```python
import json
from datetime import datetime

def custom_default(obj):
    """自定义序列化函数"""
    if isinstance(obj, datetime):
        return obj.strftime("%Y-%m-%d %H:%M:%S")
    if isinstance(obj, set):
        return list(obj)
    raise TypeError(f"Object of type {type(obj)} is not JSON serializable")

data = {
    "time": datetime.now(),
    "tags": {"python", "json", "tutorial"}
}

json_str = json.dumps(data, default=custom_default, ensure_ascii=False)
print(json_str)
```

## CSV 基础

### 什么是 CSV

CSV（Comma-Separated Values）是一种简单的表格数据格式，每行一条记录，字段之间用逗号分隔。

### 导入 csv 模块

```python

```

### 处理大文件

```python
import csv

# 分批处理大文件
def process_large_csv(filename, batch_size=1000):
    """分批处理大型 CSV 文件"""
    with open(filename, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        batch = []
        
        for row in reader:
            batch.append(row)
            
            if len(batch) >= batch_size:
                # 处理一批数据
                process_batch(batch)
                batch = []
        
        # 处理最后一批
        if batch:
            process_batch(batch)

def process_batch(batch):
    """处理一批数据"""
    print(f"处理 {len(batch)} 条记录")
```

## 实际应用

### JSON 配置文件

```python
import json
import os

class ConfigManager:
    """配置管理器"""
    
    def __init__(self, config_file="config.json"):
        self.config_file = config_file
        self.config = self.load()
    
    def load(self):
        """加载配置"""
        if os.path.exists(self.config_file):
            with open(self.config_file, "r", encoding="utf-8") as f:
                return json.load(f)
        return {}
    
    def save(self):
        """保存配置"""
        with open(self.config_file, "w", encoding="utf-8") as f:
            json.dump(self.config, f, ensure_ascii=False, indent=2)
    
    def get(self, key, default=None):
        """获取配置值"""
        return self.config.get(key, default)
    
    def set(self, key, value):
        """设置配置值"""
        self.config[key] = value
        self.save()

# 使用示例
config = ConfigManager()
config.set("host", "localhost")
config.set("port", 8080)
print(config.get("host"))  # 输出: localhost
```

### CSV 数据分析

```python
import csv
from collections import defaultdict

def analyze_csv(filename):
    """分析 CSV 文件"""
    stats = {
        "total": 0,
        "by_city": defaultdict(int),
        "age_sum": 0,
        "age_count": 0
    }
    
    with open(filename, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        
        for row in reader:
            stats["total"] += 1
            stats["by_city"][row["city"]] += 1
            stats["age_sum"] += int(row["age"])
            stats["age_count"] += 1
    
    if stats["age_count"] > 0:
        stats["avg_age"] = stats["age_sum"] / stats["age_count"]
    
    return stats
```

### CSV 与 JSON 互转

```python
import csv
import json

def csv_to_json(csv_file, json_file):
    """CSV 转 JSON"""
    data = []
    
    with open(csv_file, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for row in reader:
            data.append(dict(row))
    
    with open(json_file, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

def json_to_csv(json_file, csv_file):
    """JSON 转 CSV"""
    with open(json_file, "r", encoding="utf-8") as f:
        data = json.load(f)
    
    if not data:
        return
    
    # 获取所有字段名
    fieldnames = list(data[0].keys())
    
    with open(csv_file, "w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(data)

# 使用示例
# csv_to_json("data.csv", "data.json")
# json_to_csv("data.json", "output.csv")
```

```python
import json

def safe_json_loads(json_str):
    """安全的 JSON 解析"""
    try:
        return json.loads(json_str)
    except json.JSONDecodeError as e:
        print(f"JSON 解析错误: {e}")
```

### CSV 错误处理

```python
import csv

def safe_csv_read(filename):
    """安全的 CSV 读取"""
    try:
        with open(filename, "r", encoding="utf-8") as f:
            reader = csv.reader(f)
            return list(reader)
    except FileNotFoundError:
        print(f"文件不存在: {filename}")
        return []
    except csv.Error as e:
        print(f"CSV 读取错误: {e}")
        return []

# 处理不同编码
def read_csv_with_encoding(filename):
    """尝试不同编码读取 CSV"""
    encodings = ["utf-8", "gbk", "gb2312", "utf-8-sig"]
    
    for encoding in encodings:
        try:
            with open(filename, "r", encoding=encoding) as f:
                reader = csv.reader(f)
                return list(reader)
        except UnicodeDecodeError:
            continue
    
    raise ValueError(f"无法确定文件编码: {filename}")
```

### 练习一：学生成绩管理

```python
import json
import os

class StudentManager:
    """学生成绩管理系统"""
    
    def __init__(self, filename="students.json"):
        self.filename = filename
        self.students = self.load()
    
    def load(self):
        """从文件加载"""
        if os.path.exists(self.filename):
            with open(self.filename, "r", encoding="utf-8") as f:
                return json.load(f)
        return []
    
    def save(self):
        """保存到文件"""
        with open(self.filename, "w", encoding="utf-8") as f:
            json.dump(self.students, f, ensure_ascii=False, indent=2)
    
    def add_student(self, name, scores):
        """添加学生"""
        student = {
            "name": name,
            "scores": scores,
            "average": sum(scores.values()) / len(scores)
        }
        self.students.append(student)
        self.save()
    
    def get_student(self, name):
        """获取学生信息"""
        for student in self.students:
            if student["name"] == name:
                return student
        return None
    
    def get_ranking(self):
        """获取排名"""
        return sorted(self.students, key=lambda x: x["average"], reverse=True)

# 使用示例
manager = StudentManager()
manager.add_student("张三", {"语文": 85, "数学": 92, "英语": 88})
manager.add_student("李四", {"语文": 90, "数学": 85, "英语": 92})

print("排名:")
for i, student in enumerate(manager.get_ranking(), 1):
    print(f"{i}. {student['name']}: 平均分 {student['average']:.1f}")
```

### 练习二：CSV 数据清洗

```python
import csv

def clean_csv(input_file, output_file):
    """清洗 CSV 数据"""
    with open(input_file, "r", encoding="utf-8") as infile, \
         open(output_file, "w", encoding="utf-8", newline="") as outfile:
        
        reader = csv.DictReader(infile)
        fieldnames = reader.fieldnames
        
        writer = csv.DictWriter(outfile, fieldnames=fieldnames)
        writer.writeheader()
        
        for row in reader:
            # 清洗数据
            cleaned_row = {}
            for key, value in row.items():
                # 去除首尾空格
                value = value.strip() if value else ""
                
                # 处理空值
                if value == "" or value.lower() == "null":
                    value = ""
                
                # 处理数字字段
                if key in ["age", "score"]:
                    try:
                        value = str(int(float(value)))
                    except ValueError:
                        value = "0"
                
                cleaned_row[key] = value
            
            # 跳过空行
            if any(cleaned_row.values()):
                writer.writerow(cleaned_row)
```

### 练习三：数据导出

```python
import json
import csv

class DataExporter:
    """数据导出器"""
    
    def __init__(self, data):
        self.data = data
    
    def to_json(self, filename):
        """导出为 JSON"""
        with open(filename, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=2)
    
    def to_csv(self, filename):
        """导出为 CSV"""
        if not self.data:
            return
        
        # 获取所有字段
        fieldnames = set()
        for item in self.data:
            fieldnames.update(item.keys())
        fieldnames = sorted(fieldnames)
        
        with open(filename, "w", encoding="utf-8", newline="") as f:
            writer = csv.DictWriter(f, fieldnames=fieldnames)
            writer.writeheader()
            writer.writerows(self.data)
    
    def to_jsonl(self, filename):
        """导出为 JSON Lines 格式"""
        with open(filename, "w", encoding="utf-8") as f:
            for item in self.data:
                f.write(json.dumps(item, ensure_ascii=False) + "\n")

# 使用示例
data = [
    {"name": "张三", "age": 25, "city": "北京"},
    {"name": "李四", "age": 30, "city": "上海"}
]

exporter = DataExporter(data)
exporter.to_json("output.json")
exporter.to_csv("output.csv")
exporter.to_jsonl("output.jsonl")
```
