# 2.1 GPIO输入输出

GPIO（General Purpose Input/Output，通用输入输出）是微控制器与外部世界交互的基础接口。本节将详细介绍ESP32的GPIO使用方法。

## GPIO概述

### ESP32 GPIO特性

ESP32拥有丰富的GPIO资源，但并非所有引脚都可以自由使用。

**可自由使用的GPIO：**
- GPIO 0-19（部分有特殊功能）
- GPIO 21-23
- GPIO 25-27
- GPIO 32-33

**仅输入的GPIO：**
- GPIO 34、35、36（VP）、39（VN）
- 这些引脚没有输出驱动器，只能作为输入使用

**禁止使用的GPIO：**
- GPIO 6-11：连接内部Flash
- GPIO 20：USB OTG（部分芯片）

**特殊功能GPIO：**

| 引脚 | 特殊功能 | 使用注意 |
|------|----------|----------|
| GPIO 0 | Boot模式选择 | 启动时不能拉低 |
| GPIO 2 | 启动指示 | 启动时不能下拉 |
| GPIO 12 | Flash电压选择 | 启动时决定Flash电压 |
| GPIO 15 | 启动日志使能 | 启动时影响日志输出 |

### GPIO电气特性

| 参数 | 最小值 | 典型值 | 最大值 |
|------|--------|--------|--------|
| 输入低电平 | -0.3V | - | 0.25 x VDD |
| 输入高电平 | 0.75 x VDD | - | VDD+0.3V |
| 输出低电平 | - | - | 0.1 x VDD |
| 输出高电平 | 0.8 x VDD | - | - |
| 输出电流 | - | - | 40mA |
| 上拉/下拉电阻 | - | 45k欧姆 | - |

## GPIO输出

### 基本输出操作

```cpp
/*
 * GPIO基本输出示例
 * 控制LED亮灭
 */

#define LED_PIN 2

void setup() {
  Serial.begin(115200);
  
  // 配置GPIO为输出模式
  pinMode(LED_PIN, OUTPUT);
  
  // 设置初始状态
  digitalWrite(LED_PIN, LOW);
  
  Serial.println("GPIO输出示例启动");
}

void loop() {
  // 点亮LED（高电平）
  digitalWrite(LED_PIN, HIGH);
  Serial.println("LED ON (HIGH)");
  delay(1000);
  
  // 熄灭LED（低电平）
  digitalWrite(LED_PIN, LOW);
  Serial.println("LED OFF (LOW)");
  delay(1000);
}
```

### 多LED控制

```cpp
/*
 * 多LED控制示例
 * 实现流水灯效果
 */

// LED引脚数组
const int ledPins[] = {2, 4, 5, 18, 19};
const int ledCount = 5;

// 流水灯速度
const int delayMs = 200;

void setup() {
  Serial.begin(115200);
  
  // 初始化所有LED引脚
  for (int i = 0; i < ledCount; i++) {
    pinMode(ledPins[i], OUTPUT);
    digitalWrite(ledPins[i], LOW);
  }
  
  Serial.println("多LED控制示例启动");
}

void loop() {
  // 正向流水
  for (int i = 0; i < ledCount; i++) {
    setSingleLED(i);
    delay(delayMs);
  }
  
  // 反向流水
  for (int i = ledCount - 1; i >= 0; i--) {
    setSingleLED(i);
    delay(delayMs);
  }
  
  // 全亮
  setAllLEDs(HIGH);
  delay(500);
  
  // 全灭
  setAllLEDs(LOW);
  delay(500);
}

// 设置单个LED亮，其他灭
void setSingleLED(int index) {
  for (int i = 0; i < ledCount; i++) {
    digitalWrite(ledPins[i], (i == index) ? HIGH : LOW);
  }
}

// 设置所有LED状态
void setAllLEDs(int state) {
  for (int i = 0; i < ledCount; i++) {
    digitalWrite(ledPins[i], state);
  }
}
```

### 继电器控制

```cpp
/*
 * 继电器控制示例
 * 用于控制高功率设备
 */

#define RELAY_PIN 26

// 继电器状态
bool relayState = false;

void setup() {
  Serial.begin(115200);
  
  // 配置继电器引脚
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW);  // 初始关闭
  
  Serial.println("继电器控制示例");
  Serial.println("输入 'on' 打开，'off' 关闭");
}

void loop() {
  // 串口控制
  if (Serial.available() > 0) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    cmd.toLowerCase();
    
    if (cmd == "on") {
      relayState = true;
      digitalWrite(RELAY_PIN, HIGH);
      Serial.println("继电器已打开");
    } else if (cmd == "off") {
      relayState = false;
      digitalWrite(RELAY_PIN, LOW);
      Serial.println("继电器已关闭");
    } else if (cmd == "toggle") {
      relayState = !relayState;
      digitalWrite(RELAY_PIN, relayState ? HIGH : LOW);
      Serial.printf("继电器状态: %s\n", relayState ? "开" : "关");
    }
  }
}
```

/*

#define BUTTON_PIN 0  // BOOT按钮

void setup() {

  Serial.println("GPIO输入示例启动");

}

void loop() {

    Serial.println("按钮被按下");

  }
}

### 按键消抖

```cpp
/*
 * 按键消抖示例
 * 软件消抖和状态检测
 */

#define BUTTON_PIN 0

// 消抖参数
#define DEBOUNCE_DELAY 50  // 消抖延时(ms)

// 按键状态变量
int lastButtonState = HIGH;
int buttonState = HIGH;
unsigned long lastDebounceTime = 0;

// 按键事件计数
int pressCount = 0;

void setup() {
  Serial.begin(115200);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  
  Serial.println("按键消抖示例启动");
}

void loop() {
  // 读取当前状态
  int reading = digitalRead(BUTTON_PIN);
  
  // 检测状态变化
  if (reading != lastButtonState) {
    lastDebounceTime = millis();  // 重置消抖计时器
  }
  
  // 状态稳定后处理
  if ((millis() - lastDebounceTime) > DEBOUNCE_DELAY) {
    // 如果状态发生变化
    if (reading != buttonState) {
      buttonState = reading;
      
      // 检测按下事件（低电平有效）
      if (buttonState == LOW) {
        pressCount++;
        Serial.printf("按键按下 (第%d次)\n", pressCount);
      } else {
        Serial.println("按键释放");
      }
    }
  }
  
  lastButtonState = reading;
}
```

### 按键类封装

```cpp
/*
 * 按键类封装
 * 支持短按、长按、双击检测
 */

class Button {
private:
  int pin;
  int lastState;
  int currentState;
  unsigned long lastDebounceTime;
  unsigned long pressStartTime;
  
  // 配置参数
  int debounceDelay;
  int longPressTime;
  int doubleClickTime;
  
  // 状态标志
  bool pressed;
  bool longPressTriggered;
  int clickCount;
  unsigned long lastClickTime;
  
public:
  // 构造函数
  Button(int buttonPin, int mode = INPUT_PULLUP) {
    pin = buttonPin;
    pinMode(pin, mode);
    
    lastState = HIGH;
    currentState = HIGH;
    lastDebounceTime = 0;
    pressStartTime = 0;
    
    debounceDelay = 50;
    longPressTime = 1000;
    doubleClickTime = 400;
    
    pressed = false;
    longPressTriggered = false;
    clickCount = 0;
    lastClickTime = 0;
  }
  
  // 更新状态（在loop中调用）
  void update() {
    int reading = digitalRead(pin);
    
    // 消抖处理
    if (reading != lastState) {
      lastDebounceTime = millis();
    }
    
    if ((millis() - lastDebounceTime) > debounceDelay) {
      if (reading != currentState) {
        currentState = reading;
        
        if (currentState == LOW) {
          // 按键按下
          pressed = true;
          pressStartTime = millis();
          longPressTriggered = false;
        } else {
          // 按键释放
          if (pressed && !longPressTriggered) {
            // 短按事件
            handleClick();
          }
          pressed = false;
        }
      }
    }
    
    // 长按检测
    if (pressed && !longPressTriggered) {
      if ((millis() - pressStartTime) > longPressTime) {
        longPressTriggered = true;
        onLongPress();
      }
    }
    
    // 双击超时检测
    if (clickCount > 0 && (millis() - lastClickTime) > doubleClickTime) {
      if (clickCount == 1) {
        onSingleClick();
      }
      clickCount = 0;
    }
    
    lastState = reading;
  }
  
private:
  void handleClick() {
    clickCount++;
    lastClickTime = millis();
    
    if (clickCount == 2) {
      onDoubleClick();
      clickCount = 0;
    }
  }
  
  // 虚函数，由子类重写或通过函数指针设置
public:
  virtual void onSingleClick() {}
  virtual void onDoubleClick() {}
  virtual void onLongPress() {}
};

// 使用示例
Button myButton(0);

void setup() {
  Serial.begin(115200);
  Serial.println("按键类示例启动");
}

void loop() {
  myButton.update();
}

// 重写事件处理（需要继承Button类或使用回调）
```

### 使用中断的输入

```cpp
/*
 * GPIO中断示例
 * 使用中断检测按键
 */

#define BUTTON_PIN 0
#define LED_PIN 2

// 中断中使用的变量必须加volatile
volatile bool buttonPressed = false;
volatile unsigned long lastInterruptTime = 0;

// 中断服务函数（IRAM_ATTR放在RAM中）
void IRAM_ATTR buttonISR() {
  unsigned long currentTime = millis();
  
  // 消抖：两次中断间隔大于50ms
  if (currentTime - lastInterruptTime > 50) {
    buttonPressed = true;
    lastInterruptTime = currentTime;
  }
}

void setup() {
  Serial.begin(115200);
  
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  
  // 配置中断
  // 触发方式：FALLING(下降沿)、RISING(上升沿)、CHANGE(变化)、HIGH/LOW(电平)
  attachInterrupt(digitalPinToInterrupt(BUTTON_PIN), buttonISR, FALLING);
  
  Serial.println("中断示例启动");
}

void loop() {
  if (buttonPressed) {
    buttonPressed = false;
    
    // 切换LED
    static bool ledState = false;
    ledState = !ledState;
    digitalWrite(LED_PIN, ledState);
    
    Serial.printf("按键中断触发，LED: %s\n", ledState ? "开" : "关");
  }
  
  // 其他任务可以在这里执行
}
```

### 引脚配置详解

```cpp
/*
 * GPIO完整配置示例
 */

void setup() {
  Serial.begin(115200);
  
  // 方式1：简单配置
  pinMode(2, OUTPUT);
  
  // 方式2：带上拉/下拉配置
  pinMode(4, INPUT_PULLUP);   // 输入+内部上拉
  pinMode(5, INPUT_PULLDOWN); // 输入+内部下拉
  pinMode(18, INPUT);         // 浮空输入
  
  // 方式3：开漏输出
  pinMode(19, OUTPUT_OPEN_DRAIN);
  
  // 方式4：使用ESP-IDF API（更灵活）
  gpio_config_t io_conf = {
    .pin_bit_mask = (1ULL << GPIO_NUM_21),  // 引脚位掩码
    .mode = GPIO_MODE_OUTPUT,                // 模式
    .pull_up_en = GPIO_PULLUP_DISABLE,       // 上拉
    .pull_down_en = GPIO_PULLDOWN_DISABLE,   // 下拉
    .intr_type = GPIO_INTR_DISABLE           // 中断类型
  };
  gpio_config(&io_conf);
  
  Serial.println("GPIO配置完成");
}
```

## 实用示例

### 交通灯控制器

```cpp
/*
 * 交通灯控制器示例
 */

// 定义引脚
#define RED_PIN    25
#define YELLOW_PIN 26
#define GREEN_PIN  27

// 定义状态持续时间（毫秒）
#define RED_TIME     10000  // 红灯10秒
#define GREEN_TIME   8000   // 绿灯8秒
#define YELLOW_TIME  3000   // 黄灯3秒

enum TrafficLightState {
  RED,
  GREEN,
  YELLOW
};

TrafficLightState currentState = RED;
unsigned long stateStartTime = 0;

void setup() {
  Serial.begin(115200);
  
  // 配置引脚
  pinMode(RED_PIN, OUTPUT);
  pinMode(YELLOW_PIN, OUTPUT);
  pinMode(GREEN_PIN, OUTPUT);
  
  // 初始状态
  setState(RED);
  
  Serial.println("交通灯控制器启动");
}

void loop() {
  unsigned long currentTime = millis();
  unsigned long elapsed = currentTime - stateStartTime;
  
  switch (currentState) {
    case RED:
      if (elapsed >= RED_TIME) {
        setState(GREEN);
      }
      break;
      
    case GREEN:
      if (elapsed >= GREEN_TIME) {
        setState(YELLOW);
      }
      break;
      
    case YELLOW:
      if (elapsed >= YELLOW_TIME) {
        setState(RED);
      }
      break;
  }
  
  // 打印状态
  static unsigned long lastPrint = 0;
  if (currentTime - lastPrint > 1000) {
    lastPrint = currentTime;
    printState();
  }
}

void setState(TrafficLightState state) {
  currentState = state;
  stateStartTime = millis();
  
  // 关闭所有灯
  digitalWrite(RED_PIN, LOW);
  digitalWrite(YELLOW_PIN, LOW);
  digitalWrite(GREEN_PIN, LOW);
  
  // 打开对应灯
  switch (state) {
    case RED:
      digitalWrite(RED_PIN, HIGH);
      Serial.println("状态: 红灯");
      break;
    case GREEN:
      digitalWrite(GREEN_PIN, HIGH);
      Serial.println("状态: 绿灯");
      break;
    case YELLOW:
      digitalWrite(YELLOW_PIN, HIGH);
      Serial.println("状态: 黄灯");
      break;
  }
}

void printState() {
  const char* stateName;
  int remaining;
  unsigned long elapsed = millis() - stateStartTime;
  
  switch (currentState) {
    case RED:
      stateName = "红灯";
      remaining = (RED_TIME - elapsed) / 1000;
      break;
    case GREEN:
      stateName = "绿灯";
      remaining = (GREEN_TIME - elapsed) / 1000;
      break;
    case YELLOW:
      stateName = "黄灯";
      remaining = (YELLOW_TIME - elapsed) / 1000;
      break;
  }
  
  Serial.printf("[%s] 剩余: %d秒\n", stateName, remaining);
}
```

### 矩阵键盘扫描

```cpp
/*
 * 4x4矩阵键盘扫描示例
 */

// 行引脚（输出）
const int rowPins[4] = {13, 12, 14, 27};
// 列引脚（输入）
const int colPins[4] = {26, 25, 33, 32};

// 按键映射
const char keyMap[4][4] = {
  {'1', '2', '3', 'A'},
  {'4', '5', '6', 'B'},
  {'7', '8', '9', 'C'},
  {'*', '0', '#', 'D'}
};

void setup() {
  Serial.begin(115200);
  
  // 配置行引脚为输出
  for (int i = 0; i < 4; i++) {
    pinMode(rowPins[i], OUTPUT);
    digitalWrite(rowPins[i], HIGH);
  }
  
  // 配置列引脚为输入（上拉）
  for (int i = 0; i < 4; i++) {
    pinMode(colPins[i], INPUT_PULLUP);
  }
  
  Serial.println("矩阵键盘扫描启动");
}

void loop() {
  char key = scanKeypad();
  
  if (key != 0) {
    Serial.printf("按键: %c\n", key);
    delay(300);  // 防止重复触发
  }
  
  delay(10);
}

char scanKeypad() {
  for (int row = 0; row < 4; row++) {
    // 拉低当前行
    digitalWrite(rowPins[row], LOW);
    
    // 扫描列
    for (int col = 0; col < 4; col++) {
      if (digitalRead(colPins[col]) == LOW) {
        // 检测到按键按下
        digitalWrite(rowPins[row], HIGH);
        return keyMap[row][col];
      }
    }
    
    // 恢复行状态
    digitalWrite(rowPins[row], HIGH);
  }
  
  return 0;  // 无按键按下
}
```
