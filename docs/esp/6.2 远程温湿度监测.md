# 6.2 远程温湿度监测

本节将实现一个远程温湿度监测系统，支持数据采集、存储和可视化展示。

## 项目概述

### 功能需求

- 多点温湿度采集
- 数据本地存储
- MQTT数据上报
- Web数据展示
- 历史数据查询

### 硬件清单

| 组件 | 数量 | 说明 |
|------|------|------|
| ESP32开发板 | 1 | 主控制器 |
| DHT22传感器 | 2 | 温湿度检测 |
| OLED显示屏 | 1 | 本地显示 |

## 完整实现

```cpp
/*
 * 远程温湿度监测系统
 */

#include <WiFi.h>
#include <WebServer.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>
#include <DHT.h>
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <Preferences.h>

// ==================== 配置参数 ====================
#define DHT_PIN_1 4
#define DHT_PIN_2 5
#define DHT_TYPE DHT22

#define OLED_SDA 21
#define OLED_SCL 22

#define SAMPLE_INTERVAL 60000    // 采样间隔(ms)
#define REPORT_INTERVAL 300000   // 上报间隔(ms)
#define MAX_RECORDS 1440         // 最大记录数(24小时)

// WiFi配置
const char* ssid = "Your_SSID";
const char* password = "Your_Password";

// MQTT配置
const char* mqtt_server = "broker.emqx.io";
const int mqtt_port = 1883;
const char* mqtt_topic = "home/sensors";

// ==================== 全局对象 ====================
WiFiClient espClient;
PubSubClient mqtt(espClient);
WebServer server(80);
DHT dht1(DHT_PIN_1, DHT_TYPE);
DHT dht2(DHT_PIN_2, DHT_TYPE);
Adafruit_SSD1306 display(128, 64, &Wire, -1);
Preferences preferences;

// ==================== 数据结构 ====================
struct SensorData {
  float temperature1;
  float humidity1;
  float temperature2;
  float humidity2;
  unsigned long timestamp;
};

struct DataRecord {
  float temp1, humi1, temp2, humi2;
  time_t time;
};

// 数据存储
DataRecord records[MAX_RECORDS];
int recordIndex = 0;
int recordCount = 0;

// 当前数据
SensorData currentData;

// ==================== 初始化 ====================
void setup() {
  Serial.begin(115200);
  
  initHardware();
  initWiFi();
  initMQTT();
  initWebServer();
  initDisplay();
  loadRecords();
  
  Serial.println("温湿度监测系统已启动");
}

void initHardware() {
  dht1.begin();
  dht2.begin();
  
  // 初始化NTP时间
  configTime(8 * 3600, 0, "ntp.aliyun.com", "time.windows.com");
}

void initWiFi() {
  WiFi.begin(ssid, password);
  
  Serial.print("连接WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.printf("\nIP: %s\n", WiFi.localIP().toString().c_str());
}

void initMQTT() {
  mqtt.setServer(mqtt_server, mqtt_port);
}

void initWebServer() {
  server.on("/", handleRoot);
  server.on("/api/current", handleCurrent);
  server.on("/api/history", handleHistory);
  server.on("/api/stats", handleStats);
  
  server.begin();
}

void initDisplay() {
  Wire.begin(OLED_SDA, OLED_SCL);
  display.begin(SSD1306_SWITCHCAPVCC, 0x3C);
  updateDisplay();
}

// ==================== 数据采集 ====================
void readSensors() {
  currentData.temperature1 = dht1.readTemperature();
  currentData.humidity1 = dht1.readHumidity();
  currentData.temperature2 = dht2.readTemperature();
  currentData.humidity2 = dht2.readHumidity();
  currentData.timestamp = millis();
  
  // 保存记录
  saveRecord();
  
  // 更新显示
  updateDisplay();
}

void saveRecord() {
  time_t now;
  time(&now);
  
  records[recordIndex].temp1 = currentData.temperature1;
  records[recordIndex].humi1 = currentData.humidity1;
  records[recordIndex].temp2 = currentData.temperature2;
  records[recordIndex].humi2 = currentData.humidity2;
  records[recordIndex].time = now;
  
  recordIndex = (recordIndex + 1) % MAX_RECORDS;
  if (recordCount < MAX_RECORDS) {
    recordCount++;
  }
  
  // 定期保存到Flash
  static int saveCounter = 0;
  if (++saveCounter >= 10) {
    saveCounter = 0;
    persistRecords();
  }
}

void loadRecords() {
  preferences.begin("sensor-data", true);
  recordCount = preferences.getInt("count", 0);
  recordIndex = recordCount % MAX_RECORDS;
  preferences.end();
}

void persistRecords() {
  preferences.begin("sensor-data", false);
  preferences.putInt("count", recordCount);
  preferences.end();
}

// ==================== MQTT上报 ====================
void reportData() {
  if (!mqtt.connected()) {
    reconnectMQTT();
  }
  
  if (mqtt.connected()) {
    StaticJsonDocument<256> doc;
    
    doc["device"] = "ESP32-Monitor";
    doc["timestamp"] = time(nullptr);
    
    JsonObject sensor1 = doc.createNestedObject("sensor1");
    sensor1["temperature"] = currentData.temperature1;
    sensor1["humidity"] = currentData.humidity1;
    
    JsonObject sensor2 = doc.createNestedObject("sensor2");
    sensor2["temperature"] = currentData.temperature2;
    sensor2["humidity"] = currentData.humidity2;
    
    String payload;
    serializeJson(doc, payload);
    
    mqtt.publish(mqtt_topic, payload.c_str());
    Serial.println("数据已上报: " + payload);
  }
}

void reconnectMQTT() {
  String clientId = "ESP32-Monitor-" + String(random(0xffff), HEX);
  
  if (mqtt.connect(clientId.c_str())) {
    Serial.println("MQTT已连接");
  }
}

// ==================== Web服务 ====================
void handleRoot() {
  String html = generateWebPage();
  server.send(200, "text/html", html);
}

void handleCurrent() {
  StaticJsonDocument<256> doc;
  
  doc["temperature1"] = currentData.temperature1;
  doc["humidity1"] = currentData.humidity1;
  doc["temperature2"] = currentData.temperature2;
  doc["humidity2"] = currentData.humidity2;
  doc["timestamp"] = time(nullptr);
  doc["uptime"] = millis() / 1000;
  
  String response;
  serializeJson(doc, response);
  
  server.send(200, "application/json", response);
}

void handleHistory() {
  int hours = server.hasArg("hours") ? server.arg("hours").toInt() : 24;
  int count = min(hours * 60, recordCount);
  
  StaticJsonDocument<4096> doc;
  JsonArray array = doc.to<JsonArray>();
  
  int startIdx = (recordIndex - count + MAX_RECORDS) % MAX_RECORDS;
  
  for (int i = 0; i < count; i++) {
    int idx = (startIdx + i) % MAX_RECORDS;
    JsonObject obj = array.createNestedObject();
    obj["time"] = records[idx].time;
    obj["temp1"] = records[idx].temp1;
    obj["humi1"] = records[idx].humi1;
    obj["temp2"] = records[idx].temp2;
    obj["humi2"] = records[idx].humi2;
  }
  
  String response;
  serializeJson(doc, response);
  
  server.send(200, "application/json", response);
}

void handleStats() {
  float minTemp1 = 100, maxTemp1 = -100, avgTemp1 = 0;
  float minTemp2 = 100, maxTemp2 = -100, avgTemp2 = 0;
  
  for (int i = 0; i < recordCount; i++) {
    if (records[i].temp1 < minTemp1) minTemp1 = records[i].temp1;
    if (records[i].temp1 > maxTemp1) maxTemp1 = records[i].temp1;
    avgTemp1 += records[i].temp1;
    
    if (records[i].temp2 < minTemp2) minTemp2 = records[i].temp2;
    if (records[i].temp2 > maxTemp2) maxTemp2 = records[i].temp2;
    avgTemp2 += records[i].temp2;
  }
  
  if (recordCount > 0) {
    avgTemp1 /= recordCount;
    avgTemp2 /= recordCount;
  }
  
  StaticJsonDocument<256> doc;
  
  JsonObject s1 = doc.createNestedObject("sensor1");
  s1["min"] = minTemp1;
  s1["max"] = maxTemp1;
  s1["avg"] = avgTemp1;
  
  JsonObject s2 = doc.createNestedObject("sensor2");
  s2["min"] = minTemp2;
  s2["max"] = maxTemp2;
  s2["avg"] = avgTemp2;
  
  doc["recordCount"] = recordCount;
  
  String response;
  serializeJson(doc, response);
  
  server.send(200, "application/json", response);
}

// ==================== 显示 ====================
void updateDisplay() {
  display.clearDisplay();
  
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  
  display.setCursor(0, 0);
  display.println("Temp Monitor");
  
  display.drawLine(0, 10, 127, 10, SSD1306_WHITE);
  
  display.setCursor(0, 14);
  display.printf("S1: %.1fC %.1f%%", currentData.temperature1, currentData.humidity1);
  
  display.setCursor(0, 28);
  display.printf("S2: %.1fC %.1f%%", currentData.temperature2, currentData.humidity2);
  
  display.setCursor(0, 42);
  display.printf("Records: %d", recordCount);
  
  display.setCursor(0, 56);
  display.printf("IP: %s", WiFi.localIP().toString().c_str());
  
  display.display();
}

// ==================== 主循环 ====================
void loop() {
  server.handleClient();
  mqtt.loop();
  
  // 定期采集数据
  static unsigned long lastSample = 0;
  if (millis() - lastSample >= SAMPLE_INTERVAL) {
    lastSample = millis();
    readSensors();
  }
  
  // 定期上报数据
  static unsigned long lastReport = 0;
  if (millis() - lastReport >= REPORT_INTERVAL) {
    lastReport = millis();
    reportData();
  }
  
  delay(10);
}

// ==================== Web页面 ====================
String generateWebPage() {
  return R"rawliteral(
<!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>温湿度监测系统</title>
  <script src="https://cdn.jsdelivr.net/npm/chart.js"></script>
  <style>
    body { font-family: Arial; max-width: 800px; margin: 0 auto; padding: 20px; }
    .card { background: #f5f5f5; padding: 20px; margin: 10px 0; border-radius: 10px; }
    .sensor { display: flex; justify-content: space-around; text-align: center; }
    .value { font-size: 32px; font-weight: bold; }
    .label { color: #666; }
    canvas { max-height: 300px; }
  </style>
</head>
<body>
  <h1>温湿度监测系统</h1>
  
  <div class="card">
    <h2>实时数据</h2>
    <div class="sensor">
      <div>
        <div class="value" id="temp1">--</div>
        <div class="label">传感器1 温度(C)</div>
      </div>
      <div>
        <div class="value" id="humi1">--</div>
        <div class="label">传感器1 湿度(%)</div>
      </div>
    </div>
    <div class="sensor" style="margin-top:20px">
      <div>
        <div class="value" id="temp2">--</div>
        <div class="label">传感器2 温度(C)</div>
      </div>
      <div>
        <div class="value" id="humi2">--</div>
        <div class="label">传感器2 湿度(%)</div>
      </div>
    </div>
  </div>
  
  <div class="card">
    <h2>历史趋势</h2>
    <canvas id="chart"></canvas>
  </div>
  
  <script>
    var chart = new Chart(document.getElementById('chart'), {
      type: 'line',
      data: {
        labels: [],
        datasets: [
          { label: '温度1', data: [], borderColor: 'red', fill: false },
          { label: '温度2', data: [], borderColor: 'blue', fill: false }
        ]
      },
      options: {
        responsive: true,
        scales: { x: { display: false } }
      }
    });
    
    function updateData() {
      fetch('/api/current')
        .then(r => r.json())
        .then(d => {
          document.getElementById('temp1').textContent = d.temperature1.toFixed(1);
          document.getElementById('humi1').textContent = d.humidity1.toFixed(1);
          document.getElementById('temp2').textContent = d.temperature2.toFixed(1);
          document.getElementById('humi2').textContent = d.humidity2.toFixed(1);
        });
      
      fetch('/api/history?hours=6')
        .then(r => r.json())
        .then(d => {
          chart.data.labels = d.map(x => new Date(x.time * 1000).toLocaleTimeString());
          chart.data.datasets[0].data = d.map(x => x.temp1);
          chart.data.datasets[1].data = d.map(x => x.temp2);
          chart.update();
        });
    }
    
    updateData();
    setInterval(updateData, 60000);
  </script>
</body>
</html>
)rawliteral";
}
```

## 小结

本节实现了一个远程温湿度监测系统，支持多点采集、数据存储、MQTT上报和Web可视化。这个系统可以扩展为更完整的环境监测平台。
