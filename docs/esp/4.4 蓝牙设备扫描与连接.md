# 4.4 蓝牙设备扫描与连接

本节将介绍ESP32作为BLE中心设备，扫描和连接其他BLE设备的方法。

## BLE扫描

### 基本扫描

```cpp
/*
 * BLE扫描示例
 * 扫描周围的BLE设备
 */

#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEScan.h>

int scanTime = 5;  // 扫描时间（秒）
BLEScan* pBLEScan;

// 扫描回调类
class MyAdvertisedDeviceCallbacks: public BLEAdvertisedDeviceCallbacks {
  void onResult(BLEAdvertisedDevice advertisedDevice) {
    Serial.printf("发现设备: %s\n", advertisedDevice.toString().c_str());
  }
};

void setup() {
  Serial.begin(115200);
  
  // 初始化BLE
  BLEDevice::init("");
  
  // 创建扫描对象
  pBLEScan = BLEDevice::getScan();
  pBLEScan->setAdvertisedDeviceCallbacks(new MyAdvertisedDeviceCallbacks());
  pBLEScan->setActiveScan(true);  // 主动扫描
  pBLEScan->setInterval(100);
  pBLEScan->setWindow(99);
  
  Serial.println("BLE扫描示例启动");
}

void loop() {
  Serial.println("开始扫描...");
  
  // 开始扫描
  BLEScanResults foundDevices = pBLEScan->start(scanTime, false);
  
  Serial.printf("扫描完成，发现 %d 个设备\n", foundDevices.getCount());
  
  // 打印设备列表
  printScanResults(foundDevices);
  
  // 清除结果
  pBLEScan->clearResults();
  
  delay(5000);
}

void printScanResults(BLEScanResults results) {
  Serial.println("\n=== 扫描结果 ===");
  
  for (int i = 0; i < results.getCount(); i++) {
    BLEAdvertisedDevice device = results.getDevice(i);
    
    Serial.printf("%d: %s\n", i + 1, device.getName().c_str());
    Serial.printf("   地址: %s\n", device.getAddress().toString().c_str());
    Serial.printf("   RSSI: %d dBm\n", device.getRSSI());
    
    if (device.haveServiceUUID()) {
      Serial.printf("   服务UUID: %s\n", device.getServiceUUID().toString().c_str());
    }
    
    if (device.haveManufacturerData()) {
      String mfgData = device.getManufacturerData();
      Serial.printf("   厂商数据: %s\n", mfgData.c_str());
    }
    
    Serial.println();
  }
}
```

### 过滤扫描

```cpp
/*
 * 过滤扫描示例
 * 只扫描特定设备
 */

#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEScan.h>

// 目标设备UUID或名称
#define TARGET_SERVICE_UUID "4fafc201-1fb5-459e-8fcc-c5c9c331914b"
#define TARGET_DEVICE_NAME "ESP32-BLE"

BLEScan* pBLEScan;
bool deviceFound = false;
BLEAdvertisedDevice targetDevice;

// 过滤回调类
class MyCallbacks: public BLEAdvertisedDeviceCallbacks {
  void onResult(BLEAdvertisedDevice advertisedDevice) {
    // 检查设备名称
    if (advertisedDevice.haveName() && 
        advertisedDevice.getName() == TARGET_DEVICE_NAME) {
      Serial.printf("找到目标设备: %s\n", advertisedDevice.getName().c_str());
      targetDevice = advertisedDevice;
      deviceFound = true;
      pBLEScan->stop();  // 停止扫描
      return;
    }
    
    // 检查服务UUID
    if (advertisedDevice.haveServiceUUID() &&
        advertisedDevice.isAdvertisingService(BLEUUID(TARGET_SERVICE_UUID))) {
      Serial.printf("找到目标服务设备: %s\n", 
                    advertisedDevice.getAddress().toString().c_str());
      targetDevice = advertisedDevice;
      deviceFound = true;
      pBLEScan->stop();
    }
  }
};

void setup() {
  Serial.begin(115200);
  
  BLEDevice::init("");
  
  pBLEScan = BLEDevice::getScan();
  pBLEScan->setAdvertisedDeviceCallbacks(new MyCallbacks());
  pBLEScan->setActiveScan(true);
  
  Serial.println("开始搜索目标设备...");
}

void loop() {
  if (!deviceFound) {
    BLEScanResults results = pBLEScan->start(5, false);
    pBLEScan->clearResults();
  } else {
    Serial.println("目标设备已找到！");
    Serial.printf("地址: %s\n", targetDevice.getAddress().toString().c_str());
    
    // 可以在这里连接设备
    // connectToDevice(targetDevice);
    
    deviceFound = false;  // 重置以继续扫描
    delay(5000);
  }
}
```

## BLE连接

### 连接外围设备

```cpp
/*
 * BLE客户端连接示例
 */

#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEClient.h>
#include <BLEScan.h>

#define TARGET_SERVICE_UUID "4fafc201-1fb5-459e-8fcc-c5c9c331914b"
#define TARGET_CHAR_UUID    "beb5483e-36e1-4688-b7f5-ea07361b26a8"

BLEClient* pClient;
BLEScan* pBLEScan;
bool connected = false;
bool doConnect = false;
BLEAdvertisedDevice* myDevice;

// 连接回调
class MyClientCallback : public BLEClientCallbacks {
  void onConnect(BLEClient* pclient) {
    Serial.println("已连接到服务器");
  }
  
  void onDisconnect(BLEClient* pclient) {
    connected = false;
    Serial.println("已断开连接");
  }
};

// 扫描回调
class MyAdvertisedDeviceCallbacks: public BLEAdvertisedDeviceCallbacks {
  void onResult(BLEAdvertisedDevice advertisedDevice) {
    if (advertisedDevice.haveServiceUUID() &&
        advertisedDevice.isAdvertisingService(BLEUUID(TARGET_SERVICE_UUID))) {
      Serial.printf("找到目标设备: %s\n", 
                    advertisedDevice.getAddress().toString().c_str());
      
      BLEDevice::getScan()->stop();
      myDevice = new BLEAdvertisedDevice(advertisedDevice);
      doConnect = true;
    }
  }
};

void setup() {
  Serial.begin(115200);
  
  BLEDevice::init("ESP32-Client");
  
  // 创建客户端
  pClient = BLEDevice::createClient();
  pClient->setClientCallbacks(new MyClientCallback());
  
  // 配置扫描
  pBLEScan = BLEDevice::getScan();
  pBLEScan->setAdvertisedDeviceCallbacks(new MyAdvertisedDeviceCallbacks());
  pBLEScan->setActiveScan(true);
  
  Serial.println("开始扫描...");
  pBLEScan->start(5, false);
}

void loop() {
  if (doConnect) {
    doConnect = false;
    
    if (connectToServer()) {
      Serial.println("连接成功！");
      connected = true;
    } else {
      Serial.println("连接失败！");
    }
  }
  
  if (connected) {
    // 读取特征值
    readCharacteristic();
    delay(5000);
  }
  
  delay(10);
}

bool connectToServer() {
  Serial.printf("连接到 %s...\n", myDevice->getAddress().toString().c_str());
  
  if (!pClient->connect(myDevice)) {
    Serial.println("连接失败");
    return false;
  }
  
  // 获取服务
  BLERemoteService* pRemoteService = pClient->getService(BLEUUID(TARGET_SERVICE_UUID));
  if (pRemoteService == nullptr) {
    Serial.println("找不到服务");
    pClient->disconnect();
    return false;
  }
  
  // 获取特征值
  BLERemoteCharacteristic* pRemoteCharacteristic = 
    pRemoteService->getCharacteristic(BLEUUID(TARGET_CHAR_UUID));
  if (pRemoteCharacteristic == nullptr) {
    Serial.println("找不到特征值");
    pClient->disconnect();
    return false;
  }
  
  // 读取特征值
  if (pRemoteCharacteristic->canRead()) {
    String value = pRemoteCharacteristic->readValue().c_str();
    Serial.printf("特征值: %s\n", value.c_str());
  }
  
  return true;
}

void readCharacteristic() {
  BLERemoteService* pService = pClient->getService(TARGET_SERVICE_UUID);
  if (pService) {
    BLERemoteCharacteristic* pChar = pService->getCharacteristic(TARGET_CHAR_UUID);
    if (pChar && pChar->canRead()) {
      String value = pChar->readValue().c_str();
      Serial.printf("读取: %s\n", value.c_str());
    }
  }
}
```

### 订阅通知

```cpp
/*
 * BLE通知订阅示例
 */

#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEClient.h>

#define SERVICE_UUID        "4fafc201-1fb5-459e-8fcc-c5c9c331914b"
#define NOTIFY_CHAR_UUID    "beb5483e-36e1-4688-b7f5-ea07361b26a8"

BLEClient* pClient;
BLERemoteCharacteristic* pNotifyChar;
bool connected = false;

// 通知回调
void notifyCallback(BLERemoteCharacteristic* pBLERemoteCharacteristic, 
                    uint8_t* pData, size_t length, bool isNotify) {
  Serial.print("收到通知: ");
  for (int i = 0; i < length; i++) {
    Serial.printf("%02X ", pData[i]);
  }
  Serial.println();
  
  // 如果是字符串数据
  String strData = String((char*)pData, length);
  Serial.printf("字符串: %s\n", strData.c_str());
}

// 连接回调
class MyClientCallback : public BLEClientCallbacks {
  void onConnect(BLEClient* pclient) {
    connected = true;
    Serial.println("已连接");
  }
  
  void onDisconnect(BLEClient* pclient) {
    connected = false;
    Serial.println("已断开");
  }
};

void setup() {
  Serial.begin(115200);
  
  BLEDevice::init("ESP32-Notify-Client");
  
  pClient = BLEDevice::createClient();
  pClient->setClientCallbacks(new MyClientCallback());
  
  // 连接到已知设备
  BLEAddress targetAddress("24:0a:c4:xx:xx:xx");  // 替换为目标设备地址
  
  Serial.println("连接设备...");
  if (pClient->connect(targetAddress)) {
    Serial.println("已连接");
    
    // 获取服务和特征值
    BLERemoteService* pService = pClient->getService(SERVICE_UUID);
    if (pService) {
      pNotifyChar = pService->getCharacteristic(NOTIFY_CHAR_UUID);
      
      if (pNotifyChar && pNotifyChar->canNotify()) {
        // 订阅通知
        pNotifyChar->registerForNotify(notifyCallback);
        Serial.println("已订阅通知");
      }
    }
  }
}

void loop() {
  if (!connected) {
    // 尝试重连
    static unsigned long lastReconnect = 0;
    if (millis() - lastReconnect > 5000) {
      lastReconnect = millis();
      // reconnect();
    }
  }
  
  delay(10);
}
```

## BLE客户端类封装

```cpp
/*
 * BLE客户端类封装
 */

#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEClient.h>
#include <BLEScan.h>

class BLEClientManager {
private:
  BLEClient* pClient;
  BLEScan* pScan;
  bool connected;
  String targetName;
  BLEAddress* targetAddress;
  
  // 回调函数指针
  void (*onConnectCallback)();
  void (*onDisconnectCallback)();
  void (*onNotifyCallback)(String uuid, uint8_t* data, size_t len);
  
public:
  BLEClientManager() {
    connected = false;
    targetAddress = nullptr;
    onConnectCallback = nullptr;
    onDisconnectCallback = nullptr;
    onNotifyCallback = nullptr;
  }
  
  void init(String deviceName) {
    BLEDevice::init(deviceName);
    pClient = BLEDevice::createClient();
    pScan = BLEDevice::getScan();
    
    // 设置回调
    pClient->setClientCallbacks(new ClientCallbacks(this));
  }
  
  void setTargetName(String name) {
    targetName = name;
  }
  
  void setTargetAddress(String address) {
    targetAddress = new BLEAddress(address.c_str());
  }
  
  void onConnect(void (*callback)()) {
    onConnectCallback = callback;
  }
  
  void onDisconnect(void (*callback)()) {
    onDisconnectCallback = callback;
  }
  
  void onNotify(void (*callback)(String, uint8_t*, size_t)) {
    onNotifyCallback = callback;
  }
  
  // 扫描并连接
  bool scanAndConnect(int timeout = 10) {
    pScan->setActiveScan(true);
    BLEScanResults results = pScan->start(timeout, false);
    
    for (int i = 0; i < results.getCount(); i++) {
      BLEAdvertisedDevice device = results.getDevice(i);
      
      if (targetName.length() > 0 && device.haveName()) {
        if (device.getName() == targetName) {
          return connect(device.getAddress());
        }
      }
    }
    
    return false;
  }
  
  // 连接设备
  bool connect(BLEAddress address) {
    if (pClient->connect(address)) {
      connected = true;
      return true;
    }
    return false;
  }
  
  // 断开连接
  void disconnect() {
    if (connected) {
      pClient->disconnect();
      connected = false;
    }
  }
  
  // 检查连接状态
  bool isConnected() {
    return connected;
  }
  
  // 获取服务
  BLERemoteService* getService(String uuid) {
    if (!connected) return nullptr;
    return pClient->getService(BLEUUID(uuid.c_str()));
  }
  
  // 读取特征值
  String readCharacteristic(String serviceUuid, String charUuid) {
    if (!connected) return "";
    
    BLERemoteService* pService = getService(serviceUuid);
    if (!pService) return "";
    
    BLERemoteCharacteristic* pChar = pService->getCharacteristic(BLEUUID(charUuid.c_str()));
    if (!pChar || !pChar->canRead()) return "";
    
    return pChar->readValue().c_str();
  }
  
  // 写入特征值
  bool writeCharacteristic(String serviceUuid, String charUuid, String data) {
    if (!connected) return false;
    
    BLERemoteService* pService = getService(serviceUuid);
    if (!pService) return false;
    
    BLERemoteCharacteristic* pChar = pService->getCharacteristic(BLEUUID(charUuid.c_str()));
    if (!pChar || !pChar->canWrite()) return false;
    
    pChar->writeValue(data);
    return true;
  }
  
  // 订阅通知
  bool subscribeNotify(String serviceUuid, String charUuid) {
    if (!connected) return false;
    
    BLERemoteService* pService = getService(serviceUuid);
    if (!pService) return false;
    
    BLERemoteCharacteristic* pChar = pService->getCharacteristic(BLEUUID(charUuid.c_str()));
    if (!pChar || !pChar->canNotify()) return false;
    
    pChar->registerForNotify([this](BLERemoteCharacteristic* c, uint8_t* d, size_t l, bool n) {
      if (onNotifyCallback) {
        onNotifyCallback(c->getUUID().toString(), d, l);
      }
    });
    
    return true;
  }
  
  // 内部回调类
  class ClientCallbacks : public BLEClientCallbacks {
    BLEClientManager* manager;
  public:
    ClientCallbacks(BLEClientManager* m) : manager(m) {}
    
    void onConnect(BLEClient* pclient) {
      manager->connected = true;
      if (manager->onConnectCallback) {
        manager->onConnectCallback();
      }
    }
    
    void onDisconnect(BLEClient* pclient) {
      manager->connected = false;
      if (manager->onDisconnectCallback) {
        manager->onDisconnectCallback();
      }
    }
  };
  
  friend class ClientCallbacks;
};

// 使用示例
BLEClientManager bleClient;

void setup() {
  Serial.begin(115200);
  
  bleClient.init("ESP32-Client");
  bleClient.setTargetName("ESP32-BLE");
  
  bleClient.onConnect([]() {
    Serial.println("已连接到设备");
  });
  
  bleClient.onDisconnect([]() {
    Serial.println("已断开连接");
  });
  
  bleClient.onNotify([](String uuid, uint8_t* data, size_t len) {
    Serial.printf("通知 [%s]: %s\n", uuid.c_str(), String((char*)data, len).c_str());
  });
  
  if (bleClient.scanAndConnect(10)) {
    Serial.println("连接成功");
  }
}

void loop() {
  delay(10);
}
```
