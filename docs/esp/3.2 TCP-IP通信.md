# 3.2 TCP/IP通信

TCP/IP是互联网通信的基础协议，本节将详细介绍ESP32的TCP/IP编程方法。

## TCP/IP基础

### 协议栈层次

```
+-------------------+
|    应用层         |  HTTP, MQTT, WebSocket
+-------------------+
|    传输层         |  TCP, UDP
+-------------------+
|    网络层         |  IP
+-------------------+
|    链路层         |  WiFi, Ethernet
+-------------------+
```

### TCP vs UDP

| 特性 | TCP | UDP |
|------|-----|-----|
| 连接 | 面向连接 | 无连接 |
| 可靠性 | 可靠传输 | 尽力传输 |
| 顺序 | 保证顺序 | 不保证顺序 |
| 速度 | 较慢 | 较快 |
| 应用 | 文件传输、网页 | 视频流、DNS |

## TCP客户端

### 基本TCP客户端

```cpp
/*
 * TCP客户端基本示例
 * 连接服务器并发送接收数据
 */

#include <WiFi.h>

const char* ssid = "Your_SSID";
const char* password = "Your_Password";

// 服务器配置
const char* serverHost = "192.168.1.100";
const int serverPort = 8080;

// TCP客户端
WiFiClient client;

void setup() {
  Serial.begin(115200);
  
  // 连接WiFi
  Serial.println("连接WiFi...");
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi已连接");
  
  // 连接TCP服务器
  connectToServer();
}

void loop() {
  // 检查连接状态
  if (client.connected()) {
    // 发送心跳
    static unsigned long lastHeartbeat = 0;
    if (millis() - lastHeartbeat > 10000) {
      lastHeartbeat = millis();
      client.println("HEARTBEAT");
      Serial.println("发送心跳");
    }
    
    // 接收数据
    while (client.available()) {
      String line = client.readStringUntil('\n');
      Serial.printf("收到: %s\n", line.c_str());
      
      // 处理服务器响应
      handleServerMessage(line);
    }
  } else {
    // 断线重连
    static unsigned long lastReconnect = 0;
    if (millis() - lastReconnect > 5000) {
      lastReconnect = millis();
      Serial.println("尝试重连服务器...");
      connectToServer();
    }
  }
  
  // 处理串口输入
  handleSerialInput();
  
  delay(10);
}

// 连接服务器
void connectToServer() {
  Serial.printf("连接服务器 %s:%d...\n", serverHost, serverPort);
  
  if (client.connect(serverHost, serverPort)) {
    Serial.println("服务器连接成功！");
    client.println("HELLO FROM ESP32");
  } else {
    Serial.println("服务器连接失败！");
  }
}

// 处理服务器消息
void handleServerMessage(String message) {
  message.trim();
  
  if (message == "PING") {
    client.println("PONG");
    Serial.println("响应PONG");
  } else if (message.startsWith("ECHO:")) {
    String echo = message.substring(5);
    client.println("ECHO:" + echo);
    Serial.printf("回显: %s\n", echo.c_str());
  }
}

// 处理串口输入
void handleSerialInput() {
  if (Serial.available() > 0) {
    String input = Serial.readStringUntil('\n');
    input.trim();
    
    if (input.length() > 0 && client.connected()) {
      client.println(input);
      Serial.printf("发送: %s\n", input.c_str());
    }
  }
}
```

### TCP客户端类封装

```cpp
/*
 * TCP客户端类封装
 * 提供更易用的接口
 */

#include <WiFi.h>

class TCPClient {
private:
  WiFiClient client;
  String host;
  int port;
  bool autoReconnect;
  unsigned long reconnectInterval;
  unsigned long lastReconnectAttempt;
  
  // 回调函数
  void (*onConnectCallback)();
  void (*onDisconnectCallback)();
  void (*onMessageCallback)(String message);
  
public:
  TCPClient() {
    autoReconnect = true;
    reconnectInterval = 5000;
    lastReconnectAttempt = 0;
    onConnectCallback = NULL;
    onDisconnectCallback = NULL;
    onMessageCallback = NULL;
  }
  
  // 设置服务器
  void setServer(String serverHost, int serverPort) {
    host = serverHost;
    port = serverPort;
  }
  
  // 设置自动重连
  void setAutoReconnect(bool enable, unsigned long interval = 5000) {
    autoReconnect = enable;
    reconnectInterval = interval;
  }
  
  // 设置回调
  void onConnect(void (*callback)()) {
    onConnectCallback = callback;
  }
  
  void onDisconnect(void (*callback)()) {
    onDisconnectCallback = callback;
  }
  
  void onMessage(void (*callback)(String)) {
    onMessageCallback = callback;
  }
  
  // 连接服务器
  bool connect() {
    if (client.connected()) {
      return true;
    }
    
    Serial.printf("连接 %s:%d...\n", host.c_str(), port);
    
    if (client.connect(host.c_str(), port)) {
      Serial.println("连接成功");
      if (onConnectCallback) {
        onConnectCallback();
      }
      return true;
    } else {
      Serial.println("连接失败");
      return false;
    }
  }
  
  // 断开连接
  void disconnect() {
    if (client.connected()) {
      client.stop();
      if (onDisconnectCallback) {
        onDisconnectCallback();
      }
    }
  }
  
  // 发送数据
  bool send(String data) {
    if (!client.connected()) {
      return false;
    }
    
    client.println(data);
    return true;
  }
  
  bool send(const char* data) {
    return send(String(data));
  }
  
  // 发送原始数据
  bool sendRaw(uint8_t* data, size_t len) {
    if (!client.connected()) {
      return false;
    }
    
    client.write(data, len);
    return true;
  }
  
  // 更新（在loop中调用）
  void update() {
    // 检查连接状态
    if (!client.connected()) {
      // 触发断开回调
      static bool wasConnected = false;
      if (wasConnected && onDisconnectCallback) {
        onDisconnectCallback();
      }
      wasConnected = false;
      
      // 自动重连
      if (autoReconnect && millis() - lastReconnectAttempt > reconnectInterval) {
        lastReconnectAttempt = millis();
        connect();
      }
    } else {
      // 接收数据
      while (client.available()) {
        String line = client.readStringUntil('\n');
        line.trim();
        
        if (onMessageCallback) {
          onMessageCallback(line);
        }
      }
    }
  }
  
  // 检查连接状态
  bool isConnected() {
    return client.connected();
  }
  
  // 获取客户端对象
  WiFiClient& getClient() {
    return client;
  }
};

// 使用示例
TCPClient tcpClient;

void setup() {
  Serial.begin(115200);
  
  // 连接WiFi
  WiFi.begin("Your_SSID", "Your_Password");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  Serial.println("WiFi已连接");
  
  // 配置TCP客户端
  tcpClient.setServer("192.168.1.100", 8080);
  tcpClient.setAutoReconnect(true, 5000);
  
  // 设置回调
  tcpClient.onConnect([]() {
    Serial.println("TCP已连接");
    tcpClient.send("HELLO");
  });
  
  tcpClient.onDisconnect([]() {
    Serial.println("TCP已断开");
  });
  
  tcpClient.onMessage([](String msg) {
    Serial.printf("收到消息: %s\n", msg.c_str());
  });
  
  // 连接服务器
  tcpClient.connect();
}

void loop() {
  tcpClient.update();
  delay(10);
}
```

## TCP服务器

### 基本TCP服务器

```cpp
/*
 * TCP服务器基本示例
 * 监听端口并处理客户端连接
 */

#include <WiFi.h>

const char* ssid = "Your_SSID";
const char* password = "Your_Password";

// 服务器配置
#define SERVER_PORT 8080
#define MAX_CLIENTS 4

// TCP服务器
WiFiServer server(SERVER_PORT);

// 客户端数组
WiFiClient clients[MAX_CLIENTS];

void setup() {
  Serial.begin(115200);
  
  // 连接WiFi
  Serial.println("连接WiFi...");
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi已连接");
  Serial.printf("IP地址: %s\n", WiFi.localIP().toString().c_str());
  
  // 启动服务器
  server.begin(SERVER_PORT);
  Serial.printf("TCP服务器已启动，端口: %d\n", SERVER_PORT);
}

void loop() {
  // 接受新连接
  handleNewConnections();
  
  // 处理客户端数据
  handleClientData();
  
  delay(10);
}

// 处理新连接
void handleNewConnections() {
  if (server.hasClient()) {
    // 查找空闲槽位
    int freeSlot = -1;
    for (int i = 0; i < MAX_CLIENTS; i++) {
      if (!clients[i] || !clients[i].connected()) {
        if (freeSlot == -1) {
          freeSlot = i;
        }
      }
    }
    
    if (freeSlot != -1) {
      // 接受新客户端
      clients[freeSlot] = server.available();
      Serial.printf("新客户端连接 #%d, IP: %s\n",
                    freeSlot,
                    clients[freeSlot].remoteIP().toString().c_str());
      
      // 发送欢迎消息
      clients[freeSlot].println("Welcome to ESP32 TCP Server!");
    } else {
      // 无空闲槽位，拒绝连接
      WiFiClient rejected = server.available();
      rejected.println("Server busy, try again later");
      rejected.stop();
      Serial.println("拒绝新连接（服务器忙）");
    }
  }
}

// 处理客户端数据
void handleClientData() {
  for (int i = 0; i < MAX_CLIENTS; i++) {
    if (clients[i] && clients[i].connected()) {
      if (clients[i].available()) {
        // 读取数据
        String line = clients[i].readStringUntil('\n');
        line.trim();
        
        Serial.printf("客户端 #%d: %s\n", i, line.c_str());
        
        // 处理命令
        String response = processCommand(line);
        
        // 发送响应
        clients[i].println(response);
      }
    } else if (clients[i]) {
      // 清理断开的客户端
      Serial.printf("客户端 #%d 断开连接\n", i);
      clients[i].stop();
    }
  }
}

// 处理命令
String processCommand(String cmd) {
  cmd.toUpperCase();
  
  if (cmd == "HELLO") {
    return "HELLO! Welcome to ESP32!";
  } else if (cmd == "TIME") {
    return "TIME: " + String(millis() / 1000) + " seconds since boot";
  } else if (cmd == "STATUS") {
    int connectedCount = 0;
    for (int i = 0; i < MAX_CLIENTS; i++) {
      if (clients[i] && clients[i].connected()) {
        connectedCount++;
      }
    }
    return "STATUS: " + String(connectedCount) + " clients connected";
  } else if (cmd == "HELP") {
    return "Commands: HELLO, TIME, STATUS, HELP, ECHO <text>";
  } else if (cmd.startsWith("ECHO ")) {
    return "ECHO: " + cmd.substring(5);
  } else {
    return "Unknown command. Type HELP for help.";
  }
}
```

### 多客户端广播

```cpp
/*
 * TCP服务器广播示例
 * 向所有连接的客户端广播消息
 */

#include <WiFi.h>

#define SERVER_PORT 8080
#define MAX_CLIENTS 8

WiFiServer server(SERVER_PORT);
WiFiClient clients[MAX_CLIENTS];

void setup() {
  Serial.begin(115200);
  
  WiFi.begin("Your_SSID", "Your_Password");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  
  Serial.printf("IP: %s\n", WiFi.localIP().toString().c_str());
  
  server.begin();
  Serial.println("TCP服务器已启动");
}

void loop() {
  // 接受新连接
  if (server.hasClient()) {
    for (int i = 0; i < MAX_CLIENTS; i++) {
      if (!clients[i] || !clients[i].connected()) {
        clients[i] = server.available();
        Serial.printf("客户端 %d 连接: %s\n",
                      i, clients[i].remoteIP().toString().c_str());
        
        // 广播新客户端加入
        broadcastMessage("Client " + String(i) + " joined");
        break;
      }
    }
  }
  
  // 处理客户端消息
  for (int i = 0; i < MAX_CLIENTS; i++) {
    if (clients[i] && clients[i].connected() && clients[i].available()) {
      String msg = clients[i].readStringUntil('\n');
      msg.trim();
      
      // 广播给所有客户端
      broadcastMessage("Client " + String(i) + ": " + msg);
    }
  }
  
  // 定期广播
  static unsigned long lastBroadcast = 0;
  if (millis() - lastBroadcast > 30000) {
    lastBroadcast = millis();
    broadcastMessage("Server heartbeat: " + String(millis() / 1000) + "s");
  }
  
  delay(10);
}

// 广播消息给所有客户端
void broadcastMessage(String message) {
  Serial.printf("广播: %s\n", message.c_str());
  
  for (int i = 0; i < MAX_CLIENTS; i++) {
    if (clients[i] && clients[i].connected()) {
      clients[i].println(message);
    }
  }
}
```

## UDP通信

### UDP客户端

```cpp
/*
 * UDP客户端示例
 * 发送和接收UDP数据包
 */

#include <WiFi.h>
#include <WiFiUdp.h>

const char* ssid = "Your_SSID";
const char* password = "Your_Password";

// UDP配置
const char* udpHost = "192.168.1.100";
const int udpPort = 12345;
const int localPort = 12346;

WiFiUDP udp;

void setup() {
  Serial.begin(115200);
  
  // 连接WiFi
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi已连接");
  
  // 启动UDP
  udp.begin(localPort);
  Serial.printf("UDP已启动，本地端口: %d\n", localPort);
}

void loop() {
  // 发送UDP数据
  static unsigned long lastSend = 0;
  if (millis() - lastSend > 5000) {
    lastSend = millis();
    
    String message = "Hello from ESP32 at " + String(millis());
    sendMessage(message);
  }
  
  // 接收UDP数据
  receiveMessage();
  
  // 处理串口输入
  if (Serial.available() > 0) {
    String input = Serial.readStringUntil('\n');
    input.trim();
    if (input.length() > 0) {
      sendMessage(input);
    }
  }
  
  delay(10);
}

// 发送UDP消息
void sendMessage(String message) {
  udp.beginPacket(udpHost, udpPort);
  udp.print(message);
  udp.endPacket();
  
  Serial.printf("发送: %s -> %s:%d\n", 
                message.c_str(), udpHost, udpPort);
}

// 接收UDP消息
void receiveMessage() {
  int packetSize = udp.parsePacket();
  
  if (packetSize) {
    // 读取数据
    char packetBuffer[255];
    int len = udp.read(packetBuffer, 255);
    
    if (len > 0) {
      packetBuffer[len] = '\0';
    }
    
    // 显示发送者信息
    Serial.printf("收到UDP包: %s:%d, 长度: %d\n",
                  udp.remoteIP().toString().c_str(),
                  udp.remotePort(),
                  packetSize);
    Serial.printf("内容: %s\n", packetBuffer);
  }
}
```

### UDP广播

```cpp
/*
 * UDP广播示例
 * 发送广播消息
 */

#include <WiFi.h>
#include <WiFiUdp.h>

const char* ssid = "Your_SSID";
const char* password = "Your_Password";

const int broadcastPort = 12345;
const int localPort = 12345;

WiFiUDP udp;

void setup() {
  Serial.begin(115200);
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  Serial.println("WiFi已连接");
  
  udp.begin(localPort);
  
  Serial.println("UDP广播示例启动");
  Serial.println("输入消息进行广播");
}

void loop() {
  // 发送广播
  if (Serial.available() > 0) {
    String message = Serial.readStringUntil('\n');
    message.trim();
    
    if (message.length() > 0) {
      sendBroadcast(message);
    }
  }
  
  // 接收广播
  int packetSize = udp.parsePacket();
  if (packetSize) {
    char buffer[255];
    int len = udp.read(buffer, 255);
    if (len > 0) {
      buffer[len] = '\0';
    }
    
    Serial.printf("收到广播 [%s]: %s\n",
                  udp.remoteIP().toString().c_str(),
                  buffer);
  }
  
  delay(10);
}

// 发送广播
void sendBroadcast(String message) {
  // 获取广播地址
  IPAddress broadcastAddr = WiFi.localIP();
  broadcastAddr[3] = 255;  // 设置为广播地址
  
  udp.beginPacket(broadcastAddr, broadcastPort);
  udp.print(message);
  udp.endPacket();
  
  Serial.printf("广播: %s -> %s:%d\n",
                message.c_str(),
                broadcastAddr.toString().c_str(),
                broadcastPort);
}
```

## 实用示例

### 简单聊天服务器

```cpp
/*
 * 简单聊天服务器
 * 多客户端实时聊天
 */

#include <WiFi.h>

#define MAX_CLIENTS 8
#define SERVER_PORT 8080

WiFiServer server(SERVER_PORT);
WiFiClient clients[MAX_CLIENTS];
String clientNames[MAX_CLIENTS];

void setup() {
  Serial.begin(115200);
  
  WiFi.begin("Your_SSID", "Your_Password");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  
  Serial.printf("服务器IP: %s:%d\n", 
                WiFi.localIP().toString().c_str(), SERVER_PORT);
  
  server.begin();
}

void loop() {
  // 接受新连接
  if (server.hasClient()) {
    for (int i = 0; i < MAX_CLIENTS; i++) {
      if (!clients[i] || !clients[i].connected()) {
        clients[i] = server.available();
        clientNames[i] = "User" + String(i);
        clients[i].println("Welcome! Type /name <yourname> to set your name");
        broadcastSystem(clientNames[i] + " joined the chat");
        break;
      }
    }
  }
  
  // 处理消息
  for (int i = 0; i < MAX_CLIENTS; i++) {
    if (clients[i] && clients[i].connected() && clients[i].available()) {
      String msg = clients[i].readStringUntil('\n');
      msg.trim();
      
      if (msg.startsWith("/name ")) {
        // 设置昵称
        String oldName = clientNames[i];
        clientNames[i] = msg.substring(6);
        broadcastSystem(oldName + " is now known as " + clientNames[i]);
      } else if (msg == "/list") {
        // 列出在线用户
        String list = "Online users: ";
        for (int j = 0; j < MAX_CLIENTS; j++) {
          if (clients[j] && clients[j].connected()) {
            list += clientNames[j] + " ";
          }
        }
        clients[i].println(list);
      } else if (msg.length() > 0) {
        // 广播消息
        broadcastChat(clientNames[i], msg);
      }
    }
  }
  
  delay(10);
}

void broadcastChat(String sender, String message) {
  String fullMsg = "[" + sender + "] " + message;
  Serial.println(fullMsg);
  
  for (int i = 0; i < MAX_CLIENTS; i++) {
    if (clients[i] && clients[i].connected()) {
      clients[i].println(fullMsg);
    }
  }
}

void broadcastSystem(String message) {
  String fullMsg = "[System] " + message;
  Serial.println(fullMsg);
  
  for (int i = 0; i < MAX_CLIENTS; i++) {
    if (clients[i] && clients[i].connected()) {
      clients[i].println(fullMsg);
    }
  }
}
```
