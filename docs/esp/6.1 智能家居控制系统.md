# 6.1 智能家居控制系统

本节将通过一个完整的智能家居控制系统项目，综合运用前面学习的知识，实现一个实用的物联网应用。

## 项目概述

### 功能需求

- 多路灯光控制
- 温湿度监测
- 远程控制（WiFi + 蓝牙）
- 定时任务
- 场景模式

### 硬件清单

| 组件 | 数量 | 说明 |
|------|------|------|
| ESP32开发板 | 1 | 主控制器 |
| 继电器模块 | 4 | 灯光控制 |
| DHT22传感器 | 1 | 温湿度检测 |
| OLED显示屏 | 1 | 状态显示 |
| 按键模块 | 4 | 本地控制 |

### 系统架构

```
+------------------+     WiFi      +-------------+
|     手机APP      |<------------->|             |
+------------------+               |             |
                                   |    ESP32    |
+------------------+     BLE       |   主控制器   |

+------------------+               |             |
                                   +------+------+
                                          |
                    +---------------------+---------------------+

```

## 硬件配置

```cpp
/*
 * 智能家居控制系统
 * 硬件配置和初始化
 */

#include <WiFi.h>
#include <WebServer.h>
#include <WebSocketsServer.h>
#include <ArduinoJson.h>
#include <DHT.h>
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

// ==================== 硬件引脚定义 ====================
// 继电器引脚
#define RELAY_1 25
#define RELAY_2 26
#define RELAY_3 27
#define RELAY_4 14

// 按键引脚
#define BUTTON_1 34
#define BUTTON_2 35
#define BUTTON_3 32
#define BUTTON_4 33

// DHT传感器
#define DHT_PIN 4
#define DHT_TYPE DHT22

// OLED配置
#define OLED_SDA 21
#define OLED_SCL 22
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64

// ==================== 全局对象 ====================
WebServer server(80);
WebSocketsServer webSocket = WebSocketsServer(81);
DHT dht(DHT_PIN, DHT_TYPE);
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

// ==================== 系统状态 ====================
struct DeviceState {
  bool lights[4];
  float temperature;
  float humidity;
  int mode;  // 0:手动, 1:自动, 2:场景
  String scene;
  unsigned long lastUpdate;
} state;

// ==================== WiFi配置 ====================
const char* ssid = "Your_SSID";
const char* password = "Your_Password";
```

## 核心功能实现

### 初始化函数

```cpp
void setup() {
  Serial.begin(115200);
  
  // 初始化硬件
  initHardware();
  
  // 初始化WiFi
  initWiFi();
  
  // 初始化Web服务器
  initWebServer();
  
  // 初始化WebSocket
  initWebSocket();
  
  // 初始化显示
  initDisplay();
  
  Serial.println("智能家居系统已启动");
}

void initHardware() {
  // 配置继电器引脚
  pinMode(RELAY_1, OUTPUT);
  pinMode(RELAY_2, OUTPUT);
  pinMode(RELAY_3, OUTPUT);
  pinMode(RELAY_4, OUTPUT);
  
  // 初始状态：全部关闭
  for (int i = 0; i < 4; i++) {
    state.lights[i] = false;
    setLight(i, false);
  }
  
  // 配置按键引脚
  pinMode(BUTTON_1, INPUT_PULLUP);
  pinMode(BUTTON_2, INPUT_PULLUP);
  pinMode(BUTTON_3, INPUT_PULLUP);
  pinMode(BUTTON_4, INPUT_PULLUP);
  
  // 初始化DHT
  dht.begin();
  
  // 初始化状态
  state.temperature = 0;
  state.humidity = 0;
  state.mode = 0;
  state.scene = "normal";
}

void initWiFi() {
  WiFi.begin(ssid, password);
  
  Serial.print("连接WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.println("\nWiFi已连接");
  Serial.printf("IP地址: %s\n", WiFi.localIP().toString().c_str());
}

void initDisplay() {
  Wire.begin(OLED_SDA, OLED_SCL);
  
  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("OLED初始化失败");
    return;
  }
  
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  
  updateDisplay();
}
```

### 灯光控制

```cpp
// 设置单个灯光
void setLight(int index, bool on) {
  int pin;
  
  switch (index) {
    case 0: pin = RELAY_1; break;
    case 1: pin = RELAY_2; break;
    case 2: pin = RELAY_3; break;
    case 3: pin = RELAY_4; break;
    default: return;
  }
  
  digitalWrite(pin, on ? HIGH : LOW);
  state.lights[index] = on;
  
  Serial.printf("灯光 %d: %s\n", index + 1, on ? "开" : "关");
  
  // 通知客户端
  broadcastState();
}

// 设置所有灯光
void setAllLights(bool on) {
  for (int i = 0; i < 4; i++) {
    setLight(i, on);
  }
}

// 切换灯光状态
void toggleLight(int index) {
  setLight(index, !state.lights[index]);
}
```

### 传感器读取

```cpp
// 读取传感器数据
void readSensors() {
  float temp = dht.readTemperature();
  float humi = dht.readHumidity();
  
  if (!isnan(temp) && !isnan(humi)) {
    state.temperature = temp;
    state.humidity = humi;
    state.lastUpdate = millis();
    
    // 自动模式处理
    if (state.mode == 1) {
      autoControl();
    }
  }
}

// 自动控制逻辑
void autoControl() {
  // 温度超过阈值，打开风扇（灯光3）
  if (state.temperature > 28.0 && !state.lights[2]) {
    setLight(2, true);
  } else if (state.temperature < 26.0 && state.lights[2]) {
    setLight(2, false);
  }
}
```

### Web服务器

```cpp
void initWebServer() {
  // 静态页面
  server.on("/", HTTP_GET, handleRoot);
  
  // API接口
  server.on("/api/status", HTTP_GET, handleStatus);
  server.on("/api/light", HTTP_POST, handleLight);
  server.on("/api/mode", HTTP_POST, handleMode);
  server.on("/api/scene", HTTP_POST, handleScene);
  
  // 启动服务器
  server.begin();
  Serial.println("Web服务器已启动");
}

// 首页
void handleRoot() {
  String html = generateWebPage();
  server.send(200, "text/html", html);
}

// 状态API
void handleStatus() {
  StaticJsonDocument<256> doc;
  
  JsonArray lights = doc.createNestedArray("lights");
  for (int i = 0; i < 4; i++) {
    lights.add(state.lights[i]);
  }
  
  doc["temperature"] = state.temperature;
  doc["humidity"] = state.humidity;
  doc["mode"] = state.mode;
  doc["scene"] = state.scene;
  doc["uptime"] = millis() / 1000;
  
  String response;
  serializeJson(doc, response);
  
  server.send(200, "application/json", response);
}

// 灯光控制API
void handleLight() {
  if (!server.hasArg("plain")) {
    server.send(400, "application/json", "{\"error\":\"no data\"}");
    return;
  }
  
  StaticJsonDocument<128> doc;
  deserializeJson(doc, server.arg("plain"));
  
  int index = doc["index"];
  bool on = doc["on"];
  
  if (index >= 0 && index < 4) {
    setLight(index, on);
    server.send(200, "application/json", "{\"status\":\"ok\"}");
  } else {
    server.send(400, "application/json", "{\"error\":\"invalid index\"}");
  }
}

// 模式设置API
void handleMode() {
  if (!server.hasArg("plain")) {
    server.send(400, "application/json", "{\"error\":\"no data\"}");
    return;
  }
  
  StaticJsonDocument<64> doc;
  deserializeJson(doc, server.arg("plain"));
  
  state.mode = doc["mode"];
  
  server.send(200, "application/json", "{\"status\":\"ok\"}");
}

// 场景设置API
void handleScene() {
  if (!server.hasArg("plain")) {
    server.send(400, "application/json", "{\"error\":\"no data\"}");
    return;
  }
  
  StaticJsonDocument<64> doc;
  deserializeJson(doc, server.arg("plain"));
  
  String scene = doc["scene"];
  applyScene(scene);
  
  server.send(200, "application/json", "{\"status\":\"ok\"}");
}

// 应用场景
void applyScene(String scene) {
  state.scene = scene;
  
  if (scene == "home") {
    // 回家模式
    setLight(0, true);
    setLight(1, true);
    setLight(2, false);
    setLight(3, true);
  } else if (scene == "away") {
    // 离家模式
    setAllLights(false);
  } else if (scene == "sleep") {
    // 睡眠模式
    setAllLights(false);
    setLight(0, true);  // 只留夜灯
  } else if (scene == "movie") {
    // 观影模式
    setAllLights(false);
    setLight(3, true);  // 氛围灯
  }
  
  broadcastState();
}
```

### WebSocket通信

```cpp
void initWebSocket() {
  webSocket.begin();
  webSocket.onEvent(webSocketEvent);
  
  Serial.println("WebSocket已启动");
}

void webSocketEvent(uint8_t num, WStype_t type, uint8_t* payload, size_t length) {
  switch (type) {
    case WStype_DISCONNECTED:
      Serial.printf("[%u] 断开连接\n", num);
      break;
      
    case WStype_CONNECTED:
      Serial.printf("[%u] 已连接\n", num);
      // 发送当前状态
      sendState(num);
      break;
      
    case WStype_TEXT:
      // 处理消息
      handleWebSocketMessage(num, payload, length);
      break;
  }
}

void handleWebSocketMessage(uint8_t num, uint8_t* payload, size_t length) {
  StaticJsonDocument<128> doc;
  deserializeJson(doc, payload, length);
  
  String cmd = doc["cmd"];
  
  if (cmd == "light") {
    int index = doc["index"];
    bool on = doc["on"];
    setLight(index, on);
  } else if (cmd == "mode") {
    state.mode = doc["mode"];
    broadcastState();
  } else if (cmd == "scene") {
    applyScene(doc["scene"]);
  }
}

// 发送状态给指定客户端
void sendState(uint8_t num) {
  StaticJsonDocument<256> doc;
  
  JsonArray lights = doc.createNestedArray("lights");
  for (int i = 0; i < 4; i++) {
    lights.add(state.lights[i]);
  }
  
  doc["temperature"] = state.temperature;
  doc["humidity"] = state.humidity;
  doc["mode"] = state.mode;
  doc["scene"] = state.scene;
  
  String message;
  serializeJson(doc, message);
  
  webSocket.sendTXT(num, message);
}

// 广播状态给所有客户端
void broadcastState() {
  StaticJsonDocument<256> doc;
  
  JsonArray lights = doc.createNestedArray("lights");
  for (int i = 0; i < 4; i++) {
    lights.add(state.lights[i]);
  }
  
  doc["temperature"] = state.temperature;
  doc["humidity"] = state.humidity;
  doc["mode"] = state.mode;
  doc["scene"] = state.scene;
  
  String message;
  serializeJson(doc, message);
  
  webSocket.broadcastTXT(message);
}
```

### 显示更新

```cpp
void updateDisplay() {
  display.clearDisplay();
  
  // 标题
  display.setCursor(0, 0);
  display.println("Smart Home System");
  
  // 分隔线
  display.drawLine(0, 10, 127, 10, SSD1306_WHITE);
  
  // 温湿度
  display.setCursor(0, 14);
  display.printf("T: %.1fC  H: %.1f%%", state.temperature, state.humidity);
  
  // 灯光状态
  display.setCursor(0, 28);
  display.print("Lights: ");
  for (int i = 0; i < 4; i++) {
    display.printf("%d", state.lights[i] ? 1 : 0);
    if (i < 3) display.print(" ");
  }
  
  // 模式
  display.setCursor(0, 42);
  display.printf("Mode: %s", state.mode == 0 ? "Manual" : (state.mode == 1 ? "Auto" : "Scene"));
  
  // IP地址
  display.setCursor(0, 56);
  display.printf("IP: %s", WiFi.localIP().toString().c_str());
  
  display.display();
}
```

### 主循环

```cpp
void loop() {
  // 处理Web服务器
  server.handleClient();
  
  // 处理WebSocket
  webSocket.loop();
  
  // 处理按键
  handleButtons();
  
  // 定期读取传感器
  static unsigned long lastSensorRead = 0;
  if (millis() - lastSensorRead > 5000) {
    lastSensorRead = millis();
    readSensors();
    updateDisplay();
  }
  
  delay(10);
}

// 处理按键
void handleButtons() {
  static int lastButtonState[4] = {HIGH, HIGH, HIGH, HIGH};
  static unsigned long lastDebounce[4] = {0, 0, 0, 0};
  
  int buttonPins[4] = {BUTTON_1, BUTTON_2, BUTTON_3, BUTTON_4};
  
  for (int i = 0; i < 4; i++) {
    int reading = digitalRead(buttonPins[i]);
    
    if (reading != lastButtonState[i]) {
      lastDebounce[i] = millis();
    }
    
    if ((millis() - lastDebounce[i]) > 50) {
      if (reading == LOW && lastButtonState[i] == HIGH) {
        // 按键按下
        toggleLight(i);
      }
    }
    
    lastButtonState[i] = reading;
  }
}
```
