# 5.3 UART串口通信

UART（Universal Asynchronous Receiver Transmitter）是最常用的串行通信方式之一。本节将详细介绍ESP32的UART编程。

## UART基础

### UART特点

- 异步通信，无需时钟线
- 全双工通信
- 可配置波特率
- 支持奇偶校验
- 简单可靠

### ESP32 UART资源

ESP32有3个UART控制器：

| UART | TX引脚 | RX引脚 | 用途 |
|------|--------|--------|------|
| UART0 | GPIO1 | GPIO3 | USB下载/调试 |
| UART1 | GPIO10 | GPIO9 | 用户使用 |
| UART2 | GPIO17 | GPIO16 | 用户使用 |

### UART参数

| 参数 | 常用值 |
|------|--------|
| 波特率 | 9600, 115200, 921600 |
| 数据位 | 5, 6, 7, 8 |
| 停止位 | 1, 1.5, 2 |
| 校验位 | 无, 奇, 偶 |

## 串口基本使用

### 串口初始化

```cpp
/*
 * UART基本示例
 */

#define UART_TX 17
#define UART_RX 16

void setup() {
  // 初始化调试串口
  Serial.begin(115200);
  
  // 初始化UART2
  Serial2.begin(9600, SERIAL_8N1, UART_RX, UART_TX);
  
  Serial.println("UART示例启动");
  Serial.printf("UART2: TX=GPIO%d, RX=GPIO%d\n", UART_TX, UART_RX);
}

void loop() {
  // 从UART2读取数据
  while (Serial2.available()) {
    int data = Serial2.read();
    Serial.printf("收到: 0x%02X (%c)\n", data, data);
  }
  
  // 从调试串口读取并发送到UART2
  while (Serial.available()) {
    int data = Serial.read();
    Serial2.write(data);
  }
  
  delay(10);
}
```

### 自定义串口

```cpp
/*
 * 自定义串口示例
 * 使用任意GPIO
 */

#include <HardwareSerial.h>

#define UART_TX 4
#define UART_RX 5

HardwareSerial mySerial(1);  // 使用UART1

void setup() {
  Serial.begin(115200);
  
  // 初始化自定义串口
  // 参数：波特率, 模式, RX引脚, TX引脚
  mySerial.begin(115200, SERIAL_8N1, UART_RX, UART_TX);
  
  Serial.println("自定义串口已启动");
}

void loop() {
  if (mySerial.available()) {
    String data = mySerial.readStringUntil('\n');
    data.trim();
    Serial.printf("收到: %s\n", data.c_str());
    
    // 回显
    mySerial.println("Echo: " + data);
  }
  
  delay(10);
}
```

## 串口通信协议

### 数据帧格式

```cpp
/*
 * 自定义数据帧格式
 * 帧头 + 命令 + 长度 + 数据 + 校验
 */

#define FRAME_HEADER 0xAA
#define FRAME_TAIL   0x55

struct DataFrame {
  uint8_t header;
  uint8_t command;
  uint8_t length;
  uint8_t data[32];
  uint8_t checksum;
  uint8_t tail;
};

// 发送数据帧
void sendFrame(HardwareSerial& serial, uint8_t cmd, uint8_t* data, uint8_t len) {
  DataFrame frame;
  frame.header = FRAME_HEADER;
  frame.command = cmd;
  frame.length = len;
  
  memcpy(frame.data, data, len);
  
  // 计算校验和
  frame.checksum = cmd + len;
  for (int i = 0; i < len; i++) {
    frame.checksum += data[i];
  }
  
  frame.tail = FRAME_TAIL;
  
  // 发送
  serial.write((uint8_t*)&frame, 6 + len);
}

// 接收数据帧
bool receiveFrame(HardwareSerial& serial, DataFrame* frame) {
  static enum {WAIT_HEADER, READ_CMD, READ_LEN, READ_DATA, READ_CHECK, READ_TAIL} state = WAIT_HEADER;
  static uint8_t dataIndex = 0;
  
  while (serial.available()) {
    uint8_t byte = serial.read();
    
    switch (state) {
      case WAIT_HEADER:
        if (byte == FRAME_HEADER) {
          frame->header = byte;
          state = READ_CMD;
        }
        break;
        
      case READ_CMD:
        frame->command = byte;
        state = READ_LEN;
        break;
        
      case READ_LEN:
        frame->length = byte;
        dataIndex = 0;
        state = READ_DATA;
        break;
        
      case READ_DATA:
        frame->data[dataIndex++] = byte;
        if (dataIndex >= frame->length) {
          state = READ_CHECK;
        }
        break;
        
      case READ_CHECK:
        frame->checksum = byte;
        state = READ_TAIL;
        break;
        
      case READ_TAIL:
        frame->tail = byte;
        state = WAIT_HEADER;
        
        // 验证校验和
        uint8_t sum = frame->command + frame->length;
        for (int i = 0; i < frame->length; i++) {
          sum += frame->data[i];
        }
        
        return (sum == frame->checksum && frame->tail == FRAME_TAIL);
    }
  }
  
  return false;
}
```

### Modbus RTU协议

```cpp
/*
 * Modbus RTU从站示例
 */

#include <ModbusSlave.h>

#define RS485_TX 17
#define RS485_RX 16
#define RS485_DE 4  // 发送使能引脚

ModbusSlave modbus(Serial2, 0x01);  // 从站地址1

// 保持寄存器
uint16_t holdingRegs[10] = {0};

void setup() {
  Serial.begin(115200);
  
  // 初始化RS485
  Serial2.begin(9600, SERIAL_8N1, RS485_RX, RS485_TX);
  pinMode(RS485_DE, OUTPUT);
  digitalWrite(RS485_DE, LOW);
  
  // 配置Modbus
  modbus.cbVector[CB_READ_HOLDING] = readHolding;
  modbus.cbVector[CB_WRITE_HOLDING] = writeHolding;
  
  modbus.begin(9600);
  
  Serial.println("Modbus RTU从站已启动");
}

void loop() {
  modbus.poll();
  
  // 更新寄存器值
  holdingRegs[0] = analogRead(34);  // ADC值
  holdingRegs[1] = ESP.getFreeHeap() / 1024;  // 空闲内存(KB)
}

// 读保持寄存器回调
uint8_t readHolding(uint8_t fc, uint16_t address, uint16_t length) {
  for (int i = 0; i < length; i++) {
    modbus.writeRegister(address + i, holdingRegs[address + i]);
  }
  return STATUS_OK;
}

// 写保持寄存器回调
uint8_t writeHolding(uint8_t fc, uint16_t address, uint16_t length) {
  for (int i = 0; i < length; i++) {
    holdingRegs[address + i] = modbus.readRegister(address + i);
  }
  
  // 处理写入
  if (address == 2) {
    digitalWrite(2, holdingRegs[2] ? HIGH : LOW);
  }
  
  return STATUS_OK;
}
```

## GPS模块通信

```cpp
/*
 * GPS模块通信示例
 */

#include <TinyGPS++.h>

#define GPS_TX 16
#define GPS_RX 17

TinyGPSPlus gps;
HardwareSerial GPSSerial(2);

void setup() {
  Serial.begin(115200);
  GPSSerial.begin(9600, SERIAL_8N1, GPS_RX, GPS_TX);
  
  Serial.println("GPS模块已连接");
}

void loop() {
  while (GPSSerial.available() > 0) {
    char c = GPSSerial.read();
    gps.encode(c);
    
    // 显示原始数据
    // Serial.write(c);
  }
  
  // 显示位置信息
  if (gps.location.isValid()) {
    Serial.printf("位置: %.6f, %.6f\n", 
                  gps.location.lat(), 
                  gps.location.lng());
    Serial.printf("海拔: %.2f m\n", gps.altitude.meters());
    Serial.printf("速度: %.2f km/h\n", gps.speed.kmph());
    Serial.printf("航向: %.2f 度\n", gps.course.deg());
  }
  
  // 显示时间
  if (gps.time.isValid()) {
    Serial.printf("时间: %02d:%02d:%02d\n", 
                  gps.time.hour(), 
                  gps.time.minute(), 
                  gps.time.second());
  }
  
  // 显示卫星数
  Serial.printf("卫星数: %d\n", gps.satellites.value());
  
  delay(1000);
}
```

## RS485通信

```cpp
/*
 * RS485通信示例
 */

#define RS485_TX 17
#define RS485_RX 16
#define RS485_DE 4  // 发送使能

void setup() {
  Serial.begin(115200);
  
  Serial2.begin(9600, SERIAL_8N1, RS485_RX, RS485_TX);
  pinMode(RS485_DE, OUTPUT);
  digitalWrite(RS485_DE, LOW);  // 接收模式
  
  Serial.println("RS485通信已启动");
}

void loop() {
  // 发送数据
  rs485Send("Hello RS485");
  delay(100);
  
  // 接收数据
  String data = rs485Receive();
  if (data.length() > 0) {
    Serial.printf("收到: %s\n", data.c_str());
  }
  
  delay(1000);
}

void rs485Send(String data) {
  digitalWrite(RS485_DE, HIGH);  // 发送模式
  delay(1);
  
  Serial2.print(data);
  Serial2.flush();
  
  delay(1);
  digitalWrite(RS485_DE, LOW);  // 接收模式
}

String rs485Receive() {
  String data = "";
  
  while (Serial2.available()) {
    data += (char)Serial2.read();
  }
  
  return data;
}
```
