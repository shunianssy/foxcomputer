# 3.4 MQTT协议应用

MQTT（Message Queuing Telemetry Transport）是一种轻量级的发布/订阅消息传输协议，非常适合物联网应用。本节将详细介绍ESP32的MQTT编程方法。

## MQTT基础

### MQTT特点

- 轻量级：协议头部最小仅2字节
- 发布/订阅模式：解耦消息生产者和消费者
- QoS支持：三种消息质量等级
- 遗嘱消息：客户端异常断开时通知
- 持久会话：保持订阅状态

### QoS等级

| QoS等级 | 说明 |
|---------|------|
| 0 | 最多一次，消息可能丢失 |
| 1 | 至少一次，消息可能重复 |
| 2 | 恰好一次，保证送达且不重复 |

## MQTT客户端

### 基本MQTT客户端

```cpp
/*
 * MQTT基本客户端示例
 */

#include <WiFi.h>
#include <PubSubClient.h>

const char* ssid = "Your_SSID";
const char* password = "Your_Password";

// MQTT服务器配置
const char* mqtt_server = "broker.emqx.io";
const int mqtt_port = 1883;
const char* mqtt_user = "";
const char* mqtt_password = "";

WiFiClient espClient;
PubSubClient mqttClient(espClient);

// 消息缓冲
char msgBuffer[256];

void setup() {
  Serial.begin(115200);
  
  // 连接WiFi
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi已连接");
  
  // 配置MQTT
  mqttClient.setServer(mqtt_server, mqtt_port);
  mqttClient.setCallback(mqttCallback);
  
  // 连接MQTT
  connectMQTT();
}

void loop() {
  // 保持MQTT连接
  if (!mqttClient.connected()) {
    connectMQTT();
  }
  
  mqttClient.loop();
  
  // 定期发布消息
  static unsigned long lastPublish = 0;
  if (millis() - lastPublish > 5000) {
    lastPublish = millis();
    publishSensorData();
  }
  
  delay(10);
}

// 连接MQTT服务器
void connectMQTT() {
  Serial.print("连接MQTT服务器...");
  
  // 生成客户端ID
  String clientId = "ESP32-" + String(random(0xffff), HEX);
  
  // 连接（带遗嘱消息）
  if (mqttClient.connect(clientId.c_str(), mqtt_user, mqtt_password,
                         "home/status", 1, true, "offline")) {
    Serial.println("已连接");
    
    // 发布在线状态
    mqttClient.publish("home/status", "online", true);
    
    // 订阅主题
    mqttClient.subscribe("home/living_room/light/control");
    mqttClient.subscribe("home/+/temperature");
    
    Serial.println("已订阅主题");
  } else {
    Serial.printf("失败，状态码: %d\n", mqttClient.state());
    delay(5000);
  }
}

// MQTT消息回调
void mqttCallback(char* topic, byte* payload, unsigned int length) {
  Serial.printf("收到消息 [%s]: ", topic);
  
  // 解析消息内容
  String message;
  for (int i = 0; i < length; i++) {
    message += (char)payload[i];
  }
  Serial.println(message);
  
  // 处理不同主题的消息
  if (String(topic) == "home/living_room/light/control") {
    handleLightControl(message);
  }
}

// 处理灯光控制
void handleLightControl(String command) {
  if (command == "ON") {
    digitalWrite(2, HIGH);
    Serial.println("灯光已打开");
  } else if (command == "OFF") {
    digitalWrite(2, LOW);
    Serial.println("灯光已关闭");
  }
}

// 发布传感器数据
void publishSensorData() {
  // 模拟传感器数据
  float temperature = 25.0 + random(-20, 20) / 10.0;
  float humidity = 60.0 + random(-10, 10);
  
  // 发布温度
  snprintf(msgBuffer, sizeof(msgBuffer), "%.1f", temperature);
  mqttClient.publish("home/living_room/temperature", msgBuffer);
  
  // 发布湿度
  snprintf(msgBuffer, sizeof(msgBuffer), "%.1f", humidity);
  mqttClient.publish("home/living_room/humidity", msgBuffer);
  
  Serial.printf("已发布: 温度=%.1f, 湿度=%.1f\n", temperature, humidity);
}
```

### MQTT客户端类封装

```cpp
/*
 * MQTT客户端类封装
 */

#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>

class MQTTManager {
private:
  WiFiClient wifiClient;
  PubSubClient mqttClient;
  
  String server;
  int port;
  String user;
  String password;
  String clientId;
  String willTopic;
  String willMessage;
  
  // 回调函数指针
  void (*messageCallback)(String topic, String payload);
  
public:
  MQTTManager() : mqttClient(wifiClient) {
    messageCallback = NULL;
  }
  
  // 配置服务器
  void setServer(String mqttServer, int mqttPort) {
    server = mqttServer;
    port = mqttPort;
    mqttClient.setServer(server.c_str(), port);
  }
  
  // 配置认证
  void setCredentials(String mqttUser, String mqttPassword) {
    user = mqttUser;
    password = mqttPassword;
  }
  
  // 设置客户端ID
  void setClientId(String id) {
    clientId = id;
  }
  
  // 设置遗嘱
  void setWill(String topic, String message) {
    willTopic = topic;
    willMessage = message;
  }
  
  // 设置消息回调
  void setMessageCallback(void (*callback)(String, String)) {
    messageCallback = callback;
    mqttClient.setCallback([this](char* topic, byte* payload, unsigned int length) {
      String topicStr = String(topic);
      String payloadStr;
      for (int i = 0; i < length; i++) {
        payloadStr += (char)payload[i];
      }
      
      if (messageCallback) {
        messageCallback(topicStr, payloadStr);
      }
    });
  }
  
  // 连接
  bool connect() {
    if (clientId.length() == 0) {
      clientId = "ESP32-" + String(random(0xffff), HEX);
    }
    
    bool connected;
    
    if (willTopic.length() > 0) {
      connected = mqttClient.connect(clientId.c_str(), 
                                     user.c_str(), password.c_str(),
                                     willTopic.c_str(), 1, true, 
                                     willMessage.c_str());
    } else {
      connected = mqttClient.connect(clientId.c_str(),
                                     user.c_str(), password.c_str());
    }
    
    return connected;
  }
  
  // 断开连接
  void disconnect() {
    mqttClient.disconnect();
  }
  
  // 检查连接状态
  bool isConnected() {
    return mqttClient.connected();
  }
  
  // 订阅主题
  bool subscribe(String topic, uint8_t qos = 0) {
    return mqttClient.subscribe(topic.c_str(), qos);
  }
  
  // 取消订阅
  bool unsubscribe(String topic) {
    return mqttClient.unsubscribe(topic.c_str());
  }
  
  // 发布消息
  bool publish(String topic, String payload, bool retained = false) {
    return mqttClient.publish(topic.c_str(), payload.c_str(), retained);
  }
  
  // 发布JSON
  bool publishJson(String topic, JsonObject& json, bool retained = false) {
    String payload;
    serializeJson(json, payload);
    return publish(topic, payload, retained);
  }
  
  // 更新（在loop中调用）
  void update() {
    if (!mqttClient.connected()) {
      static unsigned long lastReconnect = 0;
      if (millis() - lastReconnect > 5000) {
        lastReconnect = millis();
        if (connect()) {
          Serial.println("MQTT已重连");
        }
      }
    }
    
    mqttClient.loop();
  }
  
  // 获取状态码
  int getState() {
    return mqttClient.state();
  }
  
  // 状态码转字符串
  String getStateString() {
    int state = mqttClient.state();
    switch (state) {
      case -4: return "MQTT_CONNECTION_TIMEOUT";
      case -3: return "MQTT_CONNECTION_LOST";
      case -2: return "MQTT_CONNECT_FAILED";
      case -1: return "MQTT_DISCONNECTED";
      case 0: return "MQTT_CONNECTED";
      case 1: return "MQTT_CONNECT_BAD_PROTOCOL";
      case 2: return "MQTT_CONNECT_BAD_CLIENT_ID";
      case 3: return "MQTT_CONNECT_UNAVAILABLE";
      case 4: return "MQTT_CONNECT_BAD_CREDENTIALS";
      case 5: return "MQTT_CONNECT_UNAUTHORIZED";
      default: return "UNKNOWN";
    }
  }
};

// 使用示例
MQTTManager mqtt;

void setup() {
  Serial.begin(115200);
  
  WiFi.begin("Your_SSID", "Your_Password");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  
  // 配置MQTT
  mqtt.setServer("broker.emqx.io", 1883);
  mqtt.setClientId("ESP32-Client");
  mqtt.setWill("device/status", "offline");
  
  // 设置消息回调
  mqtt.setMessageCallback([](String topic, String payload) {
    Serial.printf("收到 [%s]: %s\n", topic.c_str(), payload.c_str());
  });
  
  // 连接
  if (mqtt.connect()) {
    Serial.println("MQTT已连接");
    mqtt.subscribe("test/topic");
    mqtt.publish("device/status", "online", true);
  }
}

void loop() {
  mqtt.update();
  delay(10);
}
```

## JSON消息处理

```cpp
/*
 * MQTT JSON消息处理示例
 */

#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>

WiFiClient espClient;
PubSubClient mqtt(espClient);

// 发布JSON数据
void publishSensorData() {
  StaticJsonDocument<256> doc;
  
  doc["device_id"] = "ESP32_001";
  doc["timestamp"] = millis();
  
  JsonObject sensors = doc.createNestedObject("sensors");
  sensors["temperature"] = 25.5;
  sensors["humidity"] = 60.0;
  sensors["light"] = 800;
  sensors["motion"] = false;
  
  String payload;
  serializeJson(doc, payload);
  
  mqtt.publish("sensors/data", payload.c_str());
  Serial.printf("发布: %s\n", payload.c_str());
}

// 解析JSON消息
void parseCommand(String payload) {
  StaticJsonDocument<256> doc;
  DeserializationError error = deserializeJson(doc, payload);
  
  if (error) {
    Serial.printf("JSON解析失败: %s\n", error.c_str());
    return;
  }
  
  String cmd = doc["command"];
  
  if (cmd == "set_led") {
    bool state = doc["state"];
    digitalWrite(2, state ? HIGH : LOW);
    Serial.printf("LED设置为: %s\n", state ? "开" : "关");
  } else if (cmd == "get_status") {
    publishStatus();
  } else if (cmd == "set_interval") {
    int interval = doc["interval"];
    // 更新采样间隔
    Serial.printf("采样间隔设置为: %d ms\n", interval);
  }
}

// 发布状态
void publishStatus() {
  StaticJsonDocument<256> doc;
  
  doc["device_id"] = "ESP32_001";
  doc["uptime"] = millis() / 1000;
  doc["free_heap"] = ESP.getFreeHeap();
  doc["wifi_rssi"] = WiFi.RSSI();
  doc["ip"] = WiFi.localIP().toString();
  
  String payload;
  serializeJson(doc, payload);
  
  mqtt.publish("sensors/status", payload.c_str());
}

void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String message;
  for (int i = 0; i < length; i++) {
    message += (char)payload[i];
  }
  
  Serial.printf("收到 [%s]: %s\n", topic, message.c_str());
  
  if (String(topic) == "sensors/command") {
    parseCommand(message);
  }
}

void setup() {
  Serial.begin(115200);
  
  WiFi.begin("Your_SSID", "Your_Password");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  
  mqtt.setServer("broker.emqx.io", 1883);
  mqtt.setCallback(mqttCallback);
  
  pinMode(2, OUTPUT);
  
  if (mqtt.connect("ESP32-Client")) {
    mqtt.subscribe("sensors/command");
    Serial.println("MQTT已连接");
  }
}

void loop() {
  if (!mqtt.connected()) {
    mqtt.connect("ESP32-Client");
    mqtt.subscribe("sensors/command");
  }
  
  mqtt.loop();
  
  static unsigned long lastPub = 0;
  if (millis() - lastPub > 10000) {
    lastPub = millis();
    publishSensorData();
  }
  
  delay(10);
}
```

## 智能家居示例

```cpp
/*
 * MQTT智能家居示例
 */

#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>

WiFiClient espClient;
PubSubClient mqtt(espClient);

// 设备状态
struct DeviceState {
  bool light1;
  bool light2;
  int fanSpeed;  // 0-3
  float temperature;
  float humidity;
} deviceState;

// 主题定义
const char* topic_light1 = "home/living_room/light1";
const char* topic_light2 = "home/living_room/light2";
const char* topic_fan = "home/living_room/fan";
const char* topic_sensors = "home/living_room/sensors";
const char* topic_status = "home/living_room/status";
const char* topic_command = "home/living_room/command";

void setup() {
  Serial.begin(115200);
  
  // 初始化设备状态
  deviceState = {false, false, 0, 25.0, 60.0};
  
  // 连接WiFi
  WiFi.begin("Your_SSID", "Your_Password");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  Serial.println("WiFi已连接");
  
  // 配置MQTT
  mqtt.setServer("broker.emqx.io", 1883);
  mqtt.setCallback(mqttCallback);
  
  // 连接MQTT
  connectMQTT();
}

void loop() {
  if (!mqtt.connected()) {
    connectMQTT();
  }
  
  mqtt.loop();
  
  // 定期发布传感器数据
  static unsigned long lastSensor = 0;
  if (millis() - lastSensor > 30000) {
    lastSensor = millis();
    readSensors();
    publishSensorData();
  }
  
  delay(10);
}

void connectMQTT() {
  String clientId = "ESP32-SmartHome-" + String(random(0xffff), HEX);
  
  if (mqtt.connect(clientId.c_str(), topic_status, 1, true, "offline")) {
    Serial.println("MQTT已连接");
    
    // 发布在线状态
    mqtt.publish(topic_status, "online", true);
    
    // 订阅控制主题
    mqtt.subscribe(topic_light1);
    mqtt.subscribe(topic_light2);
    mqtt.subscribe(topic_fan);
    mqtt.subscribe(topic_command);
    
    // 发布当前状态
    publishDeviceState();
  }
}

void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String message;
  for (int i = 0; i < length; i++) {
    message += (char)payload[i];
  }
  
  Serial.printf("收到 [%s]: %s\n", topic, message.c_str());
  
  // 处理灯光1
  if (String(topic) == topic_light1) {
    deviceState.light1 = (message == "ON");
    digitalWrite(25, deviceState.light1 ? HIGH : LOW);
    publishDeviceState();
  }
  
  // 处理灯光2
  else if (String(topic) == topic_light2) {
    deviceState.light2 = (message == "ON");
    digitalWrite(26, deviceState.light2 ? HIGH : LOW);
    publishDeviceState();
  }
  
  // 处理风扇
  else if (String(topic) == topic_fan) {
    deviceState.fanSpeed = message.toInt();
    setFanSpeed(deviceState.fanSpeed);
    publishDeviceState();
  }
  
  // 处理命令
  else if (String(topic) == topic_command) {
    handleCommand(message);
  }
}

void handleCommand(String command) {
  StaticJsonDocument<256> doc;
  deserializeJson(doc, command);
  
  String cmd = doc["cmd"];
  
  if (cmd == "set_all_lights") {
    bool state = doc["state"];
    deviceState.light1 = state;
    deviceState.light2 = state;
    digitalWrite(25, state ? HIGH : LOW);
    digitalWrite(26, state ? HIGH : LOW);
    publishDeviceState();
  } else if (cmd == "get_state") {
    publishDeviceState();
  }
}

void setFanSpeed(int speed) {
  // 使用PWM控制风扇速度
  int pwmValue = speed * 85;  // 0, 85, 170, 255
  ledcWrite(0, pwmValue);
}

void readSensors() {
  // 模拟读取传感器
  deviceState.temperature = 25.0 + random(-20, 20) / 10.0;
  deviceState.humidity = 60.0 + random(-10, 10);
}

void publishSensorData() {
  StaticJsonDocument<128> doc;
  doc["temperature"] = deviceState.temperature;
  doc["humidity"] = deviceState.humidity;
  
  String payload;
  serializeJson(doc, payload);
  
  mqtt.publish(topic_sensors, payload.c_str());
}

void publishDeviceState() {
  StaticJsonDocument<256> doc;
  doc["light1"] = deviceState.light1;
  doc["light2"] = deviceState.light2;
  doc["fan_speed"] = deviceState.fanSpeed;
  doc["temperature"] = deviceState.temperature;
  doc["humidity"] = deviceState.humidity;
  doc["uptime"] = millis() / 1000;
  
  String payload;
  serializeJson(doc, payload);
  
  mqtt.publish(topic_status, payload.c_str(), true);
}
```
