# 6.3 智能插座制作

本节将介绍如何使用ESP32制作一个智能插座，支持远程控制、定时开关和功率监测。

## 项目概述

### 功能需求

- 远程开关控制
- 定时开关功能
- 功率监测
- 过载保护
- 状态指示

### 硬件清单

| 组件 | 数量 | 说明 |
|------|------|------|
| ESP32开发板 | 1 | 主控制器 |
| 继电器模块(5V) | 1 | 控制插座 |
| HLW8032功率模块 | 1 | 功率检测 |
| 按钮开关 | 1 | 本地控制 |
| LED指示灯 | 2 | 状态显示 |

### 安全警告

制作智能插座涉及220V高压电，请务必注意安全：

1. 确保断电后再进行接线
2. 使用绝缘良好的外壳
3. 继电器额定电流要大于负载电流
4. 建议使用成品功率模块而非自制电路

## 完整实现

```cpp
/*
 * 智能插座控制器
 */

#include <WiFi.h>
#include <WebServer.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>
#include <Preferences.h>

// ==================== 硬件配置 ====================
#define RELAY_PIN 25       // 继电器控制
#define BUTTON_PIN 0       // 按钮开关
#define LED_PIN 2          // 状态LED
#define CF_PIN 34          // 功率模块CF引脚
#define CF1_PIN 35         // 功率模块CF1引脚
#define SEL_PIN 26         // 功率模块SEL引脚

// ==================== WiFi配置 ====================
const char* ssid = "Your_SSID";
const char* password = "Your_Password";

// ==================== MQTT配置 ====================
const char* mqtt_server = "broker.emqx.io";
const int mqtt_port = 1883;
const char* mqtt_topic = "home/smartplug";

// ==================== 全局对象 ====================
WiFiClient espClient;
PubSubClient mqtt(espClient);
WebServer server(80);
Preferences preferences;

// ==================== 状态变量 ====================
struct PlugState {
  bool power;              // 开关状态
  float voltage;           // 电压
  float current;           // 电流
  float power;             // 功率
  float energy;            // 电能(kWh)
  unsigned long onTime;    // 开启时间
  unsigned long totalOnTime; // 总开启时间
} state;

// 定时任务
struct TimerTask {
  bool enabled;
  int hour;
  int minute;
  bool action;  // true=开, false=关
} timer;

// ==================== 初始化 ====================
void setup() {
  Serial.begin(115200);
  
  initHardware();
  initWiFi();
  initMQTT();
  initWebServer();
  loadConfig();
  
  Serial.println("智能插座已启动");
}

void initHardware() {
  pinMode(RELAY_PIN, OUTPUT);
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  pinMode(CF_PIN, INPUT);
  pinMode(CF1_PIN, INPUT);
  pinMode(SEL_PIN, OUTPUT);
  
  // 初始状态：关闭
  setPower(false);
}

void initWiFi() {
  WiFi.begin(ssid, password);
  
  Serial.print("连接WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.printf("\nIP: %s\n", WiFi.localIP().toString().c_str());
}

void initMQTT() {
  mqtt.setServer(mqtt_server, mqtt_port);
  mqtt.setCallback(mqttCallback);
}

void initWebServer() {
  server.on("/", handleRoot);
  server.on("/api/status", handleStatus);
  server.on("/api/power", handlePower);
  server.on("/api/timer", handleTimer);
  server.on("/api/energy", handleEnergy);
  
  server.begin();
}

void loadConfig() {
  preferences.begin("smartplug", true);
  state.totalOnTime = preferences.getULong("totalOn", 0);
  state.energy = preferences.getFloat("energy", 0);
  timer.enabled = preferences.getBool("timerEn", false);
  timer.hour = preferences.getInt("timerHour", 0);
  timer.minute = preferences.getInt("timerMin", 0);
  timer.action = preferences.getBool("timerAct", true);
  preferences.end();
}

void saveConfig() {
  preferences.begin("smartplug", false);
  preferences.putULong("totalOn", state.totalOnTime);
  preferences.putFloat("energy", state.energy);
  preferences.putBool("timerEn", timer.enabled);
  preferences.putInt("timerHour", timer.hour);
  preferences.putInt("timerMin", timer.minute);
  preferences.putBool("timerAct", timer.action);
  preferences.end();
}

// ==================== 电源控制 ====================
void setPower(bool on) {
  state.power = on;
  digitalWrite(RELAY_PIN, on ? HIGH : LOW);
  digitalWrite(LED_PIN, on ? HIGH : LOW);
  
  if (on) {
    state.onTime = millis();
  } else if (state.onTime > 0) {
    state.totalOnTime += (millis() - state.onTime) / 1000;
    state.onTime = 0;
    saveConfig();
  }
  
  Serial.printf("电源: %s\n", on ? "开" : "关");
  
  // 发布状态
  publishState();
}

void togglePower() {
  setPower(!state.power);
}

// ==================== 功率监测 ====================
void readPower() {
  // 简化的功率计算
  // 实际使用HLW8032需要更复杂的计算
  
  unsigned long cfCount = 0;
  unsigned long startTime = millis();
  
  // 计数CF脉冲
  while (millis() - startTime < 1000) {
    if (digitalRead(CF_PIN) == HIGH) {
      while (digitalRead(CF_PIN) == HIGH);
      cfCount++;
    }
  }
  
  // 根据脉冲频率计算功率
  // 需要根据实际模块校准
  if (cfCount > 0) {
    state.power = cfCount * 0.5;  // 校准系数
    state.current = state.power / 220.0;
    state.voltage = 220.0;
    
    // 累计电能
    state.energy += state.power / 3600000.0;  // kWh
  } else {
    state.power = 0;
    state.current = 0;
  }
}

// ==================== MQTT通信 ====================
void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String message;
  for (int i = 0; i < length; i++) {
    message += (char)payload[i];
  }
  
  StaticJsonDocument<128> doc;
  deserializeJson(doc, message);
  
  String cmd = doc["cmd"];
  
  if (cmd == "power") {
    setPower(doc["state"]);
  } else if (cmd == "toggle") {
    togglePower();
  }
}

void reconnectMQTT() {
  String clientId = "SmartPlug-" + String(random(0xffff), HEX);
  
  if (mqtt.connect(clientId.c_str())) {
    Serial.println("MQTT已连接");
    mqtt.subscribe(mqtt_topic "/cmd");
  }
}

void publishState() {
  if (!mqtt.connected()) return;
  
  StaticJsonDocument<256> doc;
  
  doc["power"] = state.power;
  doc["voltage"] = state.voltage;
  doc["current"] = state.current;
  doc["wattage"] = state.power;
  doc["energy"] = state.energy;
  doc["uptime"] = millis() / 1000;
  
  String payload;
  serializeJson(doc, payload);
  
  mqtt.publish(mqtt_topic "/state", payload.c_str());
}

// ==================== Web服务 ====================
void handleRoot() {
  String html = generateWebPage();
  server.send(200, "text/html", html);
}

void handleStatus() {
  StaticJsonDocument<256> doc;
  
  doc["power"] = state.power;
  doc["voltage"] = state.voltage;
  doc["current"] = state.current;
  doc["wattage"] = state.power;
  doc["energy"] = state.energy;
  doc["totalOnTime"] = state.totalOnTime;
  
  String response;
  serializeJson(doc, response);
  
  server.send(200, "application/json", response);
}

void handlePower() {
  if (!server.hasArg("plain")) {
    server.send(400, "application/json", "{\"error\":\"no data\"}");
    return;
  }
  
  StaticJsonDocument<64> doc;
  deserializeJson(doc, server.arg("plain"));
  
  bool power = doc["power"];
  setPower(power);
  
  server.send(200, "application/json", "{\"status\":\"ok\"}");
}

void handleTimer() {
  if (server.method() == HTTP_GET) {
    StaticJsonDocument<128> doc;
    
    doc["enabled"] = timer.enabled;
    doc["hour"] = timer.hour;
    doc["minute"] = timer.minute;
    doc["action"] = timer.action;
    
    String response;
    serializeJson(doc, response);
    
    server.send(200, "application/json", response);
  } else {
    StaticJsonDocument<128> doc;
    deserializeJson(doc, server.arg("plain"));
    
    timer.enabled = doc["enabled"];
    timer.hour = doc["hour"];
    timer.minute = doc["minute"];
    timer.action = doc["action"];
    
    saveConfig();
    
    server.send(200, "application/json", "{\"status\":\"ok\"}");
  }
}

void handleEnergy() {
  StaticJsonDocument<128> doc;
  
  doc["energy"] = state.energy;
  doc["totalOnTime"] = state.totalOnTime;
  doc["currentPower"] = state.power;
  
  String response;
  serializeJson(doc, response);
  
  server.send(200, "application/json", response);
}

// ==================== 定时任务 ====================
void checkTimer() {
  if (!timer.enabled) return;
  
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) return;
  
  if (timeinfo.tm_hour == timer.hour && timeinfo.tm_min == timer.minute) {
    static bool executed = false;
    if (!executed) {
      setPower(timer.action);
      executed = true;
    }
  } else {
    static int lastMin = -1;
    if (timeinfo.tm_min != lastMin) {
      lastMin = timeinfo.tm_min;
      executed = false;
    }
  }
}

// ==================== 主循环 ====================
void loop() {
  server.handleClient();
  
  if (!mqtt.connected()) {
    static unsigned long lastReconnect = 0;
    if (millis() - lastReconnect > 5000) {
      lastReconnect = millis();
      reconnectMQTT();
    }
  }
  mqtt.loop();
  
  // 处理按钮
  handleButton();
  
  // 定期读取功率
  static unsigned long lastPowerRead = 0;
  if (millis() - lastPowerRead > 5000) {
    lastPowerRead = millis();
    readPower();
  }
  
  // 检查定时任务
  checkTimer();
  
  delay(10);
}

void handleButton() {
  static int lastState = HIGH;
  static unsigned long lastDebounce = 0;
  
  int reading = digitalRead(BUTTON_PIN);
  
  if (reading != lastState) {
    lastDebounce = millis();
  }
  
  if ((millis() - lastDebounce) > 50) {
    if (reading == LOW && lastState == HIGH) {
      togglePower();
    }
  }
  
  lastState = reading;
}

// ==================== Web页面 ====================
String generateWebPage() {
  return R"rawliteral(
<!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>智能插座</title>
  <style>
    body { font-family: Arial; max-width: 400px; margin: 0 auto; padding: 20px; text-align: center; }
    .switch { width: 200px; height: 200px; border-radius: 50%; border: none; font-size: 24px; cursor: pointer; margin: 20px; }
    .switch.on { background: #4CAF50; color: white; }
    .switch.off { background: #f44336; color: white; }
    .info { background: #f5f5f5; padding: 15px; border-radius: 10px; margin: 10px 0; }
    .timer { display: flex; justify-content: center; gap: 10px; margin: 10px 0; }
    input, select { padding: 10px; font-size: 16px; }
  </style>
</head>
<body>
  <h1>智能插座控制</h1>
  
  <button id="switch" class="switch off" onclick="togglePower()">OFF</button>
  
  <div class="info">
    <p>功率: <span id="wattage">0</span> W</p>
    <p>电流: <span id="current">0</span> A</p>
    <p>电能: <span id="energy">0</span> kWh</p>
  </div>
  
  <div class="info">
    <h3>定时开关</h3>
    <div class="timer">
      <input type="time" id="timerTime">
      <select id="timerAction">
        <option value="true">开启</option>
        <option value="false">关闭</option>
      </select>
    </div>
    <label><input type="checkbox" id="timerEnabled"> 启用定时</label>
    <button onclick="setTimer()">设置</button>
  </div>
  
  <script>
    function updateStatus() {
      fetch('/api/status')
        .then(r => r.json())
        .then(d => {
          var btn = document.getElementById('switch');
          btn.className = 'switch ' + (d.power ? 'on' : 'off');
          btn.textContent = d.power ? 'ON' : 'OFF';
          document.getElementById('wattage').textContent = d.wattage.toFixed(1);
          document.getElementById('current').textContent = d.current.toFixed(2);
          document.getElementById('energy').textContent = d.energy.toFixed(3);
        });
    }
    
    function togglePower() {
      var btn = document.getElementById('switch');
      var newState = btn.classList.contains('off');
      
      fetch('/api/power', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({power: newState})
      }).then(updateStatus);
    }
    
    function setTimer() {
      var time = document.getElementById('timerTime').value.split(':');
      var data = {
        enabled: document.getElementById('timerEnabled').checked,
        hour: parseInt(time[0]),
        minute: parseInt(time[1]),
        action: document.getElementById('timerAction').value === 'true'
      };
      
      fetch('/api/timer', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify(data)
      });
    }
    
    updateStatus();
    setInterval(updateStatus, 5000);
  </script>
</body>
</html>
)rawliteral";
}
```

本节实现了智能插座的核心功能，包括远程控制、功率监测和定时开关。制作时请注意用电安全。
