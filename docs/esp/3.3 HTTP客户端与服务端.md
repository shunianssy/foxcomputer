# 3.3 HTTP客户端与服务端

HTTP是互联网最常用的应用层协议，本节将详细介绍ESP32的HTTP编程方法。

## HTTP协议基础

### HTTP请求结构

```
GET /api/data HTTP/1.1
Host: example.com
User-Agent: ESP32
Content-Type: application/json
Accept: application/json

[请求体]
```

### HTTP响应结构

```
HTTP/1.1 200 OK
Content-Type: application/json
Content-Length: 123

{"status": "success", "data": {...}}
```

### 常见HTTP方法

| 方法 | 用途 |
|------|------|
| GET | 获取资源 |
| POST | 创建资源 |
| PUT | 更新资源 |
| DELETE | 删除资源 |
| PATCH | 部分更新 |

## HTTP客户端

### 基本HTTP GET请求

```cpp
/*
 * HTTP GET请求示例
 */

#include <WiFi.h>
#include <HTTPClient.h>

const char* ssid = "Your_SSID";
const char* password = "Your_Password";

void setup() {
  Serial.begin(115200);
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi已连接");
  
  // 发送GET请求
  httpGet("http://httpbin.org/get");
}

void loop() {
  delay(10000);
}

void httpGet(const char* url) {
  HTTPClient http;
  
  Serial.printf("GET请求: %s\n", url);
  
  // 开始请求
  http.begin(url);
  
  // 添加请求头
  http.addHeader("User-Agent", "ESP32");
  http.addHeader("Accept", "application/json");
  
  // 发送请求
  int httpCode = http.GET();
  
  // 检查响应
  if (httpCode > 0) {
    Serial.printf("响应码: %d\n", httpCode);
    
    if (httpCode == HTTP_CODE_OK) {
      String payload = http.getString();
      Serial.println("响应内容:");
      Serial.println(payload);
    }
  } else {
    Serial.printf("请求失败: %s\n", http.errorToString(httpCode).c_str());
  }
  
  http.end();
}
```

### HTTP POST请求

```cpp
/*
 * HTTP POST请求示例
 */

#include <WiFi.h>
#include <HTTPClient.h>

void httpPost(const char* url, String jsonData) {
  HTTPClient http;
  
  Serial.printf("POST请求: %s\n", url);
  
  http.begin(url);
  
  // 设置请求头
  http.addHeader("Content-Type", "application/json");
  
  // 发送POST请求
  int httpCode = http.POST(jsonData);
  
  if (httpCode > 0) {
    Serial.printf("响应码: %d\n", httpCode);
    
    String response = http.getString();
    Serial.println("响应内容:");
    Serial.println(response);
  } else {
    Serial.printf("请求失败: %s\n", http.errorToString(httpCode).c_str());
  }
  
  http.end();
}

void setup() {
  Serial.begin(115200);
  
  WiFi.begin("Your_SSID", "Your_Password");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  
  // 发送JSON数据
  String jsonData = "{\"device\":\"ESP32\",\"temperature\":25.5,\"humidity\":60}";
  httpPost("http://httpbin.org/post", jsonData);
}

void loop() {}
```

### HTTP客户端类封装

```cpp
/*
 * HTTP客户端类封装
 */

#include <WiFi.h>
#include <HTTPClient.h>

class SimpleHTTPClient {
private:
  String baseUrl;
  String authToken;
  int timeout;
  
public:
  SimpleHTTPClient(String base = "") {
    baseUrl = base;
    timeout = 10000;
  }
  
  void setBaseUrl(String base) {
    baseUrl = base;
  }
  
  void setAuthToken(String token) {
    authToken = token;
  }
  
  void setTimeout(int ms) {
    timeout = ms;
  }
  
  // GET请求
  String get(String endpoint) {
    HTTPClient http;
    String url = baseUrl + endpoint;
    
    http.begin(url);
    http.setTimeout(timeout);
    
    if (authToken.length() > 0) {
      http.addHeader("Authorization", "Bearer " + authToken);
    }
    
    int code = http.GET();
    String response = "";
    
    if (code == HTTP_CODE_OK) {
      response = http.getString();
    }
    
    http.end();
    return response;
  }
  
  // POST请求
  String post(String endpoint, String body, String contentType = "application/json") {
    HTTPClient http;
    String url = baseUrl + endpoint;
    
    http.begin(url);
    http.setTimeout(timeout);
    http.addHeader("Content-Type", contentType);
    
    if (authToken.length() > 0) {
      http.addHeader("Authorization", "Bearer " + authToken);
    }
    
    int code = http.POST(body);
    String response = "";
    
    if (code > 0) {
      response = http.getString();
    }
    
    http.end();
    return response;
  }
  
  // PUT请求
  String put(String endpoint, String body) {
    HTTPClient http;
    String url = baseUrl + endpoint;
    
    http.begin(url);
    http.addHeader("Content-Type", "application/json");
    
    if (authToken.length() > 0) {
      http.addHeader("Authorization", "Bearer " + authToken);
    }
    
    int code = http.PUT(body);
    String response = "";
    
    if (code > 0) {
      response = http.getString();
    }
    
    http.end();
    return response;
  }
  
  // DELETE请求
  int del(String endpoint) {
    HTTPClient http;
    String url = baseUrl + endpoint;
    
    http.begin(url);
    
    if (authToken.length() > 0) {
      http.addHeader("Authorization", "Bearer " + authToken);
    }
    
    int code = http.sendRequest("DELETE");
    http.end();
    
    return code;
  }
};

// 使用示例
SimpleHTTPClient httpClient;

void setup() {
  Serial.begin(115200);
  
  WiFi.begin("Your_SSID", "Your_Password");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  
  httpClient.setBaseUrl("http://api.example.com");
  httpClient.setAuthToken("your_token_here");
  
  // GET请求
  String response = httpClient.get("/api/data");
  Serial.println(response);
  
  // POST请求
  String result = httpClient.post("/api/data", "{\"key\":\"value\"}");
  Serial.println(result);
}

void loop() {}
```

### JSON数据处理

```cpp
/*
 * JSON数据处理示例
 * 使用ArduinoJson库
 */

#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>

void setup() {
  Serial.begin(115200);
  
  WiFi.begin("Your_SSID", "Your_Password");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  
  // 获取JSON数据
  fetchJsonData();
  
  // 发送JSON数据
  sendJsonData();
}

void loop() {}

void fetchJsonData() {
  HTTPClient http;
  http.begin("http://httpbin.org/get");
  
  int code = http.GET();
  
  if (code == HTTP_CODE_OK) {
    String response = http.getString();
    
    // 解析JSON
    StaticJsonDocument<1024> doc;
    DeserializationError error = deserializeJson(doc, response);
    
    if (!error) {
      // 读取数据
      String origin = doc["origin"];
      String url = doc["url"];
      
      Serial.println("=== JSON解析结果 ===");
      Serial.printf("Origin: %s\n", origin.c_str());
      Serial.printf("URL: %s\n", url.c_str());
      
      // 嵌套对象
      JsonObject headers = doc["headers"];
      String host = headers["Host"];
      Serial.printf("Host: %s\n", host.c_str());
    } else {
      Serial.printf("JSON解析失败: %s\n", error.c_str());
    }
  }
  
  http.end();
}

void sendJsonData() {
  // 创建JSON文档
  StaticJsonDocument<256> doc;
  
  doc["device"] = "ESP32";
  doc["timestamp"] = millis();
  
  JsonObject data = doc.createNestedObject("data");
  data["temperature"] = 25.5;
  data["humidity"] = 60;
  data["light"] = 800;
  
  JsonArray sensors = doc.createNestedArray("sensors");
  sensors.add("DHT22");
  sensors.add("BH1750");
  
  // 序列化为字符串
  String jsonStr;
  serializeJson(doc, jsonStr);
  
  Serial.println("=== 发送JSON ===");
  Serial.println(jsonStr);
  
  // 发送HTTP POST
  HTTPClient http;
  http.begin("http://httpbin.org/post");
  http.addHeader("Content-Type", "application/json");
  
  int code = http.POST(jsonStr);
  
  if (code > 0) {
    Serial.printf("响应码: %d\n", code);
  }
  
  http.end();
}
```

## HTTP服务器

### 基本HTTP服务器

```cpp
/*
 * 基本HTTP服务器示例
 */

#include <WiFi.h>
#include <WebServer.h>

const char* ssid = "Your_SSID";
const char* password = "Your_Password";

WebServer server(80);

void setup() {
  Serial.begin(115200);
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  Serial.printf("IP: %s\n", WiFi.localIP().toString().c_str());
  
  // 注册路由处理
  server.on("/", handleRoot);
  server.on("/api/status", handleStatus);
  server.on("/api/data", HTTP_GET, handleGetData);
  server.on("/api/data", HTTP_POST, handlePostData);
  server.onNotFound(handleNotFound);
  
  // 启动服务器
  server.begin();
  Serial.println("HTTP服务器已启动");
}

void loop() {
  server.handleClient();
  delay(10);
}

// 首页处理
void handleRoot() {
  String html = "<!DOCTYPE html>"
                "<html>"
                "<head><title>ESP32 Server</title></head>"
                "<body>"
                "<h1>ESP32 HTTP Server</h1>"
                "<p>API Endpoints:</p>"
                "<ul>"
                "<li><a href='/api/status'>/api/status</a> - 系统状态</li>"
                "<li><a href='/api/data'>/api/data</a> - 数据获取</li>"
                "</ul>"
                "</body>"
                "</html>";
  
  server.send(200, "text/html", html);
}

// 状态API
void handleStatus() {
  String json = "{";
  json += "\"chip\":\"" + String(ESP.getChipModel()) + "\",";
  json += "\"free_heap\":" + String(ESP.getFreeHeap()) + ",";
  json += "\"uptime\":" + String(millis() / 1000) + ",";
  json += "\"wifi_rssi\":" + String(WiFi.RSSI());
  json += "}";
  
  server.send(200, "application/json", json);
}

// GET数据
void handleGetData() {
  // 返回传感器数据
  String json = "{";
  json += "\"temperature\":25.5,";
  json += "\"humidity\":60.0,";
  json += "\"light\":800";
  json += "}";
  
  server.send(200, "application/json", json);
}

// POST数据
void handlePostData() {
  if (server.hasArg("plain")) {
    String body = server.arg("plain");
    Serial.printf("收到POST数据: %s\n", body.c_str());
    
    // 返回响应
    server.send(200, "application/json", "{\"status\":\"ok\"}");
  } else {
    server.send(400, "application/json", "{\"error\":\"no data\"}");
  }
}

// 404处理
void handleNotFound() {
  server.send(404, "text/plain", "Not Found");
}
```

### RESTful API服务器

```cpp
/*
 * RESTful API服务器示例
 */

#include <WiFi.h>
#include <WebServer.h>
#include <ArduinoJson.h>

WebServer server(80);

// 模拟数据存储
struct Device {
  int id;
  String name;
  float temperature;
  float humidity;
  bool status;
};

Device devices[5];
int deviceCount = 0;

void setup() {
  Serial.begin(115115);
  
  WiFi.begin("Your_SSID", "Your_Password");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  Serial.printf("IP: %s\n", WiFi.localIP().toString().c_str());
  
  // 初始化数据
  initDevices();
  
  // RESTful路由
  // 设备列表
  server.on("/api/devices", HTTP_GET, getDevices);
  
  // 创建设备
  server.on("/api/devices", HTTP_POST, createDevice);
  
  // 获取单个设备
  server.on("/api/devices/[]", HTTP_GET, getDevice);
  
  // 更新设备
  server.on("/api/devices/[]", HTTP_PUT, updateDevice);
  
  // 删除设备
  server.on("/api/devices/[]", HTTP_DELETE, deleteDevice);
  
  // 启用CORS
  server.enableCORS(true);
  
  server.begin();
  Serial.println("RESTful API服务器已启动");
}

void loop() {
  server.handleClient();
  delay(10);
}

void initDevices() {
  devices[0] = {1, "Living Room", 25.5, 60.0, true};
  devices[1] = {2, "Bedroom", 23.0, 55.0, true};
  deviceCount = 2;
}

// GET /api/devices - 获取所有设备
void getDevices() {
  StaticJsonDocument<512> doc;
  JsonArray array = doc.to<JsonArray>();
  
  for (int i = 0; i < deviceCount; i++) {
    JsonObject obj = array.createNestedObject();
    obj["id"] = devices[i].id;
    obj["name"] = devices[i].name;
    obj["temperature"] = devices[i].temperature;
    obj["humidity"] = devices[i].humidity;
    obj["status"] = devices[i].status;
  }
  
  String response;
  serializeJson(doc, response);
  
  server.send(200, "application/json", response);
}

// POST /api/devices - 创建设备
void createDevice() {
  if (!server.hasArg("plain")) {
    server.send(400, "application/json", "{\"error\":\"no data\"}");
    return;
  }
  
  StaticJsonDocument<256> doc;
  deserializeJson(doc, server.arg("plain"));
  
  if (deviceCount < 5) {
    devices[deviceCount].id = deviceCount + 1;
    devices[deviceCount].name = doc["name"] | "New Device";
    devices[deviceCount].temperature = doc["temperature"] | 0;
    devices[deviceCount].humidity = doc["humidity"] | 0;
    devices[deviceCount].status = true;
    
    deviceCount++;
    
    String response;
    serializeJson(doc, response);
    server.send(201, "application/json", response);
  } else {
    server.send(507, "application/json", "{\"error\":\"storage full\"}");
  }
}

// GET /api/devices/{id} - 获取单个设备
void getDevice() {
  int id = server.pathArg(0).toInt();
  
  for (int i = 0; i < deviceCount; i++) {
    if (devices[i].id == id) {
      StaticJsonDocument<256> doc;
      doc["id"] = devices[i].id;
      doc["name"] = devices[i].name;
      doc["temperature"] = devices[i].temperature;
      doc["humidity"] = devices[i].humidity;
      doc["status"] = devices[i].status;
      
      String response;
      serializeJson(doc, response);
      
      server.send(200, "application/json", response);
      return;
    }
  }
  
  server.send(404, "application/json", "{\"error\":\"not found\"}");
}

// PUT /api/devices/{id} - 更新设备
void updateDevice() {
  int id = server.pathArg(0).toInt();
  
  for (int i = 0; i < deviceCount; i++) {
    if (devices[i].id == id) {
      StaticJsonDocument<256> doc;
      deserializeJson(doc, server.arg("plain"));
      
      if (doc.containsKey("name")) {
        devices[i].name = doc["name"].as<String>();
      }
      if (doc.containsKey("temperature")) {
        devices[i].temperature = doc["temperature"];
      }
      if (doc.containsKey("humidity")) {
        devices[i].humidity = doc["humidity"];
      }
      if (doc.containsKey("status")) {
        devices[i].status = doc["status"];
      }
      
      server.send(200, "application/json", "{\"status\":\"updated\"}");
      return;
    }
  }
  
  server.send(404, "application/json", "{\"error\":\"not found\"}");
}

// DELETE /api/devices/{id} - 删除设备
void deleteDevice() {
  int id = server.pathArg(0).toInt();
  
  for (int i = 0; i < deviceCount; i++) {
    if (devices[i].id == id) {
      // 移动后面的元素
      for (int j = i; j < deviceCount - 1; j++) {
        devices[j] = devices[j + 1];
      }
      deviceCount--;
      
      server.send(200, "application/json", "{\"status\":\"deleted\"}");
      return;
    }
  }
  
  server.send(404, "application/json", "{\"error\":\"not found\"}");
}
```
