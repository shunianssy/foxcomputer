# 1.4 项目结构与配置

本节将详细介绍ESP项目的组织结构和配置方法，帮助你建立良好的开发习惯，为后续复杂项目开发打下基础。

## Arduino项目结构

### 基本项目结构

一个典型的Arduino ESP项目结构如下：

```
my_esp_project/
|-- my_esp_project.ino    # 主程序文件（必须与文件夹同名）
|-- config.h              # 配置头文件
|-- utils.h               # 工具函数头文件
|-- utils.cpp             # 工具函数实现
|-- data/                 # SPIFFS/LittleFS数据文件
|   |-- config.json       # 配置文件
|   |-- index.html        # 网页文件
|-- lib/                  # 本地库目录
|   |-- MySensor/
|       |-- MySensor.h
|       |-- MySensor.cpp
|-- README.md             # 项目说明
```

**utils.cpp - 工具函数实现：**

```cpp
/*
 * 工具函数实现
 */

#include "utils.h"
#include "config.h"
#include <FS.h>
#ifdef ESP32
  #include <SPIFFS.h>
#endif

// 打印内存信息
void printMemoryInfo() {
  LOG_INFO("=== 内存信息 ===");
  LOG_INFO("可用堆内存: %d 字节", ESP.getFreeHeap());
  LOG_INFO("最小可用堆: %d 字节", ESP.getMinFreeHeap());
  LOG_INFO("最大分配块: %d 字节", ESP.getMaxAllocHeap());
}

// 获取可用堆内存
uint32_t getFreeHeap() {
  return ESP.getFreeHeap();
}

// 格式化时间
String formatTime(unsigned long milliseconds) {
  unsigned long seconds = milliseconds / 1000;
  unsigned long minutes = seconds / 60;
  unsigned long hours = minutes / 60;
  unsigned long days = hours / 24;
  
  char buffer[32];
  sprintf(buffer, "%lud %02lu:%02lu:%02lu", 
          days, hours % 24, minutes % 60, seconds % 60);
  return String(buffer);
}

// URL编码
String urlEncode(String str) {
  String encoded = "";
  char c;
  char code0;
  char code1;
  
  for (unsigned int i = 0; i < str.length(); i++) {
    c = str.charAt(i);
    if (c == ' ') {
      encoded += '+';
    } else if (isalnum(c)) {
      encoded += c;
    } else {
      code1 = (c & 0xf) + '0';
      if ((c & 0xf) > 9) {
        code1 = (c & 0xf) - 10 + 'A';
      }
      c = (c >> 4) & 0xf;
      code0 = c + '0';
      if (c > 9) {
        code0 = c - 10 + 'A';
      }
      encoded += '%';
      encoded += code0;
      encoded += code1;
    }
  }
  return encoded;
}

// 初始化文件系统
bool initFileSystem() {
  #ifdef ESP32
    if (!SPIFFS.begin(true)) {
      LOG_ERROR("SPIFFS挂载失败");
      return false;
    }
  #else
    if (!SPIFFS.begin()) {
      LOG_ERROR("SPIFFS挂载失败");
      return false;
    }
  #endif
  
  LOG_INFO("文件系统初始化成功");
  return true;
}

// 保存配置到文件
bool saveConfig(const String& filename, const String& data) {
  File file = SPIFFS.open("/" + filename, "w");
  if (!file) {
    LOG_ERROR("无法打开文件写入: %s", filename.c_str());
    return false;
  }
  
  size_t written = file.print(data);
  file.close();
  
  LOG_INFO("写入 %d 字节到 %s", written, filename.c_str());
  return written > 0;
}

// 从文件加载配置
String loadConfig(const String& filename) {
  File file = SPIFFS.open("/" + filename, "r");
  if (!file) {
    LOG_ERROR("无法打开文件读取: %s", filename.c_str());
    return "";
  }
  
  String data = file.readString();
  file.close();
  
  LOG_INFO("从 %s 读取 %d 字节", filename.c_str(), data.length());
  return data;
}
```

**主程序文件：**

```cpp
/*
 * ESP32 项目主程序
 * 演示完整的项目结构
 */

#include <Arduino.h>
#include "config.h"
#include "utils.h"

// 全局状态
struct AppState {
  bool wifiConnected;
  bool sensorReady;
  unsigned long startTime;
  uint32_t readCount;
} state;

void setup() {
  // 初始化串口
  Serial.begin(SERIAL_BAUD_RATE);
  delay(1000);
  
  LOG_INFO("========================================");
  LOG_INFO("项目启动");
  LOG_INFO("========================================");
  
  // 初始化文件系统
  initFileSystem();
  
  // 打印系统信息
  printSystemInfo();
  
  // 初始化硬件
  initHardware();
  
  // 初始化网络
  initNetwork();
  
  // 记录启动时间
  state.startTime = millis();
  
  LOG_INFO("初始化完成，进入主循环");
}

void loop() {
  // 主循环逻辑
  handleSensor();
  handleNetwork();
  handleSerial();
  
  // 定期打印状态
  static unsigned long lastStatusTime = 0;
  if (millis() - lastStatusTime > 30000) {
    lastStatusTime = millis();
    printStatus();
  }
}

// 初始化硬件
void initHardware() {
  LOG_INFO("初始化硬件...");
  
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  
  LOG_INFO("硬件初始化完成");
}

// 初始化网络
void initNetwork() {
  LOG_INFO("初始化网络...");
  
  // WiFi初始化代码
  // ...
  
  LOG_INFO("网络初始化完成");
}

// 处理传感器
void handleSensor() {
  static unsigned long lastRead = 0;
  
  if (millis() - lastRead >= SENSOR_READ_INTERVAL) {
    lastRead = millis();
    
    // 读取传感器
    int value = analogRead(SENSOR_PIN);
    LOG_DEBUG("传感器读数: %d", value);
    
    state.readCount++;
  }
}

// 处理网络
void handleNetwork() {
  // 网络处理逻辑
}

// 处理串口命令
void handleSerial() {
  if (Serial.available() > 0) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    cmd.toLowerCase();
    
    if (cmd == "status") {
      printStatus();
    } else if (cmd == "memory") {
      printMemoryInfo();
    } else if (cmd == "reset") {
      ESP.restart();
    }
  }
}

// 打印系统信息
void printSystemInfo() {
  LOG_INFO("=== 系统信息 ===");
  LOG_INFO("芯片型号: %s", ESP.getChipModel());
  LOG_INFO("芯片版本: %d", ESP.getChipRevision());
  LOG_INFO("CPU核心数: %d", ESP.getChipCores());
  LOG_INFO("CPU频率: %d MHz", ESP.getCpuFreqMHz());
  LOG_INFO("Flash大小: %d MB", ESP.getFlashChipSize() / 1024 / 1024);
  LOG_INFO("SDK版本: %s", ESP.getSdkVersion());
}

// 打印运行状态
void printStatus() {
  LOG_INFO("=== 运行状态 ===");
  LOG_INFO("运行时间: %s", formatTime(millis() - state.startTime).c_str());
  LOG_INFO("读取次数: %d", state.readCount);
  LOG_INFO("WiFi状态: %s", state.wifiConnected ? "已连接" : "未连接");
  LOG_INFO("可用内存: %d 字节", getFreeHeap());
}
```

## 配置管理最佳实践

### 使用Preferences库（Arduino）

```cpp
/*
 * 使用Preferences存储配置
 * 类似于Android的SharedPreferences
 */

#include <Preferences.h>

Preferences preferences;

// 保存配置
void saveConfig() {
  // 以读写模式打开命名空间
  preferences.begin("my-app", false);
  
  // 存储各种类型的数据
  preferences.putString("ssid", "MyWiFi");
  preferences.putString("password", "MyPassword");
  preferences.putInt("interval", 5000);
  preferences.putBool("enabled", true);
  preferences.putFloat("calibration", 1.234);
  
  // 关闭命名空间
  preferences.end();
  
  Serial.println("配置已保存");
}

// 读取配置
void loadConfig() {
  // 以只读模式打开
  preferences.begin("my-app", true);
  
  // 读取数据，可指定默认值
  String ssid = preferences.getString("ssid", "");
  String password = preferences.getString("password", "");
  int interval = preferences.getInt("interval", 1000);
  bool enabled = preferences.getBool("enabled", false);
  float calibration = preferences.getFloat("calibration", 1.0);
  
  preferences.end();
  
  Serial.println("=== 加载配置 ===");
  Serial.printf("SSID: %s\n", ssid.c_str());
  Serial.printf("间隔: %d ms\n", interval);
  Serial.printf("启用: %s\n", enabled ? "是" : "否");
  Serial.printf("校准值: %.3f\n", calibration);
}

// 清除配置
void clearConfig() {
  preferences.begin("my-app", false);
  preferences.clear();  // 清除当前命名空间
  // preferences.remove("ssid");  // 删除单个键
  preferences.end();
  
  Serial.println("配置已清除");
}
```

### 使用JSON配置文件

```cpp
/*
 * JSON配置文件管理
 */

#include <ArduinoJson.h>
#include <SPIFFS.h>

struct AppConfig {
  char ssid[32];
  char password[64];
  char server[64];
  int port;
  int interval;
  bool enabled;
};

// 保存配置到JSON文件
bool saveJsonConfig(const AppConfig& config) {
  // 创建JSON文档
  StaticJsonDocument<512> doc;
  
  doc["wifi"]["ssid"] = config.ssid;
  doc["wifi"]["password"] = config.password;
  doc["server"]["host"] = config.server;
  doc["server"]["port"] = config.port;
  doc["sensor"]["interval"] = config.interval;
  doc["sensor"]["enabled"] = config.enabled;
  
  // 打开文件
  File file = SPIFFS.open("/config.json", "w");
  if (!file) {
    Serial.println("无法打开配置文件");
    return false;
  }
  
  // 写入JSON
  serializeJsonPretty(doc, file);
  file.close();
  
  Serial.println("配置已保存到JSON文件");
  return true;
}

// 从JSON文件加载配置
bool loadJsonConfig(AppConfig& config) {
  File file = SPIFFS.open("/config.json", "r");
  if (!file) {
    Serial.println("配置文件不存在，使用默认值");
    return false;
  }
  
  // 解析JSON
  StaticJsonDocument<512> doc;
  DeserializationError error = deserializeJson(doc, file);
  file.close();
  
  if (error) {
    Serial.printf("JSON解析失败: %s\n", error.c_str());
    return false;
  }
  
  // 读取配置
  strlcpy(config.ssid, doc["wifi"]["ssid"] | "", sizeof(config.ssid));
  strlcpy(config.password, doc["wifi"]["password"] | "", sizeof(config.password));
  strlcpy(config.server, doc["server"]["host"] | "localhost", sizeof(config.server));
  config.port = doc["server"]["port"] | 8080;
  config.interval = doc["sensor"]["interval"] | 5000;
  config.enabled = doc["sensor"]["enabled"] | true;
  
  return true;
}

// 打印配置
void printConfig(const AppConfig& config) {
  Serial.println("=== 当前配置 ===");
  Serial.printf("WiFi SSID: %s\n", config.ssid);
  Serial.printf("服务器: %s:%d\n", config.server, config.port);
  Serial.printf("采样间隔: %d ms\n", config.interval);
  Serial.printf("传感器启用: %s\n", config.enabled ? "是" : "否");
}
```
