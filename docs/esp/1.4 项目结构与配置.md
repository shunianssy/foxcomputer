# 项目结构与配置

## 1. ESP-IDF项目结构

### 1.1 标准项目目录结构

```
my_esp_project/
├── CMakeLists.txt          # 项目CMake配置文件
├── sdkconfig               # 项目配置文件
├── sdkconfig.defaults      # 默认配置文件
├── main/                   # 主程序目录
│   ├── CMakeLists.txt      # 主程序CMake配置
│   ├── main.c              # 主程序源文件
│   ├── component.mk        # 组件Makefile（可选）
│   └── include/            # 头文件目录
│       └── main.h
├── components/             # 自定义组件目录
│   └── my_component/
│       ├── CMakeLists.txt
│       ├── include/
│       │   └── my_component.h
│       └── src/
│           └── my_component.c
├── managed_components/     # 管理的组件（idf.py）
├── build/                  # 构建输出目录
├── docs/                   # 文档目录
├── test/                   # 测试代码目录
└── README.md               # 项目说明文件
```

### 1.2 主要文件说明

#### CMakeLists.txt（项目根目录）

```cmake
# 最低CMake版本要求
cmake_minimum_required(VERSION 3.16)

# 包含ESP-IDF的构建系统
include($ENV{IDF_PATH}/tools/cmake/project.cmake)

# 项目名称
project(my_esp_project)
```

#### main/CMakeLists.txt

```cmake
# 注册组件
idf_component_register(
    SRCS "main.c"           # 源文件列表
    INCLUDE_DIRS "include"  # 头文件目录
    REQUIRES driver         # 依赖组件
)
```

#### main/main.c

```c
#include <stdio.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "esp_system.h"
#include "esp_log.h"
#include "main.h"

static const char *TAG = "MAIN";

void app_main(void)
{
    ESP_LOGI(TAG, "项目启动");
    
    // 主循环
    while (1) {
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}
```

## 3. 组件开发

### 3.1 创建自定义组件

#### 组件目录结构

```
components/
└── my_driver/
    ├── CMakeLists.txt      # 组件CMake配置
    ├── Kconfig             # 组件配置选项
    ├── include/            # 公共头文件
    │   └── my_driver.h
    ├── src/                # 源文件
    │   └── my_driver.c
    └── test/               # 测试代码
        └── test_my_driver.c
```

#### CMakeLists.txt

```cmake
idf_component_register(
    SRCS "src/my_driver.c"
    INCLUDE_DIRS "include"
    REQUIRES driver esp_timer
    PRIV_REQUIRES esp_log
)
```

#### Kconfig

```kconfig
config MY_DRIVER_ENABLED
    bool "Enable My Driver"
    default y
    help
        Enable or disable my custom driver.

config MY_DRIVER_BUFFER_SIZE
    int "Driver buffer size"
    default 256
    help
        Set the buffer size for the driver.
```

#### include/my_driver.h

```c
#ifndef MY_DRIVER_H
#define MY_DRIVER_H

#include "esp_err.h"

// 初始化驱动
esp_err_t my_driver_init(void);

// 反初始化驱动
esp_err_t my_driver_deinit(void);

// 读取数据
esp_err_t my_driver_read(uint8_t *data, size_t len);

// 写入数据
esp_err_t my_driver_write(const uint8_t *data, size_t len);

#endif // MY_DRIVER_H
```

#### src/my_driver.c

```c
#include "my_driver.h"
#include "esp_log.h"
#include "driver/gpio.h"

static const char *TAG = "MY_DRIVER";

#ifdef CONFIG_MY_DRIVER_ENABLED

// 私有变量
static bool s_initialized = false;

esp_err_t my_driver_init(void)
{
    if (s_initialized) {
        ESP_LOGW(TAG, "驱动已初始化");
        return ESP_OK;
    }
    
    ESP_LOGI(TAG, "初始化驱动");
    
    // 初始化代码
    // ...
    
    s_initialized = true;
    return ESP_OK;
}

esp_err_t my_driver_deinit(void)
{
    if (!s_initialized) {
        ESP_LOGW(TAG, "驱动未初始化");
        return ESP_OK;
    }
    
    ESP_LOGI(TAG, "反初始化驱动");
    
    // 反初始化代码
    // ...
    
    s_initialized = false;
    return ESP_OK;
}

esp_err_t my_driver_read(uint8_t *data, size_t len)
{
    if (!s_initialized) {
        ESP_LOGE(TAG, "驱动未初始化");
        return ESP_ERR_INVALID_STATE;
    }
    
    // 读取数据代码
    // ...
    
    return ESP_OK;
}

esp_err_t my_driver_write(const uint8_t *data, size_t len)
{
    if (!s_initialized) {
        ESP_LOGE(TAG, "驱动未初始化");
        return ESP_ERR_INVALID_STATE;
    }
    
    // 写入数据代码
    // ...
    
    return ESP_OK;
}

#endif // CONFIG_MY_DRIVER_ENABLED
```

### 4.2 主程序文件

```cpp
// my_arduino_project.ino

#include <Arduino.h>
#include "config.h"
#include "my_functions.h"

// 全局变量
int globalCounter = 0;

void setup() {
    Serial.begin(115200);
    Serial.println("项目启动");
    
    // 初始化代码
    initPins();
    initSensors();
}

void loop() {
    // 主循环代码
    readSensors();
    processData();
    updateOutputs();
    
    delay(1000);
}
```

// my_functions.h

#ifndef MY_FUNCTIONS_H

#endif // MY_FUNCTIONS_H

```cpp
// my_functions.cpp

#include <Arduino.h>
#include "config.h"
#include "my_functions.h"

void initPins(void) {
    pinMode(LED_PIN, OUTPUT);
    pinMode(BUTTON_PIN, INPUT_PULLUP);
    pinMode(SENSOR_PIN, INPUT);
}

void initSensors(void) {
    // 传感器初始化代码
}

void readSensors(void) {
    // 读取传感器数据
    int sensorValue = analogRead(SENSOR_PIN);
    Serial.printf("传感器值: %d\n", sensorValue);
}

void processData(void) {
    // 数据处理代码
}

void updateOutputs(void) {
    // 更新输出
    digitalWrite(LED_PIN, !digitalRead(LED_PIN));
}
```

### 5.2 platformio.ini配置

```ini
; PlatformIO项目配置文件

[env:esp32dev]
platform = espressif32
board = esp32dev
framework = arduino

; 监控配置
monitor_speed = 115200
monitor_filters = esp32_exception_decoder

; 上传配置
upload_speed = 921600
upload_port = COM3

; 编译配置
build_flags = 
    -DCORE_DEBUG_LEVEL=3
    -DBOARD_HAS_PSRAM
    -mfix-esp32-psram-cache-issue

; 库依赖
lib_deps = 
    adafruit/Adafruit SSD1306 @ ^2.5.7
    adafruit/Adafruit BME280 Library @ ^2.2.2
    knolleary/PubSubClient @ ^2.8

; 多环境配置
[env:esp32dev-debug]
extends = env:esp32dev
build_type = debug

[env:esp32dev-release]
extends = env:esp32dev
build_type = release
build_flags = 
    -Os
    -DNDEBUG
```

## 7. 构建和烧录

### 7.1 ESP-IDF构建命令

```bash
# 配置项目
idf.py menuconfig

# 设置目标芯片
idf.py set-target esp32

# 编译项目
idf.py build

# 清理构建
idf.py fullclean

# 烧录固件
idf.py -p COM3 flash

# 监视串口输出
idf.py -p COM3 monitor

# 烧录并监视
idf.py -p COM3 flash monitor

# 擦除Flash
idf.py -p COM3 erase-flash
```

### 7.2 PlatformIO构建命令

```bash
# 编译项目
pio run

# 烧录固件
pio run --target upload

# 清理构建
pio run --target clean

# 监视串口
pio device monitor

# 运行测试
pio test

# 上传文件系统
pio run --target uploadfs
```
