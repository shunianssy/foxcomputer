# 5.4 常用传感器集成

本节将介绍ESP32与常用传感器的集成方法，包括温湿度传感器、运动传感器、气体传感器等。

## 温湿度传感器

### DHT11/DHT22

```cpp
/*
 * DHT温湿度传感器示例
 */

#include <DHT.h>

#define DHT_PIN 4
#define DHT_TYPE DHT22  // DHT11, DHT22, DHT21

DHT dht(DHT_PIN, DHT_TYPE);

void setup() {
  Serial.begin(115200);
  dht.begin();
  
  Serial.println("DHT传感器已初始化");
}

void loop() {
  // 读取温度（摄氏度）
  float temp = dht.readTemperature();
  
  // 读取湿度（百分比）
  float humi = dht.readHumidity();
  
  // 检查读取是否成功
  if (isnan(temp) || isnan(humi)) {
    Serial.println("读取DHT传感器失败！");
    delay(2000);
    return;
  }
  
  // 计算热指数
  float heatIndex = dht.computeHeatIndex(temp, humi, false);
  
  Serial.printf("温度: %.1f C\n", temp);
  Serial.printf("湿度: %.1f %%\n", humi);
  Serial.printf("热指数: %.1f C\n", heatIndex);
  
  delay(2000);
}
```

## 气压传感器

### BMP280

```cpp
/*
 * BMP280气压传感器示例
 */

#include <Wire.h>
#include <Adafruit_BMP280.h>

#define BMP280_ADDRESS 0x76

Adafruit_BMP280 bmp;

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);
  
  if (!bmp.begin(BMP280_ADDRESS)) {
    Serial.println("BMP280未找到！");
    while (1);
  }
  
  // 配置参数
  bmp.setSampling(Adafruit_BMP280::MODE_NORMAL,
                  Adafruit_BMP280::SAMPLING_X2,
                  Adafruit_BMP280::SAMPLING_X16,
                  Adafruit_BMP280::FILTER_X16,
                  Adafruit_BMP280::STANDBY_MS_500);
  
  Serial.println("BMP280已初始化");
}

void loop() {
  float temp = bmp.readTemperature();
  float pressure = bmp.readPressure() / 100.0F;  // hPa
  float altitude = bmp.readAltitude(1013.25);    // 海平面气压
  
  Serial.printf("温度: %.1f C\n", temp);
  Serial.printf("气压: %.1f hPa\n", pressure);
  Serial.printf("海拔: %.1f m\n", altitude);
  
  delay(2000);
}
```

## 运动传感器

### MPU6050

```cpp
/*
 * MPU6050六轴传感器示例
 */

#include <Wire.h>
#include <MPU6050.h>

MPU6050 mpu;

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);
  
  mpu.initialize();
  
  if (!mpu.testConnection()) {
    Serial.println("MPU6050连接失败！");
    while (1);
  }
  
  Serial.println("MPU6050已初始化");
}

void loop() {
  int16_t ax, ay, az;
  int16_t gx, gy, gz;
  
  // 读取加速度和陀螺仪
  mpu.getMotion6(&ax, &ay, &az, &gx, &gy, &gz);
  
  // 转换为物理单位
  float accelX = ax / 16384.0;  // g
  float accelY = ay / 16384.0;
  float accelZ = az / 16384.0;
  
  float gyroX = gx / 131.0;     // deg/s
  float gyroY = gy / 131.0;
  float gyroZ = gz / 131.0;
  
  Serial.printf("加速度: X=%.2f, Y=%.2f, Z=%.2f g\n", accelX, accelY, accelZ);
  Serial.printf("角速度: X=%.2f, Y=%.2f, Z=%.2f deg/s\n", gyroX, gyroY, gyroZ);
  
  // 计算倾斜角度
  float pitch = atan2(accelX, sqrt(accelY * accelY + accelZ * accelZ)) * 180 / PI;
  float roll = atan2(accelY, sqrt(accelX * accelX + accelZ * accelZ)) * 180 / PI;
  
  Serial.printf("俯仰角: %.1f, 横滚角: %.1f\n", pitch, roll);
  
  delay(500);
}
```

## 气体传感器

### MQ系列

```cpp
/*
 * MQ气体传感器示例
 * MQ-2: 可燃气体
 * MQ-135: 空气质量
 */

#define MQ_PIN 34
#define MQ_TYPE "MQ-2"

// 传感器参数
#define RL_VALUE 5.0    // 负载电阻(KΩ)
#define RO_VALUE 10.0   // 清洁空气中的Ro值(KΩ)

float mqRead(int pin) {
  int i;
  float rs = 0;
  
  // 多次采样取平均
  for (i = 0; i < 100; i++) {
    rs += analogRead(pin);
    delay(10);
  }
  
  rs = rs / 100.0;
  
  // 计算Rs
  float vrl = rs * 3.3 / 4095.0;
  rs = (3.3 - vrl) / vrl * RL_VALUE;
  
  return rs;
}

float mqGetRatio(float rs_ro_ratio) {
  // 根据传感器类型计算气体浓度
  // 这里使用简化的公式，实际需要根据数据手册校准
  
  if (MQ_TYPE == "MQ-2") {
    // LPG, CO, Smoke
    return pow(10, ((log10(rs_ro_ratio) - 0.5) / -0.45));
  }
  
  return rs_ro_ratio;
}

void setup() {
  Serial.begin(115200);
  analogSetAttenuation(ADC_11db);
  
  Serial.println("MQ传感器预热中...");
  delay(60000);  // 预热60秒
  Serial.println("预热完成");
}

void loop() {
  float rs = mqRead(MQ_PIN);
  float ratio = rs / RO_VALUE;
  float ppm = mqGetRatio(ratio);
  
  Serial.printf("Rs: %.2f KΩ\n", rs);
  Serial.printf("Rs/Ro: %.2f\n", ratio);
  Serial.printf("浓度: %.1f ppm\n", ppm);
  
  delay(1000);
}
```

## 距离传感器

### HC-SR04超声波

```cpp
/*
 * HC-SR04超声波测距示例
 */

#define TRIG_PIN 4
#define ECHO_PIN 5

#define SOUND_SPEED 0.034  // cm/us

void setup() {
  Serial.begin(115200);
  
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  
  Serial.println("超声波测距已初始化");
}

void loop() {
  float distance = measureDistance();
  
  Serial.printf("距离: %.1f cm\n", distance);
  
  delay(500);
}

float measureDistance() {
  // 发送触发信号
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  
  // 测量回波时间
  unsigned long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  
  // 计算距离
  if (duration == 0) {
    return -1;  // 超时
  }
  
  float distance = duration * SOUND_SPEED / 2;
  
  return distance;
}
```

## 传感器管理类

```cpp
/*
 * 传感器管理类
 * 统一管理多个传感器
 */

#include <Arduino.h>
#include <Wire.h>

class SensorManager {
private:
  struct Sensor {
    String name;
    float value;
    unsigned long lastRead;
    unsigned long interval;
    float (*readFunc)();
  };
  
  Sensor sensors[10];
  int sensorCount = 0;
  
public:
  // 添加传感器
  void addSensor(String name, unsigned long interval, float (*readFunc)()) {
    if (sensorCount < 10) {
      sensors[sensorCount].name = name;
      sensors[sensorCount].interval = interval;
      sensors[sensorCount].readFunc = readFunc;
      sensors[sensorCount].lastRead = 0;
      sensors[sensorCount].value = 0;
      sensorCount++;
    }
  }
  
  // 更新传感器
  void update() {
    unsigned long now = millis();
    
    for (int i = 0; i < sensorCount; i++) {
      if (now - sensors[i].lastRead >= sensors[i].interval) {
        sensors[i].value = sensors[i].readFunc();
        sensors[i].lastRead = now;
      }
    }
  }
  
  // 获取传感器值
  float getValue(String name) {
    for (int i = 0; i < sensorCount; i++) {
      if (sensors[i].name == name) {
        return sensors[i].value;
      }
    }
    return NAN;
  }
  
  // 打印所有传感器值
  void printAll() {
    Serial.println("=== 传感器数据 ===");
    for (int i = 0; i < sensorCount; i++) {
      Serial.printf("%s: %.2f\n", sensors[i].name.c_str(), sensors[i].value);
    }
  }
};

// 传感器读取函数
float readTemperature() {
  return 25.0 + random(-20, 20) / 10.0;
}

float readHumidity() {
  return 60.0 + random(-10, 10);
}

float readLight() {
  return random(100, 1000);
}

SensorManager sensorMgr;

void setup() {
  Serial.begin(115200);
  
  // 添加传感器
  sensorMgr.addSensor("温度", 2000, readTemperature);
  sensorMgr.addSensor("湿度", 2000, readHumidity);
  sensorMgr.addSensor("光照", 5000, readLight);
}

void loop() {
  sensorMgr.update();
  
  static unsigned long lastPrint = 0;
  if (millis() - lastPrint > 5000) {
    lastPrint = millis();
    sensorMgr.printAll();
  }
  
  delay(100);
}
```
