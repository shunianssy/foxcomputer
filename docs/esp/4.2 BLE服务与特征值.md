# 4.2 BLE服务与特征值

本节将详细介绍BLE服务和特征值的创建与使用，这是BLE开发的核心内容。

## BLE服务创建

### 基本BLE服务器

```cpp
/*
 * BLE基本服务器示例
 * 创建一个简单的BLE外围设备
 */

#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEServer.h>

// 服务和特征值UUID
#define SERVICE_UUID           "4fafc201-1fb5-459e-8fcc-c5c9c331914b"
#define CHARACTERISTIC_UUID    "beb5483e-36e1-4688-b7f5-ea07361b26a8"

BLEServer* pServer = NULL;
BLECharacteristic* pCharacteristic = NULL;
bool deviceConnected = false;

// 服务器回调类
class MyServerCallbacks: public BLEServerCallbacks {
  void onConnect(BLEServer* pServer) {
    deviceConnected = true;
    Serial.println("设备已连接");
  };
  
  void onDisconnect(BLEServer* pServer) {
    deviceConnected = false;
    Serial.println("设备已断开");
  }
};

void setup() {
  Serial.begin(115200);
  
  // 初始化BLE
  BLEDevice::init("ESP32-BLE");  // 设备名称
  
  // 创建BLE服务器
  pServer = BLEDevice::createServer();
  pServer->setCallbacks(new MyServerCallbacks());
  
  // 创建BLE服务
  BLEService *pService = pServer->createService(SERVICE_UUID);
  
  // 创建特征值
  pCharacteristic = pService->createCharacteristic(
    CHARACTERISTIC_UUID,
    BLECharacteristic::PROPERTY_READ |
    BLECharacteristic::PROPERTY_WRITE |
    BLECharacteristic::PROPERTY_NOTIFY
  );
  
  // 设置特征值初始值
  pCharacteristic->setValue("Hello World");
  
  // 启动服务
  pService->start();
  
  // 开始广播
  BLEAdvertising *pAdvertising = BLEDevice::getAdvertising();
  pAdvertising->addServiceUUID(SERVICE_UUID);
  pAdvertising->setScanResponse(false);
  pAdvertising->setMinPreferred(0x06);  // 设置广播间隔
  pAdvertising->setMinPreferred(0x12);
  BLEDevice::startAdvertising();
  
  Serial.println("BLE服务器已启动，等待连接...");
}

void loop() {
  // 连接状态下发送通知
  if (deviceConnected) {
    static int counter = 0;
    String value = "Counter: " + String(counter++);
    pCharacteristic->setValue(value.c_str());
    pCharacteristic->notify();  // 发送通知
    
    delay(3000);  // 每3秒发送一次
  }
  
  delay(10);
}
```

### 多服务多特征值

```cpp
/*
 * 多服务和特征值示例
 */

#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEServer.h>

// 服务UUID
#define SERVICE_UUID_ENVIRONMENT  "181A"  // 环境感知服务（标准UUID）
#define SERVICE_UUID_DEVICE       "180A"  // 设备信息服务

// 特征值UUID
#define CHAR_UUID_TEMPERATURE     "2A6E"  // 温度
#define CHAR_UUID_HUMIDITY        "2A6F"  // 湿度
#define CHAR_UUID_PRESSURE        "2A6D"  // 气压
#define CHAR_UUID_MANUFACTURER    "2A29"  // 制造商名称
#define CHAR_UUID_MODEL           "2A24"  // 型号名称

BLEServer* pServer;
BLEService* pEnvService;
BLEService* pDevService;

BLECharacteristic* pTempChar;
BLECharacteristic* pHumidityChar;
BLECharacteristic* pPressureChar;
BLECharacteristic* pManufacturerChar;
BLECharacteristic* pModelChar;

void setup() {
  Serial.begin(115200);
  
  // 初始化BLE
  BLEDevice::init("ESP32-Environment");
  
  // 创建服务器
  pServer = BLEDevice::createServer();
  
  // 创建环境感知服务
  pEnvService = pServer->createService(SERVICE_UUID_ENVIRONMENT);
  
  // 创建温度特征值
  pTempChar = pEnvService->createCharacteristic(
    CHAR_UUID_TEMPERATURE,
    BLECharacteristic::PROPERTY_READ |
    BLECharacteristic::PROPERTY_NOTIFY
  );
  
  // 创建湿度特征值
  pHumidityChar = pEnvService->createCharacteristic(
    CHAR_UUID_HUMIDITY,
    BLECharacteristic::PROPERTY_READ |
    BLECharacteristic::PROPERTY_NOTIFY
  );
  
  // 创建气压特征值
  pPressureChar = pEnvService->createCharacteristic(
    CHAR_UUID_PRESSURE,
    BLECharacteristic::PROPERTY_READ |
    BLECharacteristic::PROPERTY_NOTIFY
  );
  
  // 创建设备信息服务
  pDevService = pServer->createService(SERVICE_UUID_DEVICE);
  
  // 创建制造商特征值
  pManufacturerChar = pDevService->createCharacteristic(
    CHAR_UUID_MANUFACTURER,
    BLECharacteristic::PROPERTY_READ
  );
  pManufacturerChar->setValue("Espressif");
  
  // 创建型号特征值
  pModelChar = pDevService->createCharacteristic(
    CHAR_UUID_MODEL,
    BLECharacteristic::PROPERTY_READ
  );
  pModelChar->setValue("ESP32");
  
  // 启动服务
  pEnvService->start();
  pDevService->start();
  
  // 开始广播
  BLEAdvertising* pAdvertising = BLEDevice::getAdvertising();
  pAdvertising->addServiceUUID(SERVICE_UUID_ENVIRONMENT);
  pAdvertising->addServiceUUID(SERVICE_UUID_DEVICE);
  BLEDevice::startAdvertising();
  
  Serial.println("BLE多服务服务器已启动");
}

void loop() {
  // 更新传感器数据
  updateSensorData();
  
  delay(10000);  // 每10秒更新一次
}

void updateSensorData() {
  // 模拟传感器数据
  int16_t temperature = 25 * 100;  // 25.00 C (单位0.01C)
  uint16_t humidity = 60 * 100;    // 60.00% (单位0.01%)
  uint32_t pressure = 101325;      // 1013.25 hPa
  
  // 设置特征值（小端序）
  pTempChar->setValue((uint8_t*)&temperature, sizeof(temperature));
  pHumidityChar->setValue((uint8_t*)&humidity, sizeof(humidity));
  pPressureChar->setValue((uint8_t*)&pressure, sizeof(pressure));
  
  // 发送通知
  pTempChar->notify();
  pHumidityChar->notify();
  pPressureChar->notify();
  
  Serial.println("传感器数据已更新");
}
```

## 特征值回调

### 读写回调

```cpp
/*
 * 特征值读写回调示例
 */

#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEServer.h>

#define SERVICE_UUID        "4fafc201-1fb5-459e-8fcc-c5c9c331914b"
#define CHAR_UUID_WRITE     "beb5483e-36e1-4688-b7f5-ea07361b26a8"
#define CHAR_UUID_READ      "beb5483e-36e1-4688-b7f5-ea07361b26a9"

// 特征值回调类
class MyCallbacks: public BLECharacteristicCallbacks {
  void onWrite(BLECharacteristic *pCharacteristic) {
    String value = pCharacteristic->getValue().c_str();
    Serial.printf("写入数据: %s\n", value.c_str());
    
    // 处理写入的数据
    if (value.length() > 0) {
      handleCommand(value);
    }
  }
  
  void onRead(BLECharacteristic *pCharacteristic) {
    Serial.println("读取数据");
    
    // 动态生成读取数据
    String status = getDeviceStatus();
    pCharacteristic->setValue(status.c_str());
  }
};

BLECharacteristic* pWriteChar;
BLECharacteristic* pReadChar;

void setup() {
  Serial.begin(115200);
  
  BLEDevice::init("ESP32-BLE-Callback");
  
  BLEServer* pServer = BLEDevice::createServer();
  BLEService* pService = pServer->createService(SERVICE_UUID);
  
  // 创建可写特征值
  pWriteChar = pService->createCharacteristic(
    CHAR_UUID_WRITE,
    BLECharacteristic::PROPERTY_WRITE
  );
  pWriteChar->setCallbacks(new MyCallbacks());
  
  // 创建可读特征值
  pReadChar = pService->createCharacteristic(
    CHAR_UUID_READ,
    BLECharacteristic::PROPERTY_READ
  );
  pReadChar->setCallbacks(new MyCallbacks());
  
  pService->start();
  
  BLEAdvertising* pAdvertising = BLEDevice::getAdvertising();
  pAdvertising->addServiceUUID(SERVICE_UUID);
  BLEDevice::startAdvertising();
  
  Serial.println("BLE回调示例已启动");
}

void loop() {
  delay(10);
}

void handleCommand(String cmd) {
  cmd.trim();
  cmd.toUpperCase();
  
  if (cmd == "LED_ON") {
    digitalWrite(2, HIGH);
    Serial.println("LED已打开");
  } else if (cmd == "LED_OFF") {
    digitalWrite(2, LOW);
    Serial.println("LED已关闭");
  } else if (cmd.startsWith("SET_INTERVAL ")) {
    int interval = cmd.substring(13).toInt();
    Serial.printf("间隔设置为: %d\n", interval);
  }
}

String getDeviceStatus() {
  return "LED:" + String(digitalRead(2) ? "ON" : "OFF") + 
         ",UPTIME:" + String(millis() / 1000);
}
```

## 通知与指示

### 通知实现

```cpp
/*
 * BLE通知示例
 * 主动推送数据到客户端
 */

#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEServer.h>

#define SERVICE_UUID        "4fafc201-1fb5-459e-8fcc-c5c9c331914b"
#define CHAR_UUID_NOTIFY    "beb5483e-36e1-4688-b7f5-ea07361b26a8"

BLECharacteristic* pNotifyChar;
bool deviceConnected = false;

class MyServerCallbacks: public BLEServerCallbacks {
  void onConnect(BLEServer* pServer) {
    deviceConnected = true;
    Serial.println("设备已连接");
  };
  
  void onDisconnect(BLEServer* pServer) {
    deviceConnected = false;
    Serial.println("设备已断开");
    // 重新开始广播
    BLEDevice::startAdvertising();
  }
};

void setup() {
  Serial.begin(115200);
  
  BLEDevice::init("ESP32-Notify");
  
  BLEServer* pServer = BLEDevice::createServer();
  pServer->setCallbacks(new MyServerCallbacks());
  
  BLEService* pService = pServer->createService(SERVICE_UUID);
  
  // 创建支持通知的特征值
  pNotifyChar = pService->createCharacteristic(
    CHAR_UUID_NOTIFY,
    BLECharacteristic::PROPERTY_NOTIFY
  );
  
  // 创建描述符（CCCD）
  pNotifyChar->addDescriptor(new BLE2902());
  
  pService->start();
  
  BLEAdvertising* pAdvertising = BLEDevice::getAdvertising();
  pAdvertising->addServiceUUID(SERVICE_UUID);
  BLEDevice::startAdvertising();
  
  Serial.println("BLE通知示例已启动");
}

void loop() {
  if (deviceConnected) {
    // 发送传感器数据通知
    sendSensorNotification();
    delay(5000);  // 每5秒发送一次
  }
  
  delay(10);
}

void sendSensorNotification() {
  // 创建数据结构
  struct SensorData {
    float temperature;
    float humidity;
    uint32_t timestamp;
  };
  
  SensorData data;
  data.temperature = 25.0 + random(-20, 20) / 10.0;
  data.humidity = 60.0 + random(-10, 10);
  data.timestamp = millis();
  
  // 设置特征值并发送通知
  pNotifyChar->setValue((uint8_t*)&data, sizeof(data));
  pNotifyChar->notify();
  
  Serial.printf("发送通知: T=%.1f, H=%.1f\n", 
                data.temperature, data.humidity);
}
```

## 自定义服务示例

### 智能灯服务

```cpp
/*
 * 智能灯BLE服务示例
 */

#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEServer.h>

// 自定义服务UUID
#define LIGHT_SERVICE_UUID          "0000FFE0-0000-1000-8000-00805F9B34FB"
#define LIGHT_STATE_CHAR_UUID       "0000FFE1-0000-1000-8000-00805F9B34FB"
#define LIGHT_BRIGHTNESS_CHAR_UUID  "0000FFE2-0000-1000-8000-00805F9B34FB"
#define LIGHT_COLOR_CHAR_UUID       "0000FFE3-0000-1000-8000-00805F9B34FB"

// 灯光状态
struct LightState {
  bool power;
  uint8_t brightness;  // 0-100
  uint8_t color[3];    // RGB
} lightState;

BLECharacteristic* pStateChar;
BLECharacteristic* pBrightnessChar;
BLECharacteristic* pColorChar;

// 状态特征值回调
class StateCallbacks: public BLECharacteristicCallbacks {
  void onWrite(BLECharacteristic *pCharacteristic) {
    uint8_t* data = pCharacteristic->getData();
    lightState.power = data[0] != 0;
    
    Serial.printf("灯光状态: %s\n", lightState.power ? "开" : "关");
    updateLight();
  }
};

// 亮度特征值回调
class BrightnessCallbacks: public BLECharacteristicCallbacks {
  void onWrite(BLECharacteristic *pCharacteristic) {
    uint8_t* data = pCharacteristic->getData();
    lightState.brightness = data[0];
    
    Serial.printf("亮度: %d%%\n", lightState.brightness);
    updateLight();
  }
};

// 颜色特征值回调
class ColorCallbacks: public BLECharacteristicCallbacks {
  void onWrite(BLECharacteristic *pCharacteristic) {
    uint8_t* data = pCharacteristic->getData();
    lightState.color[0] = data[0];  // R
    lightState.color[1] = data[1];  // G
    lightState.color[2] = data[2];  // B
    
    Serial.printf("颜色: R=%d, G=%d, B=%d\n", 
                  lightState.color[0],
                  lightState.color[1],
                  lightState.color[2]);
    updateLight();
  }
};

void setup() {
  Serial.begin(115200);
  
  // 初始化灯光状态
  lightState = {true, 100, {255, 255, 255}};
  
  BLEDevice::init("ESP32-SmartLight");
  
  BLEServer* pServer = BLEDevice::createServer();
  BLEService* pService = pServer->createService(LIGHT_SERVICE_UUID);
  
  // 创建状态特征值
  pStateChar = pService->createCharacteristic(
    LIGHT_STATE_CHAR_UUID,
    BLECharacteristic::PROPERTY_READ |
    BLECharacteristic::PROPERTY_WRITE |
    BLECharacteristic::PROPERTY_NOTIFY
  );
  pStateChar->setCallbacks(new StateCallbacks());
  pStateChar->addDescriptor(new BLE2902());
  
  // 创建亮度特征值
  pBrightnessChar = pService->createCharacteristic(
    LIGHT_BRIGHTNESS_CHAR_UUID,
    BLECharacteristic::PROPERTY_READ |
    BLECharacteristic::PROPERTY_WRITE |
    BLECharacteristic::PROPERTY_NOTIFY
  );
  pBrightnessChar->setCallbacks(new BrightnessCallbacks());
  pBrightnessChar->addDescriptor(new BLE2902());
  
  // 创建颜色特征值
  pColorChar = pService->createCharacteristic(
    LIGHT_COLOR_CHAR_UUID,
    BLECharacteristic::PROPERTY_READ |
    BLECharacteristic::PROPERTY_WRITE |
    BLECharacteristic::PROPERTY_NOTIFY
  );
  pColorChar->setCallbacks(new ColorCallbacks());
  pColorChar->addDescriptor(new BLE2902());
  
  pService->start();
  
  BLEAdvertising* pAdvertising = BLEDevice::getAdvertising();
  pAdvertising->addServiceUUID(LIGHT_SERVICE_UUID);
  BLEDevice::startAdvertising();
  
  Serial.println("智能灯BLE服务已启动");
}

void loop() {
  delay(10);
}

void updateLight() {
  // 更新实际灯光输出
  // 这里使用PWM控制RGB LED
  
  if (lightState.power) {
    // 计算PWM值
    int r = lightState.color[0] * lightState.brightness / 100;
    int g = lightState.color[1] * lightState.brightness / 100;
    int b = lightState.color[2] * lightState.brightness / 100;
    
    ledcWrite(0, r);  // 红色通道
    ledcWrite(1, g);  // 绿色通道
    ledcWrite(2, b);  // 蓝色通道
  } else {
    ledcWrite(0, 0);
    ledcWrite(1, 0);
    ledcWrite(2, 0);
  }
  
  // 更新特征值
  uint8_t state = lightState.power ? 1 : 0;
  pStateChar->setValue(&state, 1);
  pBrightnessChar->setValue(&lightState.brightness, 1);
  pColorChar->setValue(lightState.color, 3);
}
```
