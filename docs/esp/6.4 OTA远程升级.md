# 6.4 OTA远程升级

OTA（Over-The-Air）升级允许通过WiFi远程更新ESP32的固件，无需物理连接。本节将详细介绍OTA升级的实现方法。

## OTA概述

### OTA优势

- 无需物理连接即可更新
- 支持远程维护
- 减少维护成本
- 提高产品可维护性

### OTA类型

| 类型 | 说明 |
|------|------|
| Arduino OTA | Arduino IDE直接上传 |
| Web OTA | 通过Web页面上传 |
| HTTP OTA | 从服务器下载固件 |
| MQTT OTA | 通过MQTT下发固件 |

## Arduino OTA

### 基本配置

```cpp
/*
 * Arduino OTA示例
 * 支持Arduino IDE直接上传
 */

#include <WiFi.h>
#include <ArduinoOTA.h>

const char* ssid = "Your_SSID";
const char* password = "Your_Password";

void setup() {
  Serial.begin(115200);
  
  // 连接WiFi
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.printf("\nIP: %s\n", WiFi.localIP().toString().c_str());
  
  // 配置OTA
  ArduinoOTA.setHostname("ESP32-OTA");  // 设备名称
  ArduinoOTA.setPassword("123456");      // OTA密码
  
  // OTA开始回调
  ArduinoOTA.onStart([]() {
    String type;
    if (ArduinoOTA.getCommand() == U_FLASH) {
      type = "sketch";
    } else {
      type = "filesystem";
    }
    Serial.println("开始OTA更新: " + type);
  });
  
  // OTA结束回调
  ArduinoOTA.onEnd([]() {
    Serial.println("\nOTA更新完成");
  });
  
  // OTA进度回调
  ArduinoOTA.onProgress([](unsigned int progress, unsigned int total) {
    Serial.printf("进度: %u%%\r", (progress / (total / 100)));
  });
  
  // OTA错误回调
  ArduinoOTA.onError([](ota_error_t error) {
    Serial.printf("OTA错误[%u]: ", error);
    if (error == OTA_AUTH_ERROR) {
      Serial.println("认证失败");
    } else if (error == OTA_BEGIN_ERROR) {
      Serial.println("开始失败");
    } else if (error == OTA_CONNECT_ERROR) {
      Serial.println("连接失败");
    } else if (error == OTA_RECEIVE_ERROR) {
      Serial.println("接收失败");
    } else if (error == OTA_END_ERROR) {
      Serial.println("结束失败");
    }
  });
  
  // 启动OTA
  ArduinoOTA.begin();
  
  Serial.println("OTA已就绪");
  Serial.printf("设备名: %s\n", "ESP32-OTA");
}

void loop() {
  ArduinoOTA.handle();
  
  // 其他任务
  delay(10);
}
```

### 使用方法

1. 将上述代码上传到ESP32
2. 打开Arduino IDE
3. 在"工具" -> "端口"中选择ESP32的网络端口
4. 点击上传按钮

## Web OTA

### Web页面上传固件

```cpp
/*
 * Web OTA示例
 * 通过Web页面上传固件
 */

#include <WiFi.h>
#include <WebServer.h>
#include <Update.h>

const char* ssid = "Your_SSID";
const char* password = "Your_Password";

WebServer server(80);

void setup() {
  Serial.begin(115200);
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  Serial.printf("IP: %s\n", WiFi.localIP().toString().c_str());
  
  // OTA页面
  server.on("/", HTTP_GET, []() {
    server.sendHeader("Connection", "close");
    server.send(200, "text/html", otaPage);
  });
  
  // OTA上传处理
  server.on("/update", HTTP_POST, []() {
    server.sendHeader("Connection", "close");
    server.send(200, "text/plain", (Update.hasError()) ? "FAIL" : "OK");
    ESP.restart();
  }, []() {
    HTTPUpload& upload = server.upload();
    
    if (upload.status == UPLOAD_FILE_START) {
      Serial.printf("开始更新: %s\n", upload.filename.c_str());
      
      if (!Update.begin(UPDATE_SIZE_UNKNOWN)) {
        Update.printError(Serial);
      }
    } else if (upload.status == UPLOAD_FILE_WRITE) {
      if (Update.write(upload.buf, upload.currentSize) != upload.currentSize) {
        Update.printError(Serial);
      }
    } else if (upload.status == UPLOAD_FILE_END) {
      if (Update.end(true)) {
        Serial.printf("更新成功: %u bytes\n", upload.totalSize);
      } else {
        Update.printError(Serial);
      }
    }
  });
  
  server.begin();
  Serial.println("Web OTA已启动");
}

void loop() {
  server.handleClient();
  delay(10);
}
```

## HTTP OTA

### 从服务器下载固件

```cpp
/*
 * HTTP OTA示例
 * 从HTTP服务器下载固件
 */

#include <WiFi.h>
#include <HTTPClient.h>
#include <Update.h>

const char* ssid = "Your_SSID";
const char* password = "Your_Password";

// 固件服务器地址
const char* firmwareUrl = "http://your-server.com/firmware.bin";
const char* versionUrl = "http://your-server.com/version.txt";

String currentVersion = "1.0.0";

void setup() {
  Serial.begin(115200);
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  
  Serial.printf("当前版本: %s\n", currentVersion.c_str());
  
  // 检查更新
  checkUpdate();
}

void loop() {
  // 定期检查更新
  static unsigned long lastCheck = 0;
  if (millis() - lastCheck > 3600000) {  // 每小时检查
    lastCheck = millis();
    checkUpdate();
  }
  
  delay(10000);
}

void checkUpdate() {
  Serial.println("检查更新...");
  
  HTTPClient http;
  http.begin(versionUrl);
  
  int httpCode = http.GET();
  
  if (httpCode == HTTP_CODE_OK) {
    String latestVersion = http.getString();
    latestVersion.trim();
    
    Serial.printf("最新版本: %s\n", latestVersion.c_str());
    
    if (latestVersion != currentVersion) {
      Serial.println("发现新版本，开始下载...");
      downloadUpdate();
    } else {
      Serial.println("已是最新版本");
    }
  }
  
  http.end();
}

void downloadUpdate() {
  HTTPClient http;
  http.begin(firmwareUrl);
  
  int httpCode = http.GET();
  
  if (httpCode == HTTP_CODE_OK) {
    int contentLength = http.getSize();
    
    if (contentLength > 0) {
      Serial.printf("固件大小: %d bytes\n", contentLength);
      
      if (Update.begin(contentLength)) {
        WiFiClient* client = http.getStreamPtr();
        
        size_t written = Update.writeStream(*client);
        
        if (written == contentLength) {
          Serial.println("写入完成");
          
          if (Update.end()) {
            Serial.println("更新成功，重启...");
            ESP.restart();
          } else {
            Update.printError(Serial);
          }
        } else {
          Serial.printf("写入不完整: %d / %d\n", written, contentLength);
        }
      } else {
        Update.printError(Serial);
      }
    }
  }
  
  http.end();
}
```

## MQTT OTA

### 通过MQTT下发固件

```cpp
/*
 * MQTT OTA示例
 * 通过MQTT接收固件更新
 */

#include <WiFi.h>
#include <PubSubClient.h>
#include <Update.h>

const char* ssid = "Your_SSID";
const char* password = "Your_Password";

const char* mqtt_server = "broker.emqx.io";
const int mqtt_port = 1883;

WiFiClient espClient;
PubSubClient mqtt(espClient);

String deviceId = "ESP32-" + String((uint32_t)ESP.getEfuseMac(), HEX);
String firmwareTopic = "device/" + deviceId + "/firmware";
String commandTopic = "device/" + deviceId + "/command";

// 固件接收缓冲
uint8_t firmwareBuffer[4096];
int firmwareIndex = 0;
int firmwareSize = 0;
bool updating = false;

void setup() {
  Serial.begin(115200);
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  
  mqtt.setServer(mqtt_server, mqtt_port);
  mqtt.setCallback(mqttCallback);
  
  connectMQTT();
}

void loop() {
  if (!mqtt.connected()) {
    connectMQTT();
  }
  mqtt.loop();
  
  delay(10);
}

void connectMQTT() {
  if (mqtt.connect(deviceId.c_str())) {
    Serial.println("MQTT已连接");
    mqtt.subscribe(commandTopic.c_str());
    mqtt.subscribe(firmwareTopic.c_str());
  }
}

void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String topicStr = String(topic);
  
  if (topicStr == commandTopic) {
    handleCommand(payload, length);
  } else if (topicStr == firmwareTopic) {
    handleFirmware(payload, length);
  }
}

void handleCommand(byte* payload, unsigned int length) {
  String cmd = String((char*)payload, length);
  
  if (cmd == "check_update") {
    // 请求检查更新
    mqtt.publish(("device/" + deviceId + "/status").c_str(), 
                 ("version:1.0.0,ready_for_update").c_str());
  } else if (cmd.startsWith("update_size:")) {
    // 开始更新
    firmwareSize = cmd.substring(12).toInt();
    firmwareIndex = 0;
    updating = true;
    
    if (Update.begin(firmwareSize)) {
      Serial.printf("开始OTA更新，大小: %d\n", firmwareSize);
    } else {
      updating = false;
      Update.printError(Serial);
    }
  }
}

void handleFirmware(byte* payload, unsigned int length) {
  if (!updating) return;
  
  // 写入固件数据
  if (Update.write(payload, length) == length) {
    firmwareIndex += length;
    
    // 报告进度
    int progress = (firmwareIndex * 100) / firmwareSize;
    String status = "progress:" + String(progress);
    mqtt.publish(("device/" + deviceId + "/status").c_str(), status.c_str());
    
    // 检查是否完成
    if (firmwareIndex >= firmwareSize) {
      if (Update.end(true)) {
        Serial.println("OTA更新成功，重启...");
        mqtt.publish(("device/" + deviceId + "/status").c_str(), "update_success");
        delay(1000);
        ESP.restart();
      } else {
        Update.printError(Serial);
        mqtt.publish(("device/" + deviceId + "/status").c_str(), "update_failed");
      }
      updating = false;
    }
  } else {
    Update.printError(Serial);
    updating = false;
  }
}
```

## OTA管理类

```cpp
/*
 * OTA管理类
 * 统一管理各种OTA方式
 */

#include <ArduinoOTA.h>
#include <WebServer.h>
#include <HTTPClient.h>
#include <Update.h>

class OTAManager {
private:
  String deviceName;
  String currentVersion;
  String firmwareUrl;
  WebServer* server;
  
  void (*onStartCallback)();
  void (*onEndCallback)();
  void (*onProgressCallback)(int);
  void (*onErrorCallback)(String);
  
public:
  OTAManager(String name, String version) {
    deviceName = name;
    currentVersion = version;
    onStartCallback = NULL;
    onEndCallback = NULL;
    onProgressCallback = NULL;
    onErrorCallback = NULL;
  }
  
  // 设置回调
  void onStart(void (*callback)()) { onStartCallback = callback; }
  void onEnd(void (*callback)()) { onEndCallback = callback; }
  void onProgress(void (*callback)(int)) { onProgressCallback = callback; }
  void onError(void (*callback)(String)) { onErrorCallback = callback; }
  
  // 初始化Arduino OTA
  void initArduinoOTA(String password = "") {
    ArduinoOTA.setHostname(deviceName.c_str());
    
    if (password.length() > 0) {
      ArduinoOTA.setPassword(password.c_str());
    }
    
    ArduinoOTA.onStart([this]() {
      if (onStartCallback) onStartCallback();
    });
    
    ArduinoOTA.onEnd([this]() {
      if (onEndCallback) onEndCallback();
    });
    
    ArduinoOTA.onProgress([this](unsigned int progress, unsigned int total) {
      if (onProgressCallback) {
        onProgressCallback(progress * 100 / total);
      }
    });
    
    ArduinoOTA.onError([this](ota_error_t error) {
      String msg;
      switch (error) {
        case OTA_AUTH_ERROR: msg = "认证失败"; break;
        case OTA_BEGIN_ERROR: msg = "开始失败"; break;
        case OTA_CONNECT_ERROR: msg = "连接失败"; break;
        case OTA_RECEIVE_ERROR: msg = "接收失败"; break;
        case OTA_END_ERROR: msg = "结束失败"; break;
      }
      if (onErrorCallback) onErrorCallback(msg);
    });
    
    ArduinoOTA.begin();
  }
  
  // 初始化Web OTA
  void initWebOTA(WebServer* webServer) {
    server = webServer;
    
    server->on("/ota", HTTP_GET, [this]() {
      server->send(200, "text/html", getWebOTAPage());
    });
    
    server->on("/update", HTTP_POST, [this]() {
      server->send(200, "text/plain", Update.hasError() ? "FAIL" : "OK");
      delay(1000);
      ESP.restart();
    }, [this]() {
      handleUpload();
    });
  }
  
  // 处理上传
  void handleUpload() {
    HTTPUpload& upload = server->upload();
    
    if (upload.status == UPLOAD_FILE_START) {
      if (onStartCallback) onStartCallback();
      
      if (!Update.begin(UPDATE_SIZE_UNKNOWN)) {
        Update.printError(Serial);
      }
    } else if (upload.status == UPLOAD_FILE_WRITE) {
      if (Update.write(upload.buf, upload.currentSize) != upload.currentSize) {
        Update.printError(Serial);
      }
    } else if (upload.status == UPLOAD_FILE_END) {
      if (Update.end(true)) {
        if (onEndCallback) onEndCallback();
      } else {
        if (onErrorCallback) onErrorCallback("更新失败");
      }
    }
  }
  
  // HTTP OTA更新
  bool updateFromURL(String url) {
    HTTPClient http;
    http.begin(url);
    
    int httpCode = http.GET();
    
    if (httpCode == HTTP_CODE_OK) {
      int len = http.getSize();
      
      if (Update.begin(len)) {
        if (onStartCallback) onStartCallback();
        
        WiFiClient* stream = http.getStreamPtr();
        size_t written = Update.writeStream(*stream);
        
        if (written == len) {
          if (Update.end()) {
            if (onEndCallback) onEndCallback();
            return true;
          }
        }
        
        Update.printError(Serial);
      }
    }
    
    if (onErrorCallback) onErrorCallback("下载失败");
    return false;
  }
  
  // 处理循环
  void handle() {
    ArduinoOTA.handle();
  }
  
  // 获取版本
  String getVersion() {
    return currentVersion;
  }
  
  // Web页面
  String getWebOTAPage() {
    return R"rawliteral(
<!DOCTYPE html>
<html>
<head><meta charset="UTF-8"><title>OTA更新</title></head>
<body style="text-align:center;font-family:Arial;padding:50px">
  <h1>固件更新</h1>
  <form method="POST" action="/update" enctype="multipart/form-data">
    <input type="file" name="update" accept=".bin"><br><br>
    <button type="submit" style="padding:10px 20px;font-size:16px">上传更新</button>
  </form>
</body>
</html>
)rawliteral";
  }
};

// 使用示例
OTAManager ota("ESP32-Device", "1.0.0");
WebServer server(80);

void setup() {
  Serial.begin(115200);
  
  // 连接WiFi...
  
  ota.onStart([]() { Serial.println("开始更新..."); });
  ota.onEnd([]() { Serial.println("更新完成！"); });
  ota.onProgress([](int p) { Serial.printf("进度: %d%%\n", p); });
  ota.onError([](String e) { Serial.println("错误: " + e); });
  
  ota.initArduinoOTA("123456");
  ota.initWebOTA(&server);
  
  server.begin();
}

void loop() {
  ota.handle();
  server.handleClient();
}
```
