# 4.3 蓝牙串口通信

蓝牙串口（SPP，Serial Port Profile）是经典蓝牙的一种配置文件，可以模拟串口通信。本节将介绍ESP32的蓝牙串口编程。

## 蓝牙串口概述

### SPP特点

- 模拟标准串口通信
- 支持双向数据传输
- 兼容传统串口应用
- 配对后自动连接

### 适用场景

- 无线串口调试
- 数据采集传输
- 设备配置管理
- 传统设备改造

## 基本蓝牙串口

### 蓝牙串口服务器

```cpp
/*
 * 蓝牙串口服务器示例
 * ESP32作为蓝牙串口设备
 */

#include "BluetoothSerial.h"

BluetoothSerial SerialBT;

void setup() {
  Serial.begin(115200);
  
  // 初始化蓝牙串口
  // 参数：设备名称
  SerialBT.begin("ESP32-Serial");  // 蓝牙设备名称
  
  Serial.println("蓝牙串口已启动");
  Serial.println("设备名称: ESP32-Serial");
  Serial.println("请配对并连接");
}

void loop() {
  // 从蓝牙串口读取数据并发送到硬件串口
  if (SerialBT.available()) {
    Serial.write(SerialBT.read());
  }
  
  // 从硬件串口读取数据并发送到蓝牙串口
  if (Serial.available()) {
    SerialBT.write(Serial.read());
  }
  
  delay(10);
}
```

### 带回调的蓝牙串口

```cpp
/*
 * 带连接回调的蓝牙串口
 */

#include "BluetoothSerial.h"

BluetoothSerial SerialBT;

// 连接状态回调
void callback(esp_spp_cb_event_t event, esp_spp_cb_param_t *param) {
  if (event == ESP_SPP_SRV_OPEN_EVT) {
    Serial.println("蓝牙客户端已连接");
    SerialBT.println("欢迎连接ESP32蓝牙串口！");
  }
  
  if (event == ESP_SPP_CLOSE_EVT) {
    Serial.println("蓝牙客户端已断开");
  }
}

void setup() {
  Serial.begin(115200);
  
  // 注册回调
  SerialBT.register_callback(callback);
  
  // 启动蓝牙串口
  SerialBT.begin("ESP32-BT-Serial");
  
  Serial.println("蓝牙串口已启动，等待连接...");
}

void loop() {
  // 处理蓝牙数据
  if (SerialBT.available()) {
    String data = SerialBT.readStringUntil('\n');
    data.trim();
    
    Serial.printf("收到: %s\n", data.c_str());
    
    // 处理命令
    String response = processCommand(data);
    SerialBT.println(response);
  }
  
  // 处理硬件串口数据
  if (Serial.available()) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    SerialBT.println("Local: " + cmd);
  }
  
  delay(10);
}

String processCommand(String cmd) {
  cmd.toUpperCase();
  
  if (cmd == "HELLO") {
    return "Hello from ESP32!";
  } else if (cmd == "STATUS") {
    return "Status: OK, Uptime: " + String(millis() / 1000) + "s";
  } else if (cmd == "HELP") {
    return "Commands: HELLO, STATUS, HELP, LED ON/OFF";
  } else if (cmd == "LED ON") {
    digitalWrite(2, HIGH);
    return "LED已打开";
  } else if (cmd == "LED OFF") {
    digitalWrite(2, LOW);
    return "LED已关闭";
  } else {
    return "未知命令: " + cmd;
  }
}
```

## 蓝牙串口数据传输

### 二进制数据传输

```cpp
/*
 * 蓝牙串口二进制数据传输示例
 */

#include "BluetoothSerial.h"

BluetoothSerial SerialBT;

// 数据包结构
struct DataPacket {
  uint8_t header;      // 包头 0xAA
  uint8_t command;     // 命令
  uint8_t length;      // 数据长度
  uint8_t data[32];    // 数据
  uint8_t checksum;    // 校验和
};

void setup() {
  Serial.begin(115200);
  SerialBT.begin("ESP32-Binary");
  
  Serial.println("蓝牙二进制传输已启动");
}

void loop() {
  // 接收数据包
  if (SerialBT.available() >= 5) {  // 最小包长度
    if (SerialBT.read() == 0xAA) {  // 检查包头
      DataPacket packet;
      packet.header = 0xAA;
      packet.command = SerialBT.read();
      packet.length = SerialBT.read();
      
      // 读取数据
      if (packet.length <= 32) {
        SerialBT.readBytes(packet.data, packet.length);
        packet.checksum = SerialBT.read();
        
        // 验证校验和
        if (verifyChecksum(&packet)) {
          processPacket(&packet);
        }
      }
    }
  }
  
  delay(10);
}

bool verifyChecksum(DataPacket* packet) {
  uint8_t sum = packet->header + packet->command + packet->length;
  for (int i = 0; i < packet->length; i++) {
    sum += packet->data[i];
  }
  return (sum == packet->checksum);
}

void processPacket(DataPacket* packet) {
  Serial.printf("收到数据包: CMD=0x%02X, LEN=%d\n", 
                packet->command, packet->length);
  
  switch (packet->command) {
    case 0x01:  // 读传感器
      sendSensorData();
      break;
    case 0x02:  // 设置参数
      setParameter(packet->data, packet->length);
      break;
    case 0x03:  // 控制LED
      controlLED(packet->data[0]);
      break;
  }
}

void sendSensorData() {
  DataPacket response;
  response.header = 0xAA;
  response.command = 0x01;
  response.length = 8;
  
  // 填充传感器数据
  float temp = 25.5;
  float humi = 60.0;
  memcpy(&response.data[0], &temp, 4);
  memcpy(&response.data[4], &humi, 4);
  
  // 计算校验和
  response.checksum = response.header + response.command + response.length;
  for (int i = 0; i < response.length; i++) {
    response.checksum += response.data[i];
  }
  
  // 发送响应
  SerialBT.write((uint8_t*)&response, 5 + response.length);
}

void setParameter(uint8_t* data, int len) {
  // 处理参数设置
  Serial.println("设置参数");
}

void controlLED(uint8_t state) {
  digitalWrite(2, state ? HIGH : LOW);
  Serial.printf("LED: %s\n", state ? "开" : "关");
}
```

### 高速数据传输

```cpp
/*
 * 蓝牙高速数据传输示例
 * 使用缓冲区提高传输效率
 */

#include "BluetoothSerial.h"

BluetoothSerial SerialBT;

#define BUFFER_SIZE 512

uint8_t txBuffer[BUFFER_SIZE];
uint8_t rxBuffer[BUFFER_SIZE];
int rxIndex = 0;

void setup() {
  Serial.begin(115200);
  SerialBT.begin("ESP32-HighSpeed");
  
  // 设置蓝牙串口缓冲区大小
  SerialBT.setPin("1234");  // 设置配对PIN码（可选）
  
  Serial.println("高速蓝牙串口已启动");
}

void loop() {
  // 批量读取数据
  while (SerialBT.available() && rxIndex < BUFFER_SIZE) {
    rxBuffer[rxIndex++] = SerialBT.read();
  }
  
  // 处理完整数据包
  if (rxIndex > 0) {
    processBuffer(rxBuffer, rxIndex);
    rxIndex = 0;
  }
  
  // 定期发送数据
  static unsigned long lastSend = 0;
  if (millis() - lastSend > 100) {
    lastSend = millis();
    sendBufferedData();
  }
  
  delay(1);
}

void processBuffer(uint8_t* buffer, int len) {
  Serial.printf("收到 %d 字节\n", len);
  
  // 回显数据
  SerialBT.write(buffer, len);
}

void sendBufferedData() {
  // 填充发送缓冲区
  int txLen = 0;
  
  // 添加时间戳
  uint32_t timestamp = millis();
  memcpy(&txBuffer[txLen], &timestamp, 4);
  txLen += 4;
  
  // 添加传感器数据
  float temp = 25.0 + random(-20, 20) / 10.0;
  memcpy(&txBuffer[txLen], &temp, 4);
  txLen += 4;
  
  // 批量发送
  SerialBT.write(txBuffer, txLen);
}
```

## 蓝牙串口应用示例

### 无线调试终端

```cpp
/*
 * 无线调试终端示例
 * 通过蓝牙进行设备调试
 */

#include "BluetoothSerial.h"

BluetoothSerial SerialBT;

// 命令处理函数类型
typedef void (*CommandHandler)(String args);

// 命令结构
struct Command {
  String name;
  String description;
  CommandHandler handler;
};

// 命令列表
Command commands[20];
int commandCount = 0;

void setup() {
  Serial.begin(115200);
  SerialBT.begin("ESP32-Debug");
  
  // 注册命令
  registerCommand("help", "显示帮助", cmdHelp);
  registerCommand("status", "显示状态", cmdStatus);
  registerCommand("reset", "重启设备", cmdReset);
  registerCommand("gpio", "GPIO控制 (gpio <pin> <0|1>)", cmdGPIO);
  registerCommand("pwm", "PWM控制 (pwm <pin> <value>)", cmdPWM);
  registerCommand("adc", "读取ADC (adc <pin>)", cmdADC);
  registerCommand("mem", "显示内存信息", cmdMem);
  registerCommand("wifi", "WiFi信息", cmdWiFi);
  
  Serial.println("调试终端已启动");
  SerialBT.println("ESP32 Debug Terminal v1.0");
  SerialBT.println("输入 'help' 查看可用命令");
}

void loop() {
  // 蓝牙命令处理
  if (SerialBT.available()) {
    String line = SerialBT.readStringUntil('\n');
    line.trim();
    processCommand(line);
  }
  
  // 串口命令处理
  if (Serial.available()) {
    String line = Serial.readStringUntil('\n');
    line.trim();
    processCommand(line);
  }
  
  delay(10);
}

// 注册命令
void registerCommand(String name, String desc, CommandHandler handler) {
  if (commandCount < 20) {
    commands[commandCount].name = name;
    commands[commandCount].description = desc;
    commands[commandCount].handler = handler;
    commandCount++;
  }
}

// 处理命令
void processCommand(String line) {
  line.trim();
  if (line.length() == 0) return;
  
  // 解析命令和参数
  String cmd = line;
  String args = "";
  int spaceIndex = line.indexOf(' ');
  
  if (spaceIndex > 0) {
    cmd = line.substring(0, spaceIndex);
    args = line.substring(spaceIndex + 1);
  }
  
  cmd.toLowerCase();
  
  // 查找并执行命令
  bool found = false;
  for (int i = 0; i < commandCount; i++) {
    if (commands[i].name == cmd) {
      commands[i].handler(args);
      found = true;
      break;
    }
  }
  
  if (!found) {
    SerialBT.println("未知命令: " + cmd);
    SerialBT.println("输入 'help' 查看可用命令");
  }
}

// 命令实现
void cmdHelp(String args) {
  SerialBT.println("\n=== 可用命令 ===");
  for (int i = 0; i < commandCount; i++) {
    SerialBT.printf("%-10s - %s\n", 
                    commands[i].name.c_str(), 
                    commands[i].description.c_str());
  }
}

void cmdStatus(String args) {
  SerialBT.println("\n=== 系统状态 ===");
  SerialBT.printf("运行时间: %lu 秒\n", millis() / 1000);
  SerialBT.printf("可用内存: %d 字节\n", ESP.getFreeHeap());
  SerialBT.printf("CPU频率: %d MHz\n", ESP.getCpuFreqMHz());
}

void cmdReset(String args) {
  SerialBT.println("正在重启...");
  delay(100);
  ESP.restart();
}

void cmdGPIO(String args) {
  int pin, value;
  if (sscanf(args.c_str(), "%d %d", &pin, &value) == 2) {
    pinMode(pin, OUTPUT);
    digitalWrite(pin, value ? HIGH : LOW);
    SerialBT.printf("GPIO %d 设置为 %d\n", pin, value);
  } else {
    SerialBT.println("用法: gpio <pin> <0|1>");
  }
}

void cmdPWM(String args) {
  int pin, value;
  if (sscanf(args.c_str(), "%d %d", &pin, &value) == 2) {
    ledcAttachPin(pin, 0);
    ledcSetup(0, 5000, 8);
    ledcWrite(0, value);
    SerialBT.printf("GPIO %d PWM设置为 %d\n", pin, value);
  } else {
    SerialBT.println("用法: pwm <pin> <0-255>");
  }
}

void cmdADC(String args) {
  int pin = args.toInt();
  if (pin > 0) {
    int value = analogRead(pin);
    float voltage = value * 3.3 / 4095.0;
    SerialBT.printf("ADC %d: %d (%.2fV)\n", pin, value, voltage);
  } else {
    SerialBT.println("用法: adc <pin>");
  }
}

void cmdMem(String args) {
  SerialBT.println("\n=== 内存信息 ===");
  SerialBT.printf("可用堆: %d 字节\n", ESP.getFreeHeap());
  SerialBT.printf("最小可用堆: %d 字节\n", ESP.getMinFreeHeap());
  SerialBT.printf("最大分配块: %d 字节\n", ESP.getMaxAllocHeap());
}

void cmdWiFi(String args) {
  SerialBT.println("\n=== WiFi信息 ===");
  SerialBT.printf("状态: %s\n", 
                  WiFi.status() == WL_CONNECTED ? "已连接" : "未连接");
  if (WiFi.status() == WL_CONNECTED) {
    SerialBT.printf("SSID: %s\n", WiFi.SSID().c_str());
    SerialBT.printf("IP: %s\n", WiFi.localIP().toString().c_str());
    SerialBT.printf("RSSI: %d dBm\n", WiFi.RSSI());
  }
}
```
