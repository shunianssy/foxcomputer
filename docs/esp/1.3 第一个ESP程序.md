# 第一个ESP程序

## 1. Hello World程序

### 1.1 ESP-IDF版本

```c
#include <stdio.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "esp_system.h"
#include "esp_spi_flash.h"
#include "esp_log.h"

static const char *TAG = "HELLO_WORLD";

void app_main(void)
{
    // 打印欢迎信息
    ESP_LOGI(TAG, "Hello World!");
    
    // 打印芯片信息
    ESP_LOGI(TAG, "芯片信息:");
    esp_chip_info_t chip_info;
    esp_chip_info(&chip_info);
    ESP_LOGI(TAG, "芯片型号: %s", 
             chip_info.model == CHIP_ESP32 ? "ESP32" : 
             chip_info.model == CHIP_ESP32S2 ? "ESP32-S2" : 
             chip_info.model == CHIP_ESP32S3 ? "ESP32-S3" : "Unknown");
    ESP_LOGI(TAG, "CPU核心数: %d", chip_info.cores);
    ESP_LOGI(TAG, "CPU频率: %d MHz", esp_clk_cpu_freq() / 1000000);
    
    // 打印Flash信息
    ESP_LOGI(TAG, "Flash大小: %d MB", 
             spi_flash_get_chip_size() / (1024 * 1024));
    
    // 打印内存信息
    ESP_LOGI(TAG, "可用内存: %d KB", 
             esp_get_free_heap_size() / 1024);
    
    // 主循环
    while (1) {
        ESP_LOGI(TAG, "系统运行中...");
        vTaskDelay(pdMS_TO_TICKS(1000));  // 延迟1秒
    }
}
```

## 2. LED闪烁程序

### 2.1 基础LED闪烁（ESP-IDF）

```c
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/gpio.h"
#include "esp_log.h"

static const char *TAG = "LED_BLINK";

#define LED_PIN GPIO_NUM_2

void app_main(void)
{
    // 配置GPIO
    gpio_config_t io_conf = {
        .pin_bit_mask = (1ULL << LED_PIN),
        .mode = GPIO_MODE_OUTPUT,
        .pull_up_en = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE,
    };
    gpio_config(&io_conf);
    
    ESP_LOGI(TAG, "LED闪烁程序启动");
    
    // 主循环
    while (1) {
        // 点亮LED
        gpio_set_level(LED_PIN, 1);
        ESP_LOGI(TAG, "LED ON");
        vTaskDelay(pdMS_TO_TICKS(1000));
        
        // 熄灭LED
        gpio_set_level(LED_PIN, 0);
        ESP_LOGI(TAG, "LED OFF");
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}
```

### 2.2 可调频率LED闪烁（Arduino）

```cpp
#include <Arduino.h>

#define LED_PIN 2
#define POT_PIN 34  // 电位器引脚

void setup() {
    Serial.begin(115200);
    
    // 配置引脚
    pinMode(LED_PIN, OUTPUT);
    pinMode(POT_PIN, INPUT);
    
    Serial.println("可调频率LED闪烁程序");
}

void loop() {
    // 读取电位器值
    int potValue = analogRead(POT_PIN);
    
    // 计算延迟时间（100ms - 2000ms）
    int delayTime = map(potValue, 0, 4095, 100, 2000);
    
    // 点亮LED
    digitalWrite(LED_PIN, HIGH);
    Serial.printf("LED ON, 延迟: %d ms\n", delayTime);
    delay(delayTime);
    
    // 熄灭LED
    digitalWrite(LED_PIN, LOW);
    Serial.printf("LED OFF, 延迟: %d ms\n", delayTime);
    delay(delayTime);
}
```

## 3. 按键控制程序

### 3.1 按键控制LED（ESP-IDF）

```c
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/gpio.h"
#include "esp_log.h"

static const char *TAG = "BUTTON_CONTROL";

#define LED_PIN GPIO_NUM_2
#define BUTTON_PIN GPIO_NUM_0

void app_main(void)
{
    // 配置LED引脚
    gpio_config_t led_conf = {
        .pin_bit_mask = (1ULL << LED_PIN),
        .mode = GPIO_MODE_OUTPUT,
        .pull_up_en = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE,
    };
    gpio_config(&led_conf);
    
    // 配置按键引脚
    gpio_config_t button_conf = {
        .pin_bit_mask = (1ULL << BUTTON_PIN),
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_ENABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE,
    };
    gpio_config(&button_conf);
    
    ESP_LOGI(TAG, "按键控制LED程序启动");
    
    int led_state = 0;
    int last_button_state = 1;
    
    // 主循环
    while (1) {
        // 读取按键状态
        int button_state = gpio_get_level(BUTTON_PIN);
        
        // 检测按键按下（下降沿）
        if (button_state == 0 && last_button_state == 1) {
            // 切换LED状态
            led_state = !led_state;
            gpio_set_level(LED_PIN, led_state);
            ESP_LOGI(TAG, "LED状态: %s", led_state ? "ON" : "OFF");
            
            // 消抖延迟
            vTaskDelay(pdMS_TO_TICKS(50));
        }
        
        last_button_state = button_state;
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}
```

### 3.2 按键计数器（Arduino）

```cpp
#include <Arduino.h>

#define LED_PIN 2
#define BUTTON_PIN 0

int buttonCount = 0;
int lastButtonState = HIGH;
unsigned long lastDebounceTime = 0;
unsigned long debounceDelay = 50;

void setup() {
    Serial.begin(115200);
    
    // 配置引脚
    pinMode(LED_PIN, OUTPUT);
    pinMode(BUTTON_PIN, INPUT_PULLUP);
    
    Serial.println("按键计数器程序");
    Serial.println("按下按键计数");
}

void loop() {
    // 读取按键状态
    int buttonState = digitalRead(BUTTON_PIN);
    
    // 检测按键按下
    if (buttonState != lastButtonState) {
        lastDebounceTime = millis();
    }
    
    // 消抖处理
    if ((millis() - lastDebounceTime) > debounceDelay) {
        if (buttonState == LOW && lastButtonState == HIGH) {
            buttonCount++;
            
            // 打印计数
            Serial.printf("按键次数: %d\n", buttonCount);
            
            // LED闪烁指示
            digitalWrite(LED_PIN, HIGH);
            delay(100);
            digitalWrite(LED_PIN, LOW);
        }
    }
    
    lastButtonState = buttonState;
}
```

## 4. PWM调光程序

### 4.1 LED呼吸灯（ESP-IDF）

```c
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/ledc.h"
#include "esp_log.h"

static const char *TAG = "LED_BREATH";

#define LED_PIN GPIO_NUM_2
#define LEDC_TIMER LEDC_TIMER_0
#define LEDC_MODE LEDC_LOW_SPEED_MODE
#define LEDC_CHANNEL LEDC_CHANNEL_0
#define LEDC_DUTY_RES LEDC_TIMER_13_BIT  // 13位分辨率
#define LEDC_DUTY (4095)  // 最大占空比
#define LEDC_FREQUENCY (5000)  // 5kHz频率

void app_main(void)
{
    // 配置LEDC定时器
    ledc_timer_config_t ledc_timer = {
        .speed_mode = LEDC_MODE,
        .duty_resolution = LEDC_DUTY_RES,
        .timer_num = LEDC_TIMER,
        .freq_hz = LEDC_FREQUENCY,
        .clk_cfg = LEDC_AUTO_CLK,
    };
    ledc_timer_config(&ledc_timer);
    
    // 配置LEDC通道
    ledc_channel_config_t ledc_channel = {
        .gpio_num = LED_PIN,
        .speed_mode = LEDC_MODE,
        .channel = LEDC_CHANNEL,
        .intr_type = LEDC_INTR_DISABLE,
        .timer_sel = LEDC_TIMER,
        .duty = 0,
        .hpoint = 0,
    };
    ledc_channel_config(&ledc_channel);
    
    ESP_LOGI(TAG, "LED呼吸灯程序启动");
    
    int brightness = 0;
    int direction = 1;
    
    // 主循环
    while (1) {
        // 更新亮度
        brightness += direction * 10;
        
        // 反转方向
        if (brightness >= LEDC_DUTY || brightness <= 0) {
            direction = -direction;
        }
        
        // 限制亮度范围
        brightness = (brightness < 0) ? 0 : brightness;
        brightness = (brightness > LEDC_DUTY) ? LEDC_DUTY : brightness;
        
        // 设置PWM占空比
        ledc_set_duty(LEDC_MODE, LEDC_CHANNEL, brightness);
        ledc_update_duty(LEDC_MODE, LEDC_CHANNEL);
        
        // 延迟
        vTaskDelay(pdMS_TO_TICKS(20));
    }
}
```

### 4.2 可调亮度LED（Arduino）

```cpp
#include <Arduino.h>

#define LED_PIN 2
#define POT_PIN 34

void setup() {
    Serial.begin(115200);
    
    // 配置PWM
    ledcSetup(0, 5000, 8);  // 通道0, 5kHz, 8位分辨率
    ledcAttachPin(LED_PIN, 0);
    
    Serial.println("可调亮度LED程序");
}

void loop() {
    // 读取电位器值
    int potValue = analogRead(POT_PIN);
    
    // 映射到PWM值（0-255）

    // 设置PWM
    ledcWrite(0, pwmValue);
    
    // 打印信息
    Serial.printf("电位器值: %d, PWM值: %d\n", potValue, pwmValue);
    
    delay(100);
}
```

## 5. 串口通信程序

### 5.1 串口回显程序（ESP-IDF）

```c
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/uart.h"
#include "esp_log.h"

static const char *TAG = "UART_ECHO";

#define UART_NUM UART_NUM_0
#define BUF_SIZE (1024)
#define RD_BUF_SIZE (BUF_SIZE)

void app_main(void)
{
    // 配置UART参数
    uart_config_t uart_config = {
        .baud_rate = 115200,
        .data_bits = UART_DATA_8_BITS,
        .parity = UART_PARITY_DISABLE,
        .stop_bits = UART_STOP_BITS_1,
        .flow_ctrl = UART_HW_FLOWCTRL_DISABLE,
        .source_clk = UART_SCLK_APB,
    };
    
    // 配置UART
    ESP_ERROR_CHECK(uart_driver_install(UART_NUM, BUF_SIZE * 2, BUF_SIZE * 2, 0, NULL, 0));
    ESP_ERROR_CHECK(uart_param_config(UART_NUM, &uart_config));
    ESP_ERROR_CHECK(uart_set_pin(UART_NUM, UART_PIN_NO_CHANGE, UART_PIN_NO_CHANGE, UART_PIN_NO_CHANGE, UART_PIN_NO_CHANGE));
    
    ESP_LOGI(TAG, "串口回显程序启动");
    ESP_LOGI(TAG, "请输入字符，将回显显示");
    
    uint8_t *data = (uint8_t *) malloc(RD_BUF_SIZE + 1);
    
    // 主循环
    while (1) {
        // 读取串口数据
        int len = uart_read_bytes(UART_NUM, data, RD_BUF_SIZE, 20 / portTICK_PERIOD_MS);
        
        if (len > 0) {
            // 添加字符串结束符
            data[len] = '\0';
            
            // 打印接收到的数据
            ESP_LOGI(TAG, "接收到: %s", data);
            
            // 回显数据
            uart_write_bytes(UART_NUM, (const char *)data, len);
        }
    }
    
    free(data);
}
```

### 5.2 串口命令解析（Arduino）

```cpp
#include <Arduino.h>

#define LED_PIN 2

String inputString = "";
bool stringComplete = false;

void setup() {
    Serial.begin(115200);
    
    pinMode(LED_PIN, OUTPUT);
    
    Serial.println("串口命令解析程序");
    Serial.println("可用命令:");
    Serial.println("  ON  - 点亮LED");
    Serial.println("  OFF - 熄灭LED");
    Serial.println("  BLINK - LED闪烁");
}

void loop() {
    // 处理串口命令
    if (stringComplete) {
        // 去除空格和换行符
        inputString.trim();
        
        // 转换为大写
        inputString.toUpperCase();
        
        // 解析命令
        if (inputString == "ON") {
            digitalWrite(LED_PIN, HIGH);
            Serial.println("LED已点亮");
        } else if (inputString == "OFF") {
            digitalWrite(LED_PIN, LOW);
            Serial.println("LED已熄灭");
        } else if (inputString == "BLINK") {
            for (int i = 0; i < 5; i++) {
                digitalWrite(LED_PIN, HIGH);
                delay(200);
                digitalWrite(LED_PIN, LOW);
                delay(200);
            }
            Serial.println("LED闪烁完成");
        } else {
            Serial.println("未知命令: " + inputString);
        }
        
        // 清空字符串
        inputString = "";
        stringComplete = false;
    }
}

// 串口接收中断
void serialEvent() {
    while (Serial.available()) {
        char inChar = (char)Serial.read();
        inputString += inChar;
        
        if (inChar == '\n') {
            stringComplete = true;
        }
    }
}
```

## 6. 定时器程序

### 6.1 硬件定时器（ESP-IDF）

```c
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/gptimer.h"
#include "esp_log.h"

static const char *TAG = "TIMER_EXAMPLE";

#define LED_PIN GPIO_NUM_2
#define TIMER_RESOLUTION_HZ 1000000  // 1MHz分辨率
#define TIMER_INTERVAL_SEC 1  // 1秒间隔

static bool IRAM_ATTR timer_on_alarm_cb(gptimer_handle_t timer, const gptimer_alarm_event_data_t *edata, void *user_ctx)
{
    BaseType_t high_task_awoken = pdFALSE;
    
    // 切换LED状态
    static int led_state = 0;
    led_state = !led_state;
    gpio_set_level(LED_PIN, led_state);
    
    return (high_task_awoken == pdTRUE);
}

void app_main(void)
{
    // 配置LED引脚
    gpio_config_t io_conf = {
        .pin_bit_mask = (1ULL << LED_PIN),
        .mode = GPIO_MODE_OUTPUT,
        .pull_up_en = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE,
    };
    gpio_config(&io_conf);
    
    // 配置定时器
    gptimer_handle_t gptimer = NULL;
    gptimer_config_t timer_config = {
        .clk_src = GPTIMER_CLK_SRC_APB,
        .direction = GPTIMER_COUNT_UP,
        .resolution_hz = TIMER_RESOLUTION_HZ,
    };
    ESP_ERROR_CHECK(gptimer_new_timer(&timer_config, &gptimer));
    
    // 配置定时器中断
    gptimer_alarm_config_t alarm_config = {
        .alarm_count = TIMER_INTERVAL_SEC * TIMER_RESOLUTION_HZ,
        .reload_count = 0,
        .flags.auto_reload_on_alarm = true,
    };
    ESP_ERROR_CHECK(gptimer_set_alarm_action(gptimer, &alarm_config));
    
    // 注册回调函数
    gptimer_event_callbacks_t cbs = {
        .on_alarm = timer_on_alarm_cb,
    };
    ESP_ERROR_CHECK(gptimer_register_event_callbacks(gptimer, &cbs, NULL));
    
    // 启动定时器
    ESP_ERROR_CHECK(gptimer_enable(gptimer));
    ESP_ERROR_CHECK(gptimer_start(gptimer));
    
    ESP_LOGI(TAG, "硬件定时器程序启动");
    
    // 主循环
    while (1) {
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}
```
