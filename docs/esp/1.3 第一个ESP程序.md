# 1.3 第一个ESP程序

本节将带你编写并运行第一个ESP程序，通过LED闪烁和串口输出这两个经典的入门示例，让你快速熟悉ESP开发流程。

## 程序一：LED闪烁

LED闪烁是嵌入式开发的"Hello World"，让我们从这个最简单的程序开始。

### 硬件准备

- ESP32开发板一块
- USB数据线
- （可选）外接LED和220欧姆电阻

### 电路连接

大多数ESP32开发板都有板载LED，通常连接在GPIO2。如果你想外接LED：

```
ESP32 GPIO2 --[220欧姆电阻]-- LED正极 -- LED负极 -- GND
```

### Arduino代码实现

```cpp
/*
 * ESP32 LED闪烁程序
 * 功能：控制板载LED以1秒间隔闪烁
 * 
 * 硬件连接：
 * - 使用板载LED（GPIO2）
 * - 或外接LED到任意GPIO
 */

// 定义LED引脚
#define LED_PIN 2

// 定义闪烁间隔（毫秒）
#define BLINK_INTERVAL 1000

// 全局变量：记录LED状态
bool ledState = false;

// 上一次切换的时间
unsigned long previousMillis = 0;

void setup() {
  // 初始化串口通信
  Serial.begin(115200);
  Serial.println("\n--- LED闪烁程序启动 ---");
  
  // 配置LED引脚为输出模式
  pinMode(LED_PIN, OUTPUT);
  
  // 初始状态：LED熄灭
  digitalWrite(LED_PIN, LOW);
  Serial.println("LED初始化完成");
}

void loop() {
  // 获取当前时间
  unsigned long currentMillis = millis();
  
  // 检查是否到达切换时间
  // 使用非阻塞方式，不影响其他操作
  if (currentMillis - previousMillis >= BLINK_INTERVAL) {
    // 保存上次切换时间
    previousMillis = currentMillis;
    
    // 切换LED状态
    ledState = !ledState;
    digitalWrite(LED_PIN, ledState);
    
    // 输出状态到串口
    Serial.printf("[%lu ms] LED状态: %s\n", 
                  currentMillis, 
                  ledState ? "开" : "关");
  }
  
  // 这里可以添加其他非阻塞代码
}
```

### ESP-IDF代码实现

```c
/*
 * ESP32 LED闪烁程序 - ESP-IDF版本
 * 使用FreeRTOS任务实现
 */

#include <stdio.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/gpio.h"
#include "esp_log.h"

// 定义LED引脚
#define LED_PIN GPIO_NUM_2

// 日志标签
static const char *TAG = "LED_BLINK";

void app_main(void) {
    // 输出启动信息
    ESP_LOGI(TAG, "LED闪烁程序启动");
    
    // 配置GPIO
    gpio_config_t io_conf = {
        .pin_bit_mask = (1ULL << LED_PIN),  // 引脚位掩码
        .mode = GPIO_MODE_OUTPUT,            // 输出模式
        .pull_up_en = GPIO_PULLUP_DISABLE,   // 禁用上拉
        .pull_down_en = GPIO_PULLDOWN_DISABLE, // 禁用下拉
        .intr_type = GPIO_INTR_DISABLE       // 禁用中断
    };
    
    // 应用配置
    gpio_config(&io_conf);
    
    ESP_LOGI(TAG, "GPIO配置完成，开始闪烁");
    
    // 主循环
    while (1) {
        // 点亮LED
        gpio_set_level(LED_PIN, 1);
        ESP_LOGI(TAG, "LED ON");
        
        // 延时1000ms
        vTaskDelay(pdMS_TO_TICKS(1000));
        
        // 熄灭LED
        gpio_set_level(LED_PIN, 0);
        ESP_LOGI(TAG, "LED OFF");
        
        // 延时1000ms
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}
```

## 程序二：串口通信

串口是ESP与电脑通信的主要方式，学会使用串口对于调试非常重要。

### Arduino串口通信示例

```cpp
/*
 * ESP32 串口通信示例
 * 功能：接收串口命令并执行相应操作
 * 
 * 命令列表：
 * - "on"  : 打开LED
 * - "off" : 关闭LED
 * - "blink" : LED闪烁模式
 * - "status" : 显示当前状态
 * - "help" : 显示帮助信息
 */

#define LED_PIN 2
#define MAX_BLINK_COUNT 10

// LED状态枚举
enum LedMode {
  MODE_OFF,
  MODE_ON,
  MODE_BLINK
};

// 全局变量
LedMode currentMode = MODE_OFF;
bool ledState = false;
unsigned long lastBlinkTime = 0;
const long blinkInterval = 500;

void setup() {
  // 初始化串口
  Serial.begin(115200);
  
  // 等待串口连接（用于USB CDC）
  delay(1000);
  
  // 配置LED
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);
  
  // 打印欢迎信息
  printWelcome();
}

void loop() {
  // 处理串口接收
  handleSerialInput();
  
  // 处理闪烁模式
  handleBlinkMode();
}

// 打印欢迎信息
void printWelcome() {
  Serial.println("\n========================================");
  Serial.println("    ESP32 串口控制程序 v1.0");
  Serial.println("========================================");
  Serial.println("输入 'help' 查看可用命令");
  Serial.println("----------------------------------------");
  Serial.printf("芯片型号: %s\n", ESP.getChipModel());
  Serial.printf("CPU频率: %d MHz\n", ESP.getCpuFreqMHz());
  Serial.printf("Flash大小: %d MB\n", ESP.getFlashChipSize() / 1024 / 1024);
  Serial.printf("可用内存: %d 字节\n", ESP.getFreeHeap());
  Serial.println("========================================\n");
}

// 处理串口输入
void handleSerialInput() {
  // 检查是否有数据可读
  if (Serial.available() > 0) {
    // 读取一行数据
    String input = Serial.readStringUntil('\n');
    
    // 去除首尾空白
    input.trim();
    
    // 转换为小写
    input.toLowerCase();
    
    // 处理命令
    processCommand(input);
  }
}

// 处理命令
void processCommand(String cmd) {
  Serial.printf("\n收到命令: \"%s\"\n", cmd.c_str());
  
  if (cmd == "on") {
    currentMode = MODE_ON;
    digitalWrite(LED_PIN, HIGH);
    Serial.println("LED已打开");
    
  } else if (cmd == "off") {
    currentMode = MODE_OFF;
    digitalWrite(LED_PIN, LOW);
    Serial.println("LED已关闭");
    
  } else if (cmd == "blink") {
    currentMode = MODE_BLINK;
    Serial.println("LED闪烁模式已启动");
    
  } else if (cmd == "status") {
    printStatus();
    
  } else if (cmd == "help") {
    printHelp();
    
  } else if (cmd == "info") {
    printSystemInfo();
    
  } else if (cmd.length() > 0) {
    Serial.println("未知命令！输入 'help' 查看帮助");
  }
}

// 处理闪烁模式
void handleBlinkMode() {
  if (currentMode == MODE_BLINK) {
    unsigned long currentTime = millis();
    
    if (currentTime - lastBlinkTime >= blinkInterval) {
      lastBlinkTime = currentTime;
      ledState = !ledState;
      digitalWrite(LED_PIN, ledState);
    }
  }
}

// 打印状态
void printStatus() {
  Serial.println("\n--- 当前状态 ---");
  Serial.printf("LED模式: ");
  switch (currentMode) {
    case MODE_OFF: Serial.println("关闭"); break;
    case MODE_ON: Serial.println("常亮"); break;
    case MODE_BLINK: Serial.println("闪烁"); break;
  }
  Serial.printf("LED状态: %s\n", digitalRead(LED_PIN) ? "开" : "关");
  Serial.printf("运行时间: %lu 秒\n", millis() / 1000);
  Serial.printf("可用内存: %d 字节\n", ESP.getFreeHeap());
}

// 打印帮助信息
void printHelp() {
  Serial.println("\n--- 可用命令 ---");
  Serial.println("on      - 打开LED");
  Serial.println("off     - 关闭LED");
  Serial.println("blink   - LED闪烁模式");
  Serial.println("status  - 显示当前状态");
  Serial.println("info    - 显示系统信息");
  Serial.println("help    - 显示此帮助");
}

// 打印系统信息
void printSystemInfo() {
  Serial.println("\n--- 系统信息 ---");
  Serial.printf("芯片型号: %s\n", ESP.getChipModel());
  Serial.printf("芯片版本: %d\n", ESP.getChipRevision());
  Serial.printf("CPU核心数: %d\n", ESP.getChipCores());
  Serial.printf("CPU频率: %d MHz\n", ESP.getCpuFreqMHz());
  Serial.printf("Flash大小: %d MB\n", ESP.getFlashChipSize() / 1024 / 1024);
  Serial.printf("Flash速度: %d MHz\n", ESP.getFlashChipSpeed() / 1000000);
  Serial.printf("SDK版本: %s\n", ESP.getSdkVersion());
  Serial.printf("可用内存: %d 字节\n", ESP.getFreeHeap());
  Serial.printf("运行时间: %lu 毫秒\n", millis());
}
```

### 串口通信要点

**波特率选择：**

| 波特率 | 适用场景 |
|--------|----------|
| 9600 | 传统设备兼容 |
| 115200 | 常用默认值 |
| 230400 | 更高速传输 |
| 921600 | 高速调试 |

**数据格式：**
- 数据位：8位（默认）
- 停止位：1位（默认）
- 校验位：无（默认）

**接收数据方式：**

```cpp
// 方式1：读取单个字符
if (Serial.available() > 0) {
  char c = Serial.read();
  Serial.printf("收到字符: %c (ASCII: %d)\n", c, c);
}

// 方式2：读取一行
String line = Serial.readStringUntil('\n');

// 方式3：读取指定长度
byte buffer[64];
int len = Serial.readBytes(buffer, sizeof(buffer));

// 方式4：超时读取
Serial.setTimeout(1000);  // 设置超时1000ms
String data = Serial.readString();
```

## 程序三：按键检测

学习如何读取GPIO输入，实现按键控制LED。

### 硬件连接

```
按键接线方式（两种方式）：

方式1 - 内部上拉：
GPIO0 ---- 按键 ---- GND
（配置INPUT_PULLUP，按下时读取到LOW）

方式2 - 外部下拉：
3.3V ---- 电阻(10K) ---- GPIO0 ---- 按键 ---- GND
（配置INPUT，按下时读取到HIGH）
```

### Arduino按键示例

```cpp
/*
 * ESP32 按键控制LED程序
 * 功能：按键控制LED，支持短按和长按检测
 */

#define LED_PIN 2
#define BUTTON_PIN 0  // 通常开发板上的BOOT按钮

// 按键状态
enum ButtonState {
  BUTTON_IDLE,
  BUTTON_PRESSED,
  BUTTON_HELD
};

// 全局变量
ButtonState buttonState = BUTTON_IDLE;
unsigned long buttonPressTime = 0;
bool ledState = false;

// 长按阈值（毫秒）
#define LONG_PRESS_TIME 1000

// 消抖延时（毫秒）
#define DEBOUNCE_TIME 50

void setup() {
  Serial.begin(115200);
  Serial.println("\n--- 按键控制程序启动 ---");
  
  // 配置LED
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);
  
  // 配置按键（使用内部上拉电阻）
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  
  Serial.println("按下BOOT按钮控制LED");
  Serial.println("短按：切换LED状态");
  Serial.println("长按：LED闪烁3次");
}

void loop() {
  handleButton();
}

// 处理按键
void handleButton() {
  static int lastButtonState = HIGH;
  static unsigned long lastDebounceTime = 0;
  
  // 读取当前按键状态
  int reading = digitalRead(BUTTON_PIN);
  
  // 消抖处理
  if (reading != lastButtonState) {
    lastDebounceTime = millis();
  }
  
  // 状态稳定后处理
  if ((millis() - lastDebounceTime) > DEBOUNCE_TIME) {
    // 检测按键按下
    if (reading == LOW && buttonState == BUTTON_IDLE) {
      buttonState = BUTTON_PRESSED;
      buttonPressTime = millis();
      Serial.println("按键按下");
    }
    
    // 检测长按
    if (reading == LOW && buttonState == BUTTON_PRESSED) {
      if ((millis() - buttonPressTime) > LONG_PRESS_TIME) {
        buttonState = BUTTON_HELD;
        Serial.println("检测到长按");
        handleLongPress();
      }
    }
    
    // 检测按键释放
    if (reading == HIGH && buttonState != BUTTON_IDLE) {
      if (buttonState == BUTTON_PRESSED) {
        // 短按
        handleShortPress();
      }
      buttonState = BUTTON_IDLE;
      Serial.println("按键释放");
    }
  }
  
  lastButtonState = reading;
}

// 处理短按
void handleShortPress() {
  ledState = !ledState;
  digitalWrite(LED_PIN, ledState);
  Serial.printf("LED状态: %s\n", ledState ? "开" : "关");
}

// 处理长按
void handleLongPress() {
  Serial.println("LED闪烁3次");
  for (int i = 0; i < 3; i++) {
    digitalWrite(LED_PIN, HIGH);
    delay(200);
    digitalWrite(LED_PIN, LOW);
    delay(200);
  }
  // 恢复之前状态
  digitalWrite(LED_PIN, ledState);
}
```

### 使用中断的按键检测

```cpp
/*
 * ESP32 按键中断示例
 * 使用中断方式检测按键，更高效
 */

#define LED_PIN 2
#define BUTTON_PIN 0

// 全局变量
volatile bool buttonPressed = false;
volatile unsigned long lastInterruptTime = 0;

// 中断服务函数
void IRAM_ATTR buttonISR() {
  // 消抖：两次中断间隔需大于50ms
  unsigned long currentTime = millis();
  if (currentTime - lastInterruptTime > 50) {
    buttonPressed = true;
    lastInterruptTime = currentTime;
  }
}

void setup() {
  Serial.begin(115200);
  
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  
  // 配置中断：下降沿触发
  attachInterrupt(digitalPinToInterrupt(BUTTON_PIN), buttonISR, FALLING);
  
  Serial.println("中断方式按键检测已启动");
}

void loop() {
  if (buttonPressed) {
    buttonPressed = false;
    
    // 切换LED
    bool state = digitalRead(LED_PIN);
    digitalWrite(LED_PIN, !state);
    
    Serial.printf("按键触发，LED: %s\n", !state ? "开" : "关");
  }
}
```

## 程序四：多任务处理

ESP32支持FreeRTOS，可以创建多个并行任务。

```cpp
/*
 * ESP32 FreeRTOS多任务示例
 * 创建两个并行运行的任务
 */

#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

#define LED_PIN 2

// 任务1：LED闪烁
void TaskBlink(void *parameter) {
  Serial.println("任务1：LED闪烁启动");
  
  pinMode(LED_PIN, OUTPUT);
  
  while (1) {
    digitalWrite(LED_PIN, HIGH);
    vTaskDelay(500 / portTICK_PERIOD_MS);
    digitalWrite(LED_PIN, LOW);
    vTaskDelay(500 / portTICK_PERIOD_MS);
  }
}

// 任务2：串口输出
void TaskSerial(void *parameter) {
  Serial.println("任务2：串口输出启动");
  
  int counter = 0;
  
  while (1) {
    Serial.printf("计数器: %d, 可用内存: %d 字节\n", 
                  counter++, 
                  ESP.getFreeHeap());
    vTaskDelay(2000 / portTICK_PERIOD_MS);
  }
}

void setup() {
  Serial.begin(115200);
  delay(1000);
  
  Serial.println("\n--- FreeRTOS多任务示例 ---");
  
  // 创建任务1（优先级1）
  xTaskCreate(
    TaskBlink,        // 任务函数
    "Blink",          // 任务名称
    2048,             // 栈大小（字）
    NULL,             // 参数
    1,                // 优先级
    NULL              // 任务句柄
  );
  
  // 创建任务2（优先级1）
  xTaskCreate(
    TaskSerial,
    "Serial",
    4096,
    NULL,
    1,
    NULL
  );
  
  Serial.println("两个任务已创建并启动");
}

void loop() {
  // 主循环可以留空，任务在后台运行
  vTaskDelay(1000 / portTICK_PERIOD_MS);
}
```

## 上传与调试

### 上传步骤

1. **连接开发板**：用USB线连接ESP32和电脑
2. **选择端口**：工具 -> 端口 -> 选择对应COM口
3. **上传程序**：点击上传按钮或按Ctrl+U
4. **观察输出**：上传完成后打开串口监视器

### 常见上传问题

**问题1：上传卡在"Connecting..."**

解决方法：
1. 按住BOOT按钮
2. 按一下RST复位按钮
3. 松开BOOT按钮
4. 等待上传完成

**问题2：上传失败 "Failed to connect"**

解决方法：
1. 降低上传速度到115200
2. 更换USB线
3. 检查驱动是否正确安装

### 调试技巧
