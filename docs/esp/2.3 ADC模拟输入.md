# 2.3 ADC模拟输入

ADC（Analog-to-Digital Converter，模数转换器）是将模拟信号转换为数字信号的器件。ESP32内置了功能强大的ADC模块，本节将详细介绍其使用方法。

## ADC概述

### ESP32 ADC特性

ESP32集成了两个12位SAR ADC：

**ADC1（通道0-7）：**
- GPIO 36（VP）、39（VN）、34、35、32、33、4、0
- 可在任何模式下使用

**ADC2（通道0-9）：**
- GPIO 2、13、12、14、15、25、26、27、13、13
- WiFi启用时不可使用

**主要特性：**
- 12位分辨率（0-4095）
- 可配置衰减（0dB、2.5dB、6dB、11dB）
- 多通道输入
- 支持连续采样
- 内置霍尔传感器和温度传感器

### ADC引脚分配

```
ESP32 ADC引脚分配：

ADC1:
  GPIO36 (VP)  - ADC1_CH0
  GPIO39 (VN)  - ADC1_CH3
  GPIO34       - ADC1_CH6
  GPIO35       - ADC1_CH7
  GPIO32       - ADC1_CH4
  GPIO33       - ADC1_CH5
  GPIO4        - ADC1_CH0 (部分芯片)
  GPIO0        - ADC1_CH1 (部分芯片)

ADC2（WiFi启用时不可用）:
  GPIO2        - ADC2_CH2
  GPIO13       - ADC2_CH4
  GPIO12       - ADC2_CH5
  GPIO14       - ADC2_CH6
  GPIO15       - ADC2_CH3
  GPIO25       - ADC2_CH8
  GPIO26       - ADC2_CH9
  GPIO27       - ADC2_CH7
```

### 电压范围与衰减

| 衰减设置 | 可测电压范围 | 说明 |
|----------|--------------|------|
| 0dB | 0 - 1.1V | 默认，精度最高 |
| 2.5dB | 0 - 1.5V | |
| 6dB | 0 - 2.2V | |
| 11dB | 0 - 3.3V | 测量范围最大 |

## 基本ADC读取

### 简单读取

```cpp
/*
 * ADC基本读取示例
 */

#define ADC_PIN 34  // ADC1_CH6

void setup() {
  Serial.begin(115200);
  
  // 设置ADC衰减（可选，默认11dB）
  analogSetAttenuation(ADC_11db);
  
  // 设置ADC分辨率（可选，默认12位）
  analogReadResolution(12);  // 12位：0-4095
  
  // 设置采样宽度（可选）
  analogSetWidth(12);
  
  Serial.println("ADC基本读取示例启动");
  Serial.printf("ADC引脚: GPIO%d\n", ADC_PIN);
}

void loop() {
  // 读取ADC原始值
  int rawValue = analogRead(ADC_PIN);
  
  // 转换为电压
  float voltage = rawValue * 3.3 / 4095.0;
  
  Serial.printf("原始值: %d, 电压: %.3f V\n", rawValue, voltage);
  
  delay(1000);
}
```

## ADC校准

ESP32的ADC存在非线性误差，需要进行校准以提高精度。

```cpp
/*
 * ADC校准示例
 */

#include "esp_adc_cal.h"

#define ADC_PIN 34

// 校准参数
esp_adc_cal_characteristics_t *adc_chars;

void setup() {
  Serial.begin(115200);
  
  // 配置ADC
  analogReadResolution(12);
  analogSetAttenuation(ADC_11db);
  
  // 校准ADC
  calibrateADC();
  
  Serial.println("ADC校准示例启动");
}

void loop() {
  // 读取并校准
  uint32_t raw = analogRead(ADC_PIN);
  uint32_t voltage = esp_adc_cal_raw_to_voltage(raw, adc_chars);
  
  Serial.printf("原始值: %d, 校准电压: %d mV\n", raw, voltage);
  
  delay(1000);
}

void calibrateADC() {
  // 检查eFuse校准数据
  if (esp_adc_cal_check_efuse(ESP_ADC_CAL_VAL_EFUSE_VREF) == ESP_OK) {
    Serial.println("使用eFuse Vref校准");
  } else if (esp_adc_cal_check_efuse(ESP_ADC_CAL_VAL_EFUSE_TP) == ESP_OK) {
    Serial.println("使用eFuse两点校准");
  } else {
    Serial.println("使用默认校准");
  }
  
  // 分配校准参数结构
  adc_chars = (esp_adc_cal_characteristics_t *)calloc(1, sizeof(esp_adc_cal_characteristics_t));
  
  // 初始化校准参数
  esp_adc_cal_value_t val_type = esp_adc_cal_characterize(
    ADC_UNIT_1,        // ADC单元
    ADC_ATTEN_DB_11,   // 衰减
    ADC_WIDTH_BIT_12,  // 位宽
    1100,              // 默认Vref（mV）
    adc_chars          // 校准参数
  );
  
  if (val_type == ESP_ADC_CAL_VAL_EFUSE_VREF) {
    Serial.printf("eFuse Vref: %d mV\n", adc_chars->vref);
  } else if (val_type == ESP_ADC_CAL_VAL_EFUSE_TP) {
    Serial.println("使用eFuse两点校准");
  } else {
    Serial.println("使用默认Vref");
  }
}
```

## 多次采样平均

```cpp
/*
 * ADC多次采样平均示例
 * 提高测量精度和稳定性
 */

#define ADC_PIN 34
#define SAMPLE_COUNT 64  // 采样次数

void setup() {
  Serial.begin(115200);
  
  analogReadResolution(12);
  analogSetAttenuation(ADC_11db);
  
  Serial.println("ADC多次采样平均示例");
}

void loop() {
  // 方法1：简单平均
  uint32_t sum = 0;
  for (int i = 0; i < SAMPLE_COUNT; i++) {
    sum += analogRead(ADC_PIN);
    delayMicroseconds(100);
  }
  int avgValue = sum / SAMPLE_COUNT;
  float avgVoltage = avgValue * 3.3 / 4095.0;
  
  // 方法2：中值滤波（去除极值后平均）
  int medianValue = readADCMedian(ADC_PIN, SAMPLE_COUNT);
  float medianVoltage = medianValue * 3.3 / 4095.0;
  
  // 方法3：移动平均
  static float movingAvg = 0;
  float alpha = 0.1;  // 平滑系数
  int current = analogRead(ADC_PIN);
  movingAvg = alpha * current + (1 - alpha) * movingAvg;
  float movingVoltage = movingAvg * 3.3 / 4095.0;
  
  Serial.printf("简单平均: %.3fV, 中值滤波: %.3fV, 移动平均: %.3fV\n",
                avgVoltage, medianVoltage, movingVoltage);
  
  delay(1000);
}

// 中值滤波读取
int readADCMedian(int pin, int samples) {
  int readings[samples];
  
  // 采集数据
  for (int i = 0; i < samples; i++) {
    readings[i] = analogRead(pin);
    delayMicroseconds(100);
  }
  
  // 排序
  for (int i = 0; i < samples - 1; i++) {
    for (int j = i + 1; j < samples; j++) {
      if (readings[i] > readings[j]) {
        int temp = readings[i];
        readings[i] = readings[j];
        readings[j] = temp;
      }
    }
  }
  
  // 去除最大最小各10%，取中间80%平均
  int start = samples * 0.1;
  int end = samples * 0.9;
  int sum = 0;
  int count = 0;
  
  for (int i = start; i < end; i++) {
    sum += readings[i];
    count++;
  }
  
  return sum / count;
}
```

## 实用传感器示例

### 光敏电阻

```cpp
/*
 * 光敏电阻读取示例
 * 使用分压电路测量光照强度
 */

#define LDR_PIN 34

// 分压电阻值（欧姆）
#define R_FIXED 10000  // 10K欧姆

void setup() {
  Serial.begin(115200);
  
  analogReadResolution(12);
  analogSetAttenuation(ADC_11db);
  
  Serial.println("光敏电阻读取示例");
}

void loop() {
  int raw = analogRead(LDR_PIN);
  float voltage = raw * 3.3 / 4095.0;
  
  // 计算光敏电阻阻值
  // Vout = Vin * R_LDR / (R_FIXED + R_LDR)
  // R_LDR = R_FIXED * Vout / (Vin - Vout)
  float rLDR = R_FIXED * voltage / (3.3 - voltage);
  
  // 计算光照强度（简化模型）
  // 光照强度与电阻成反比
  int lightPercent = map(raw, 0, 4095, 100, 0);
  
  Serial.printf("原始值: %d, 电压: %.3fV, 电阻: %.0f欧姆, 光照: %d%%\n",
                raw, voltage, rLDR, lightPercent);
  
  delay(1000);
}
```

### 电压测量

```cpp
/*
 * 电压测量示例
 * 使用分压电路测量高于3.3V的电压
 */

#define VOLTAGE_PIN 34

// 分压电阻（欧姆）
#define R1 100000  // 100K欧姆（连接被测电压）
#define R2 10000   // 10K欧姆（连接GND）

// 计算分压比
#define VOLTAGE_DIVIDER ((float)(R1 + R2) / R2)

// 校准系数（根据实际测量调整）
#define CALIBRATION_FACTOR 1.0

void setup() {
  Serial.begin(115200);
  
  analogReadResolution(12);
  analogSetAttenuation(ADC_11db);
  
  Serial.println("电压测量示例");
  Serial.printf("分压比: %.2f\n", VOLTAGE_DIVIDER);
  Serial.println("最大可测电压: " + String(3.3 * VOLTAGE_DIVIDER) + "V");
}

void loop() {
  // 多次采样平均
  uint32_t sum = 0;
  for (int i = 0; i < 100; i++) {
    sum += analogRead(VOLTAGE_PIN);
    delayMicroseconds(100);
  }
  int avgRaw = sum / 100;
  
  // 计算电压
  float adcVoltage = avgRaw * 3.3 / 4095.0;
  float measuredVoltage = adcVoltage * VOLTAGE_DIVIDER * CALIBRATION_FACTOR;
  
  Serial.printf("ADC电压: %.3fV, 测量电压: %.2fV\n", 
                adcVoltage, measuredVoltage);
  
  delay(1000);
}
```

### 电流测量（ACS712）

```cpp
/*
 * ACS712电流传感器读取示例
 * 测量交流或直流电流
 */

#define CURRENT_PIN 34

// ACS712灵敏度（mV/A）
// 5A型号: 185mV/A
// 20A型号: 100mV/A
// 30A型号: 66mV/A
#define SENSITIVITY 185

// 采样次数
#define SAMPLES 1000

void setup() {
  Serial.begin(115200);
  
  analogReadResolution(12);
  analogSetAttenuation(ADC_11db);
  
  Serial.println("ACS712电流传感器示例");
}

void loop() {
  // 测量DC电流
  float dcCurrent = measureDCCurrent();
  Serial.printf("DC电流: %.3f A\n", dcCurrent);
  
  // 测量AC电流
  float acCurrent = measureACCurrent();
  Serial.printf("AC电流: %.3f A\n", acCurrent);
  
  delay(1000);
}

// 测量DC电流
float measureDCCurrent() {
  // 多次采样取平均
  uint32_t sum = 0;
  for (int i = 0; i < SAMPLES; i++) {
    sum += analogRead(CURRENT_PIN);
    delayMicroseconds(100);
  }
  int avgRaw = sum / SAMPLES;
  
  // 转换为电压
  float voltage = avgRaw * 3.3 / 4095.0;
  
  // ACS712在无电流时输出VCC/2
  // 电流 = (电压 - VCC/2) / 灵敏度
  float current = (voltage - 1.65) / (SENSITIVITY / 1000.0);
  
  return current;
}

// 测量AC电流
float measureACCurrent() {
  float maxValue = 0;
  float minValue = 4095;
  
  unsigned long startTime = millis();
  
  // 采样一个完整周期（假设50Hz，20ms）
  while (millis() - startTime < 100) {  // 采样100ms
    int raw = analogRead(CURRENT_PIN);
    float voltage = raw * 3.3 / 4095.0;
    
    if (voltage > maxValue) maxValue = voltage;
    if (voltage < minValue) minValue = voltage;
    
    delayMicroseconds(200);
  }
  
  // 计算峰峰值
  float peakToPeak = maxValue - minValue;
  
  // 计算RMS电流
  float rmsVoltage = peakToPeak * 0.3536;  // Vrms = Vpp / (2 * sqrt(2))
  float rmsCurrent = rmsVoltage / (SENSITIVITY / 1000.0);
  
  return rmsCurrent;
}
```

## ADC类封装

```cpp
/*
 * ADC读取类封装
 * 提供便捷的ADC读取接口
 */

class ADCReader {
private:
  int pin;
  int samples;
  adc_atten_t attenuation;
  adc_bits_width_t width;
  
public:
  // 构造函数
  ADCReader(int adcPin, int sampleCount = 64) {
    pin = adcPin;
    samples = sampleCount;
    attenuation = ADC_ATTEN_DB_11;
    width = ADC_WIDTH_BIT_12;
    
    // 配置ADC
    analogReadResolution(12);
    analogSetAttenuation(ADC_11db);
  }
  
  // 设置衰减
  void setAttenuation(adc_atten_t atten) {
    attenuation = atten;
    analogSetAttenuation(atten);
  }
  
  // 读取原始值
  int readRaw() {
    return analogRead(pin);
  }
  
  // 读取平均值
  int readAverage() {
    uint32_t sum = 0;
    for (int i = 0; i < samples; i++) {
      sum += analogRead(pin);
      delayMicroseconds(100);
    }
    return sum / samples;
  }
  
  // 读取电压
  float readVoltage(float vref = 3.3) {
    int raw = readAverage();
    return raw * vref / 4095.0;
  }
  
  // 读取百分比
  float readPercent() {
    int raw = readAverage();
    return raw * 100.0 / 4095.0;
  }
  
  // 映射到范围
  int mapToRange(int minVal, int maxVal) {
    int raw = readAverage();
    return map(raw, 0, 4095, minVal, maxVal);
  }
  
  // 检测阈值
  bool isAboveThreshold(float threshold) {
    return readVoltage() > threshold;
  }
};

// 使用示例
ADCReader myADC(34);

void setup() {
  Serial.begin(115200);
  Serial.println("ADC类示例");
}

void loop() {
  Serial.printf("电压: %.3fV, 百分比: %.1f%%\n", 
                myADC.readVoltage(), 
                myADC.readPercent());
  delay(1000);
}
```
