# 2.4 定时器与中断

定时器和中断是嵌入式系统的核心概念，它们使得程序能够精确控制时间并在事件发生时及时响应。本节将详细介绍ESP32的定时器和中断功能。

## 定时器概述

### ESP32定时器资源

ESP32内置4个64位通用定时器，分为两组：

**定时器组0：**
- 定时器0（TIMER_0）
- 定时器1（TIMER_1）

**定时器组1：**
- 定时器0（TIMER_0）
- 定时器1（TIMER_1）

**主要特性：**
- 64位计数器
- 16位预分频器
- 可配置为递增或递减
- 支持自动重载
- 支持报警中断
- 支持软件控制暂停/恢复

### 定时器时钟

ESP32定时器使用APB时钟（通常80MHz）作为时钟源：

```
定时器频率 = APB时钟 / (预分频值 + 1)

例如：
- 预分频值 = 79 -> 定时器频率 = 80MHz / 80 = 1MHz
- 预分频值 = 7999 -> 定时器频率 = 80MHz / 8000 = 10kHz
```

## 硬件定时器

### 基本定时器使用

```cpp
/*
 * 硬件定时器基本示例
 * 使用定时器产生精确延时
 */

#include "esp_timer.h"

// 定时器句柄
esp_timer_handle_t periodic_timer;

// 定时器回调函数
void IRAM_ATTR onTimer(void* arg) {
  // 在这里执行定时任务
  // 注意：此函数在中断上下文中执行，应尽量简短
  
  static int counter = 0;
  counter++;
  
  // 可以在这里设置标志，在主循环中处理
  // gpio_set_level(GPIO_NUM_2, counter % 2);
}

void setup() {
  Serial.begin(115200);
  
  // 配置定时器
  const esp_timer_create_args_t periodic_timer_args = {
    .callback = &onTimer,       // 回调函数
    .arg = NULL,                // 传递给回调的参数
    .dispatch_method = ESP_TIMER_TASK,  // 调度方法
    .name = "periodic"          // 定时器名称（调试用）
  };
  
  // 创建定时器
  ESP_ERROR_CHECK(esp_timer_create(&periodic_timer_args, &periodic_timer));
  
  // 启动周期定时器（每1秒触发一次）
  ESP_ERROR_CHECK(esp_timer_start_periodic(periodic_timer, 1000000));  // 微秒
  
  Serial.println("硬件定时器已启动（周期1秒）");
}

void loop() {
  // 主循环可以做其他事情
  delay(100);
}

// 清理
void cleanup() {
  esp_timer_stop(periodic_timer);
  esp_timer_delete(periodic_timer);
}
```

### Arduino定时器方式

```cpp
/*
 * Arduino风格的定时器使用
 * 使用Ticker库
 */

#include <Ticker.h>

// 创建Ticker对象
Ticker timer1;
Ticker timer2;

// 定时器计数器
volatile int counter1 = 0;
volatile int counter2 = 0;

// 定时器回调函数
void IRAM_ATTR onTimer1() {
  counter1++;
}

void IRAM_ATTR onTimer2() {
  counter2++;
}

void setup() {
  Serial.begin(115200);
  
  // 启动定时器1：每1秒触发
  timer1.attach(1.0, onTimer1);
  
  // 启动定时器2：每0.5秒触发
  timer2.attach(0.5, onTimer2);
  
  Serial.println("定时器已启动");
}

void loop() {
  // 打印计数器
  static int lastCounter1 = 0;
  static int lastCounter2 = 0;
  
  if (counter1 != lastCounter1 || counter2 != lastCounter2) {
    Serial.printf("定时器1: %d次, 定时器2: %d次\n", counter1, counter2);
    lastCounter1 = counter1;
    lastCounter2 = counter2;
  }
  
  delay(100);
}
```

### 定时器控制LED闪烁

```cpp
/*
 * 使用定时器控制LED闪烁
 * 不同频率闪烁
 */

#include <Ticker.h>

#define LED1_PIN 2
#define LED2_PIN 4

Ticker blinker1;
Ticker blinker2;

// LED状态
volatile bool led1State = false;
volatile bool led2State = false;

// 闪烁回调
void IRAM_ATTR blinkLED1() {
  led1State = !led1State;
  digitalWrite(LED1_PIN, led1State);
}

void IRAM_ATTR blinkLED2() {
  led2State = !led2State;
  digitalWrite(LED2_PIN, led2State);
}

void setup() {
  Serial.begin(115200);
  
  pinMode(LED1_PIN, OUTPUT);
  pinMode(LED2_PIN, OUTPUT);
  
  // LED1每500ms闪烁
  blinker1.attach(0.5, blinkLED1);
  
  // LED2每200ms闪烁
  blinker2.attach(0.2, blinkLED2);
  
  Serial.println("LED闪烁定时器已启动");
}

void loop() {
  // 可以通过串口改变闪烁频率
  if (Serial.available() > 0) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    
    if (cmd.startsWith("led1 ")) {
      float interval = cmd.substring(5).toFloat();
      blinker1.detach();
      blinker1.attach(interval, blinkLED1);
      Serial.printf("LED1闪烁间隔: %.2f秒\n", interval);
    } else if (cmd.startsWith("led2 ")) {
      float interval = cmd.substring(5).toFloat();
      blinker2.detach();
      blinker2.attach(interval, blinkLED2);
      Serial.printf("LED2闪烁间隔: %.2f秒\n", interval);
    }
  }
  
  delay(100);
}
```

## 中断系统

### 中断概述

中断是一种硬件机制，当特定事件发生时，CPU暂停当前任务，转而执行中断服务程序（ISR）。

**ESP32中断类型：**
- GPIO中断（外部中断）
- 定时器中断
- UART中断
- SPI中断
- WiFi中断
- 蓝牙中断

### GPIO中断

```cpp
/*
 * GPIO中断示例
 * 检测按键按下
 */

#define BUTTON_PIN 0
#define LED_PIN 2

// 中断标志
volatile bool buttonPressed = false;
volatile unsigned long pressTime = 0;

// 中断服务函数
void IRAM_ATTR buttonISR() {
  buttonPressed = true;
  pressTime = millis();
}

void setup() {
  Serial.begin(115200);
  
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  
  // 配置中断
  // 触发模式：FALLING(下降沿)、RISING(上升沿)、CHANGE(变化)、HIGH(高电平)、LOW(低电平)
  attachInterrupt(digitalPinToInterrupt(BUTTON_PIN), buttonISR, FALLING);
  
  Serial.println("GPIO中断示例启动");
  Serial.println("按下BOOT按钮触发中断");
}

void loop() {
  if (buttonPressed) {
    buttonPressed = false;
    
    Serial.printf("按钮中断触发，时间: %lu ms\n", pressTime);
    
    // 切换LED
    static bool ledState = false;
    ledState = !ledState;
    digitalWrite(LED_PIN, ledState);
  }
}
```

### 中断消抖

```cpp
/*
 * 中断消抖示例
 * 防止机械抖动导致多次触发
 */

#define BUTTON_PIN 0

// 消抖时间（毫秒）
#define DEBOUNCE_TIME 50

// 中断变量
volatile unsigned long lastInterruptTime = 0;
volatile int interruptCount = 0;

void IRAM_ATTR buttonISR() {
  unsigned long currentTime = millis();
  
  // 消抖：两次中断间隔必须大于DEBOUNCE_TIME
  if (currentTime - lastInterruptTime > DEBOUNCE_TIME) {
    interruptCount++;
    lastInterruptTime = currentTime;
  }
}

void setup() {
  Serial.begin(115200);
  
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(BUTTON_PIN), buttonISR, FALLING);
  
  Serial.println("中断消抖示例启动");
}

void loop() {
  static int lastCount = 0;
  
  if (interruptCount != lastCount) {
    Serial.printf("有效中断次数: %d\n", interruptCount);
    lastCount = interruptCount;
  }
  
  delay(100);
}
```

### 多引脚中断

```cpp
/*
 * 多引脚中断示例
 * 使用同一个ISR处理多个引脚
 */

// 按钮引脚
const int buttonPins[] = {0, 35, 34, 39};
const int buttonCount = 4;
const char* buttonNames[] = {"BOOT", "BTN1", "BTN2", "BTN3"};

// 中断状态
volatile int triggeredButton = -1;
volatile unsigned long triggerTime = 0;

// 中断服务函数
void IRAM_ATTR buttonISR() {
  // 检查哪个按钮被按下
  for (int i = 0; i < buttonCount; i++) {
    if (digitalRead(buttonPins[i]) == LOW) {
      triggeredButton = i;
      triggerTime = millis();
      break;
    }
  }
}

void setup() {
  Serial.begin(115200);
  
  // 配置所有按钮引脚
  for (int i = 0; i < buttonCount; i++) {
    pinMode(buttonPins[i], INPUT_PULLUP);
    attachInterrupt(digitalPinToInterrupt(buttonPins[i]), buttonISR, FALLING);
  }
  
  Serial.println("多引脚中断示例启动");
  Serial.println("按下任意按钮触发中断");
}

void loop() {
  if (triggeredButton >= 0) {
    int btn = triggeredButton;
    triggeredButton = -1;
    
    Serial.printf("按钮 %s 被按下，时间: %lu ms\n", 
                  buttonNames[btn], triggerTime);
  }
  
  delay(10);
}
```

## 定时器中断

```cpp
/*
 * 定时器中断示例
 * 使用硬件定时器产生精确中断
 */

#include "driver/timer.h"

#define TIMER_DIVIDER 80        // 预分频（80MHz/80 = 1MHz）
#define TIMER_SCALE 1000000     // 1MHz = 1微秒精度
#define TIMER_INTERVAL_SEC 1    // 中断间隔（秒）

// 中断计数
volatile int timerInterruptCount = 0;

// 定时器中断处理函数
void IRAM_ATTR timerGroup0ISR(void* para) {
  // 获取定时器组和中断状态
  timer_group_t timer_group = (timer_group_t)(int)para;
  uint32_t timer_idx = TIMER_0;
  
  // 清除中断状态
  timer_group_intr_clr_in_isr(timer_group, timer_idx);
  
  // 重新加载报警值
  timer_group_enable_alarm_in_isr(timer_group, timer_idx);
  
  // 更新计数
  timerInterruptCount++;
}

void setup() {
  Serial.begin(115200);
  
  // 配置定时器
  timer_config_t config = {
    .divider = TIMER_DIVIDER,
    .counter_dir = TIMER_COUNT_UP,
    .counter_en = TIMER_PAUSE,
    .alarm_en = TIMER_ALARM_EN,
    .auto_reload = TIMER_AUTORELOAD_EN,
  };
  
  // 初始化定时器
  timer_init(TIMER_GROUP_0, TIMER_0, &config);
  
  // 设置计数器初始值
  timer_set_counter_value(TIMER_GROUP_0, TIMER_0, 0);
  
  // 设置报警值
  timer_set_alarm_value(TIMER_GROUP_0, TIMER_0, TIMER_INTERVAL_SEC * TIMER_SCALE);
  
  // 启用定时器中断
  timer_enable_intr(TIMER_GROUP_0, TIMER_0);
  
  // 注册中断处理函数
  timer_isr_register(TIMER_GROUP_0, TIMER_0, timerGroup0ISR, 
                     (void*)TIMER_GROUP_0, ESP_INTR_FLAG_IRAM, NULL);
  
  // 启动定时器
  timer_start(TIMER_GROUP_0, TIMER_0);
  
  Serial.println("定时器中断已启动（每1秒）");
}

void loop() {
  static int lastCount = 0;
  
  if (timerInterruptCount != lastCount) {
    Serial.printf("定时器中断次数: %d\n", timerInterruptCount);
    lastCount = timerInterruptCount;
  }
  
  delay(100);
}
```

## 中断优先级

```cpp
/*
 * 中断优先级示例
 * ESP32支持1-7级优先级（数字越大优先级越高）
 */

#include "esp_intr_alloc.h"

#define HIGH_PRIORITY_PIN 0
#define LOW_PRIORITY_PIN 35

// 中断计数
volatile int highPriorityCount = 0;
volatile int lowPriorityCount = 0;

// 高优先级中断处理函数
void IRAM_ATTR highPriorityISR() {
  highPriorityCount++;
}

// 低优先级中断处理函数
void IRAM_ATTR lowPriorityISR() {
  lowPriorityCount++;
}

void setup() {
  Serial.begin(115200);
  
  pinMode(HIGH_PRIORITY_PIN, INPUT_PULLUP);
  pinMode(LOW_PRIORITY_PIN, INPUT_PULLUP);
  
  // 配置高优先级中断
  attachInterruptArg(digitalPinToInterrupt(HIGH_PRIORITY_PIN), 
                     highPriorityISR, NULL, FALLING);
  
  // 配置低优先级中断
  attachInterruptArg(digitalPinToInterrupt(LOW_PRIORITY_PIN), 
                     lowPriorityISR, NULL, FALLING);
  
  Serial.println("中断优先级示例启动");
}

void loop() {
  static int lastHigh = 0, lastLow = 0;
  
  if (highPriorityCount != lastHigh || lowPriorityCount != lastLow) {
    Serial.printf("高优先级: %d, 低优先级: %d\n", 
                  highPriorityCount, lowPriorityCount);
    lastHigh = highPriorityCount;
    lastLow = lowPriorityCount;
  }
  
  delay(100);
}
```

## 实用示例

### 精确脉冲生成

```cpp
/*
 * 精确脉冲生成示例
 * 使用定时器生成精确的脉冲序列
 */

#include "driver/timer.h"
#include "driver/gpio.h"

#define PULSE_PIN 2
#define TIMER_DIVIDER 80
#define TIMER_SCALE 1000000

// 脉冲参数
volatile int pulseCount = 0;
volatile int targetPulses = 10;
volatile bool pulseActive = false;

// 定时器中断
void IRAM_ATTR pulseTimerISR(void* arg) {
  timer_group_intr_clr_in_isr(TIMER_GROUP_0, TIMER_0);
  
  if (pulseCount < targetPulses * 2) {
    // 切换引脚状态
    int level = gpio_get_level(GPIO_NUM_2);
    gpio_set_level(GPIO_NUM_2, !level);
    
    pulseCount++;
    
    // 设置下一个脉冲时间
    uint64_t alarm_value;
    timer_get_alarm_value(TIMER_GROUP_0, TIMER_0, &alarm_value);
    timer_set_alarm_value(TIMER_GROUP_0, TIMER_0, alarm_value + 100);  // 100us
    timer_group_enable_alarm_in_isr(TIMER_GROUP_0, TIMER_0);
  } else {
    pulseActive = false;
  }
}

void setup() {
  Serial.begin(115200);
  
  // 配置脉冲引脚
  gpio_config_t io_conf = {
    .pin_bit_mask = (1ULL << PULSE_PIN),
    .mode = GPIO_MODE_OUTPUT,
    .pull_up_en = GPIO_PULLUP_DISABLE,
    .pull_down_en = GPIO_PULLDOWN_DISABLE,
    .intr_type = GPIO_INTR_DISABLE
  };
  gpio_config(&io_conf);
  
  // 配置定时器
  timer_config_t config = {
    .divider = TIMER_DIVIDER,
    .counter_dir = TIMER_COUNT_UP,
    .counter_en = TIMER_PAUSE,
    .alarm_en = TIMER_ALARM_EN,
    .auto_reload = TIMER_AUTORELOAD_DIS
  };
  
  timer_init(TIMER_GROUP_0, TIMER_0, &config);
  timer_set_counter_value(TIMER_GROUP_0, TIMER_0, 0);
  timer_enable_intr(TIMER_GROUP_0, TIMER_0);
  timer_isr_register(TIMER_GROUP_0, TIMER_0, pulseTimerISR, NULL, 
                     ESP_INTR_FLAG_IRAM, NULL);
  
  Serial.println("精确脉冲生成示例");
  Serial.println("输入脉冲数量开始生成");
}

void loop() {
  if (Serial.available() > 0) {
    int count = Serial.parseInt();
    if (count > 0 && !pulseActive) {
      targetPulses = count;
      pulseCount = 0;
      pulseActive = true;
      
      // 启动定时器
      timer_set_alarm_value(TIMER_GROUP_0, TIMER_0, 100);
      timer_start(TIMER_GROUP_0, TIMER_0);
      
      Serial.printf("开始生成 %d 个脉冲\n", count);
    }
  }
  
  if (!pulseActive && pulseCount > 0) {
    Serial.println("脉冲生成完成");
    pulseCount = 0;
    timer_pause(TIMER_GROUP_0, TIMER_0);
  }
  
  delay(100);
}
```

### 频率计

```cpp
/*
 * 频率计示例
 * 使用中断测量输入信号频率
 */

#define FREQ_PIN 34

// 测量变量
volatile unsigned long pulseCount = 0;
unsigned long lastMeasureTime = 0;
float frequency = 0;

// 中断服务函数
void IRAM_ATTR pulseISR() {
  pulseCount++;
}

void setup() {
  Serial.begin(115200);
  
  pinMode(FREQ_PIN, INPUT);
  
  // 配置中断
  attachInterrupt(digitalPinToInterrupt(FREQ_PIN), pulseISR, RISING);
  
  lastMeasureTime = millis();
  
  Serial.println("频率计示例启动");
  Serial.println("将信号连接到GPIO34");
}

void loop() {
  unsigned long currentTime = millis();
  
  // 每秒测量一次
  if (currentTime - lastMeasureTime >= 1000) {
    // 禁用中断以安全读取计数
    noInterrupts();
    unsigned long count = pulseCount;
    pulseCount = 0;
    interrupts();
    
    // 计算频率
    float elapsed = (currentTime - lastMeasureTime) / 1000.0;
    frequency = count / elapsed;
    
    Serial.printf("频率: %.2f Hz\n", frequency);
    
    lastMeasureTime = currentTime;
  }
}
```

### 实时时钟模拟

```cpp
/*
 * 简单实时时钟示例
 * 使用定时器维护时间
 */

#include <Ticker.h>

Ticker clockTicker;

// 时间结构
struct Time {
  int hours;
  int minutes;
  int seconds;
} currentTime = {0, 0, 0};

// 更新时间
void IRAM_ATTR updateTime() {
  currentTime.seconds++;
  
  if (currentTime.seconds >= 60) {
    currentTime.seconds = 0;
    currentTime.minutes++;
    
    if (currentTime.minutes >= 60) {
      currentTime.minutes = 0;
      currentTime.hours++;
      
      if (currentTime.hours >= 24) {
        currentTime.hours = 0;
      }
    }
  }
}

void setup() {
  Serial.begin(115200);
  
  // 设置初始时间
  currentTime = {12, 0, 0};  // 12:00:00
  
  // 启动定时器（每秒更新）
  clockTicker.attach(1.0, updateTime);
  
  Serial.println("实时时钟示例启动");
  Serial.println("初始时间: 12:00:00");
}

void loop() {
  // 打印时间
  static int lastSecond = -1;
  
  if (currentTime.seconds != lastSecond) {
    Serial.printf("当前时间: %02d:%02d:%02d\n", 
                  currentTime.hours, 
                  currentTime.minutes, 
                  currentTime.seconds);
    lastSecond = currentTime.seconds;
  }
  
  delay(100);
}
```

## 中断编程注意事项

### ISR编程规则

```cpp
/*
 * 中断服务函数编程规则
 */

// 规则1：ISR必须使用IRAM_ATTR宏
void IRAM_ATTR correctISR() {
  // 正确
}

// 规则2：ISR中不能使用可能阻塞的函数
void IRAM_ATTR badISR() {
  // 错误：Serial.print可能阻塞
  // Serial.println("In ISR");  
  
  // 错误：delay会阻塞
  // delay(10);
  
  // 正确：使用标志变量
  static volatile bool flag = false;
  flag = true;
}

// 规则3：ISR中访问的变量必须使用volatile
volatile int isrCounter = 0;

void IRAM_ATTR counterISR() {
  isrCounter++;  // 正确
}

// 规则4：ISR应该尽量简短
void IRAM_ATTR goodISR() {
  // 只做必要的事情
  static volatile bool state = false;
  state = !state;
  // 复杂处理放在主循环中
}

// 规则5：共享数据需要保护
volatile int sharedData = 0;

void IRAM_ATTR dataISR() {
  sharedData++;
}

void processData() {
  // 读取共享数据时禁用中断
  noInterrupts();
  int data = sharedData;
  interrupts();
  
  // 处理数据
  Serial.println(data);
}
```
