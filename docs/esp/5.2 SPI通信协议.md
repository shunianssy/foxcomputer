# 5.2 SPI通信协议

SPI（Serial Peripheral Interface）是一种高速同步串行通信协议，常用于连接Flash、显示屏、SD卡等高速外设。本节将详细介绍ESP32的SPI编程。

## SPI基础

### SPI特点

- 四线制：MOSI、MISO、SCK、CS
- 全双工通信
- 主从模式
- 高速传输（可达80MHz）
- 简单协议，无地址机制

### ESP32 SPI引脚

ESP32有4个SPI控制器：

| SPI | 用途 | 默认引脚 |
|-----|------|----------|
| SPI0 | Flash | 内部使用 |
| SPI1 | Flash | 内部使用 |
| SPI2 (HSPI) | 用户使用 | MOSI:13, MISO:12, SCK:14, CS:15 |
| SPI3 (VSPI) | 用户使用 | MOSI:23, MISO:19, SCK:18, CS:5 |

### SPI模式

| 模式 | CPOL | CPHA | 说明 |
|------|------|------|------|
| 0 | 0 | 0 | 空闲低电平，第一边沿采样 |
| 1 | 0 | 1 | 空闲低电平，第二边沿采样 |
| 2 | 1 | 0 | 空闲高电平，第一边沿采样 |
| 3 | 1 | 1 | 空闲高电平，第二边沿采样 |

## SPI基本使用

### 使用VSPI

```cpp
/*
 * SPI基本示例
 * 使用VSPI（SPI3）
 */

#include <SPI.h>

// VSPI引脚定义
#define VSPI_MOSI 23
#define VSPI_MISO 19
#define VSPI_SCK  18
#define VSPI_CS   5

// SPI设置
SPIClass* vspi = NULL;
#define SPI_SPEED 1000000  // 1MHz

void setup() {
  Serial.begin(115200);
  
  // 初始化SPI
  vspi = new SPIClass(VSPI);
  vspi->begin(VSPI_SCK, VSPI_MISO, VSPI_MOSI, VSPI_CS);
  
  // 配置CS引脚
  pinMode(VSPI_CS, OUTPUT);
  digitalWrite(VSPI_CS, HIGH);
  
  Serial.println("SPI已初始化");
}

void loop() {
  // 发送并接收数据
  uint8_t txData = 0xAA;
  uint8_t rxData = spiTransfer(txData);
  
  Serial.printf("发送: 0x%02X, 接收: 0x%02X\n", txData, rxData);
  
  delay(1000);
}

// SPI传输单字节
uint8_t spiTransfer(uint8_t data) {
  vspi->beginTransaction(SPISettings(SPI_SPEED, MSBFIRST, SPI_MODE0));
  
  digitalWrite(VSPI_CS, LOW);
  uint8_t rx = vspi->transfer(data);
  digitalWrite(VSPI_CS, HIGH);
  
  vspi->endTransaction();
  
  return rx;
}

// SPI传输多字节
void spiTransferBuffer(uint8_t* txBuf, uint8_t* rxBuf, size_t len) {
  vspi->beginTransaction(SPISettings(SPI_SPEED, MSBFIRST, SPI_MODE0));
  
  digitalWrite(VSPI_CS, LOW);
  vspi->transferBytes(txBuf, rxBuf, len);
  digitalWrite(VSPI_CS, HIGH);
  
  vspi->endTransaction();
}
```

### 使用HSPI

```cpp
/*
 * HSPI使用示例
 */

#include <SPI.h>

#define HSPI_MOSI 13
#define HSPI_MISO 12
#define HSPI_SCK  14
#define HSPI_CS   15

SPIClass* hspi = NULL;

void setup() {
  Serial.begin(115200);
  
  hspi = new SPIClass(HSPI);
  hspi->begin(HSPI_SCK, HSPI_MISO, HSPI_MOSI, HSPI_CS);
  
  pinMode(HSPI_CS, OUTPUT);
  digitalWrite(HSPI_CS, HIGH);
  
  Serial.println("HSPI已初始化");
}

void loop() {
  // 使用HSPI通信
  hspi->beginTransaction(SPISettings(1000000, MSBFIRST, SPI_MODE0));
  
  digitalWrite(HSPI_CS, LOW);
  uint8_t data = hspi->transfer(0x55);
  digitalWrite(HSPI_CS, HIGH);
  
  hspi->endTransaction();
  
  Serial.printf("接收: 0x%02X\n", data);
  delay(1000);
}
```

## SPI设备驱动

### MFRC522 RFID读写器

```cpp
/*
 * MFRC522 RFID驱动示例
 */

#include <SPI.h>
#include <MFRC522.h>

#define RST_PIN  4
#define SS_PIN   5

MFRC522 rfid(SS_PIN, RST_PIN);

void setup() {
  Serial.begin(115200);
  SPI.begin(18, 19, 23, 5);
  
  rfid.PCD_Init();
  
  Serial.println("RFID读取器已初始化");
  Serial.println("请将卡片靠近读取器...");
}

void loop() {
  // 检查是否有新卡片
  if (!rfid.PICC_IsNewCardPresent()) {
    return;
  }
  
  // 读取卡片
  if (!rfid.PICC_ReadCardSerial()) {
    return;
  }
  
  // 打印卡片UID
  Serial.print("卡片UID: ");
  for (byte i = 0; i < rfid.uid.size; i++) {
    Serial.printf("%02X ", rfid.uid.uidByte[i]);
  }
  Serial.println();
  
  // 获取卡片类型
  MFRC522::PICC_Type piccType = rfid.PICC_GetType(rfid.uid.sak);
  Serial.printf("卡片类型: %s\n", rfid.PICC_GetTypeName(piccType).c_str());
  
  delay(1000);
}
```

### SD卡读写

```cpp
/*
 * SD卡读写示例
 */

#include <SPI.h>
#include <SD.h>

#define SD_CS 5

void setup() {
  Serial.begin(115200);
  
  // 初始化SD卡
  if (!SD.begin(SD_CS)) {
    Serial.println("SD卡初始化失败！");
    return;
  }
  
  Serial.println("SD卡初始化成功");
  
  // 获取SD卡信息
  uint8_t cardType = SD.cardType();
  Serial.print("SD卡类型: ");
  switch (cardType) {
    case CARD_NONE: Serial.println("无卡"); break;
    case CARD_MMC:  Serial.println("MMC"); break;
    case CARD_SD:   Serial.println("SD"); break;
    case CARD_SDHC: Serial.println("SDHC"); break;
    default:        Serial.println("未知"); break;
  }
  
  Serial.printf("总容量: %llu MB\n", SD.totalBytes() / (1024 * 1024));
  Serial.printf("已使用: %llu MB\n", SD.usedBytes() / (1024 * 1024));
  
  // 写入文件
  writeFile("/test.txt", "Hello, SD Card!");
  
  // 读取文件
  readFile("/test.txt");
  
  // 列出目录
  listDir("/");
}

void loop() {}

void writeFile(const char* path, const char* message) {
  File file = SD.open(path, FILE_WRITE);
  
  if (!file) {
    Serial.printf("无法打开文件写入: %s\n", path);
    return;
  }
  
  if (file.println(message)) {
    Serial.printf("写入成功: %s\n", path);
  } else {
    Serial.printf("写入失败: %s\n", path);
  }
  
  file.close();
}

void readFile(const char* path) {
  File file = SD.open(path);
  
  if (!file) {
    Serial.printf("无法打开文件读取: %s\n", path);
    return;
  }
  
  Serial.printf("文件内容: %s\n", path);
  while (file.available()) {
    Serial.write(file.read());
  }
  
  file.close();
}

void listDir(const char* dirname) {
  File root = SD.open(dirname);
  
  if (!root) {
    Serial.println("无法打开目录");
    return;
  }
  
  if (!root.isDirectory()) {
    Serial.println("不是目录");
    return;
  }
  
  File file = root.openNextFile();
  while (file) {
    if (file.isDirectory()) {
      Serial.printf("  DIR: %s\n", file.name());
    } else {
      Serial.printf("  FILE: %s, SIZE: %d\n", file.name(), file.size());
    }
    file = root.openNextFile();
  }
}
```

## SPI类封装

```cpp
/*
 * SPI设备类封装
 */

#include <SPI.h>

class SPIDevice {
private:
  SPIClass* spi;
  int csPin;
  uint32_t speed;
  uint8_t mode;
  bool msbFirst;
  
public:
  SPIDevice(SPIClass* spiInstance, int chipSelect, 
            uint32_t clockSpeed = 1000000, 
            uint8_t spiMode = SPI_MODE0,
            bool msb = true) {
    spi = spiInstance;
    csPin = chipSelect;
    speed = clockSpeed;
    mode = spiMode;
    msbFirst = msb;
    
    pinMode(csPin, OUTPUT);
    digitalWrite(csPin, HIGH);
  }
  
  // 传输单字节
  uint8_t transfer(uint8_t data) {
    beginTransaction();
    uint8_t rx = spi->transfer(data);
    endTransaction();
    return rx;
  }
  
  // 传输缓冲区
  void transferBuffer(uint8_t* txBuf, uint8_t* rxBuf, size_t len) {
    beginTransaction();
    spi->transferBytes(txBuf, rxBuf, len);
    endTransaction();
  }
  
  // 写入数据（忽略接收）
  void write(uint8_t* data, size_t len) {
    beginTransaction();
    digitalWrite(csPin, LOW);
    for (size_t i = 0; i < len; i++) {
      spi->transfer(data[i]);
    }
    digitalWrite(csPin, HIGH);
    endTransaction();
  }
  
  // 读取数据
  void read(uint8_t* buffer, size_t len) {
    beginTransaction();
    digitalWrite(csPin, LOW);
    for (size_t i = 0; i < len; i++) {
      buffer[i] = spi->transfer(0xFF);
    }
    digitalWrite(csPin, HIGH);
    endTransaction();
  }
  
  // 写入寄存器
  void writeRegister(uint8_t reg, uint8_t value) {
    beginTransaction();
    digitalWrite(csPin, LOW);
    spi->transfer(reg);
    spi->transfer(value);
    digitalWrite(csPin, HIGH);
    endTransaction();
  }
  
  // 读取寄存器
  uint8_t readRegister(uint8_t reg) {
    beginTransaction();
    digitalWrite(csPin, LOW);
    spi->transfer(reg | 0x80);  // 读标志
    uint8_t value = spi->transfer(0xFF);
    digitalWrite(csPin, HIGH);
    endTransaction();
    return value;
  }
  
private:
  void beginTransaction() {
    spi->beginTransaction(SPISettings(speed, 
                                       msbFirst ? MSBFIRST : LSBFIRST, 
                                       mode));
    digitalWrite(csPin, LOW);
  }
  
  void endTransaction() {
    digitalWrite(csPin, HIGH);
    spi->endTransaction();
  }
};

// 使用示例
SPIClass vspi(VSPI);
SPIDevice* device;

void setup() {
  Serial.begin(115115);
  
  vspi.begin(18, 19, 23, 5);
  device = new SPIDevice(&vspi, 5, 1000000, SPI_MODE0);
  
  // 使用设备
  uint8_t id = device->readRegister(0x00);
  Serial.printf("设备ID: 0x%02X\n", id);
}

void loop() {}
```
