# 2.2 PWM与模拟输出

PWM（Pulse Width Modulation，脉冲宽度调制）是一种通过改变脉冲宽度来模拟模拟信号的技术。ESP32拥有强大的PWM功能，本节将详细介绍其使用方法。

## PWM基础概念

### 什么是PWM

PWM通过快速开关数字信号，控制输出信号的平均电压。主要参数包括：

- **频率（Frequency）**：每秒脉冲的次数，单位Hz
- **占空比（Duty Cycle）**：高电平时间占周期的比例，0%-100%
- **分辨率（Resolution）**：占空比的精度，用位数表示

```
占空比示意：

25%占空比：  |__|__|__|__|__|__|__|__|__|__|__|__
             |--|

50%占空比：  |____|____|____|____|____|____|____
             |----|

75%占空比：  |______|______|______|______|______
             |------|
```

### ESP32 PWM特性

ESP32的LED PWM控制器特性：

- 16个独立通道（通道0-15）
- 可配置频率和占空比
- 最高16位分辨率
- 频率范围：1Hz - 40MHz
- 支持渐变效果
- 可映射到任意GPIO

## LEDC库使用

### 基本PWM输出

```cpp
/*
 * ESP32 LEDC基本示例
 * 控制LED亮度
 */

// LED引脚
#define LED_PIN 2

// PWM参数
#define PWM_CHANNEL 0      // PWM通道（0-15）
#define PWM_FREQ 5000      // PWM频率（Hz）
#define PWM_RESOLUTION 8   // 分辨率（1-16位）

void setup() {
  Serial.begin(115200);
  
  // 配置LED PWM
  // 参数：通道、频率、分辨率
  ledcSetup(PWM_CHANNEL, PWM_FREQ, PWM_RESOLUTION);
  
  // 将通道绑定到GPIO
  ledcAttachPin(LED_PIN, PWM_CHANNEL);
  
  Serial.println("PWM示例启动");
  Serial.printf("通道: %d, 频率: %d Hz, 分辨率: %d 位\n", 
                PWM_CHANNEL, PWM_FREQ, PWM_RESOLUTION);
}

void loop() {
  // 呼吸灯效果
  // 渐亮
  for (int duty = 0; duty <= 255; duty++) {
    ledcWrite(PWM_CHANNEL, duty);
    delay(10);
  }
  
  // 渐暗
  for (int duty = 255; duty >= 0; duty--) {
    ledcWrite(PWM_CHANNEL, duty);
    delay(10);
  }
}
```

### 多通道PWM

```cpp
/*
 * 多通道PWM示例
 * RGB LED控制
 */

// RGB LED引脚
#define RED_PIN   25
#define GREEN_PIN 26
#define BLUE_PIN  27

// PWM通道
#define RED_CHANNEL   0
#define GREEN_CHANNEL 1
#define BLUE_CHANNEL  2

// PWM参数
#define PWM_FREQ 5000
#define PWM_RESOLUTION 8

void setup() {
  Serial.begin(115200);
  
  // 配置三个PWM通道
  ledcSetup(RED_CHANNEL, PWM_FREQ, PWM_RESOLUTION);
  ledcSetup(GREEN_CHANNEL, PWM_FREQ, PWM_RESOLUTION);
  ledcSetup(BLUE_CHANNEL, PWM_FREQ, PWM_RESOLUTION);
  
  // 绑定引脚
  ledcAttachPin(RED_PIN, RED_CHANNEL);
  ledcAttachPin(GREEN_PIN, GREEN_CHANNEL);
  ledcAttachPin(BLUE_PIN, BLUE_CHANNEL);
  
  Serial.println("RGB LED控制启动");
}

void loop() {
  // 红色渐变
  Serial.println("红色渐变");
  for (int i = 0; i <= 255; i++) {
    setRGB(i, 0, 0);
    delay(10);
  }
  
  // 绿色渐变
  Serial.println("绿色渐变");
  for (int i = 0; i <= 255; i++) {
    setRGB(0, i, 0);
    delay(10);
  }
  
  // 蓝色渐变
  Serial.println("蓝色渐变");
  for (int i = 0; i <= 255; i++) {
    setRGB(0, 0, i);
    delay(10);
  }
  
  // 彩虹效果
  Serial.println("彩虹效果");
  for (int hue = 0; hue < 360; hue++) {
    int r, g, b;
    HSVtoRGB(hue, 100, 100, r, g, b);
    setRGB(r, g, b);
    delay(20);
  }
}

// 设置RGB颜色
void setRGB(int r, int g, int b) {
  ledcWrite(RED_CHANNEL, r);
  ledcWrite(GREEN_CHANNEL, g);
  ledcWrite(BLUE_CHANNEL, b);
}

// HSV转RGB
// h: 0-360, s: 0-100, v: 0-100
void HSVtoRGB(int h, int s, int v, int& r, int& g, int& b) {
  float hh, ss, vv;
  float c, x, m;
  
  hh = h / 60.0;
  ss = s / 100.0;
  vv = v / 100.0;
  
  c = vv * ss;
  x = c * (1 - abs(fmod(hh, 2) - 1));
  m = vv - c;
  
  float rf, gf, bf;
  
  if (hh < 1) {
    rf = c; gf = x; bf = 0;
  } else if (hh < 2) {
    rf = x; gf = c; bf = 0;
  } else if (hh < 3) {
    rf = 0; gf = c; bf = x;
  } else if (hh < 4) {
    rf = 0; gf = x; bf = c;
  } else if (hh < 5) {
    rf = x; gf = 0; bf = c;
  } else {
    rf = c; gf = 0; bf = x;
  }
  
  r = (rf + m) * 255;
  g = (gf + m) * 255;
  b = (bf + m) * 255;
}
```

### PWM渐变功能

```cpp
/*
 * PWM渐变功能示例
 * 使用硬件渐变实现平滑过渡
 */

#define LED_PIN 2
#define PWM_CHANNEL 0
#define PWM_FREQ 5000
#define PWM_RESOLUTION 8

void setup() {
  Serial.begin(115200);
  
  ledcSetup(PWM_CHANNEL, PWM_FREQ, PWM_RESOLUTION);
  ledcAttachPin(LED_PIN, PWM_CHANNEL);
  
  Serial.println("PWM渐变示例启动");
}

void loop() {
  // 方式1：软件渐变
  Serial.println("软件渐变");
  ledcFade(LED_PIN, 0, 255, 2000);  // 2秒渐亮
  delay(500);
  ledcFade(LED_PIN, 255, 0, 2000);  // 2秒渐暗
  delay(500);
  
  // 方式2：手动控制
  Serial.println("手动渐变");
  for (int i = 0; i <= 255; i += 5) {
    ledcWrite(PWM_CHANNEL, i);
    delay(20);
  }
  for (int i = 255; i >= 0; i -= 5) {
    ledcWrite(PWM_CHANNEL, i);
    delay(20);
  }
}

// 软件实现渐变
void ledcFade(int pin, int startDuty, int endDuty, int duration) {
  int steps = abs(endDuty - startDuty);
  int stepDelay = duration / steps;
  int step = (endDuty > startDuty) ? 1 : -1;
  
  for (int duty = startDuty; duty != endDuty; duty += step) {
    ledcWrite(PWM_CHANNEL, duty);
    delay(stepDelay);
  }
  
  ledcWrite(PWM_CHANNEL, endDuty);
}
```

## PWM频率与分辨率

### 频率与分辨率关系

PWM频率和分辨率存在权衡关系：

```
最大分辨率 = log2(时钟频率 / PWM频率)

例如：
- 80MHz时钟，5kHz PWM -> 最大分辨率 = log2(16000) = 13.97位
- 80MHz时钟，10kHz PWM -> 最大分辨率 = log2(8000) = 12.97位
- 80MHz时钟，1kHz PWM -> 最大分辨率 = log2(80000) = 16.29位
```

### 不同应用场景的PWM配置

/*
 * 不同场景的PWM配置示例

void setup() {
  Serial.begin(115200);

  // 电机控制：中频中分辨率
  // 频率：10-20kHz，分辨率：8-10位
  ledcSetup(1, 10000, 10);  // 通道1：电机PWM
  
  // 蜂鸣器：音频频率
  // 频率：20Hz-20kHz，分辨率：8位
  ledcSetup(2, 2000, 8);  // 通道2：蜂鸣器
  
  // 伺服电机：固定频率
  // 频率：50Hz，分辨率：16位
  ledcSetup(3, 50, 16);  // 通道3：伺服电机
  
  Serial.println("多场景PWM配置完成");
}

void loop() {
  // 演示不同应用
  demoLed();
  demoMotor();
  demoBuzzer();
  demoServo();
}

void demoLed() {
  Serial.println("LED调光演示");
  ledcAttachPin(2, 0);
  ledcWrite(0, 32768);  // 50%亮度（16位分辨率）
  delay(1000);
}

void demoMotor() {
  Serial.println("电机控制演示");
  ledcAttachPin(4, 1);
  ledcWrite(1, 512);  // 50%速度（10位分辨率）
  delay(1000);
}

void demoBuzzer() {
  Serial.println("蜂鸣器演示");
  ledcAttachPin(5, 2);
  ledcWriteTone(2, 1000);  // 1kHz音调
  delay(500);
  ledcWriteTone(2, 0);  // 关闭
}

void demoServo() {
  Serial.println("伺服电机演示");
  ledcAttachPin(18, 3);
  // 50Hz，16位分辨率
  // 1ms脉冲 = 3276 (约0度)
  // 1.5ms脉冲 = 4915 (约90度)
  // 2ms脉冲 = 6553 (约180度)
  ledcWrite(3, 4915);  // 中间位置
  delay(1000);
}

## 伺服电机控制

```cpp
/*
 * 伺服电机控制示例
 * 使用PWM控制SG90等常见舵机
 */

#define SERVO_PIN 18
#define SERVO_CHANNEL 3
#define SERVO_FREQ 50
#define SERVO_RESOLUTION 16

// 伺服电机脉冲范围（16位分辨率，50Hz）
#define SERVO_MIN 1638   // 0.5ms脉冲（约0度）
#define SERVO_MID 4915   // 1.5ms脉冲（约90度）
#define SERVO_MAX 8192   // 2.5ms脉冲（约180度）

void setup() {
  Serial.begin(115200);
  
  // 配置伺服PWM
  ledcSetup(SERVO_CHANNEL, SERVO_FREQ, SERVO_RESOLUTION);
  ledcAttachPin(SERVO_PIN, SERVO_CHANNEL);
  
  Serial.println("伺服电机控制启动");
  Serial.println("输入0-180控制角度");
}

void loop() {
  // 串口控制
  if (Serial.available() > 0) {
    int angle = Serial.parseInt();
    if (angle >= 0 && angle <= 180) {
      setServoAngle(angle);
      Serial.printf("角度: %d度\n", angle);
    }
  }
  
  // 演示扫描
  static unsigned long lastScan = 0;
  static int angle = 0;
  static bool direction = true;
  
  if (millis() - lastScan > 20) {
    lastScan = millis();
    
    if (direction) {
      angle++;
      if (angle >= 180) direction = false;
    } else {
      angle--;
      if (angle <= 0) direction = true;
    }
    
    setServoAngle(angle);
  }
}

// 设置伺服电机角度
void setServoAngle(int angle) {
  // 将角度映射到脉冲宽度
  int pulse = map(angle, 0, 180, SERVO_MIN, SERVO_MAX);
  ledcWrite(SERVO_CHANNEL, pulse);
}
```

## 直流电机控制

```cpp
/*
 * 直流电机PWM控制示例
 * 使用L298N驱动模块
 */

// 电机引脚
#define MOTOR_ENA 25  // PWM速度控制
#define MOTOR_IN1 26  // 方向控制1
#define MOTOR_IN2 27  // 方向控制2

// PWM参数
#define MOTOR_CHANNEL 0
#define MOTOR_FREQ 10000
#define MOTOR_RESOLUTION 8

// 电机方向
enum MotorDirection {
  FORWARD,
  BACKWARD,
  STOP
};

void setup() {
  Serial.begin(115200);
  
  // 配置PWM
  ledcSetup(MOTOR_CHANNEL, MOTOR_FREQ, MOTOR_RESOLUTION);
  ledcAttachPin(MOTOR_ENA, MOTOR_CHANNEL);
  
  // 配置方向引脚
  pinMode(MOTOR_IN1, OUTPUT);
  pinMode(MOTOR_IN2, OUTPUT);
  
  // 初始停止
  motorStop();
  
  Serial.println("电机控制启动");
  Serial.println("命令: f[速度]前进, b[速度]后退, s停止");
  Serial.println("例如: f128 表示以50%速度前进");
}

void loop() {
  if (Serial.available() > 0) {
    char cmd = Serial.read();
    int speed = Serial.parseInt();
    
    switch (cmd) {
      case 'f':
        motorRun(FORWARD, speed);
        Serial.printf("前进，速度: %d\n", speed);
        break;
      case 'b':
        motorRun(BACKWARD, speed);
        Serial.printf("后退，速度: %d\n", speed);
        break;
      case 's':
        motorStop();
        Serial.println("停止");
        break;
    }
  }
}

// 电机运行
void motorRun(MotorDirection dir, int speed) {
  // 限制速度范围
  speed = constrain(speed, 0, 255);
  
  // 设置方向
  switch (dir) {
    case FORWARD:
      digitalWrite(MOTOR_IN1, HIGH);
      digitalWrite(MOTOR_IN2, LOW);
      break;
    case BACKWARD:
      digitalWrite(MOTOR_IN1, LOW);
      digitalWrite(MOTOR_IN2, HIGH);
      break;
    case STOP:
      digitalWrite(MOTOR_IN1, LOW);
      digitalWrite(MOTOR_IN2, LOW);
      speed = 0;
      break;
  }
  
  // 设置速度
  ledcWrite(MOTOR_CHANNEL, speed);
}

// 电机停止
void motorStop() {
  motorRun(STOP, 0);
}

// 电机刹车
void motorBrake() {
  digitalWrite(MOTOR_IN1, HIGH);
  digitalWrite(MOTOR_IN2, HIGH);
  ledcWrite(MOTOR_CHANNEL, 255);
}
```

## 音频输出

```cpp
/*
 * PWM音频输出示例
 * 简单的蜂鸣器音乐播放
 */

#define BUZZER_PIN 25
#define BUZZER_CHANNEL 4

// 音符频率定义
#define NOTE_C4  262
#define NOTE_D4  294
#define NOTE_E4  330
#define NOTE_F4  349
#define NOTE_G4  392
#define NOTE_A4  440
#define NOTE_B4  494
#define NOTE_C5  523

// 旋律（小星星）
int melody[] = {
  NOTE_C4, NOTE_C4, NOTE_G4, NOTE_G4, NOTE_A4, NOTE_A4, NOTE_G4,
  NOTE_F4, NOTE_F4, NOTE_E4, NOTE_E4, NOTE_D4, NOTE_D4, NOTE_C4
};

// 音符时值（4=四分音符，8=八分音符）
int noteDurations[] = {
  4, 4, 4, 4, 4, 4, 2,
  4, 4, 4, 4, 4, 4, 2
};

void setup() {
  Serial.begin(115200);
  
  // 配置蜂鸣器PWM
  ledcSetup(BUZZER_CHANNEL, 2000, 8);
  ledcAttachPin(BUZZER_PIN, BUZZER_CHANNEL);
  
  Serial.println("音频输出示例启动");
  Serial.println("输入 'p' 播放音乐");
}

void loop() {
  if (Serial.available() > 0) {
    char cmd = Serial.read();
    if (cmd == 'p') {
      playMelody();
    }
  }
}

// 播放旋律
void playMelody() {
  Serial.println("播放音乐...");
  
  int noteCount = sizeof(melody) / sizeof(melody[0]);
  
  for (int i = 0; i < noteCount; i++) {
    // 计算音符持续时间
    int noteDuration = 1000 / noteDurations[i];
    
    // 播放音符
    ledcWriteTone(BUZZER_CHANNEL, melody[i]);
    
    // 等待音符播放
    delay(noteDuration * 0.9);
    
    // 短暂停顿
    ledcWriteTone(BUZZER_CHANNEL, 0);
    delay(noteDuration * 0.1);
  }
  
  Serial.println("播放完成");
}

// 播放单个音符
void playNote(int frequency, int duration) {
  ledcWriteTone(BUZZER_CHANNEL, frequency);
  delay(duration);
  ledcWriteTone(BUZZER_CHANNEL, 0);
}

// 播放警报声
void playAlarm() {
  for (int i = 0; i < 3; i++) {
    ledcWriteTone(BUZZER_CHANNEL, 1000);
    delay(200);
    ledcWriteTone(BUZZER_CHANNEL, 1500);
    delay(200);
  }
  ledcWriteTone(BUZZER_CHANNEL, 0);
}
```

## DAC输出

ESP32内置两个8位DAC通道，可以输出真正的模拟信号。

```cpp
/*
 * DAC输出示例
 * ESP32有两个DAC通道：GPIO25(DAC1)和GPIO26(DAC2)
 */

#define DAC1_PIN 25
#define DAC2_PIN 26

void setup() {
  Serial.begin(115200);
  Serial.println("DAC输出示例启动");
  Serial.println("DAC1: GPIO25, DAC2: GPIO26");
}

void loop() {
  // 生成正弦波
  sineWave();
  
  // 生成三角波
  triangleWave();
  
  // 生成锯齿波
  sawtoothWave();
}

// 正弦波
void sineWave() {
  Serial.println("正弦波输出");
  for (int i = 0; i < 360; i += 5) {
    int value = 128 + 127 * sin(i * PI / 180);
    dacWrite(DAC1_PIN, value);
    delayMicroseconds(100);
  }
}

// 三角波
void triangleWave() {
  Serial.println("三角波输出");
  // 上升
  for (int i = 0; i < 256; i++) {
    dacWrite(DAC1_PIN, i);
    delayMicroseconds(100);
  }
  // 下降
  for (int i = 255; i >= 0; i--) {
    dacWrite(DAC1_PIN, i);
    delayMicroseconds(100);
  }
}

// 锯齿波
void sawtoothWave() {
  Serial.println("锯齿波输出");
  for (int i = 0; i < 256; i++) {
    dacWrite(DAC1_PIN, i);
    delayMicroseconds(100);
  }
}

// 双通道输出
void dualChannel() {
  // DAC1和DAC2输出不同波形
  for (int i = 0; i < 256; i++) {
    dacWrite(DAC1_PIN, i);           // 锯齿波
    dacWrite(DAC2_PIN, 255 - i);     // 反向锯齿波
    delayMicroseconds(100);
  }
}
```
