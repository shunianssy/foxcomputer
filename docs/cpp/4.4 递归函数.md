# 4.4 递归函数

递归是函数调用自身的编程技术。本节介绍递归的原理与应用。

## 递归基础

```cpp
#include <iostream>

// 递归计算阶乘
int factorial(int n) {
 // 基本情况（终止条件）
 if (n <= 1) {
 return 1;
 }
 // 递归情况
 return n * factorial(n - 1);
}

// 递归计算斐波那契数
int fibonacci(int n) {
 if (n <= 1) {
 return n;
 }
 return fibonacci(n - 1) + fibonacci(n - 2);
}

int main() {
 // 阶乘
 for (int i = 0; i <= 5; ++i) {
 std::cout << i << "! = " << factorial(i) << std::endl;
 }
 
 // 斐波那契
 std::cout << "\n斐波那契数列:" << std::endl;
 for (int i = 0; i < 10; ++i) {
 std::cout << fibonacci(i) << " ";
 }
 std::cout << std::endl;
 
 return 0;
}
```

## 递归求和

```cpp
#include <iostream>

// 递归求和 1 + 2 + ... + n
int sum(int n) {
 if (n <= 0) {
 return 0;
 }
 return n + sum(n - 1);
}

// 递归求数组和
int arraySum(int arr[], int size) {
 if (size <= 0) {
 return 0;
 }
 return arr[size - 1] + arraySum(arr, size - 1);
}

int main() {
 std::cout << "1+2+...+10 = " << sum(10) << std::endl;
 
 int arr[] = {1, 2, 3, 4, 5};
 int size = sizeof(arr) / sizeof(arr[0]);
 std::cout << "数组和: " << arraySum(arr, size) << std::endl;
 
 return 0;
}
```

## 汉诺塔问题

```cpp
#include <iostream>

// 汉诺塔递归解法
void hanoi(int n, char from, char to, char aux) {
 if (n == 1) {
 std::cout << "移动盘子 1 从 " << from << " 到 " << to << std::endl;
 return;
 }
 
 // 将 n-1 个盘子从 from 移到 aux
 hanoi(n - 1, from, aux, to);
 
 // 将第 n 个盘子从 from 移到 to
 std::cout << "移动盘子 " << n << " 从 " << from << " 到 " << to << std::endl;
 
 // 将 n-1 个盘子从 aux 移到 to
 hanoi(n - 1, aux, to, from);
}

int main() {
 int n = 3;
 std::cout << "汉诺塔 " << n << " 个盘子:" << std::endl;
 hanoi(n, 'A', 'C', 'B');
 
 return 0;
}
```

## 二分查找

```cpp
#include <iostream>

// 递归二分查找
int binarySearch(int arr[], int left, int right, int target) {
 if (left > right) {
 return -1; // 未找到
 }
 
 int mid = left + (right - left) / 2;
 
 if (arr[mid] == target) {
 return mid;
 } else if (arr[mid] > target) {
 return binarySearch(arr, left, mid - 1, target);
 } else {
 return binarySearch(arr, mid + 1, right, target);
 }
}

int main() {
 int arr[] = {1, 3, 5, 7, 9, 11, 13, 15};
 int size = sizeof(arr) / sizeof(arr[0]);
 int target = 7;
 
 int index = binarySearch(arr, 0, size - 1, target);
 
 if (index != -1) {
 std::cout << "找到 " << target << " 在位置 " << index << std::endl;
 } else {
 std::cout << "未找到 " << target << std::endl;
 }
 
 return 0;
}
```

## 尾递归优化

```cpp
#include <iostream>

// 普通递归
int factorialNormal(int n) {
 if (n <= 1) return 1;
 return n * factorialNormal(n - 1); // 乘法在递归调用之后
}

// 尾递归（可以优化为循环）
int factorialTail(int n, int acc = 1) {
 if (n <= 1) return acc;
 return factorialTail(n - 1, n * acc); // 递归调用是最后操作
}

int main() {
 std::cout << "普通递归: " << factorialNormal(5) << std::endl;
 std::cout << "尾递归: " << factorialTail(5) << std::endl;
 
 return 0;
}
```

## 小结

递归函数要点：

1. **基本情况**：必须有终止条件
2. **递归情况**：问题规模逐渐缩小
3. **栈溢出**：递归深度过大会导致栈溢出
4. **尾递归**：可以优化为循环，避免栈溢出

> **提示**：递归代码简洁，但要注意终止条件和递归深度。深度递归考虑使用迭代替代。
