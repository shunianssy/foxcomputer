# 1.3 Hello World

本节将带你编写第一个 C++ 程序，并详细解析程序的每个部分。

## 第一个 C++ 程序

```cpp
// my_first_program.cpp
// 我的第一个 C++ 程序

#include <iostream>

int main() {
    std::cout << "Hello, World!" << std::endl;
    return 0;
}
```

## 程序解析

### 1. 注释

```cpp
// 这是单行注释

/*
 * 这是多行注释
 * 可以跨越多行
 */

/**
 * 这是文档注释
 * 通常用于函数和类的说明
 * @param argc 参数个数
 * @param argv 参数数组
 * @return 程序退出码
 */
```

### 2. 头文件包含

```cpp
#include <iostream>  // 输入输出流库

// 其他常用头文件
#include <string>    // 字符串
#include <vector>    // 动态数组
#include <map>       // 映射容器
#include <algorithm> // 算法库
#include <cmath>     // 数学函数
```

### 3. main 函数

```cpp
// 标准形式
int main() {
    return 0;
}

// 带命令行参数的形式
int main(int argc, char* argv[]) {
    // argc: 参数个数（包括程序名）
    // argv: 参数字符串数组
    return 0;
}

// 另一种写法
int main(int argc, char** argv) {
    return 0;
}
```

### 4. 输出语句

```cpp
#include <iostream>

int main() {
    // 基本输出
    std::cout << "Hello";
    
    // 输出多个内容
    std::cout << "Hello" << ", " << "World" << "!" << std::endl;
    
    // 输出变量
    int age = 25;
    std::cout << "年龄: " << age << std::endl;
    
    // 换行符
    std::cout << "第一行" << std::endl;
    std::cout << "第二行\n";  // \n 也可以换行
    
    return 0;
}
```

## 输入输出详解

### 标准输出 (cout)

```cpp
#include <iostream>
#include <iomanip>  // 格式化输出

int main() {
    // 基本输出
    std::cout << "Hello, World!" << std::endl;
    
    // 输出数字
    int integer = 42;
    double pi = 3.14159265;
    
    std::cout << "整数: " << integer << std::endl;
    std::cout << "浮点数: " << pi << std::endl;
    
    // 格式化输出
    std::cout << std::fixed;                    // 固定小数点
    std::cout << std::setprecision(2);          // 保留 2 位小数
    std::cout << "Pi (2位小数): " << pi << std::endl;
    
    // 宽度和对齐
    std::cout << std::setw(10) << 42 << std::endl;     // 宽度 10
    std::cout << std::left << std::setw(10) << 42 << std::endl;  // 左对齐
    
    return 0;
}
```

### 标准输入 (cin)

```cpp
#include <iostream>
#include <string>

int main() {
    // 输入整数
    int age;
    std::cout << "请输入年龄: ";
    std::cin >> age;
    std::cout << "你的年龄是: " << age << std::endl;
    
    // 输入字符串（不含空格）
    std::string name;
    std::cout << "请输入名字: ";
    std::cin >> name;
    std::cout << "你好, " << name << "!" << std::endl;
    
    // 输入包含空格的字符串
    std::cin.ignore();  // 清除缓冲区
    std::string fullName;
    std::cout << "请输入全名: ";
    std::getline(std::cin, fullName);
    std::cout << "全名: " << fullName << std::endl;
    
    // 输入多个值
    int a, b;
    std::cout << "请输入两个数字（用空格分隔）: ";
    std::cin >> a >> b;
    std::cout << "和: " << a + b << std::endl;
    
    return 0;
}
```

## 命令行参数

```cpp
#include <iostream>

int main(int argc, char* argv[]) {
    // argc: 参数个数
    // argv: 参数数组
    
    std::cout << "参数个数: " << argc << std::endl;
    
    // 打印所有参数
    for (int i = 0; i < argc; ++i) {
        std::cout << "参数 " << i << ": " << argv[i] << std::endl;
    }
    
    // 检查参数
    if (argc < 2) {
        std::cerr << "用法: " << argv[0] << " <名字>" << std::endl;
        return 1;  // 非零返回值表示错误
    }
    
    std::cout << "你好, " << argv[1] << "!" << std::endl;
    
    return 0;
}
```

## 实用示例

### 简单计算器

```cpp
#include <iostream>

int main() {
    double num1, num2;
    char op;
    
    std::cout << "简单计算器" << std::endl;
    std::cout << "输入格式: 数字 运算符 数字" << std::endl;
    std::cout << "例如: 5 + 3" << std::endl;
    std::cout << "请输入: ";
    
    std::cin >> num1 >> op >> num2;
    
    double result;
    bool valid = true;
    
    switch (op) {
        case '+':
            result = num1 + num2;
            break;
        case '-':
            result = num1 - num2;
            break;
        case '*':
            result = num1 * num2;
            break;
        case '/':
            if (num2 != 0) {
                result = num1 / num2;
            } else {
                std::cout << "错误: 除数不能为零!" << std::endl;
                valid = false;
            }
            break;
        default:
            std::cout << "错误: 无效的运算符!" << std::endl;
            valid = false;
    }
    
    if (valid) {
        std::cout << num1 << " " << op << " " << num2 << " = " << result << std::endl;
    }
    
    return 0;
}
```

### 温度转换

```cpp
#include <iostream>
#include <iomanip>

int main() {
    double celsius, fahrenheit;
    
    // 摄氏转华氏
    std::cout << "请输入摄氏温度: ";
    std::cin >> celsius;
    
    fahrenheit = celsius * 9.0 / 5.0 + 32;
    
    std::cout << std::fixed << std::setprecision(2);
    std::cout << celsius << "°C = " << fahrenheit << "°F" << std::endl;
    
    // 华氏转摄氏
    std::cout << "\n请输入华氏温度: ";
    std::cin >> fahrenheit;
    
    celsius = (fahrenheit - 32) * 5.0 / 9.0;
    
    std::cout << fahrenheit << "°F = " << celsius << "°C" << std::endl;
    
    return 0;
}
```

### 交互式问候

```cpp
#include <iostream>
#include <string>

int main() {
    std::string name;
    int age;
    std::string city;
    
    std::cout << "=== 个人信息收集 ===" << std::endl;
    
    std::cout << "请输入你的名字: ";
    std::getline(std::cin, name);
    
    std::cout << "请输入你的年龄: ";
    std::cin >> age;
    
    std::cin.ignore();  // 清除换行符
    std::cout << "请输入你的城市: ";
    std::getline(std::cin, city);
    
    // 输出收集的信息
    std::cout << "\n=== 信息汇总 ===" << std::endl;
    std::cout << "姓名: " << name << std::endl;
    std::cout << "年龄: " << age << " 岁" << std::endl;
    std::cout << "城市: " << city << std::endl;
    std::cout << "\n你好, " << name << "! 欢迎来自 " << city << " 的朋友!" << std::endl;
    
    return 0;
}
```

## 返回值

```cpp
#include <iostream>
#include <cstdlib>  // EXIT_SUCCESS, EXIT_FAILURE

int main() {
    bool success = true;
    
    if (success) {
        std::cout << "程序执行成功!" << std::endl;
        return EXIT_SUCCESS;  // 返回 0
        // 或 return 0;
    } else {
        std::cerr << "程序执行失败!" << std::endl;
        return EXIT_FAILURE;  // 返回 1
        // 或 return 1;
    }
}
```

## 小结

Hello World 程序要点：

1. **程序结构**：头文件包含 → main 函数 → 语句 → 返回值
2. **输入输出**：`std::cout` 输出，`std::cin` 输入
3. **注释**：单行 `//`，多行 `/* */`
4. **编译运行**：`g++ file.cpp -o output && ./output`

> 💡 **提示**：Hello World 是学习任何编程语言的第一步。理解程序的每个部分，为后续学习打下坚实基础。
