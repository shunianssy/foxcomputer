# 2.3 运算符

运算符是执行特定操作的符号。本节介绍 C++ 中的各类运算符及其使用方法。

## 算术运算符

```cpp
#include <iostream>

int main() {
    int a = 10, b = 3;
    
    // 基本算术运算
    std::cout << "a = " << a << ", b = " << b << std::endl;
    std::cout << "a + b = " << a + b << std::endl;   // 加法：13
    std::cout << "a - b = " << a - b << std::endl;   // 减法：7
    std::cout << "a * b = " << a * b << std::endl;   // 乘法：30
    std::cout << "a / b = " << a / b << std::endl;   // 整数除法：3
    std::cout << "a % b = " << a % b << std::endl;   // 取模：1
    
    // 浮点除法
    double x = 10.0, y = 3.0;
    std::cout << "10.0 / 3.0 = " << x / y << std::endl;  // 3.333...
    
    // 整数除法注意事项
    int c = 5, d = 2;
    std::cout << "5 / 2 = " << c / d << std::endl;       // 2（整数除法）
    std::cout << "5.0 / 2 = " << 5.0 / d << std::endl;   // 2.5（浮点除法）
    
    // 取模运算
    std::cout << "10 % 3 = " << 10 % 3 << std::endl;     // 1
    std::cout << "-10 % 3 = " << -10 % 3 << std::endl;   // -1（结果符号与被除数相同）
    
    return 0;
}
```

### 自增自减运算符

```cpp
#include <iostream>

int main() {
    int a = 5;
    
    // 前置自增：先加 1，再使用
    int b = ++a;  // a = 6, b = 6
    std::cout << "前置自增: a = " << a << ", b = " << b << std::endl;
    
    // 后置自增：先使用，再加 1
    int c = a++;  // c = 6, a = 7
    std::cout << "后置自增: a = " << a << ", c = " << c << std::endl;
    
    // 前置自减
    int d = --a;  // a = 6, d = 6
    std::cout << "前置自减: a = " << a << ", d = " << d << std::endl;
    
    // 后置自减
    int e = a--;  // e = 6, a = 5
    std::cout << "后置自减: a = " << a << ", e = " << e << std::endl;
    
    // 在表达式中的使用
    int x = 5;
    int result = x++ + ++x;  // 不推荐：行为可能未定义
    std::cout << "result = " << result << std::endl;
    
    // 推荐写法
    x = 5;
    int temp1 = x;   // 5
    x++;             // x = 6
    ++x;             // x = 7
    int temp2 = x;   // 7
    result = temp1 + temp2;  // 12
    std::cout << "推荐写法: result = " << result << std::endl;
    
    return 0;
}
```

## 关系运算符

```cpp
#include <iostream>
#include <iomanip>

int main() {
    int a = 10, b = 5;
    
    std::cout << std::boolalpha;  // 输出 true/false
    
    // 关系运算符
    std::cout << "a = " << a << ", b = " << b << std::endl;
    std::cout << "a == b: " << (a == b) << std::endl;  // 等于：false
    std::cout << "a != b: " << (a != b) << std::endl;  // 不等于：true
    std::cout << "a > b: " << (a > b) << std::endl;    // 大于：true
    std::cout << "a < b: " << (a < b) << std::endl;    // 小于：false
    std::cout << "a >= b: " << (a >= b) << std::endl;  // 大于等于：true
    std::cout << "a <= b: " << (a <= b) << std::endl;  // 小于等于：false
    
    // 浮点数比较（注意精度问题）
    double x = 0.1 + 0.2;
    double y = 0.3;
    std::cout << "\n0.1 + 0.2 == 0.3: " << (x == y) << std::endl;  // 可能是 false！
    std::cout << "实际值: " << std::setprecision(17) << x << std::endl;
    
    // 正确的浮点数比较方式
    double epsilon = 1e-9;
    bool isEqual = std::abs(x - y) < epsilon;
    std::cout << "使用 epsilon 比较: " << isEqual << std::endl;
    
    return 0;
}
```

## 逻辑运算符

```cpp
#include <iostream>

int main() {
    bool a = true, b = false;
    
    std::cout << std::boolalpha;
    
    // 逻辑运算符
    std::cout << "a = " << a << ", b = " << b << std::endl;
    std::cout << "a && b: " << (a && b) << std::endl;  // 逻辑与：false
    std::cout << "a || b: " << (a || b) << std::endl;  // 逻辑或：true
    std::cout << "!a: " << (!a) << std::endl;          // 逻辑非：false
    std::cout << "!b: " << (!b) << std::endl;          // true
    
    // 短路求值
    int x = 5;
    bool result = (x > 0) && (++x > 5);  // 第一个条件为真，继续执行
    std::cout << "x = " << x << std::endl;  // x = 6
    
    x = 5;
    result = (x < 0) && (++x > 5);  // 第一个条件为假，短路
    std::cout << "短路后 x = " << x << std::endl;  // x = 5（未执行 ++x）
    
    // 实际应用
    int age = 25;
    bool hasLicense = true;
    
    // 检查是否可以驾驶
    if (age >= 18 && hasLicense) {
        std::cout << "可以驾驶" << std::endl;
    }
    
    // 检查年龄范围
    if (age < 0 || age > 150) {
        std::cout << "年龄无效" << std::endl;
    } else {
        std::cout << "年龄有效" << std::endl;
    }
    
    return 0;
}
```

## 位运算符

```cpp
#include <iostream>
#include <bitset>

int main() {
    unsigned int a = 0b1100;  // 12
    unsigned int b = 0b1010;  // 10
    
    // 位运算符
    std::cout << "a = " << std::bitset<4>(a) << " (" << a << ")" << std::endl;
    std::cout << "b = " << std::bitset<4>(b) << " (" << b << ")" << std::endl;
    
    std::cout << "a & b = " << std::bitset<4>(a & b)   // 按位与：1000 (8)
              << " (" << (a & b) << ")" << std::endl;
    std::cout << "a | b = " << std::bitset<4>(a | b)   // 按位或：1110 (14)
              << " (" << (a | b) << ")" << std::endl;
    std::cout << "a ^ b = " << std::bitset<4>(a ^ b)   // 按位异或：0110 (6)
              << " (" << (a ^ b) << ")" << std::endl;
    std::cout << "~a = " << std::bitset<8>(~a)         // 按位取反
              << " (" << (~a) << ")" << std::endl;
    
    // 移位运算
    std::cout << "a << 2 = " << (a << 2) << std::endl;  // 左移：48
    std::cout << "a >> 2 = " << (a >> 2) << std::endl;  // 右移：3
    
    // 位运算应用
    // 1. 判断奇偶
    int num = 7;
    bool isOdd = num & 1;
    std::cout << num << " 是奇数: " << std::boolalpha << isOdd << std::endl;
    
    // 2. 交换两个数
    int x = 5, y = 10;
    x ^= y;
    y ^= x;
    x ^= y;
    std::cout << "交换后: x = " << x << ", y = " << y << std::endl;
    
    // 3. 快速乘除 2
    int value = 10;
    std::cout << value << " * 2 = " << (value << 1) << std::endl;
    std::cout << value << " / 2 = " << (value >> 1) << std::endl;
    
    // 4. 设置/清除/切换位
    unsigned char flags = 0b00000000;
    
    flags |= (1 << 2);   // 设置第 2 位
    std::cout << "设置第2位: " << std::bitset<8>(flags) << std::endl;
    
    flags &= ~(1 << 2);  // 清除第 2 位
    std::cout << "清除第2位: " << std::bitset<8>(flags) << std::endl;
    
    flags ^= (1 << 3);   // 切换第 3 位
    std::cout << "切换第3位: " << std::bitset<8>(flags) << std::endl;
    
    return 0;
}
```

## 赋值运算符

```cpp
#include <iostream>

int main() {
    int a = 10;
    
    // 基本赋值
    a = 20;
    std::cout << "a = " << a << std::endl;
    
    // 复合赋值运算符
    a += 5;   // a = a + 5 = 25
    std::cout << "a += 5: " << a << std::endl;
    
    a -= 3;   // a = a - 3 = 22
    std::cout << "a -= 3: " << a << std::endl;
    
    a *= 2;   // a = a * 2 = 44
    std::cout << "a *= 2: " << a << std::endl;
    
    a /= 4;   // a = a / 4 = 11
    std::cout << "a /= 4: " << a << std::endl;
    
    a %= 3;   // a = a % 3 = 2
    std::cout << "a %= 3: " << a << std::endl;
    
    // 位运算复合赋值
    a = 0b1100;
    a &= 0b1010;  // a = a & 0b1010
    std::cout << "a &= 0b1010: " << a << std::endl;
    
    a |= 0b0001;
    std::cout << "a |= 0b0001: " << a << std::endl;
    
    a ^= 0b1111;
    std::cout << "a ^= 0b1111: " << a << std::endl;
    
    a <<= 2;
    std::cout << "a <<= 2: " << a << std::endl;
    
    a >>= 1;
    std::cout << "a >>= 1: " << a << std::endl;
    
    return 0;
}
```

## 条件运算符（三元运算符）

```cpp
#include <iostream>
#include <string>

int main() {
    int a = 10, b = 20;
    
    // 基本用法
    int max = (a > b) ? a : b;
    std::cout << "最大值: " << max << std::endl;
    
    // 嵌套使用
    int score = 75;
    std::string grade = (score >= 90) ? "A" :
                        (score >= 80) ? "B" :
                        (score >= 70) ? "C" :
                        (score >= 60) ? "D" : "F";
    std::cout << "成绩等级: " << grade << std::endl;
    
    // 在输出中使用
    int count = 1;
    std::cout << "有 " << count << " 个项目" << std::endl;
    std::cout << "有 " << count << " 个项目" << (count > 1 ? "们" : "") << std::endl;
    
    // 条件表达式作为左值
    int x = 5, y = 10;
    int* p = nullptr;
    
    // 选择要修改的变量
    (x > y ? x : y) = 100;  // y 被修改为 100
    std::cout << "x = " << x << ", y = " << y << std::endl;
    
    return 0;
}
```

## 逗号运算符

```cpp
#include <iostream>

int main() {
    int a = 5, b = 10;
    
    // 逗号运算符：从左到右依次执行，返回最后一个表达式的值
    int result = (a++, b++, a + b);
    std::cout << "result = " << result << std::endl;  // 18
    std::cout << "a = " << a << ", b = " << b << std::endl;  // a = 6, b = 11
    
    // 在 for 循环中使用
    for (int i = 0, j = 10; i < j; ++i, --j) {
        std::cout << "i = " << i << ", j = " << j << std::endl;
    }
    
    return 0;
}
```

## sizeof 运算符

```cpp
#include <iostream>
#include <string>
#include <vector>

int main() {
    // 基本类型大小
    std::cout << "sizeof(char): " << sizeof(char) << std::endl;
    std::cout << "sizeof(int): " << sizeof(int) << std::endl;
    std::cout << "sizeof(double): " << sizeof(double) << std::endl;
    std::cout << "sizeof(bool): " << sizeof(bool) << std::endl;
    
    // 变量大小
    int arr[10];
    std::cout << "sizeof(arr): " << sizeof(arr) << std::endl;       // 40
    std::cout << "数组元素个数: " << sizeof(arr) / sizeof(arr[0]) << std::endl;  // 10
    
    // 对象大小
    std::string str = "Hello";
    std::cout << "sizeof(std::string): " << sizeof(str) << std::endl;
    
    // 容器大小（注意：不是元素总大小）
    std::vector<int> vec = {1, 2, 3, 4, 5};
    std::cout << "sizeof(std::vector<int>): " << sizeof(vec) << std::endl;
    
    // sizeof 用于表达式
    double d = 3.14;
    std::cout << "sizeof(d + 1): " << sizeof(d + 1) << std::endl;  // 8
    
    return 0;
}
```

## 运算符优先级

```cpp
#include <iostream>

int main() {
    // 运算符优先级示例
    
    // 乘除优先于加减
    int a = 2 + 3 * 4;  // 14，不是 20
    std::cout << "2 + 3 * 4 = " << a << std::endl;
    
    // 使用括号改变优先级
    int b = (2 + 3) * 4;  // 20
    std::cout << "(2 + 3) * 4 = " << b << std::endl;
    
    // 比较运算符优先级
    bool result = 5 > 3 && 2 < 4;  // true
    std::cout << "5 > 3 && 2 < 4: " << std::boolalpha << result << std::endl;
    
    // 位运算优先级较低
    int c = 5 & 3 + 1;   // 5 & 4 = 4
    std::cout << "5 & 3 + 1 = " << c << std::endl;
    
    int d = (5 & 3) + 1; // 1 + 1 = 2
    std::cout << "(5 & 3) + 1 = " << d << std::endl;
    
    // 推荐使用括号明确优先级
    int e = 5 & (3 + 1);  // 明确意图
    
    return 0;
}
```

### 优先级表（从高到低）

| 优先级 | 运算符 | 说明 |
|--------|--------|------|
| 1 | `::` | 作用域解析 |
| 2 | `()` `[]` `->` `.` `++` `--` | 后缀运算符 |
| 3 | `++` `--` `+` `-` `!` `~` `*` `&` `sizeof` | 前缀运算符 |
| 4 | `.*` `->*` | 成员指针 |
| 5 | `*` `/` `%` | 乘除 |
| 6 | `+` `-` | 加减 |
| 7 | `<<` `>>` | 移位 |
| 8 | `<` `<=` `>` `>=` | 比较 |
| 9 | `==` `!=` | 相等 |
| 10 | `&` | 按位与 |
| 11 | `^` | 按位异或 |
| 12 | `\|` | 按位或 |
| 13 | `&&` | 逻辑与 |
| 14 | `\|\|` | 逻辑或 |
| 15 | `?:` | 条件 |
| 16 | `=` `+=` `-=` `*=` `/=` 等 | 赋值 |
| 17 | `,` | 逗号 |

## 小结

运算符要点：

1. **算术运算符**：`+` `-` `*` `/` `%`，注意整数除法
2. **关系运算符**：`==` `!=` `>` `<` `>=` `<=`，浮点数比较要小心
3. **逻辑运算符**：`&&` `||` `!`，注意短路求值
4. **位运算符**：`&` `|` `^` `~` `<<` `>>`，用于位操作
5. **优先级**：不确定时使用括号

> 💡 **提示**：当不确定运算符优先级时，使用括号明确表达意图，这样代码更清晰、更安全。
