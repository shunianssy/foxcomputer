# 2.1 积木块类型详解

Scratch 扩展支持多种积木块类型，每种类型有不同的用途和表现形式。

## 积木块类型概览

| 类型 | 英文名 | 形状 | 用途 |
|-----|-------|-----|-----|
| 命令块 | COMMAND | 矩形 | 执行操作，无返回值 |
| 报告块 | REPORTER | 椭圆 | 返回值，可嵌入其他积木 |
| 布尔块 | BOOLEAN | 六边形 | 返回 true/false |
| 帽子块 | HAT | 顶部凸起 | 事件触发，程序入口 |
| 条件块 | CONDITIONAL | C 形 | 条件分支控制 |
| 循环块 | LOOP | C 形带箭头 | 循环控制 |

## 命令块 (COMMAND)

命令块用于执行操作，不返回任何值。

### 基本语法

```javascript
{
 opcode: 'moveSteps', // 操作码，对应方法名
 blockType: Scratch.BlockType.COMMAND,
 text: '移动 [STEPS] 步', // 显示文本
 arguments: { // 参数定义
 STEPS: {
 type: Scratch.ArgumentType.NUMBER,
 defaultValue: 10
 }
 }
}
```

### 完整示例

```javascript
class MotionExtension {
 constructor(runtime) {
 this.runtime = runtime;
 }

 getInfo() {
 return {
 id: 'motionExt',
 name: '运动扩展',
 color1: '#4C97FF',
 blocks: [
 // 基本命令块
 {
 opcode: 'moveForward',
 blockType: Scratch.BlockType.COMMAND,
 text: '向前移动 [DISTANCE] 步'
 },
 // 带多个参数的命令块
 {
 opcode: 'glideTo',
 blockType: Scratch.BlockType.COMMAND,
 text: '在 [SECS] 秒内滑行到 x: [X] y: [Y]'
 },
 // 无参数命令块
 {
 opcode: 'resetPosition',
 blockType: Scratch.BlockType.COMMAND,
 text: '重置位置'
 }
 ],
 arguments: {
 DISTANCE: { type: Scratch.ArgumentType.NUMBER, defaultValue: 10 },
 SECS: { type: Scratch.ArgumentType.NUMBER, defaultValue: 1 },
 X: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0 },
 Y: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0 }
 }
 };
 }

 /**
 * 向前移动
 * @param {object} args - 参数对象
 * @param {number} args.DISTANCE - 移动距离
 */
 moveForward(args) {
 const distance = Number(args.DISTANCE) || 0;
 // 执行移动逻辑
 console.log(`向前移动 ${distance} 步`);
 }

 /**
 * 滑行到指定位置
 * @param {object} args - 参数对象
 */
 glideTo(args) {
 const secs = Number(args.SECS) || 1;
 const x = Number(args.X) || 0;
 const y = Number(args.Y) || 0;
 // 执行滑行逻辑
 return new Promise(resolve => {
 setTimeout(() => {
 console.log(`滑行到 (${x}, ${y})`);
 resolve();
 }, secs * 1000);
 });
 }

 /**
 * 重置位置
 */
 resetPosition() {
 console.log('位置已重置');
 }
}
```

### 异步命令块

```javascript
/**
 * 异步命令块示例
 * 使用 Promise 实现等待效果
 */
{
 opcode: 'waitSeconds',
 blockType: Scratch.BlockType.COMMAND,
 text: '等待 [SECS] 秒'
}

waitSeconds(args) {
 const secs = Number(args.SECS) || 0;
 return new Promise(resolve => {
 setTimeout(resolve, secs * 1000);
 });
}
```

## 报告块 (REPORTER)

报告块返回一个值，可以嵌入到其他积木块中使用。

### 基本语法

```javascript
{
 opcode: 'getRandomNumber',
 blockType: Scratch.BlockType.REPORTER,
 text: '从 [FROM] 到 [TO] 的随机数',
 arguments: {
 FROM: { type: Scratch.ArgumentType.NUMBER, defaultValue: 1 },
 TO: { type: Scratch.ArgumentType.NUMBER, defaultValue: 10 }
 }
}
```

### 完整示例

```javascript
class MathExtension {
 getInfo() {
 return {
 id: 'mathExt',
 name: '数学扩展',
 color1: '#FF8C1A',
 blocks: [
 // 数字报告块
 {
 opcode: 'add',
 blockType: Scratch.BlockType.REPORTER,
 text: '[A] + [B]',
 arguments: {
 A: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0 },
 B: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0 }
 }
 },
 // 字符串报告块
 {
 opcode: 'join',
 blockType: Scratch.BlockType.REPORTER,
 text: '连接 [FIRST] 和 [SECOND]',
 arguments: {
 FIRST: { type: Scratch.ArgumentType.STRING, defaultValue: '你好' },
 SECOND: { type: Scratch.ArgumentType.STRING, defaultValue: '世界' }
 }
 },
 // 带禁用选项的报告块
 {
 opcode: 'getCurrentTime',
 blockType: Scratch.BlockType.REPORTER,
 text: '当前时间',
 disableMonitor: false // 是否禁用监视器
 }
 ]
 };
 }

 /**
 * 加法运算
 * @param {object} args
 * @returns {number}
 */
 add(args) {
 const a = Number(args.A) || 0;
 const b = Number(args.B) || 0;
 return a + b;
 }

 /**
 * 字符串连接
 * @param {object} args
 * @returns {string}
 */
 join(args) {
 return String(args.FIRST) + String(args.SECOND);
 }

 /**
 * 获取当前时间
 * @returns {string}
 */
 getCurrentTime() {
 const now = new Date();
 return now.toLocaleTimeString('zh-CN');
 }
}
```

### 异步报告块

```javascript
/**
 * 异步报告块 - 调用 API 获取数据
 */
{
 opcode: 'fetchData',
 blockType: Scratch.BlockType.REPORTER,
 text: '从 [URL] 获取数据'
}

async fetchData(args) {
 try {
 const response = await fetch(args.URL);
 const data = await response.json();
 return JSON.stringify(data);
 } catch (error) {
 console.error('获取数据失败:', error);
 return '';
 }
}
```

## 布尔块 (BOOLEAN)

布尔块返回 true 或 false，通常用于条件判断。

### 基本语法

```javascript
{
 opcode: 'isEven',
 blockType: Scratch.BlockType.BOOLEAN,
 text: '[NUM] 是偶数？',
 arguments: {
 NUM: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0 }
 }
}
```

### 完整示例

```javascript
class LogicExtension {
 getInfo() {
 return {
 id: 'logicExt',
 name: '逻辑扩展',
 color1: '#FF8C1A',
 blocks: [
 // 数字判断
 {
 opcode: 'isEven',
 blockType: Scratch.BlockType.BOOLEAN,
 text: '[NUM] 是偶数？',
 arguments: {
 NUM: { type: Scratch.ArgumentType.NUMBER, defaultValue: 2 }
 }
 },
 // 字符串判断
 {
 opcode: 'contains',
 blockType: Scratch.BlockType.BOOLEAN,
 text: '[STRING] 包含 [SUBSTRING]？',
 arguments: {
 STRING: { type: Scratch.ArgumentType.STRING, defaultValue: 'Hello World' },
 SUBSTRING: { type: Scratch.ArgumentType.STRING, defaultValue: 'World' }
 }
 },
 // 范围判断
 {
 opcode: 'inRange',
 blockType: Scratch.BlockType.BOOLEAN,
 text: '[NUM] 在 [MIN] 到 [MAX] 之间？',
 arguments: {
 NUM: { type: Scratch.ArgumentType.NUMBER, defaultValue: 5 },
 MIN: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0 },
 MAX: { type: Scratch.ArgumentType.NUMBER, defaultValue: 10 }
 }
 }
 ]
 };
 }

 /**
 * 判断是否为偶数
 * @param {object} args
 * @returns {boolean}
 */
 isEven(args) {
 const num = Number(args.NUM);
 return num % 2 === 0;
 }

 /**
 * 判断字符串是否包含子串
 * @param {object} args
 * @returns {boolean}
 */
 contains(args) {
 const str = String(args.STRING);
 const sub = String(args.SUBSTRING);
 return str.includes(sub);
 }

 /**
 * 判断数字是否在范围内
 * @param {object} args
 * @returns {boolean}
 */
 inRange(args) {
 const num = Number(args.NUM);
 const min = Number(args.MIN);
 const max = Number(args.MAX);
 return num >= min && num <= max;
 }
}
```

## 帽子块 (HAT)

帽子块是程序的入口点，当特定事件发生时触发执行。

### 基本语法

```javascript
{
 opcode: 'whenKeyPressed',
 blockType: Scratch.BlockType.HAT,
 text: '当按下 [KEY] 键',
 arguments: {
 KEY: { type: Scratch.ArgumentType.STRING, menu: 'keys' }
 },
 isEdgeActivated: false // 是否边缘激活
}
```

### 完整示例

```javascript
class EventExtension {
 constructor(runtime) {
 this.runtime = runtime;
 this.keyPressed = null;
 
 // 监听键盘事件
 document.addEventListener('keydown', (e) => {
 this.keyPressed = e.key;
 this.runtime.startHats('eventExt_whenKeyPressed', {
 KEY: e.key
 });
 });
 }

 getInfo() {
 return {
 id: 'eventExt',
 name: '事件扩展',
 color1: '#FFBF00',
 blocks: [
 // 键盘事件帽子块
 {
 opcode: 'whenKeyPressed',
 blockType: Scratch.BlockType.HAT,
 text: '当按下 [KEY] 键',
 arguments: {
 KEY: {
 type: Scratch.ArgumentType.STRING,
 menu: 'keys',
 defaultValue: 'space'
 }
 },
 isEdgeActivated: false
 },
 // 定时器帽子块
 {
 opcode: 'whenTimer',
 blockType: Scratch.BlockType.HAT,
 text: '每 [SECS] 秒',
 arguments: {
 SECS: { type: Scratch.ArgumentType.NUMBER, defaultValue: 1 }
 }
 }
 ],
 menus: {
 keys: {
 acceptReporters: true,
 items: [
 { text: '空格', value: 'space' },
 { text: '上', value: 'ArrowUp' },
 { text: '下', value: 'ArrowDown' },
 { text: '左', value: 'ArrowLeft' },
 { text: '右', value: 'ArrowRight' }
 ]
 }
 }
 };
 }

 /**
 * 键盘按下事件
 * @param {object} args
 * @returns {boolean} 是否匹配
 */
 whenKeyPressed(args) {
 return args.KEY === this.keyPressed;
 }

 /**
 * 定时器事件
 * @param {object} args
 * @returns {boolean}
 */
 whenTimer(args) {
 // 需要配合 runtime 实现定时触发
 return false;
 }
}
```

## 条件块 (CONDITIONAL)

条件块用于条件分支控制，类似 "如果...那么...否则..." 结构。

### 基本语法

```javascript
{
 opcode: 'ifThenElse',
 blockType: Scratch.BlockType.CONDITIONAL,
 text: '如果 [CONDITION] 那么',
 arguments: {
 CONDITION: { type: Scratch.ArgumentType.BOOLEAN }
 },
 branchCount: 2 // 分支数量
}
```

### 完整示例

```javascript
class ControlExtension {
 getInfo() {
 return {
 id: 'controlExt',
 name: '控制扩展',
 color1: '#FFAB19',
 blocks: [
 // 双分支条件块
 {
 opcode: 'ifElse',
 blockType: Scratch.BlockType.CONDITIONAL,
 text: '如果 [CONDITION]',
 arguments: {
 CONDITION: {
 type: Scratch.ArgumentType.BOOLEAN
 }
 },
 branchCount: 2
 },
 // 单分支条件块
 {
 opcode: 'ifOnly',
 blockType: Scratch.BlockType.CONDITIONAL,
 text: '当 [CONDITION] 时',
 arguments: {
 CONDITION: {
 type: Scratch.ArgumentType.BOOLEAN
 }
 },
 branchCount: 1
 }
 ]
 };
 }

 /**
 * 条件判断
 * @param {object} args
 * @param {object} util - Scratch 工具对象
 * @returns {boolean}
 */
 ifElse(args, util) {
 return args.CONDITION;
 }

 ifOnly(args, util) {
 return args.CONDITION;
 }
}
```

## 循环块 (LOOP)

循环块用于重复执行代码块。

### 基本语法

```javascript
{
 opcode: 'repeatTimes',
 blockType: Scratch.BlockType.LOOP,
 text: '重复 [TIMES] 次',
 arguments: {
 TIMES: { type: Scratch.ArgumentType.NUMBER, defaultValue: 10 }
 }
}
```

### 完整示例

```javascript
class LoopExtension {
 getInfo() {
 return {
 id: 'loopExt',
 name: '循环扩展',
 color1: '#FFAB19',
 blocks: [
 // 基本循环
 {
 opcode: 'repeatTimes',
 blockType: Scratch.BlockType.LOOP,
 text: '重复 [TIMES] 次',
 arguments: {
 TIMES: { type: Scratch.ArgumentType.NUMBER, defaultValue: 10 }
 }
 },
 // 条件循环
 {
 opcode: 'repeatUntil',
 blockType: Scratch.BlockType.LOOP,
 text: '重复直到 [CONDITION]',
 arguments: {
 CONDITION: { type: Scratch.ArgumentType.BOOLEAN }
 },
 isTerminal: true // 是否为终止条件
 },
 // 无限循环
 {
 opcode: 'forever',
 blockType: Scratch.BlockType.LOOP,
 text: '无限重复'
 }
 ]
 };
 }

 /**
 * 重复指定次数
 * @param {object} args
 * @returns {number} 循环次数
 */
 repeatTimes(args) {
 const times = Math.max(0, Math.floor(Number(args.TIMES) || 0));
 return times;
 }

 /**
 * 重复直到条件为真
 * @param {object} args
 * @returns {boolean} 是否继续循环
 */
 repeatUntil(args) {
 return !args.CONDITION;
 }

 /**
 * 无限循环
 * @returns {boolean} 始终返回 true
 */
 forever() {
 return true;
 }
}
```

## 组合示例

创建一个包含多种积木块类型的完整扩展：

```javascript
class CompleteExtension {
 constructor(runtime) {
 this.runtime = runtime;
 this.counter = 0;
 }

 getInfo() {
 return {
 id: 'complete',
 name: '完整示例',
 color1: '#9966FF',
 blocks: [
 // 帽子块
 {
 opcode: 'whenFlagClicked',
 blockType: Scratch.BlockType.HAT,
 text: '当开始被点击',
 isEdgeActivated: false
 },
 // 命令块
 {
 opcode: 'setCounter',
 blockType: Scratch.BlockType.COMMAND,
 text: '设置计数器为 [VALUE]'
 },
 // 报告块
 {
 opcode: 'getCounter',
 blockType: Scratch.BlockType.REPORTER,
 text: '计数器值'
 },
 // 布尔块
 {
 opcode: 'isCounterPositive',
 blockType: Scratch.BlockType.BOOLEAN,
 text: '计数器是正数？'
 },
 // 循环块
 {
 opcode: 'repeatIncrement',
 blockType: Scratch.BlockType.LOOP,
 text: '重复增加 [TIMES] 次'
 }
 ],
 arguments: {
 VALUE: { type: Scratch.ArgumentType.NUMBER, defaultValue: 0 },
 TIMES: { type: Scratch.ArgumentType.NUMBER, defaultValue: 10 }
 }
 };
 }

 whenFlagClicked() {
 return true;
 }

 setCounter(args) {
 this.counter = Number(args.VALUE) || 0;
 }

 getCounter() {
 return this.counter;
 }

 isCounterPositive() {
 return this.counter > 0;
 }

 repeatIncrement(args) {
 return Math.max(0, Math.floor(Number(args.TIMES) || 0));
 }
}
```

## 小结

本章要点：

1. **命令块**：执行操作，无返回值
2. **报告块**：返回值，可嵌入其他积木
3. **布尔块**：返回 true/false，用于条件判断
4. **帽子块**：事件触发入口
5. **条件块**：条件分支控制
6. **循环块**：重复执行控制

> **提示**：选择正确的积木块类型是设计好扩展的关键。
