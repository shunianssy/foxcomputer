# 3.2 硬件交互扩展

本节介绍如何创建与硬件设备交互的 Scratch 扩展，包括 micro:bit、Arduino 等设备。

## 硬件扩展基础

### 通信方式

| 方式 | 适用场景 | 特点 |
|-----|---------|-----|
| Web Serial API | 串口设备 | 直连，低延迟 |
| Web Bluetooth | 蓝牙设备 | 无线，便携 |
| WebSocket | 网络设备 | 跨平台，远程 |
| HTTP API | IoT 设备 | 简单，兼容性好 |

## Web Serial API 扩展

### 检测浏览器支持

```javascript
/**
 * 检测 Web Serial API 支持
 */
function checkSerialSupport() {
 if (!('serial' in navigator)) {
 console.warn('当前浏览器不支持 Web Serial API');
 console.warn('请使用 Chrome 89+ 或 Edge 89+');
 return false;
 }
 return true;
}
```

### 基本串口扩展

```javascript
/**
 * 串口通信扩展
 */
class SerialExtension {
 constructor(runtime) {
 this.runtime = runtime;
 this.port = null;
 this.reader = null;
 this.writer = null;
 this.isConnected = false;
 this.receiveBuffer = '';
 this.lastReceived = '';
 }

 getInfo() {
 return {
 id: 'serial',
 name: '串口通信',
 color1: '#4C97FF',
 color2: '#4280D7',
 color3: '#3373CC',
 blocks: [
 {
 opcode: 'connect',
 blockType: Scratch.BlockType.COMMAND,
 text: '连接串口'
 },
 {
 opcode: 'disconnect',
 blockType: Scratch.BlockType.COMMAND,
 text: '断开串口'
 },
 {
 opcode: 'isConnected',
 blockType: Scratch.BlockType.BOOLEAN,
 text: '串口已连接？'
 },
 {
 opcode: 'sendData',
 blockType: Scratch.BlockType.COMMAND,
 text: '发送数据 [DATA]'
 },
 {
 opcode: 'receiveData',
 blockType: Scratch.BlockType.REPORTER,
 text: '接收数据'
 },
 {
 opcode: 'sendAndReceive',
 blockType: Scratch.BlockType.REPORTER,
 text: '发送 [DATA] 并等待响应'
 }
 ]
 };
 }

 /**
 * 连接串口
 */
 async connect() {
 // 检查支持
 if (!checkSerialSupport()) {
 console.error('浏览器不支持串口通信');
 return;
 }

 try {
 // 请求用户选择串口
 this.port = await navigator.serial.requestPort();
 
 // 打开串口
 await this.port.open({
 baudRate: 9600, // 波特率
 dataBits: 8, // 数据位
 stopBits: 1, // 停止位
 parity: 'none' // 校验位
 });

 this.isConnected = true;
 console.log('串口已连接');

 // 开始读取数据
 this.startReading();

 } catch (error) {
 console.error('串口连接失败:', error);
 this.isConnected = false;
 }
 }

 /**
 * 断开串口
 */
 async disconnect() {
 try {
 if (this.reader) {
 await this.reader.cancel();
 this.reader = null;
 }
 
 if (this.port) {
 await this.port.close();
 this.port = null;
 }

 this.isConnected = false;
 console.log('串口已断开');
 
 } catch (error) {
 console.error('断开串口失败:', error);
 }
 }

 /**
 * 开始读取数据
 */
 async startReading() {
 while (this.port && this.port.readable) {
 this.reader = this.port.readable.getReader();
 
 try {
 while (true) {
 const { value, done } = await this.reader.read();
 
 if (done) {
 break;
 }

 // 处理接收到的数据
 const text = new TextDecoder().decode(value);
 this.receiveBuffer += text;
 
 // 检查是否有完整行（以换行符结尾）
 const lines = this.receiveBuffer.split('\n');
 if (lines.length > 1) {
 this.lastReceived = lines[0].trim();
 this.receiveBuffer = lines.slice(1).join('\n');
 }
 }
 } catch (error) {
 console.error('读取数据错误:', error);
 } finally {
 this.reader.releaseLock();
 }
 }
 }

 /**
 * 检查连接状态
 * @returns {boolean}
 */
 isConnected_() {
 return this.isConnected;
 }

 /**
 * 发送数据
 * @param {object} args
 */
 async sendData(args) {
 if (!this.isConnected || !this.port) {
 console.warn('串口未连接');
 return;
 }

 const data = String(args.DATA) + '\n';
 const encoder = new TextEncoder();

 try {
 this.writer = this.port.writable.getWriter();
 await this.writer.write(encoder.encode(data));
 this.writer.releaseLock();
 console.log('已发送:', data.trim());
 } catch (error) {
 console.error('发送数据失败:', error);
 }
 }

 /**
 * 接收数据
 * @returns {string}
 */
 receiveData() {
 return this.lastReceived;
 }

 /**
 * 发送并等待响应
 * @param {object} args
 * @returns {Promise<string>}
 */
 async sendAndReceive(args) {
 const data = String(args.DATA);
 
 // 清空之前的接收缓冲
 this.lastReceived = '';
 
 // 发送数据
 await this.sendData({ DATA: data });
 
 // 等待响应（最多 5 秒）
 const timeout = 5000;
 const startTime = Date.now();
 
 while (!this.lastReceived && Date.now() - startTime < timeout) {
 await new Promise(resolve => setTimeout(resolve, 100));
 }
 
 return this.lastReceived || '超时';
 }
}
```

### micro:bit 蓝牙扩展

```javascript
/**
 * micro:bit 蓝牙扩展
 */
class MicrobitExtension {
 constructor(runtime) {
 this.runtime = runtime;
 this.device = null;
 this.server = null;
 this.characteristics = {};
 this.isConnected = false;
 
 // micro:bit BLE 服务 UUID
 this.SERVICES = {
 UART: '6e400001-b5a3-f393-e0a9-e50e24dcca9e',
 LED: 'e95dd91d-251d-470a-a062-fa1922dfa9a8',

 TEMPERATURE: 'e95d6100-251d-470a-a062-fa1922dfa9a8'
 };

 this.CHARACTERISTICS = {
 UART_TX: '6e400002-b5a3-f393-e0a9-e50e24dcca9e',

 LED_MATRIX: 'e95d7b77-251d-470a-a062-fa1922dfa9a8',

 /**
 * 连接 micro:bit
 */
 async connect() {
 try {
 // 请求蓝牙设备
 this.device = await navigator.bluetooth.requestDevice({
 filters: [{ namePrefix: 'BBC micro:bit' }],
 optionalServices: Object.values(this.SERVICES)
 });

 console.log('找到设备:', this.device.name);

 // 连接 GATT 服务器
 this.server = await this.device.gatt.connect();
 console.log('GATT 服务器已连接');

 // 获取服务
 await this.setupServices();

 this.isConnected = true;
 console.log('micro:bit 连接成功');

 // 监听断开事件
 this.device.addEventListener('gattserverdisconnected', () => {
 this.isConnected = false;
 console.log('micro:bit 已断开');
 });

 } catch (error) {
 console.error('连接失败:', error);
 this.isConnected = false;
 }
 }

 /**
 * 设置服务
 */
 async setupServices() {
 // 获取 LED 服务
 try {
 const ledService = await this.server.getPrimaryService(this.SERVICES.LED);
 this.characteristics.ledMatrix = await ledService.getCharacteristic(
 this.CHARACTERISTICS.LED_MATRIX
 );
 this.characteristics.ledText = await ledService.getCharacteristic(
 this.CHARACTERISTICS.LED_TEXT
 );
 } catch (e) {
 console.warn('LED 服务不可用');
 }

 // 获取按钮服务
 try {
 const buttonService = await this.server.getPrimaryService(this.SERVICES.BUTTON);
 
 const buttonA = await buttonService.getCharacteristic(
 this.CHARACTERISTICS.BUTTON_A
 );
 buttonA.addEventListener('characteristicvaluechanged', (e) => {
 this.buttonA = e.target.value.getUint8(0) === 1;
 });
 await buttonA.startNotifications();

 const buttonB = await buttonService.getCharacteristic(
 this.CHARACTERISTICS.BUTTON_B
 );
 buttonB.addEventListener('characteristicvaluechanged', (e) => {
 this.buttonB = e.target.value.getUint8(0) === 1;
 });
 await buttonB.startNotifications();
 } catch (e) {
 console.warn('按钮服务不可用');
 }

 // 获取加速度计服务
 try {
 const accService = await this.server.getPrimaryService(this.SERVICES.ACCELEROMETER);
 const accChar = await accService.getCharacteristic(
 this.CHARACTERISTICS.ACCELEROMETER_DATA
 );
 accChar.addEventListener('characteristicvaluechanged', (e) => {
 const data = e.target.value;
 this.accelerometer = {
 x: data.getInt16(0, true),
 y: data.getInt16(2, true),
 z: data.getInt16(4, true)
 };
 });
 await accChar.startNotifications();
 } catch (e) {
 console.warn('加速度计服务不可用');
 }

 // 获取温度服务
 try {
 const tempService = await this.server.getPrimaryService(this.SERVICES.TEMPERATURE);
 this.characteristics.temperature = await tempService.getCharacteristic(
 this.CHARACTERISTICS.TEMPERATURE
 );
 } catch (e) {
 console.warn('温度服务不可用');
 }
 }

 /**
 * 断开连接
 */
 async disconnect() {
 if (this.device && this.device.gatt.connected) {
 this.device.gatt.disconnect();
 }
 this.isConnected = false;
 }

 /**
 * 检查连接状态
 */
 isConnected_() {
 return this.isConnected;
 }

 /**
 * 显示文本
 * @param {object} args
 */
 async showText(args) {
 if (!this.characteristics.ledText) {
 console.warn('LED 文本特征不可用');
 return;
 }

 const text = String(args.TEXT).substring(0, 20); // 最多 20 字符
 const encoder = new TextEncoder();
 const data = encoder.encode(text);
 
 await this.characteristics.ledText.writeValue(data);
 }

 /**
 * 显示图案
 * @param {object} args
 */
 async showPattern(args) {
 if (!this.characteristics.ledMatrix) {
 console.warn('LED 矩阵特征不可用');
 return;
 }

 // 将图案转换为 5x5 LED 数据
 const pattern = this.parsePattern(String(args.PATTERN));
 await this.characteristics.ledMatrix.writeValue(pattern);
 }

 /**
 * 解析图案字符串
 * @param {string} patternStr
 * @returns {Uint8Array}
 */
 parsePattern(patternStr) {
 // 期望格式: 5 行，每行 5 个 0 或 1
 const rows = patternStr.trim().split('\n');
 const data = new Uint8Array(5);
 
 for (let row = 0; row < 5; row++) {
 let byte = 0;
 const rowData = rows[row] || '00000';
 
 for (let col = 0; col < 5; col++) {
 if (rowData[col] === '1') {
 byte |= (1 << (4 - col));
 }
 }
 data[row] = byte;
 }
 
 return data;
 }

 /**
 * 清除显示
 */
 async clearDisplay() {
 await this.showPattern({ PATTERN: '00000\n00000\n00000\n00000\n00000' });
 }

 /**
 * 检测按钮状态
 * @param {object} args
 * @returns {boolean}
 */
 isButtonPressed(args) {
 return args.BUTTON === 'A' ? this.buttonA : this.buttonB;
 }

 /**
 * 获取温度
 * @returns {Promise<number>}
 */
 async getTemperature() {
 if (!this.characteristics.temperature) {
 return 0;
 }

 const value = await this.characteristics.temperature.readValue();
 return value.getInt8(0);
 }

 /**
 * 获取加速度
 * @param {object} args
 * @returns {number}
 */
 getAccelerometer(args) {
 return this.accelerometer[args.AXIS] || 0;
 }
}
```

## WebSocket 扩展

```javascript
/**
 * WebSocket 通信扩展
 */
class WebSocketExtension {
 constructor(runtime) {
 this.runtime = runtime;
 this.socket = null;
 this.isConnected = false;
 this.lastMessage = '';
 this.messageQueue = [];
 }

 getInfo() {
 return {
 id: 'websocket',
 name: 'WebSocket',
 color1: '#FF8C1A',
 color2: '#DB7A16',
 color3: '#B86912',
 blocks: [
 {
 opcode: 'connect',
 blockType: Scratch.BlockType.COMMAND,
 text: '连接到 [URL]'
 },
 {
 opcode: 'disconnect',
 blockType: Scratch.BlockType.COMMAND,
 text: '断开连接'
 },
 {
 opcode: 'isConnected',
 blockType: Scratch.BlockType.BOOLEAN,
 text: '已连接？'
 },
 {
 opcode: 'send',
 blockType: Scratch.BlockType.COMMAND,
 text: '发送消息 [MESSAGE]'
 },
 {
 opcode: 'receive',
 blockType: Scratch.BlockType.REPORTER,
 text: '接收消息'
 },
 {
 opcode: 'onMessage',
 blockType: Scratch.BlockType.HAT,
 text: '当收到消息',
 isEdgeActivated: false
 }
 ]
 };
 }

 /**
 * 连接 WebSocket
 * @param {object} args
 */
 connect(args) {
 return new Promise((resolve, reject) => {
 try {
 const url = String(args.URL);
 this.socket = new WebSocket(url);

 this.socket.onopen = () => {
 this.isConnected = true;
 console.log('WebSocket 已连接');
 resolve();
 };

 this.socket.onmessage = (event) => {
 this.lastMessage = event.data;
 this.messageQueue.push(event.data);
 
 // 触发消息事件
 this.runtime.startHats('websocket_onMessage');
 };

 this.socket.onclose = () => {
 this.isConnected = false;
 console.log('WebSocket 已断开');
 };

 this.socket.onerror = (error) => {
 console.error('WebSocket 错误:', error);
 this.isConnected = false;
 reject(error);
 };

 } catch (error) {
 reject(error);
 }
 });
 }

 /**
 * 断开连接
 */
 disconnect() {
 if (this.socket) {
 this.socket.close();
 this.socket = null;
 }
 this.isConnected = false;
 }

 /**
 * 检查连接状态
 */
 isConnected_() {
 return this.isConnected;
 }

 /**
 * 发送消息
 * @param {object} args
 */
 send(args) {
 if (this.socket && this.isConnected) {
 this.socket.send(String(args.MESSAGE));
 }
 }

 /**
 * 接收消息
 * @returns {string}
 */
 receive() {
 if (this.messageQueue.length > 0) {
 return this.messageQueue.shift();
 }
 return this.lastMessage;
 }

 /**
 * 消息事件触发
 */
 onMessage() {
 return true;
 }
}
```

## 小结

本章要点：

1. **Web Serial API**：串口设备通信
2. **Web Bluetooth**：蓝牙设备连接
3. **WebSocket**：实时双向通信
4. **异步处理**：Promise 和 async/await
5. **事件监听**：设备状态和数据变化

> **提示**：硬件扩展需要 HTTPS 环境和用户授权。
