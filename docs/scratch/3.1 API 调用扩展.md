```javascript
/**
 * 翻译扩展
 * 使用 MyMemory 免费翻译 API
 */
class TranslateExtension {
 constructor(runtime) {
 this.runtime = runtime;
 this.baseUrl = 'https://api.mymemory.translated.net/get';
 }

 getInfo() {
 return {
 id: 'translate',
 name: '翻译',
 color1: '#9966FF',
 color2: '#855CD6',
 color3: '#774DCB',
 blocks: [
 {
 opcode: 'translate',
 blockType: Scratch.BlockType.REPORTER,
 text: '将 [TEXT] 从 [FROM] 翻译为 [TO]'
 },
 {
 opcode: 'detectLanguage',
 blockType: Scratch.BlockType.REPORTER,
 text: '检测 [TEXT] 的语言'
 }
 ],
 menus: {
 languages: {
 acceptReporters: true,
 items: [
 { text: '中文', value: 'zh-CN' },
 { text: '英语', value: 'en' },
 { text: '日语', value: 'ja' },
 { text: '韩语', value: 'ko' },
 { text: '法语', value: 'fr' },
 { text: '德语', value: 'de' },
 { text: '西班牙语', value: 'es' },
 { text: '俄语', value: 'ru' }
 ]
 }
 }
 };
 }

 /**
 * 翻译文本
 * @param {object} args
 * @returns {Promise<string>}
 */
 async translate(args) {
 const text = String(args.TEXT || '');
 const from = args.FROM || 'en';
 const to = args.TO || 'zh-CN';

 if (!text.trim()) {
 return '';
 }

 try {
 // MyMemory API 格式: langpair=from|to
 const langPair = `${from}|${to}`;
 const url = `${this.baseUrl}?q=${encodeURIComponent(text)}&langpair=${langPair}`;

 const response = await fetch(url);
 const data = await response.json();

 if (data.responseStatus === 200) {
 return data.responseData.translatedText;
 }
 
 return text; // 翻译失败返回原文
 } catch (error) {
 console.error('翻译失败:', error);
 return text;
 }
 }

 /**
 * 检测语言（简单实现）
 * @param {object} args
 * @returns {string}
 */
 detectLanguage(args) {
 const text = String(args.TEXT || '');
 
 // 简单的语言检测规则
 if (/[\u4e00-\u9fa5]/.test(text)) {
 return 'zh-CN'; // 包含中文
 }
 if (/[\u3040-\u309f\u30a0-\u30ff]/.test(text)) {
 return 'ja'; // 包含日文
 }
 if (/[\uac00-\ud7af]/.test(text)) {
 return 'ko'; // 包含韩文
 }
 if (/[а-яА-ЯёЁ]/.test(text)) {
 return 'ru'; // 包含俄文
 }
 
 return 'en'; // 默认英文
 }
}
```

```javascript
/**
 * JSON 数据处理扩展
 */
class JsonExtension {
 getInfo() {
 return {
 id: 'json',
 name: 'JSON',
 color1: '#FF6680',
 color2: '#FF4D6A',
 color3: '#FF3355',
 blocks: [
 {
 opcode: 'jsonGet',
 blockType: Scratch.BlockType.REPORTER,
 text: '从 [JSON] 获取 [KEY]'
 },
 {
 opcode: 'jsonSet',
 blockType: Scratch.BlockType.REPORTER,
 text: '在 [JSON] 设置 [KEY] 为 [VALUE]'
 },
 {
 opcode: 'jsonLength',
 blockType: Scratch.BlockType.REPORTER,
 text: '[JSON] 的长度'
 },
 {
 opcode: 'jsonKeys',
 blockType: Scratch.BlockType.REPORTER,
 text: '[JSON] 的所有键'
 },
 {
 opcode: 'jsonIsValid',
 blockType: Scratch.BlockType.BOOLEAN,
 text: '[JSON] 是有效的？'
 }
 ]
 };
 }

 /**
 * 安全解析 JSON
 * @param {string} jsonStr
 * @returns {object|null}
 */
 safeParse(jsonStr) {
 try {
 return JSON.parse(String(jsonStr));
 } catch {
 return null;
 }
 }

 /**
 * 获取 JSON 值
 * @param {object} args
 * @returns {string}
 */
 jsonGet(args) {
 const data = this.safeParse(args.JSON);
 if (!data) return '';

 const key = String(args.KEY);
 const value = data[key];
 
 if (value === undefined) return '';
 if (typeof value === 'object') {
 return JSON.stringify(value);
 }
 return String(value);
 }

 /**
 * 设置 JSON 值
 * @param {object} args
 * @returns {string}
 */
 jsonSet(args) {
 const data = this.safeParse(args.JSON) || {};
 const key = String(args.KEY);
 const value = args.VALUE;
 
 data[key] = value;
 return JSON.stringify(data);
 }

 /**
 * 获取 JSON 长度
 * @param {object} args
 * @returns {number}
 */
 jsonLength(args) {
 const data = this.safeParse(args.JSON);
 if (!data) return 0;
 
 if (Array.isArray(data)) {
 return data.length;
 }
 if (typeof data === 'object') {
 return Object.keys(data).length;
 }
 return 0;
 }

 /**
 * 获取所有键
 * @param {object} args
 * @returns {string}
 */
 jsonKeys(args) {
 const data = this.safeParse(args.JSON);
 if (!data || typeof data !== 'object') {
 return '[]';
 }
 
 return JSON.stringify(Object.keys(data));
 }

 /**
 * 验证 JSON
 * @param {object} args
 * @returns {boolean}
 */
 jsonIsValid(args) {
 try {
 JSON.parse(String(args.JSON));
 return true;
 } catch {
 return false;
 }
 }
}
```

## 错误处理与重试机制

```javascript
/**
 * 带错误处理和重试的 API 请求
 */
class RobustApiExtension {
 constructor(runtime) {
 this.runtime = runtime;
 this.maxRetries = 3;
 this.retryDelay = 1000; // 毫秒
 this.timeout = 10000; // 毫秒
 }

 /**
 * 延迟函数
 * @param {number} ms
 * @returns {Promise}
 */
 delay(ms) {
 return new Promise(resolve => setTimeout(resolve, ms));
 }

 /**
 * 带超时的 fetch
 * @param {string} url
 * @param {object} options
 * @returns {Promise<Response>}
 */
 async fetchWithTimeout(url, options = {}) {
 const controller = new AbortController();
 const timeoutId = setTimeout(() => controller.abort(), this.timeout);

 try {
 const response = await fetch(url, {
 ...options,
 signal: controller.signal
 });
 clearTimeout(timeoutId);
 return response;
 } catch (error) {
 clearTimeout(timeoutId);
 throw error;
 }
 }

 /**
 * 带重试的 API 请求
 * @param {string} url
 * @param {object} options
 * @returns {Promise<object>}
 */
 async fetchWithRetry(url, options = {}) {
 let lastError;

 for (let attempt = 1; attempt <= this.maxRetries; attempt++) {
 try {
 console.log(`API 请求尝试 ${attempt}/${this.maxRetries}: ${url}`);
 
 const response = await this.fetchWithTimeout(url, options);
 
 if (!response.ok) {
 throw new Error(`HTTP ${response.status}: ${response.statusText}`);
 }
 
 const data = await response.json();
 console.log('API 请求成功');
 return data;
 
 } catch (error) {
 lastError = error;
 console.error(`尝试 ${attempt} 失败:`, error.message);
 
 // 如果不是最后一次尝试，等待后重试
 if (attempt < this.maxRetries) {
 const delayTime = this.retryDelay * attempt; // 指数退避
 console.log(`等待 ${delayTime}ms 后重试...`);
 await this.delay(delayTime);
 }
 }
 }

 throw lastError;
 }

 /**
 * 安全的 API 调用
 * @param {string} url
 * @param {*} defaultValue
 * @returns {Promise<*>}
 */
 async safeFetch(url, defaultValue = null) {
 try {
 return await this.fetchWithRetry(url);
 } catch (error) {
 console.error('API 请求最终失败:', error);
 return defaultValue;
 }
 }
}
```

## 缓存机制

```javascript
/**
 * 带缓存的 API 扩展
 */
class CachedApiExtension {
 constructor(runtime) {
 this.runtime = runtime;
 this.cache = new Map();
 this.cacheTimeout = 5 * 60 * 1000; // 5 分钟
 }

 /**
 * 生成缓存键
 * @param {string} url
 * @returns {string}
 */
 getCacheKey(url) {
 return url;
 }

 /**
 * 检查缓存是否有效
 * @param {string} key
 * @returns {boolean}
 */
 isCacheValid(key) {
 if (!this.cache.has(key)) {
 return false;
 }
 
 const cached = this.cache.get(key);
 return Date.now() - cached.timestamp < this.cacheTimeout;
 }

 /**
 * 获取缓存数据
 * @param {string} key
 * @returns {*}
 */
 getFromCache(key) {
 const cached = this.cache.get(key);
 return cached ? cached.data : null;
 }

 /**
 * 设置缓存数据
 * @param {string} key
 * @param {*} data
 */
 setToCache(key, data) {
 this.cache.set(key, {
 data: data,
 timestamp: Date.now()
 });
 }

 /**
 * 带缓存的 API 请求
 * @param {string} url
 * @returns {Promise<*>}
 */
 async fetchWithCache(url) {
 const cacheKey = this.getCacheKey(url);
 
 // 检查缓存
 if (this.isCacheValid(cacheKey)) {
 console.log('使用缓存数据');
 return this.getFromCache(cacheKey);
 }
 
 // 发起请求
 console.log('发起新请求');
 const response = await fetch(url);
 const data = await response.json();
 
 // 存入缓存
 this.setToCache(cacheKey, data);
 
 return data;
 }

 /**
 * 清除缓存
 */
 clearCache() {
 this.cache.clear();
 console.log('缓存已清除');
 }
}
```

## 小结

本章要点：

1. **API 调用**：使用 fetch 进行网络请求
2. **异步处理**：async/await 处理异步操作

> **提示**：API 扩展要注意请求频率限制和错误处理。
