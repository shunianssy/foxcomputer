# 2.1 模型定义

Django 的模型层是框架最强大的功能之一。通过 ORM（对象关系映射），你可以用 Python 类来定义数据库表结构，无需编写 SQL 语句。

## 模型基础

### 创建模型

在应用的 `models.py` 中定义模型：

```python
# blog/models.py
from django.db import models

class Article(models.Model):
    """文章模型"""
    
    title = models.CharField('标题', max_length=100)
    content = models.TextField('内容')
    created_at = models.DateTimeField('创建时间', auto_now_add=True)
    
    class Meta:
        verbose_name = '文章'
        verbose_name_plural = '文章'
        ordering = ['-created_at']
    
    def __str__(self):
        return self.title
```

### 模型映射规则

| Python 模型 | 数据库表 |
|-------------|----------|
| 类名 | 表名（app_modelname） |
| 属性 | 字段 |
| 实例 | 记录行 |

## 字段选项

### 通用选项

```python
class Article(models.Model):
    # verbose_name: 字段的可读名称
    title = models.CharField('标题', max_length=100)
    
    # null: 是否允许数据库为 NULL
    subtitle = models.CharField(null=True, max_length=100)
    
    # blank: 是否允许表单为空
    description = models.TextField(blank=True)
    
    # default: 默认值
    is_published = models.BooleanField(default=False)
    
    # choices: 选项列表
    STATUS_CHOICES = [
        ('draft', '草稿'),
        ('published', '已发布'),
        ('archived', '已归档'),
    ]
    status = models.CharField('状态', max_length=20, choices=STATUS_CHOICES, default='draft')
    
    # unique: 是否唯一
    slug = models.SlugField(unique=True)
    
    # db_index: 是否创建索引
    tags = models.CharField(max_length=200, db_index=True)
    
    # editable: 是否可编辑
    views = models.IntegerField(default=0, editable=False)
    
    # help_text: 帮助文本
    content = models.TextField(help_text='请输入文章内容')
```

### 日期字段特殊选项

class Article(models.Model):

## Meta 选项

`Meta` 类用于定义模型的元数据：

```python
class Article(models.Model):
    title = models.CharField(max_length=100)
    
    class Meta:
        # 表名（默认为 app_modelname）
        db_table = 'blog_articles'
        
        # 按创建时间降序排列
        ordering = ['-created_at']
        
        # 唯一约束
        unique_together = ['title', 'author']
        
        # 索引
        indexes = [
            models.Index(fields=['title']),
            models.Index(fields=['-created_at']),
            models.Index(fields=['author', 'status']),
        ]
        
        # 权限
        permissions = [
            ('can_publish', '可以发布文章'),
            ('can_edit', '可以编辑文章'),
        ]
        
        # 抽象基类（不会创建表）
        abstract = True
        
        # 后台管理名称
        verbose_name = '文章'
        verbose_name_plural = '文章列表'
```

## 模型方法

### 实例方法

```python
class Article(models.Model):
    title = models.CharField(max_length=100)
    content = models.TextField()
    views = models.IntegerField(default=0)
    
    def __str__(self):
        """字符串表示"""
        return self.title
    
    def get_summary(self, length=100):
        """获取摘要"""
        if len(self.content) > length:
            return self.content[:length] + '...'
        return self.content
    
    def increase_views(self):
        """增加浏览量"""
        self.views += 1
        self.save(update_fields=['views'])
```

### 类方法

```python
class Article(models.Model):
    title = models.CharField(max_length=100)
    status = models.CharField(max_length=20)
    
    @classmethod
    def get_published(cls):
        """获取已发布的文章"""
        return cls.objects.filter(status='published')
    
    @classmethod
    def get_by_author(cls, author_id):
        """获取指定作者的文章"""
        return cls.objects.filter(author_id=author_id)
```

### 属性方法

```python
class Article(models.Model):
    title = models.CharField(max_length=100)
    content = models.TextField()
    
    @property
    def word_count(self):
        """字数统计"""
        return len(self.content)
    
    @property
    def reading_time(self):
        """预估阅读时间（分钟）"""
        return max(1, self.word_count // 500)
```

## 模型继承

### 抽象基类

用于提取公共字段，不会创建数据库表：

```python
class BaseModel(models.Model):
    """抽象基类"""
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    
    class Meta:
        abstract = True  # 关键：声明为抽象类

class Article(BaseModel):
    """文章模型，继承基类"""
    title = models.CharField(max_length=100)
    content = models.TextField()

class Comment(BaseModel):
    """评论模型，继承基类"""
    content = models.TextField()
    # 自动拥有 created_at 和 updated_at 字段
```

### 多表继承

每个模型都有独立的表：

```python
class Person(models.Model):
    name = models.CharField(max_length=50)
    
class Student(Person):
    """学生模型"""
    student_id = models.CharField(max_length=20)
    # 会创建独立的表，包含 person_ptr_id 外键
```

### 代理模型

不创建新表，只修改行为：

```python
class Article(models.Model):
    title = models.CharField(max_length=100)
    status = models.CharField(max_length=20)

class PublishedArticle(Article):
    """代理模型：只包含已发布的文章"""
    class Meta:
        proxy = True
    
    def get_queryset(self):
        return super().get_queryset().filter(status='published')
```

## 完整示例

```python
# blog/models.py
from django.db import models
from django.contrib.auth.models import User
from django.utils import timezone

class Category(models.Model):
    """文章分类"""
    name = models.CharField('分类名称', max_length=50, unique=True)
    slug = models.SlugField('URL 别名', unique=True)
    description = models.TextField('分类描述', blank=True)
    
    class Meta:
        verbose_name = '分类'
        verbose_name_plural = '分类'
        ordering = ['name']
    
    def __str__(self):
        return self.name

class Tag(models.Model):
    """文章标签"""
    name = models.CharField('标签名称', max_length=30, unique=True)
    
    class Meta:
        verbose_name = '标签'
        verbose_name_plural = '标签'
    
    def __str__(self):
        return self.name

class Article(models.Model):
    """文章模型"""
    
    STATUS_CHOICES = [
        ('draft', '草稿'),
        ('published', '已发布'),
    ]
    
    title = models.CharField('标题', max_length=200)
    slug = models.SlugField('URL 别名', unique_for_date='publish')
    author = models.ForeignKey(
        User,
        on_delete=models.CASCADE,
        related_name='articles',
        verbose_name='作者'
    )
    category = models.ForeignKey(
        Category,
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name='articles',
        verbose_name='分类'
    )
    tags = models.ManyToManyField(Tag, blank=True, verbose_name='标签')
    content = models.TextField('内容')
    status = models.CharField('状态', max_length=10, choices=STATUS_CHOICES, default='draft')
    views = models.PositiveIntegerField('浏览量', default=0, editable=False)
    publish = models.DateTimeField('发布时间', default=timezone.now)
    created_at = models.DateTimeField('创建时间', auto_now_add=True)
    updated_at = models.DateTimeField('更新时间', auto_now=True)
    
    class Meta:
        verbose_name = '文章'
        verbose_name_plural = '文章'
        ordering = ['-publish']
        indexes = [
            models.Index(fields=['-publish']),
            models.Index(fields=['status']),
        ]
    
    def __str__(self):
        return self.title
    
    @property
    def word_count(self):
        """字数统计"""
        return len(self.content)
    
    def increase_views(self):
        """增加浏览量"""
        self.views += 1
        self.save(update_fields=['views'])
```

## 小结

模型定义的关键点：

1. **字段类型**：选择合适的字段类型存储数据
2. **字段选项**：配置字段的约束和行为
3. **Meta 类**：定义模型的元数据
4. **模型方法**：添加业务逻辑方法
5. **模型继承**：复用公共字段和行为

> 💡 **提示**：好的模型设计是项目成功的基础。花时间设计好数据结构，可以避免后期的重构工作。
