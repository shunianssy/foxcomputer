# 2.3 QuerySet API

Django 的 QuerySet API 是与数据库交互的核心接口。通过链式调用方法，你可以构建复杂的查询，而无需编写 SQL 语句。

## QuerySet 基础

### 获取 QuerySet

```python
from blog.models import Article

# 获取所有记录
articles = Article.objects.all()

# QuerySet 是惰性求值的
# 这行代码不会立即执行查询
queryset = Article.objects.filter(status='published')

# 只有在以下情况才会执行查询：
# 1. 迭代
for article in queryset:
    print(article.title)

# 2. 切片
articles = queryset[:10]

# 3. 转换为列表
article_list = list(queryset)

# 4. 获取数量
count = queryset.count()
```

### QuerySet 执行时机

| 操作 | 是否执行查询 |
|------|-------------|
| `filter()` | 否 |
| `exclude()` | 否 |
| `order_by()` | 否 |
| `all()` | 否 |
| `for article in queryset` | 是 |
| `list(queryset)` | 是 |
| `queryset[0]` | 是 |
| `queryset.count()` | 是 |
| `queryset.exists()` | 是 |

## 查询方法

### 基本查询

```python
# 获取所有
Article.objects.all()

# 过滤
Article.objects.filter(status='published')

# 排除
Article.objects.exclude(status='draft')

# 获取单条
article = Article.objects.get(pk=1)

# 获取单条或 None
article = Article.objects.filter(pk=1).first()

# 获取最后一条
article = Article.objects.last()

# 最新记录
article = Article.objects.latest('created_at')

# 最早记录
article = Article.objects.earliest('created_at')
```

### 字段查找

```python
# 精确匹配
Article.objects.filter(title='Hello')

# 不区分大小写的精确匹配
Article.objects.filter(title__iexact='hello')

# 包含
Article.objects.filter(title__contains='Django')

# 不区分大小写的包含
Article.objects.filter(title__icontains='django')

# 开头为
Article.objects.filter(title__startswith='Python')

# 结尾为
Article.objects.filter(title__endswith='教程')

# 正则匹配
Article.objects.filter(title__regex=r'^Python')

# 在列表中
Article.objects.filter(status__in=['published', 'draft'])

# 范围
Article.objects.filter(views__range=[100, 500])

# 大于
Article.objects.filter(views__gt=100)

# 大于等于
Article.objects.filter(views__gte=100)

# 小于
Article.objects.filter(views__lt=1000)

# 小于等于
Article.objects.filter(views__lte=1000)

# 为空
Article.objects.filter(category__isnull=True)

# 不为空
Article.objects.filter(category__isnull=False)
```

### 日期查询

```python
from datetime import datetime, timedelta

# 精确日期
Article.objects.filter(created_at__date=datetime.today())

# 年份
Article.objects.filter(created_at__year=2024)

# 月份
Article.objects.filter(created_at__month=1)

# 日
Article.objects.filter(created_at__day=15)

# 星期几（1=周日，7=周六）
Article.objects.filter(created_at__week_day=1)

# 小时
Article.objects.filter(created_at__hour=12)

# 分钟
Article.objects.filter(created_at__minute=30)

# 日期范围
Article.objects.filter(
    created_at__range=[
        datetime.now() - timedelta(days=7),
        datetime.now()
    ]
)
```

## 链式查询

```python
# 链式调用
articles = Article.objects.filter(
    status='published'
).filter(
    views__gt=100
).exclude(
    title__contains='测试'
).order_by(
    '-created_at'
)[:10]
```

### Q 对象（复杂查询）

```python
from django.db.models import Q

# OR 查询
Article.objects.filter(
    Q(status='published') | Q(status='featured')
)

# AND + OR
Article.objects.filter(
    Q(status='published') & (Q(views__gt=100) | Q(title__contains='Django'))
)

# NOT
Article.objects.filter(
    ~Q(status='draft')
)

# 组合使用
Article.objects.filter(
    Q(title__contains='Django') | Q(content__contains='Django'),
    status='published'  # AND 条件
)
```

## 排序与分组

### 排序

```python
# 升序
Article.objects.order_by('created_at')

# 降序
Article.objects.order_by('-created_at')

# 多字段排序
Article.objects.order_by('-views', 'created_at')

# 随机排序
Article.objects.order_by('?')

# 清除排序
Article.objects.order_by()
```

### 分组聚合

```python
from django.db.models import Count, Sum, Avg, Max, Min

# 统计每个作者的文章数
Article.objects.values('author').annotate(
    article_count=Count('id')
)

# 统计每个分类的总浏览量
Article.objects.values('category__name').annotate(
    total_views=Sum('views')
)

# 聚合函数
stats = Article.objects.aggregate(
    total=Count('id'),
    avg_views=Avg('views'),
    max_views=Max('views'),
    min_views=Min('views'),
    total_views=Sum('views')
)
# 返回：{'total': 100, 'avg_views': 50.5, ...}
```

### 分组示例

```python
# 每个作者的文章数和平均浏览量
from django.db.models import Count, Avg

author_stats = Article.objects.values(
    'author__username'
).annotate(
    article_count=Count('id'),
    avg_views=Avg('views')
).order_by('-article_count')

for stat in author_stats:
    print(f"{stat['author__username']}: {stat['article_count']} 篇文章")
```

## 关联查询

### 正向查询

```python
# 获取文章的作者
article = Article.objects.get(pk=1)
author = article.author  # 自动查询

# 预加载关联数据（避免 N+1 问题）
articles = Article.objects.select_related('author', 'category')

for article in articles:
    print(article.author.username)  # 不会额外查询
```

### 反向查询

```python
# 获取用户的所有文章
user = User.objects.get(pk=1)
articles = user.articles.all()  # 使用 related_name

# 预加载反向关联
users = User.objects.prefetch_related('articles')

for user in users:
    for article in user.articles.all():
        print(article.title)
```

### 多对多查询

```python
# 获取文章的所有标签
article = Article.objects.get(pk=1)
tags = article.tags.all()

# 查找包含特定标签的文章
Article.objects.filter(tags__name='Python')

# 预加载多对多
articles = Article.objects.prefetch_related('tags')
```

### 关联查询优化

```python
# select_related: 一对一、一对多（外键）
Article.objects.select_related('author', 'category')

# prefetch_related: 多对多、反向查询
Article.objects.prefetch_related('tags', 'comments')

# 组合使用
Article.objects.select_related(
    'author', 'category'
).prefetch_related(
    'tags'
)

# Prefetch 对象：自定义预加载
from django.db.models import Prefetch

Article.objects.prefetch_related(
    Prefetch(
        'comments',
        queryset=Comment.objects.filter(is_public=True),
        to_attr='public_comments'
    )
)
```

## 更新与删除

### 批量更新

```python
# 更新所有匹配的记录
Article.objects.filter(status='draft').update(status='published')

# 更新多个字段
Article.objects.filter(views=0).update(
    views=1,
    status='published'
)

# 使用 F 表达式
from django.db.models import F
Article.objects.update(views=F('views') + 1)
```

### 批量删除

```python
# 删除匹配的记录
Article.objects.filter(status='draft').delete()

# 注意：delete() 会返回删除的数量
result = Article.objects.filter(views=0).delete()
# 返回：(10, {'blog.Article': 10})
```

### 单条记录操作

```python
# 保存
article = Article(title='New Article', content='...')
article.save()

# 更新指定字段
article.title = 'Updated Title'
article.save(update_fields=['title'])

# 删除
article.delete()
```

## 创建记录

```python
# 方式1：create()
article = Article.objects.create(
    title='New Article',
    content='Content here',
    author=user
)

# 方式2：实例化 + save()
article = Article(
    title='New Article',
    content='Content here'
)
article.author = user
article.save()

# 方式3：get_or_create()
article, created = Article.objects.get_or_create(
    title='Existing Article',
    defaults={'content': 'Default content'}
)
# created 为 True 表示新创建，False 表示已存在

# 方式4：update_or_create()
article, created = Article.objects.update_or_create(
    slug='my-article',
    defaults={
        'title': 'Updated Title',
        'content': 'New content'
    }
)

# 批量创建
Article.objects.bulk_create([
    Article(title='Article 1', content='...'),
    Article(title='Article 2', content='...'),
    Article(title='Article 3', content='...'),
])
```

## 性能优化

### 只查询需要的字段

```python
# 只获取指定字段
Article.objects.only('title', 'created_at')

# 排除某些字段
Article.objects.defer('content')

# 获取值列表
titles = Article.objects.values_list('title', flat=True)

# 获取字典列表
data = Article.objects.values('title', 'author__username')
```

### 查询优化技巧

```python
# 1. 使用 exists() 代替 count()
if Article.objects.filter(author=user).exists():
    pass

# 2. 使用 iterator() 处理大量数据
for article in Article.objects.iterator():
    process(article)

# 3. 批量操作
Article.objects.bulk_create(articles, batch_size=100)
Article.objects.bulk_update(articles, ['title', 'content'], batch_size=100)

# 4. 使用 explain() 分析查询
print(Article.objects.filter(title='Django').explain())
```

## 小结

QuerySet API 的关键点：

1. **惰性求值**：QuerySet 在需要时才执行查询
2. **链式调用**：可以连续调用多个方法
3. **字段查找**：使用 `__` 语法进行复杂查询
4. **关联查询**：使用 `select_related` 和 `prefetch_related` 优化
5. **批量操作**：使用 `bulk_create` 和 `bulk_update` 提高性能

> 💡 **提示**：理解 QuerySet 的惰性求值特性，可以帮助你写出更高效的代码。注意 N+1 问题，合理使用预加载。
