# 3.2 视图函数

视图函数（Function-Based Views, FBV）是 Django 处理请求的基本方式。每个视图函数接收 HTTP 请求，返回 HTTP 响应。

## 视图函数基础

### 基本结构

```python
# blog/views.py
from django.shortcuts import render
from django.http import HttpResponse, JsonResponse

def hello(request):
 """最简单的视图"""
 return HttpResponse("Hello, Django!")

def article_list(request):
 """文章列表视图"""
 articles = Article.objects.filter(status='published')
 return render(request, 'blog/article_list.html', {
 'articles': articles
 })
```

### request 对象

```python
def request_info(request):
 """展示 request 对象的属性"""
 
 # 请求方法
 method = request.method # 'GET', 'POST', 'PUT', 'DELETE'
 
 # 请求路径
 path = request.path # '/blog/article/1/'
 full_path = request.get_full_path() # '/blog/article/1/?page=2'
 
 # GET 参数
 page = request.GET.get('page', 1) # 获取 GET 参数
 tags = request.GET.getlist('tags') # 获取多个值
 
 # POST 数据
 title = request.POST.get('title')
 
 # 请求头
 user_agent = request.META.get('HTTP_USER_AGENT')
 content_type = request.META.get('CONTENT_TYPE')
 
 # 用户信息
 user = request.user
 is_authenticated = request.user.is_authenticated
 
 # Cookies
 session_id = request.COOKIES.get('sessionid')
 
 # FILES（上传文件）
 file = request.FILES.get('avatar')
 
 return JsonResponse({
 'method': method,
 'path': path,
 'user': str(user),
 })
```

## 响应类型

### HttpResponse

```python
from django.http import HttpResponse

def simple_response(request):
 """基本响应"""
 return HttpResponse("Hello World")

def html_response(request):
 """HTML 响应"""
 html = "<h1>Hello</h1><p>This is a paragraph.</p>"
 return HttpResponse(html, content_type='text/html')

def custom_headers(request):
 """自定义响应头"""
 response = HttpResponse("Custom headers")
 response['X-Custom-Header'] = 'Custom Value'
 response['Cache-Control'] = 'no-cache'
 return response
```

### JsonResponse

```python
from django.http import JsonResponse

def json_response(request):
 """JSON 响应"""
 data = {
 'status': 'success',
 'message': '操作成功',
 'data': {
 'id': 1,
 'title': 'Django 教程'
 }
 }
 return JsonResponse(data)

def json_list(request):
 """JSON 列表响应"""
 articles = list(Article.objects.values('id', 'title'))
 return JsonResponse(articles, safe=False) # 非字典需要 safe=False
```

### HttpResponseRedirect

```python
from django.shortcuts import redirect
from django.urls import reverse

def redirect_view(request):
 """重定向"""
 # 重定向到 URL
 return redirect('/blog/')
 
 # 重定向到命名 URL
 return redirect('blog:article_list')
 
 # 重定向到命名 URL（带参数）
 return redirect('blog:article_detail', pk=1)
 
 # 永久重定向
 return redirect('blog:article_list', permanent=True)
```

### StreamingHttpResponse

```python
from django.http import StreamingHttpResponse
import csv

def download_csv(request):
 """流式响应 - 下载大文件"""
 
 def generate_csv():
 """生成 CSV 内容"""
 yield ['ID', '标题', '内容']
 for article in Article.objects.iterator():
 yield [article.id, article.title, article.content]
 
 response = StreamingHttpResponse(
 generate_csv(),
 content_type='text/csv'
 )
 response['Content-Disposition'] = 'attachment; filename="articles.csv"'
 return response
```

### FileResponse

```python
from django.http import FileResponse
```

## 常用视图模式

### 列表视图

```python
from django.shortcuts import render
from django.core.paginator import Paginator
from .models import Article

def article_list(request):
 """文章列表视图"""
 # 获取查询参数
 page = request.GET.get('page', 1)
 category_slug = request.GET.get('category')
 search_query = request.GET.get('q')
 
 # 构建查询
 articles = Article.objects.filter(status='published')
 
 if category_slug:
 articles = articles.filter(category__slug=category_slug)
 
 if search_query:
 articles = articles.filter(title__icontains=search_query)
 
 # 分页
 paginator = Paginator(articles, 10) # 每页 10 条
 page_obj = paginator.get_page(page)
 
 return render(request, 'blog/article_list.html', {
 'page_obj': page_obj,
 'category_slug': category_slug,
 'search_query': search_query,
 })
```

### 详情视图

```python
from django.shortcuts import render, get_object_or_404

def article_detail(request, pk):
 """文章详情视图"""
 article = get_object_or_404(
 Article,
 pk=pk,
 status='published'
 )
 
 # 增加浏览量
 article.increase_views()
 
 # 获取相关文章
 related_articles = Article.objects.filter(
 category=article.category
 ).exclude(pk=pk)[:3]
 
 # 获取评论
 comments = article.comments.filter(is_public=True)
 
 return render(request, 'blog/article_detail.html', {
 'article': article,
 'related_articles': related_articles,
 'comments': comments,
 })
```

### 创建视图

```python
from django.shortcuts import render, redirect
from django.contrib.auth.decorators import login_required
from django.contrib import messages
from .models import Article
from .forms import ArticleForm

@login_required
def article_create(request):
 """创建文章视图"""
 if request.method == 'POST':
 form = ArticleForm(request.POST, request.FILES)
 if form.is_valid():
 article = form.save(commit=False)
 article.author = request.user
 article.save()
 form.save_m2m() # 保存多对多关系
 messages.success(request, '文章创建成功！')
 return redirect('blog:article_detail', pk=article.pk)
 else:
 form = ArticleForm()
 
 return render(request, 'blog/article_form.html', {
 'form': form,
 'title': '创建文章',
 })
```

### 更新视图

```python
from django.contrib.auth.decorators import login_required
from django.http import HttpResponseForbidden

@login_required
def article_update(request, pk):
 """更新文章视图"""
 article = get_object_or_404(Article, pk=pk)
 
 # 权限检查
 if article.author != request.user:
 return HttpResponseForbidden("你没有权限编辑此文章")
 
 if request.method == 'POST':
 form = ArticleForm(request.POST, request.FILES, instance=article)
 if form.is_valid():
 form.save()
 messages.success(request, '文章更新成功！')
 return redirect('blog:article_detail', pk=article.pk)
 else:
 form = ArticleForm(instance=article)
 
 return render(request, 'blog/article_form.html', {
 'form': form,
 'article': article,
 'title': '编辑文章',
 })
```

### 删除视图

```python
from django.views.decorators.http import require_POST

@login_required
@require_POST
def article_delete(request, pk):
 """删除文章视图"""
 article = get_object_or_404(Article, pk=pk)
 
 # 权限检查
 if article.author != request.user:
 return HttpResponseForbidden("你没有权限删除此文章")
 
 article.delete()
 messages.success(request, '文章已删除！')
 return redirect('blog:article_list')
```

## 视图装饰器

### HTTP 方法限制

```python
from django.views.decorators.http import require_http_methods, require_GET, require_POST

@require_http_methods(["GET", "POST"])
def article_create(request):
 pass

@require_GET
def article_list(request):
 pass

@require_POST
def article_delete(request, pk):
 pass
```

### 登录验证

```python
from django.contrib.auth.decorators import login_required, permission_required

@login_required
def profile(request):
 pass

@login_required(login_url='/accounts/login/')
def dashboard(request):
 pass

@permission_required('blog.add_article', raise_exception=True)
def article_create(request):
 pass
```

### 缓存控制

```python
from django.views.decorators.cache import cache_page, never_cache

@cache_page(60 * 15) # 缓存 15 分钟
def article_list(request):
 pass

@never_cache
def profile(request):
 pass
```

### GZip 压缩

```python
from django.views.decorators.gzip import gzip_page

@gzip_page
def large_data(request):
 pass
```

### CSRF 豁免

```python
from django.views.decorators.csrf import csrf_exempt, csrf_protect

@csrf_exempt
def api_endpoint(request):
 """API 端点，不需要 CSRF 验证"""
 pass

@csrf_protect
def sensitive_action(request):
 """敏感操作，强制 CSRF 验证"""
 pass
```

## 错误处理

### 404 错误

```python
from django.http import Http404

def article_detail(request, pk):
 try:
 article = Article.objects.get(pk=pk)
 except Article.DoesNotExist:
 raise Http404("文章不存在")
 
 return render(request, 'blog/article_detail.html', {'article': article})

# 或使用快捷方式
def article_detail(request, pk):
 article = get_object_or_404(Article, pk=pk)
 return render(request, 'blog/article_detail.html', {'article': article})
```

### 自定义错误视图

```python
# views.py
def page_not_found(request, exception):
 """自定义 404 页面"""
 return render(request, 'errors/404.html', status=404)

def server_error(request):
 """自定义 500 页面"""
 return render(request, 'errors/500.html', status=500)

def permission_denied(request, exception):
 """自定义 403 页面"""
 return render(request, 'errors/403.html', status=403)

def bad_request(request, exception):
 """自定义 400 页面"""
 return render(request, 'errors/400.html', status=400)

# urls.py
handler404 = 'myapp.views.page_not_found'
handler500 = 'myapp.views.server_error'
handler403 = 'myapp.views.permission_denied'
handler400 = 'myapp.views.bad_request'
```

## 小结

视图函数的关键点：

1. **request 对象**：包含请求的所有信息
2. **响应类型**：HttpResponse、JsonResponse、Redirect 等
3. **常用模式**：列表、详情、创建、更新、删除
4. **装饰器**：限制 HTTP 方法、验证登录、缓存控制
5. **错误处理**：Http404、自定义错误页面

> **提示**：视图函数简单直观，适合简单的业务逻辑。对于复杂场景，可以考虑使用类视图（CBV）。
