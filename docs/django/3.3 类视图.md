# 3.3 类视图

类视图（Class-Based Views, CBV）是 Django 提供的另一种视图实现方式。相比函数视图，类视图更适合处理复杂的业务逻辑，并且支持继承和混入。

## 类视图基础

### 基本类视图

```python
from django.views import View
from django.http import HttpResponse

class HelloView(View):
 """最简单的类视图"""
 
 def get(self, request):
 return HttpResponse("Hello, Django!")
 
 def post(self, request):
 return HttpResponse("POST 请求")
```

### URL 配置

```python
# urls.py
from django.urls import path
from .views import HelloView, ArticleListView

urlpatterns = [
 path('hello/', HelloView.as_view(), name='hello'),
 path('articles/', ArticleListView.as_view(), name='article_list'),
]
```

### 类视图优势

| 特性 | 函数视图（FBV） | 类视图（CBV） |
|------|----------------|---------------|
| 代码组织 | 线性结构 | 按方法分离 |
| 继承 | 不支持 | 支持 |
| 混入（Mixin） | 不支持 | 支持 |
| HTTP 方法处理 | if/else | 独立方法 |
| 代码复用 | 较难 | 容易 |

## View 类详解

### HTTP 方法处理

```python
from django.views import View
from django.http import HttpResponse, JsonResponse

class ArticleAPIView(View):
 """文章 API 视图"""
 
 def get(self, request, pk=None):
 """获取文章"""
 if pk:
 article = get_object_or_404(Article, pk=pk)
 return JsonResponse({
 'id': article.id,
 'title': article.title,
 'content': article.content,
 })
 else:
 articles = list(Article.objects.values('id', 'title'))
 return JsonResponse(articles, safe=False)
 
 def post(self, request):
 """创建文章"""
 import json
 data = json.loads(request.body)
 article = Article.objects.create(**data)
 return JsonResponse({'id': article.id}, status=201)
 
 def put(self, request, pk):
 """更新文章"""
 article = get_object_or_404(Article, pk=pk)
 import json
 data = json.loads(request.body)
 Article.objects.filter(pk=pk).update(**data)
 return JsonResponse({'status': 'success'})
 
 def delete(self, request, pk):
 """删除文章"""
 article = get_object_or_404(Article, pk=pk)
 article.delete()
 return JsonResponse({'status': 'deleted'}, status=204)
```

### 类属性和方法

```python
class ArticleView(View):
 # 类属性
 model = Article
 template_name = 'blog/article_list.html'
 
 def get(self, request):
 # 使用类属性
 articles = self.model.objects.all()
 return render(request, self.template_name, {'articles': articles})
 
 @classmethod
 def as_view(cls, **kwargs):
 """自定义 as_view 方法"""
 return super().as_view(**kwargs)
 
 def dispatch(self, request, *args, **kwargs):
 """请求分发前的处理"""
 # 可以在这里添加权限检查等
 return super().dispatch(request, *args, **kwargs)
 
 def http_method_not_allowed(self, request, *args, **kwargs):
 """不支持的 HTTP 方法"""
 return JsonResponse({'error': 'Method not allowed'}, status=405)
```

## 通用视图

Django 提供了一系列通用视图，简化常见的视图模式。

### ListView

```python
from django.views.generic import ListView
from .models import Article

class ArticleListView(ListView):
 """文章列表视图"""
 model = Article
 template_name = 'blog/article_list.html'
 context_object_name = 'articles' # 模板中的变量名
 paginate_by = 10 # 分页
 ordering = ['-created_at']
 
 def get_queryset(self):
 """自定义查询集"""
 queryset = super().get_queryset()
 category = self.request.GET.get('category')
 if category:
 queryset = queryset.filter(category__slug=category)
 return queryset
 
 def get_context_data(self, **kwargs):
 """添加额外上下文"""
 context = super().get_context_data(**kwargs)
 context['categories'] = Category.objects.all()
 context['current_category'] = self.request.GET.get('category')
 return context
```

### DetailView

```python
from django.views.generic import DetailView

class ArticleDetailView(DetailView):
 """文章详情视图"""
 model = Article
 template_name = 'blog/article_detail.html'
 context_object_name = 'article'
 slug_field = 'slug' # 使用 slug 而不是 pk
 slug_url_kwarg = 'slug'
 
 def get_object(self, queryset=None):
 """自定义获取对象"""
 obj = super().get_object(queryset)
 obj.increase_views() # 增加浏览量
 return obj
 
 def get_context_data(self, **kwargs):
 context = super().get_context_data(**kwargs)
 context['related_articles'] = Article.objects.filter(
 category=self.object.category
 ).exclude(pk=self.object.pk)[:3]
 return context
```

### CreateView

```python
from django.views.generic import CreateView
from django.urls import reverse_lazy
from django.contrib.auth.mixins import LoginRequiredMixin

class ArticleCreateView(LoginRequiredMixin, CreateView):
 """创建文章视图"""
 model = Article
 template_name = 'blog/article_form.html'
 fields = ['title', 'content', 'category', 'tags']
 success_url = reverse_lazy('blog:article_list')
 
 def form_valid(self, form):
 """表单验证通过时调用"""
 form.instance.author = self.request.user
 return super().form_valid(form)
 
 def get_success_url(self):
 """成功后跳转的 URL"""
 return reverse('blog:article_detail', kwargs={'pk': self.object.pk})
```

### UpdateView

```python
from django.views.generic import UpdateView
from django.contrib.auth.mixins import LoginRequiredMixin, UserPassesTestMixin

class ArticleUpdateView(LoginRequiredMixin, UserPassesTestMixin, UpdateView):
 """更新文章视图"""
 model = Article
 template_name = 'blog/article_form.html'
 fields = ['title', 'content', 'category', 'tags']
 
 def test_func(self):
 """权限检查"""
 article = self.get_object()
 return article.author == self.request.user
 
 def get_success_url(self):
 return reverse('blog:article_detail', kwargs={'pk': self.object.pk})
```

### FormView

```python
from django.views.generic import FormView
from .forms import ContactForm

class ContactView(FormView):
 """联系表单视图"""

 return super().form_valid(form)
```

### TemplateView

```python
from django.views.generic import TemplateView

class AboutView(TemplateView):
 """关于页面"""

```

## Mixin 类

Mixin 是可重用的类，用于为视图添加额外功能。

### 自定义 Mixin

```python
class CategoryMixin:
 """分类 Mixin"""
 def get_category(self):
 slug = self.kwargs.get('slug')
 return get_object_or_404(Category, slug=slug)
 
 def get_context_data(self, **kwargs):
 context = super().get_context_data(**kwargs)
 context['category'] = self.get_category()
 return context

class CategoryArticleListView(CategoryMixin, ListView):
 """分类文章列表"""
 model = Article
 template_name = 'blog/category_articles.html'
 
 def get_queryset(self):
 return Article.objects.filter(category=self.get_category())
```

### 内置 Mixin

```python
from django.contrib.auth.mixins import (
 LoginRequiredMixin,
 PermissionRequiredMixin,
 UserPassesTestMixin
)

class ArticleCreateView(LoginRequiredMixin, PermissionRequiredMixin, CreateView):
 """需要登录和权限的创建视图"""
 model = Article
 fields = ['title', 'content']
 permission_required = 'blog.add_article'
 login_url = '/accounts/login/'

class AdminOnlyView(UserPassesTestMixin, TemplateView):
 """仅管理员可访问"""
 template_name = 'admin/dashboard.html'
 
 def test_func(self):
 return self.request.user.is_staff
```

## 通用视图对照表

| 视图类 | 用途 | 主要方法 |
|--------|------|----------|
| View | 基础视图 | get, post, put, delete |
| TemplateView | 渲染模板 | get_context_data |
| RedirectView | 重定向 | get_redirect_url |
| ListView | 列表展示 | get_queryset, get_context_data |
| DetailView | 详情展示 | get_object |
| CreateView | 创建记录 | form_valid |
| UpdateView | 更新记录 | form_valid, test_func |
| DeleteView | 删除记录 | delete |
| FormView | 表单处理 | form_valid |

## 类视图最佳实践

### 1. 合理使用 Mixin

```python
class OwnerMixin:
 """所有者检查 Mixin"""
 def test_func(self):
 obj = self.get_object()
 return obj.author == self.request.user

class ArticleUpdateView(LoginRequiredMixin, OwnerMixin, UpdateView):
 model = Article
 fields = ['title', 'content']
```

### 2. 复写方法而非修改

```python
class ArticleListView(ListView):
 model = Article
 
 # 推荐：复写方法
 def get_queryset(self):
 return super().get_queryset().filter(status='published')
 
 # 不推荐：直接修改
 # queryset = Article.objects.filter(status='published')
```

### 3. 使用 get_context_data 添加上下文

```python
class ArticleDetailView(DetailView):
 model = Article
 
 def get_context_data(self, **kwargs):
 context = super().get_context_data(**kwargs)
 context['comments'] = self.object.comments.all()
 context['related'] = self.get_related_articles()
 return context
 
 def get_related_articles(self):
 return Article.objects.filter(
 category=self.object.category
 ).exclude(pk=self.object.pk)[:5]
```

## 小结

类视图的关键点：

1. **View 类**：基础类视图，按 HTTP 方法分发
2. **通用视图**：ListView、DetailView、CreateView 等
3. **Mixin**：可重用的功能组件
4. **方法复写**：get_queryset、get_context_data 等
5. **权限控制**：LoginRequiredMixin、PermissionRequiredMixin

> **提示**：类视图适合复杂的业务场景，代码组织更清晰。对于简单的视图，函数视图仍然是好选择。
