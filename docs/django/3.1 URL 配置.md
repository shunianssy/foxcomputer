# 3.1 URL 配置

URL 配置是 Django 的路由系统，负责将 URL 路径映射到视图函数。良好的 URL 设计是 Web 应用的重要组成部分。

## URL 配置基础

### 项目级 URL 配置

```python
# mysite/urls.py
from django.contrib import admin
from django.urls import path, include

urlpatterns = [
 path('admin/', admin.site.urls),
 path('blog/', include('blog.urls')), # 包含应用的 URL
]
```

### 应用级 URL 配置

```python
# blog/urls.py
from django.urls import path
from . import views

app_name = 'blog' # 应用命名空间

urlpatterns = [
 path('', views.article_list, name='article_list'),
 path('article/<int:pk>/', views.article_detail, name='article_detail'),
 path('category/<slug:slug>/', views.category_detail, name='category_detail'),
]
```

## path() 函数

### 基本语法

```python
path(route, view, kwargs=None, name=None)
```

| 参数 | 说明 |
|------|------|
| `route` | URL 路径模式 |
| `view` | 视图函数或类视图 |
| `kwargs` | 传递给视图的额外参数 |
| `name` | URL 名称，用于反向解析 |

### 路径转换器

```python
from django.urls import path
from . import views

urlpatterns = [
 # str: 匹配非空字符串，不含 /
 path('article/<str:title>/', views.article_by_title),
 
 # int: 匹配整数
 path('article/<int:pk>/', views.article_detail),
 
 # slug: 匹配 slug 字符串（字母、数字、下划线、连字符）
 path('category/<slug:slug>/', views.category_detail),
 
 # uuid: 匹配 UUID
 path('file/<uuid:id>/', views.file_detail),
 
 # path: 匹配任意非空字符串，包含 /
 path('page/<path:url>/', views.page_detail),
]
```

### 自定义路径转换器

```python
# converters.py
class FourDigitYearConverter:
 """四位年份转换器"""
 regex = '[0-9]{4}'

 def to_python(self, value):
 return int(value)

 def to_url(self, value):
 return '%04d' % value

# urls.py
from django.urls import path, register_converter
from . import converters, views

# 注册转换器
register_converter(converters.FourDigitYearConverter, 'yyyy')

urlpatterns = [
 path('archive/<yyyy:year>/', views.archive),
]
```

## re_path() 正则路由

对于复杂的 URL 模式，可以使用正则表达式：

```python
from django.urls import path, re_path
from . import views

urlpatterns = [
 # 匹配年份和月份
 re_path(r'^archive/(?P<year>[0-9]{4})/(?P<month>[0-9]{2})/$', views.archive),
 
 # 匹配用户名（字母、数字、下划线）
 re_path(r'^user/(?P<username>\w+)/$', views.user_profile),
 
 # 匹配文件路径
 re_path(r'^download/(?P<path>.+)/$', views.download),
]
```

### path vs re_path

```python
# path() 更简洁
path('article/<int:pk>/', views.article_detail)

# re_path() 更灵活
re_path(r'^article/(?P<pk>\d+)/$', views.article_detail)
```

## URL 命名

### 命名 URL

```python
# blog/urls.py
urlpatterns = [
 path('', views.article_list, name='article_list'),
 path('article/<int:pk>/', views.article_detail, name='article_detail'),
 path('search/', views.search, name='search'),
]
```

### 命名空间

```python
# 项目 urls.py
urlpatterns = [
 path('blog/', include('blog.urls', namespace='blog')),
 path('shop/', include('shop.urls', namespace='shop')),
]

# 或者使用 app_name
# blog/urls.py
app_name = 'blog'
urlpatterns = [...]
```

## URL 反向解析

### 在视图中使用

```python
from django.urls import reverse, reverse_lazy
from django.shortcuts import redirect

def article_create(request):
 # 创建文章后重定向
 return redirect('blog:article_list')

def get_absolute_url(self):
 # 模型中定义绝对 URL
 return reverse('blog:article_detail', kwargs={'pk': self.pk})

# 带参数的反向解析
url = reverse('blog:article_detail', kwargs={'pk': 1})
# 输出: /blog/article/1/

# reverse_lazy 用于类视图
class ArticleCreateView(CreateView):
 success_url = reverse_lazy('blog:article_list')
```

### 在模板中使用

```html
<!-- 使用 url 标签 -->
<a href="{% url 'blog:article_list' %}">文章列表</a>

<!-- 带参数 -->
<a href="{% url 'blog:article_detail' pk=article.pk %}">{{ article.title }}</a>

<!-- 带查询参数 -->
<a href="{% url 'blog:search' %}?q=django">搜索 Django</a>
```

## URL 包含

### include() 函数

```python
from django.urls import path, include

urlpatterns = [
 # 包含其他 URL 配置
 path('blog/', include('blog.urls')),
 
 # 包含并添加前缀
 path('api/v1/', include([
 path('articles/', include('blog.api_urls')),
 path('users/', include('users.api_urls')),
 ])),
 
 # 包含并设置命名空间
 path('blog/', include(('blog.urls', 'blog'), namespace='blog')),
]
```

### 三方应用 URL

```python
urlpatterns = [
 path('admin/', admin.site.urls),
 path('accounts/', include('django.contrib.auth.urls')),
 path('api/', include('rest_framework.urls')),
]
```

## URL 参数传递

### 位置参数

```python
# urls.py
path('archive/<int:year>/<int:month>/', views.archive)

# views.py
def archive(request, year, month):
 pass
```

### 关键字参数

```python
# urls.py
path('article/<int:pk>/', views.article_detail, {'status': 'published'})

# views.py
def article_detail(request, pk, status):
 article = Article.objects.get(pk=pk, status=status)
 return render(request, 'article_detail.html', {'article': article})
```

### 额外参数

```python
# urls.py
urlpatterns = [
 path('about/', views.static_page, {'template': 'about.html'}, name='about'),
 path('contact/', views.static_page, {'template': 'contact.html'}, name='contact'),
]

# views.py
def static_page(request, template):
 return render(request, template)
```

## URL 配置最佳实践

### 1. 按功能模块组织

```python
# blog/urls.py
from django.urls import path
from . import views

app_name = 'blog'

urlpatterns = [
 # 文章相关
 path('', views.ArticleListView.as_view(), name='article_list'),
 path('article/<int:pk>/', views.ArticleDetailView.as_view(), name='article_detail'),
 path('article/create/', views.ArticleCreateView.as_view(), name='article_create'),
 
 # 分类相关
 path('categories/', views.CategoryListView.as_view(), name='category_list'),
 path('category/<slug:slug>/', views.CategoryDetailView.as_view(), name='category_detail'),
 
 # 标签相关
 path('tags/', views.TagListView.as_view(), name='tag_list'),
 path('tag/<slug:slug>/', views.TagDetailView.as_view(), name='tag_detail'),
]
```

### 2. 使用命名空间

```python
# 项目 urls.py
urlpatterns = [
 path('admin/', admin.site.urls),
 path('blog/', include('blog.urls')),
 path('shop/', include('shop.urls')),
 path('api/', include('api.urls')),
]

# blog/urls.py
app_name = 'blog'
urlpatterns = [...]

# 使用时
reverse('blog:article_list')
```

### 3. RESTful 风格

```python
# RESTful URL 设计
urlpatterns = [
 # 文章资源
 path('articles/', views.ArticleListCreateView.as_view(), name='article_list'),
 path('articles/<int:pk>/', views.ArticleDetailView.as_view(), name='article_detail'),
 
 # 评论资源
 path('articles/<int:article_id>/comments/', views.CommentListView.as_view(), name='comment_list'),
 path('comments/<int:pk>/', views.CommentDetailView.as_view(), name='comment_detail'),
]
```

### 4. 版本控制

```python
# API 版本控制
urlpatterns = [
 path('api/v1/', include('api.v1.urls')),
 path('api/v2/', include('api.v2.urls')),
]
```

## 完整示例

```python
# mysite/urls.py
from django.contrib import admin
from django.urls import path, include
from django.conf import settings
from django.conf.urls.static import static

urlpatterns = [
 # 管理后台
 path('admin/', admin.site.urls),
 
 # 应用路由
 path('', include('core.urls')),
 path('blog/', include('blog.urls')),
 path('accounts/', include('accounts.urls')),
 
 # API 路由
 path('api/v1/', include('api.v1.urls')),
 
 # 第三方应用
 path('markdownx/', include('markdownx.urls')),
]

# 开发环境静态文件
if settings.DEBUG:
 urlpatterns += static(settings.MEDIA_URL, document_root=settings.MEDIA_ROOT)
 urlpatterns += static(settings.STATIC_URL, document_root=settings.STATIC_ROOT)
```

## 小结

URL 配置的关键点：

1. **path()**：简单的 URL 模式匹配
2. **re_path()**：复杂的正则表达式匹配
3. **命名**：为 URL 添加名称，便于反向解析
4. **命名空间**：避免不同应用间的 URL 名称冲突
5. **include()**：模块化组织 URL 配置

> **提示**：始终使用命名 URL 和反向解析，避免在代码中硬编码 URL 路径。这样当 URL 结构变化时，只需要修改 urls.py 文件。
