# 3.4 通用视图

Django 提供了丰富的通用视图（Generic Views），可以大幅减少重复代码。本章详细介绍各种通用视图的使用方法。

## 通用视图概述

### 视图分类

```
通用视图
├── 基础视图
│ ├── View # 基础类视图
│ ├── TemplateView # 模板视图
│ └── RedirectView # 重定向视图
├── 列表视图
│ └── ListView # 列表展示
├── 详情视图
│ └── DetailView # 详情展示
├── 编辑视图
│ ├── CreateView # 创建视图
│ ├── UpdateView # 更新视图
│ └── DeleteView # 删除视图
└── 日期视图
 ├── ArchiveIndexView # 归档索引
 ├── YearArchiveView # 年归档
 ├── MonthArchiveView # 月归档
 └── DateDetailView # 日期详情
```

## ListView 详解

### 基本用法

```python
from django.views.generic import ListView
from .models import Article

class ArticleListView(ListView):
 model = Article
 template_name = 'blog/article_list.html'
 context_object_name = 'articles'
 paginate_by = 10
 ordering = ['-created_at']
```

### 自定义查询

```python
class ArticleListView(ListView):
 model = Article
 template_name = 'blog/article_list.html'
 context_object_name = 'articles'
 paginate_by = 10
 
 def get_queryset(self):
 """自定义查询集"""
 queryset = super().get_queryset()
 
 # 过滤已发布文章
 queryset = queryset.filter(status='published')
 
 # 按分类过滤
 category_slug = self.request.GET.get('category')
 if category_slug:
 queryset = queryset.filter(category__slug=category_slug)
 
 # 搜索
 search = self.request.GET.get('q')
 if search:
 queryset = queryset.filter(title__icontains=search)
 
 return queryset
 
 def get_context_data(self, **kwargs):
 """添加额外上下文"""
 context = super().get_context_data(**kwargs)
 
 # 添加分类列表
 context['categories'] = Category.objects.all()
 
 # 添加当前筛选条件
 context['current_category'] = self.request.GET.get('category')
 context['search_query'] = self.request.GET.get('q', '')
 
 return context
```

### 分页自定义

```python
class ArticleListView(ListView):
 model = Article
 paginate_by = 10
 
 def get_paginate_by(self, queryset):
 """动态设置每页数量"""
 per_page = self.request.GET.get('per_page', 10)
 return min(int(per_page), 50) # 最大 50 条
 
 def get_paginator(self, queryset, per_page, **kwargs):
 """自定义分页器"""
 from django.core.paginator import Paginator
 return Paginator(queryset, per_page, **kwargs)
```

### 使用 Slug

```python
class ArticleDetailView(DetailView):
 model = Article
 template_name = 'blog/article_detail.html'
 slug_field = 'slug' # 模型字段
 slug_url_kwarg = 'slug' # URL 参数名
```

### 自定义获取对象

```python
class ArticleDetailView(DetailView):
 model = Article
 template_name = 'blog/article_detail.html'
 
 def get_object(self, queryset=None):
 """自定义获取对象"""
 obj = super().get_object(queryset)
 
 # 增加浏览量
 obj.increase_views()
 
 # 检查权限
 if obj.status == 'draft' and obj.author != self.request.user:
 raise Http404("文章不存在")
 
 return obj
 
 def get_context_data(self, **kwargs):
 context = super().get_context_data(**kwargs)
 
 # 相关文章
 context['related_articles'] = Article.objects.filter(
 category=self.object.category
 ).exclude(pk=self.object.pk)[:5]
 
 # 评论
 context['comments'] = self.object.comments.filter(is_public=True)
 
 return context
```

## CreateView 详解

### 基本用法

```python
from django.views.generic import CreateView
from django.urls import reverse_lazy
from django.contrib.auth.mixins import LoginRequiredMixin

class ArticleCreateView(LoginRequiredMixin, CreateView):
 model = Article
 fields = ['title', 'content', 'category', 'tags']
 template_name = 'blog/article_form.html'
 success_url = reverse_lazy('blog:article_list')
```

### 自定义表单

```python
class ArticleCreateView(LoginRequiredMixin, CreateView):
 model = Article
 form_class = ArticleForm
 template_name = 'blog/article_form.html'
 
 def form_valid(self, form):
 """表单验证通过"""
 form.instance.author = self.request.user
 messages.success(self.request, '文章创建成功！')
 return super().form_valid(form)
 
 def form_invalid(self, form):
 """表单验证失败"""
 messages.error(self.request, '请检查表单内容')
 return super().form_invalid(form)
 
 def get_success_url(self):
 """成功跳转 URL"""
 return reverse('blog:article_detail', kwargs={'pk': self.object.pk})
 
 def get_context_data(self, **kwargs):
 context = super().get_context_data(**kwargs)
 context['title'] = '创建文章'
 return context
```

### 保存多对多关系

```python
class ArticleCreateView(LoginRequiredMixin, CreateView):
 model = Article
 form_class = ArticleForm
 
 def form_valid(self, form):
 form.instance.author = self.request.user
 self.object = form.save(commit=False)
 self.object.save()
 form.save_m2m() # 保存多对多关系
 return redirect(self.get_success_url())
```

## UpdateView 详解

### 基本用法

```python
from django.views.generic import UpdateView
from django.contrib.auth.mixins import LoginRequiredMixin, UserPassesTestMixin

class ArticleUpdateView(LoginRequiredMixin, UserPassesTestMixin, UpdateView):
 model = Article
 fields = ['title', 'content', 'category', 'tags']
 template_name = 'blog/article_form.html'
 
 def test_func(self):
 """权限检查"""
 article = self.get_object()
 return article.author == self.request.user
 
 def get_success_url(self):
 return reverse('blog:article_detail', kwargs={'pk': self.object.pk})
```

### 自定义表单初始值

```python
class ArticleUpdateView(UpdateView):
 model = Article
 form_class = ArticleForm
 
 def get_initial(self):
 """设置表单初始值"""
 initial = super().get_initial()
 initial['tags'] = self.object.tags.values_list('id', flat=True)
 return initial
 
 def get_context_data(self, **kwargs):
 context = super().get_context_data(**kwargs)
 context['title'] = f'编辑: {self.object.title}'
 return context
```

## DeleteView 详解

### 基本用法

```python
from django.views.generic import DeleteView

class ArticleDeleteView(
 LoginRequiredMixin,

 SuccessMessageMixin,
 DeleteView
):

 success_message = '文章已删除'
 
 def test_func(self):
 article = self.get_object()
 return article.author == self.request.user
```

### 自定义删除逻辑

```python
class ArticleDeleteView(DeleteView):
 model = Article
 success_url = reverse_lazy('blog:article_list')
 
 def delete(self, request, *args, **kwargs):
 """自定义删除操作"""
 self.object = self.get_object()
 
 # 软删除（不真正删除）
 self.object.is_deleted = True
 self.object.save()
 
 # 或者记录日志
 logger.info(f"Article {self.object.id} deleted by {request.user}")
 
 messages.success(request, '文章已删除')
 return redirect(self.success_url)
 
 def post(self, request, *args, **kwargs):
 """POST 请求处理"""
 return self.delete(request, *args, **kwargs)
```

## FormView 详解

### 基本用法

```python
from django.views.generic import FormView
from .forms import ContactForm

class ContactView(FormView):
 template_name = 'blog/contact.html'
 form_class = ContactForm
 success_url = reverse_lazy('blog:contact_success')
 
 def form_valid(self, form):
 form.send_email()
 messages.success(self.request, '邮件发送成功！')
 return super().form_valid(form)
```

### 搜索视图

```python
class SearchView(FormView):
 template_name = 'blog/search.html'
 form_class = SearchForm
 
 def form_valid(self, form):
 query = form.cleaned_data['query']
 self.results = Article.objects.filter(
 title__icontains=query
 )
 return self.render_to_response(
 self.get_context_data(form=form, results=self.results)
 )
 
 def get_context_data(self, **kwargs):
 context = super().get_context_data(**kwargs)
 context['results'] = getattr(self, 'results', [])
 return context
```

## TemplateView 详解

### 基本用法

```python
from django.views.generic import TemplateView

class AboutView(TemplateView):
 template_name = 'blog/about.html'
 
 def get_context_data(self, **kwargs):
 context = super().get_context_data(**kwargs)
 context['title'] = '关于我们'
 context['team_members'] = self.get_team_members()
 return context
 
 def get_team_members(self):
 return [
 {'name': 'Alice', 'role': 'Developer'},
 {'name': 'Bob', 'role': 'Designer'},
 ]
```

## RedirectView 详解

### 基本用法

```python
from django.views.generic import RedirectView

class GoToGoogleView(RedirectView):
 url = 'https://google.com'
 permanent = False # 302 临时重定向

class ArticleRedirectView(RedirectView):
 pattern_name = 'blog:article_detail' # 使用命名 URL
 
 def get_redirect_url(self, *args, **kwargs):
 # 动态构建 URL
 article = get_object_or_404(Article, pk=kwargs['pk'])
 return super().get_redirect_url(pk=article.pk)
```

## 组合使用示例

```python
# views.py
from django.views.generic import ListView, DetailView, CreateView, UpdateView, DeleteView
from django.contrib.auth.mixins import LoginRequiredMixin, PermissionRequiredMixin
from django.urls import reverse_lazy
from django.contrib.messages.views import SuccessMessageMixin
from .models import Article
from .forms import ArticleForm

# 列表视图
class ArticleListView(ListView):
 model = Article
 template_name = 'blog/article_list.html'
 context_object_name = 'articles'
 paginate_by = 10
 
 def get_queryset(self):
 return super().get_queryset().filter(status='published')

# 详情视图
class ArticleDetailView(DetailView):
 model = Article
 template_name = 'blog/article_detail.html'
 
 def get_object(self, queryset=None):
 obj = super().get_object(queryset)
 obj.increase_views()
 return obj

# 创建视图
class ArticleCreateView(
 LoginRequiredMixin,
 PermissionRequiredMixin,
 SuccessMessageMixin,
 CreateView
):
 model = Article
 form_class = ArticleForm
 template_name = 'blog/article_form.html'
 permission_required = 'blog.add_article'
 success_message = '文章创建成功'
 
 def form_valid(self, form):
 form.instance.author = self.request.user
 return super().form_valid(form)

# 更新视图
class ArticleUpdateView(
 LoginRequiredMixin,
 PermissionRequiredMixin,
 SuccessMessageMixin,
 UpdateView
):
 model = Article
 form_class = ArticleForm
 template_name = 'blog/article_form.html'
 permission_required = 'blog.change_article'
 success_message = '文章更新成功'

# 删除视图
class ArticleDeleteView(
 LoginRequiredMixin,
 PermissionRequiredMixin,
 SuccessMessageMixin,
 DeleteView
):
 model = Article
 success_url = reverse_lazy('blog:article_list')
 permission_required = 'blog.delete_article'
 success_message = '文章已删除'
```

## 小结

通用视图的关键点：

1. **ListView**：列表展示，支持分页
2. **DetailView**：详情展示，支持 slug
3. **CreateView**：创建记录，自动处理表单
4. **UpdateView**：更新记录，权限控制
5. **DeleteView**：删除记录，确认页面
6. **Mixin**：组合使用，扩展功能

> **提示**：通用视图可以大幅减少重复代码。通过复写方法和使用 Mixin，可以实现各种复杂需求。
