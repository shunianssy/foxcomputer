# 4.3 自定义标签

Django 允许创建自定义模板标签和过滤器，用于扩展模板功能。本章介绍如何创建和使用自定义标签。

## 自定义过滤器

### 创建过滤器文件

在应用目录下创建 `templatetags` 目录：

```text
blog/
├── __init__.py
├── models.py
├── views.py
├── templatetags/
│ ├── __init__.py
│ └── blog_tags.py
```

### 定义过滤器

```python
# blog/templatetags/blog_tags.py
from django import template
from django.utils.safestring import mark_safe

register = template.Library()

@register.filter
def multiply(value, arg):
 """乘法过滤器"""
 try:
 return float(value) * float(arg)
 except (ValueError, TypeError):
 return ''

@register.filter
def currency(value, symbol='¥'):
 """货币格式化"""
 try:
 return f"{symbol}{float(value):.2f}"
 except (ValueError, TypeError):
 return value

@register.filter
def truncate_cn(value, length):
 """中文截取（按字符数）"""
 if len(value) > length:
 return value[:length] + '...'
 return value

@register.filter
def markdown(value):
 """Markdown 转 HTML"""
 import markdown as md
 return mark_safe(md.markdown(value, extensions=['fenced_code', 'tables']))

@register.filter(name='add_class')
def add_class(field, css_class):
 """为表单字段添加 CSS 类"""
 return field.as_widget(attrs={'class': css_class})
```

### 使用自定义过滤器

```html
{% load blog_tags %}

<!-- 乘法 -->
{{ price|multiply:2 }}

<!-- 货币格式 -->
{{ price|currency }}
{{ price|currency:"$" }}

<!-- 中文截取 -->
{{ content|truncate_cn:100 }}

<!-- Markdown -->
{{ content|markdown }}

<!-- 表单字段添加类 -->
{{ form.title|add_class:"form-control" }}
```

## 自定义简单标签

### 定义简单标签

```python
# blog/templatetags/blog_tags.py
from django import template
from ..models import Category, Article

register = template.Library()

@register.simple_tag
def total_articles():
 """获取文章总数"""
 return Article.objects.count()

@register.simple_tag
def recent_articles(count=5):
 """获取最近文章"""
 return Article.objects.order_by('-created_at')[:count]

@register.simple_tag
def categories_with_count():
 """获取分类及其文章数"""
 return Category.objects.annotate(
 article_count=Count('articles')
 ).filter(article_count__gt=0)

@register.simple_tag(takes_context=True)
def current_time(context, format_string):
 """获取当前时间（使用上下文）"""
 from django.utils import timezone
 return timezone.now().strftime(format_string)

@register.simple_tag
def query_transform(request, **kwargs):
 """修改查询参数"""
 query = request.GET.copy()
 for key, value in kwargs.items():
 query[key] = value
 return query.urlencode()
```

### 使用简单标签

```html
{% load blog_tags %}

<!-- 基本使用 -->
<p>文章总数：{% total_articles %}</p>

<!-- 带参数 -->
{% recent_articles 10 as articles %}
{% for article in articles %}
 <li>{{ article.title }}</li>
{% endfor %}

<!-- 使用上下文 -->
<p>当前时间：{% current_time "%Y-%m-%d %H:%M:%S" %}</p>

<!-- 查询参数转换 -->
<a href="?{% query_transform request page=2 %}">下一页</a>
```

## 自定义包含标签

包含标签可以渲染另一个模板，适合创建可重用的组件。

### 定义包含标签

```python
# blog/templatetags/blog_tags.py
from django import template
from ..models import Category, Tag

register = template.Library()

@register.inclusion_tag('blog/_categories.html')
def show_categories():
 """显示分类列表"""
 categories = Category.objects.annotate(
 article_count=Count('articles')
 ).filter(article_count__gt=0)
 return {'categories': categories}

@register.inclusion_tag('blog/_article_list.html')
def show_article_list(articles, show_author=True):
 """显示文章列表"""
 return {
 'articles': articles,
 'show_author': show_author
 }

@register.inclusion_tag('blog/_pagination.html', takes_context=True)
def show_pagination(context, page_obj):
 """显示分页"""
 return {
 'page_obj': page_obj,
 'request': context['request']
 }

@register.inclusion_tag('blog/_tag_cloud.html')
def tag_cloud(limit=20):
 """标签云"""
 tags = Tag.objects.annotate(
 article_count=Count('articles')
 ).order_by('-article_count')[:limit]
 return {'tags': tags}
```

### 模板文件

```html
<!-- templates/blog/_categories.html -->
<div class="categories">
 <h3>分类</h3>
 <ul>
 {% for category in categories %}
 <li>
 <a href="{% url 'blog:category' category.slug %}">
 {{ category.name }}
 <span class="count">({{ category.article_count }})</span>
 </a>
 </li>
 {% endfor %}
 </ul>
</div>

<!-- templates/blog/_article_list.html -->
<div class="article-list">
{% for article in articles %}
 <article class="article-item">
 <h4><a href="{{ article.get_absolute_url }}">{{ article.title }}</a></h4>
 {% if show_author %}
 <span class="author">{{ article.author.username }}</span>
 {% endif %}
 <span class="date">{{ article.created_at|date:"Y-m-d" }}</span>
 </article>
{% empty %}
 <p>暂无文章</p>
{% endfor %}
</div>
```

### 使用包含标签

```html
{% load blog_tags %}

<!-- 显示分类 -->
{% show_categories %}

<!-- 显示文章列表 -->
{% show_article_list articles show_author=False %}

<!-- 显示分页 -->
{% show_pagination page_obj %}

<!-- 标签云 -->
{% tag_cloud 30 %}
```

## 自定义块标签

块标签用于更复杂的逻辑，可以处理标签之间的内容。

### 定义块标签

```python
# blog/templatetags/blog_tags.py
from django import template

register = template.Library()

class HighlightNode(template.Node):
 def __init__(self, nodelist, color):
 self.nodelist = nodelist
 self.color = color
 
 def render(self, context):
 output = self.nodelist.render(context)
 return f'<span style="background-color: {self.color}">{output}</span>'

@register.tag
def highlight(parser, token):
 """高亮显示内容"""
 try:
 tag_name, color = token.split_contents()
 except ValueError:
 raise template.TemplateSyntaxError(
 f"{token.contents.split()[0]} 标签需要一个颜色参数"
 )
 
 nodelist = parser.parse(('endhighlight',))
 parser.delete_first_token()
 
 return HighlightNode(nodelist, color)

class CacheNode(template.Node):
 def __init__(self, nodelist, timeout, key):
 self.nodelist = nodelist
 self.timeout = timeout
 self.key = key
 
 def render(self, context):
 from django.core.cache import cache
 key = self.key.resolve(context)
 value = cache.get(key)
 if value is None:
 value = self.nodelist.render(context)
 cache.set(key, value, self.timeout)
 return value

@register.tag
def mycache(parser, token):
 """简单缓存标签"""
 bits = token.split_contents()
 if len(bits) != 3:
 raise template.TemplateSyntaxError(
 "mycache 标签需要 timeout 和 key 两个参数"
 )
 timeout = int(bits[1])
 key = parser.compile_filter(bits[2])
 nodelist = parser.parse(('endmycache',))
 parser.delete_first_token()
 return CacheNode(nodelist, timeout, key)
```

### 使用块标签

```html
{% load blog_tags %}

<!-- 高亮显示 -->
{% highlight yellow %}
 这段文字会有黄色背景
{% endhighlight %}

<!-- 缓存 -->
{% mycache 300 "sidebar" %}
 {% show_categories %}
{% endmycache %}
```

## 实用自定义标签示例

### 面包屑导航

```python
# blog/templatetags/blog_tags.py
@register.inclusion_tag('blog/_breadcrumbs.html', takes_context=True)
def breadcrumbs(context):
 """面包屑导航"""
 request = context['request']
 path = request.path.strip('/').split('/')
 
 items = [{'title': '首页', 'url': '/'}]
 url = ''
 
 for segment in path:
 url += f'/{segment}'
 items.append({
 'title': segment.title(),
 'url': url
 })
 
 return {'breadcrumbs': items}
```

```html
<!-- templates/blog/_breadcrumbs.html -->
<nav class="breadcrumbs">
 <ol>
 {% for item in breadcrumbs %}
 {% if not forloop.last %}
 <li><a href="{{ item.url }}">{{ item.title }}</a></li>
 {% else %}
 <li class="active">{{ item.title }}</li>
 {% endif %}
 {% endfor %}
 </ol>
</nav>
```

### 读取时间估算

```python
@register.simple_tag
def reading_time(content, wpm=200):
 """估算阅读时间"""
 words = len(content.split())
 minutes = max(1, words // wpm)
 return f"{minutes} 分钟阅读"
```

```html
{% load blog_tags %}
<span class="reading-time">{% reading_time article.content %}</span>
```

### 分页链接

```python
@register.simple_tag(takes_context=True)
def pagination_url(context, page):
 """生成分页链接"""
 request = context['request']
 query = request.GET.copy()
 query['page'] = page
 return f"?{query.urlencode()}"
```

```html
{% load blog_tags %}

<nav class="pagination">
 {% if page_obj.has_previous %}
 <a href="{% pagination_url page_obj.previous_page_number %}">上一页</a>
 {% endif %}
 
 {% for num in page_obj.paginator.page_range %}
 {% if page_obj.number == num %}
 <span class="current">{{ num }}</span>
 {% else %}
 <a href="{% pagination_url num %}">{{ num }}</a>
 {% endif %}
 {% endfor %}
 
 {% if page_obj.has_next %}
 <a href="{% pagination_url page_obj.next_page_number %}">下一页</a>
 {% endif %}
</nav>
```

### 状态徽章

```python
@register.inclusion_tag('blog/_status_badge.html')
def status_badge(status):
 """状态徽章"""
 status_map = {
 'draft': {'text': '草稿', 'class': 'badge-secondary'},
 'published': {'text': '已发布', 'class': 'badge-success'},
 'archived': {'text': '已归档', 'class': 'badge-warning'},
 }
 return {'status': status_map.get(status, {})}
```

```html
<!-- templates/blog/_status_badge.html -->
<span class="badge {{ status.class }}">{{ status.text }}</span>

<!-- 使用 -->
{% status_badge article.status %}
```

## 注册方式总结

| 装饰器 | 用途 | 返回值 |
|--------|------|--------|
| `@register.filter` | 过滤器 | 处理后的值 |
| `@register.simple_tag` | 简单标签 | 字符串 |
| `@register.inclusion_tag` | 包含标签 | 字典（渲染模板） |
| `@register.tag` | 块标签 | Node 对象 |

## 小结

自定义标签的关键点：

1. **过滤器**：用于处理变量值
2. **简单标签**：返回字符串，可带参数
3. **包含标签**：渲染模板片段
4. **块标签**：处理复杂逻辑
5. **takes_context**：访问模板上下文

> **提示**：自定义标签应该保持简单，避免在模板中进行复杂的业务逻辑。复杂的逻辑应该在视图中处理。
