# 5.1 表单类

Django 的表单系统提供了强大的表单处理功能，包括字段定义、验证、渲染等。本章介绍如何创建和使用表单类。

## 表单基础

### 创建表单类

```python
# blog/forms.py
from django import forms

class ContactForm(forms.Form):
 """联系表单"""
 
 name = forms.CharField(
 label='姓名',
 max_length=100,
 required=True,
 widget=forms.TextInput(attrs={'class': 'form-control'})
 )
 
 email = forms.EmailField(
 label='邮箱',
 required=True,
 widget=forms.EmailInput(attrs={'class': 'form-control'})
 )
 
 subject = forms.CharField(
 label='主题',
 max_length=200,
 required=True
 )
 
 message = forms.CharField(
 label='消息',
 widget=forms.Textarea(attrs={'class': 'form-control', 'rows': 5})
 )
```

### 使用表单

```python
# views.py
from django.shortcuts import render, redirect
from .forms import ContactForm

def contact(request):
 if request.method == 'POST':
 form = ContactForm(request.POST)
 if form.is_valid():
 # 获取清洗后的数据
 name = form.cleaned_data['name']
 email = form.cleaned_data['email']
 subject = form.cleaned_data['subject']
 message = form.cleaned_data['message']
 
 # 处理数据（如发送邮件）
 send_contact_email(name, email, subject, message)
 
 return redirect('contact_success')
 else:
 form = ContactForm()
 
 return render(request, 'blog/contact.html', {'form': form})
```

## 字段类型

### 常用字段

```python
from django import forms

class ExampleForm(forms.Form):

 time_field = forms.TimeField(label='时间')
 datetime_field = forms.DateTimeField(label='日期时间')

 ('option1', '选项一'),

 choice_field = forms.ChoiceField(label='单选', choices=CHOICES)
 multiple_choice = forms.MultipleChoiceField(label='多选', choices=CHOICES)
```

## 字段参数

### 通用参数

```python
class ArticleForm(forms.Form):

 label='标题', # 标签文本
```

## ModelForm

ModelForm 可以根据模型自动生成表单。

### 基本用法

```python
# blog/forms.py
from django import forms
from .models import Article

class ArticleForm(forms.ModelForm):
 """文章表单"""
 
 class Meta:
 model = Article
 fields = ['title', 'content', 'category', 'tags', 'status']
 # 或排除某些字段
 # exclude = ['author', 'created_at']
 
 # 自定义标签
 labels = {
 'title': '文章标题',
 'content': '文章内容',
 }
 
 # 帮助文本
 help_texts = {
 'title': '请输入有吸引力的标题',
 }
 
 # 错误消息
 error_messages = {
 'title': {
 'required': '标题不能为空',
 'max_length': '标题太长了',
 },
 }
 
 # 控件属性
 widgets = {
 'title': forms.TextInput(attrs={'class': 'form-control'}),
 'content': forms.Textarea(attrs={'class': 'form-control', 'rows': 10}),
 'category': forms.Select(attrs={'class': 'form-select'}),
 'tags': forms.CheckboxSelectMultiple(),
 }
```

### 自定义字段

```python
class ArticleForm(forms.ModelForm):
 """文章表单（自定义字段）"""
 
 # 添加额外字段
 confirm_publish = forms.BooleanField(
 label='确认发布',
 required=True,
 widget=forms.CheckboxInput(attrs={'class': 'form-check-input'})
 )
 
 # 覆盖模型字段
 content = forms.CharField(
 label='内容',
 widget=forms.Textarea(attrs={'class': 'form-control', 'rows': 10}),
 help_text='支持 Markdown 格式'
 )
 
 class Meta:
 model = Article
 fields = ['title', 'content', 'category', 'tags', 'status']
 
 def __init__(self, *args, **kwargs):
 super().__init__(*args, **kwargs)
 # 动态修改字段
 self.fields['category'].queryset = Category.objects.filter(is_active=True)
```

### 保存 ModelForm

```python
# views.py
def article_create(request):
 if request.method == 'POST':
 form = ArticleForm(request.POST, request.FILES)
 if form.is_valid():
 # 保存但不提交到数据库
 article = form.save(commit=False)
 article.author = request.user
 article.save()
 
 # 保存多对多关系
 form.save_m2m()
 
 return redirect('article_detail', pk=article.pk)
 else:
 form = ArticleForm()
 
 return render(request, 'blog/article_form.html', {'form': form})

def article_update(request, pk):
 article = get_object_or_404(Article, pk=pk)
 
 if request.method == 'POST':
 form = ArticleForm(request.POST, instance=article)
 if form.is_valid():
 form.save()
 return redirect('article_detail', pk=article.pk)
 else:
 form = ArticleForm(instance=article)
 
 return render(request, 'blog/article_form.html', {'form': form})
```

## 表单集

表单集用于管理多个相同类型的表单。

### 创建表单集

```python
from django.forms import formset_factory

# 基于表单类
ArticleFormSet = formset_factory(ArticleForm, extra=3, max_num=10)

# 基于 ModelForm
from django.forms import modelformset_factory
ArticleFormSet = modelformset_factory(
 Article,
 fields=['title', 'content'],
 extra=3,
 can_delete=True
)
```

### 使用表单集

```python
# views.py
def manage_articles(request):
 ArticleFormSet = modelformset_factory(
 Article,
 fields=['title', 'content'],
 extra=3
 )
 
 if request.method == 'POST':
 formset = ArticleFormSet(request.POST, queryset=Article.objects.filter(author=request.user))
 if formset.is_valid():
 formset.save()
 return redirect('article_list')
 else:
 formset = ArticleFormSet(queryset=Article.objects.filter(author=request.user))
 
 return render(request, 'blog/manage_articles.html', {'formset': formset})
```

### 模板渲染


## 内联表单集

内联表单集用于处理关联对象的表单。

```python
from django.forms import inlineformset_factory

# 创建内联表单集
CommentFormSet = inlineformset_factory(
 Article, # 父模型
 Comment, # 子模型
 fields=['content', 'author'],
 extra=3,
 can_delete=True
)

# 使用
def edit_article_with_comments(request, pk):
 article = get_object_or_404(Article, pk=pk)
 
 if request.method == 'POST':
 form = ArticleForm(request.POST, instance=article)
 formset = CommentFormSet(request.POST, instance=article)
 
 if form.is_valid() and formset.is_valid():
 form.save()
 formset.save()
 return redirect('article_detail', pk=article.pk)
 else:
 form = ArticleForm(instance=article)
 formset = CommentFormSet(instance=article)
 
 return render(request, 'blog/edit_article.html', {
 'form': form,
 'formset': formset
 })
```

## 小结

表单类的关键点：

1. **Form 类**：手动定义表单字段
2. **ModelForm**：根据模型自动生成表单
3. **字段类型**：CharField、EmailField、ChoiceField 等
4. **Widget**：控制表单控件的渲染
5. **表单集**：管理多个相同类型的表单

> **提示**：优先使用 ModelForm，可以减少重复代码。对于复杂场景，可以自定义字段和验证逻辑。
