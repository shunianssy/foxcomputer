# 基本命令

## 镜像命令

### 搜索镜像

```bash
# 搜索镜像
docker search <镜像名>

# 搜索官方镜像
docker search --filter is-official=true <镜像名>

# 搜索自动构建的镜像
docker search --filter is-automated=true <镜像名>

# 限制搜索结果数量
docker search --limit 10 <镜像名>

# 示例：搜索 Nginx 镜像
docker search nginx
```

### 拉取镜像

```bash
# 拉取最新版本的镜像
docker pull <镜像名>

# 拉取指定版本的镜像
docker pull <镜像名>:<标签>

# 示例
docker pull nginx
docker pull nginx:1.24
docker pull nginx:alpine
docker pull python:3.11-slim
```

### 查看本地镜像

```bash
# 列出所有本地镜像
docker images

# 列出所有镜像（包括中间层）
docker images -a

# 只显示镜像 ID
docker images -q

# 格式化输出
docker images --format "table {{.Repository}}\t{{.Tag}}\t{{.Size}}"

# 示例：查看镜像详细信息
docker images nginx
```

### 删除镜像

```bash
# 删除指定镜像
docker rmi <镜像ID或名称>

# 删除多个镜像
docker rmi <镜像1> <镜像2> <镜像3>

# 强制删除镜像
docker rmi -f <镜像ID或名称>

# 删除所有未使用的镜像
docker image prune

# 删除所有未使用的镜像（包括没有标签的）
docker image prune -a

# 示例
docker rmi nginx:latest
docker rmi $(docker images -q)
```

### 构建镜像

```bash
# 使用 Dockerfile 构建镜像
docker build -t <镜像名>:<标签> <路径>

# 不使用缓存构建
docker build --no-cache -t <镜像名> .

# 构建时传递构建参数
docker build --build-arg ARG=value -t <镜像名> .

# 示例
docker build -t myapp:1.0 .
docker build -t myapp:latest -f Dockerfile.prod .
```

## 容器命令

### 运行容器

```bash
# 基本运行命令
docker run [选项] <镜像名> [命令]

# 常用选项
# -d: 后台运行
# -i: 保持标准输入打开
# -t: 分配伪终端
# --name: 指定容器名称
# -p: 端口映射
# -v: 挂载卷
# -e: 设置环境变量
# --rm: 容器退出后自动删除
# --restart: 重启策略

# 示例
docker run -d --name my-nginx -p 80:80 nginx
docker run -it --name my-ubuntu ubuntu bash
docker run --rm hello-world
docker run -d --restart=always nginx
```

### 查看容器

```bash
# 查看运行中的容器
docker ps

# 查看所有容器
docker ps -a

# 查看最近创建的容器
docker ps -l

# 只显示容器 ID
docker ps -q

# 显示容器大小
docker ps -s

# 格式化输出
docker ps --format "table {{.Names}}\t{{.Status}}\t{{.Ports}}"

# 示例：查看特定容器
docker ps -f name=my-nginx
docker ps -f status=exited
```

### 启动/停止/重启容器

```bash
# 启动容器
docker start <容器ID或名称>

# 停止容器
docker stop <容器ID或名称>

# 重启容器
docker restart <容器ID或名称>

# 强制停止容器
docker kill <容器ID或名称>

# 示例
docker start my-nginx
docker stop my-nginx
docker restart my-nginx
```

### 删除容器

```bash
# 删除已停止的容器
docker rm <容器ID或名称>

# 删除运行中的容器
docker rm -f <容器ID或名称>

# 删除多个容器
docker rm <容器1> <容器2> <容器3>

# 删除所有已停止的容器
docker container prune

# 删除所有容器
docker rm -f $(docker ps -aq)

# 示例
docker rm my-nginx
docker rm -f $(docker ps -aq)
```

### 进入容器

```bash
# 使用 exec 进入容器
docker exec -it <容器ID或名称> <命令>

# 示例
docker exec -it my-nginx bash
docker exec -it my-nginx sh
docker exec my-nginx ls -la
```

### 查看容器日志

```bash
# 查看容器日志
docker logs <容器ID或名称>

# 实时查看日志
docker logs -f <容器ID或名称>

# 查看最后 N 行日志
docker logs --tail 100 <容器ID或名称>

# 查看最近时间的日志
docker logs --since 10m <容器ID或名称>

# 显示时间戳
docker logs -t <容器ID或名称>

# 示例
docker logs -f --tail 100 my-nginx
docker logs --since 1h my-nginx
```

### 查看容器详细信息

```bash
# 查看容器详细信息
docker inspect <容器ID或名称>

# 查看容器 IP 地址
docker inspect --format='{{.NetworkSettings.IPAddress}}' <容器ID或名称>

# 查看容器端口映射
docker inspect --format='{{range $p, $conf := .NetworkSettings.Ports}}{{$p}} -> {{(index $conf 0).HostPort}}{{"\n"}}{{end}}' <容器ID或名称>

# 查看容器挂载的卷
docker inspect --format='{{json .Mounts}}' <容器ID或名称>

# 示例
docker inspect my-nginx
```

### 容器资源管理

```bash
# 查看容器资源使用情况
docker stats

# 查看特定容器的资源使用
docker stats <容器ID或名称>

# 不显示流式输出
docker stats --no-stream

# 示例
docker stats my-nginx
docker stats --no-stream
```

### 容器文件操作

```bash
# 从容器复制文件到宿主机
docker cp <容器ID或名称>:<容器内路径> <宿主机路径>

# 从宿主机复制文件到容器
docker cp <宿主机路径> <容器ID或名称>:<容器内路径>

# 示例
docker cp my-nginx:/etc/nginx/nginx.conf ./nginx.conf
docker cp ./index.html my-nginx:/usr/share/nginx/html/
```

### 容器导出与导入

```bash
# 导出容器为 tar 文件
docker export <容器ID或名称> > <文件名>.tar

# 从 tar 文件导入为镜像
docker import <文件名>.tar <镜像名>:<标签>

# 示例
docker export my-nginx > my-nginx.tar
docker import my-nginx.tar my-nginx:imported
```

## 网络命令

### 查看网络

```bash
# 列出所有网络
docker network ls

# 查看网络详细信息
docker network inspect <网络名称>

# 示例
docker network ls
docker network inspect bridge
```

### 创建网络

```bash
# 创建网络
docker network create <网络名称>

# 创建指定驱动类型的网络
docker network create -d bridge <网络名称>

# 创建带子网的网络
docker network create --subnet=172.20.0.0/16 <网络名称>

# 示例
docker network create my-network
docker network create --driver bridge my-bridge
```

### 连接容器到网络

```bash
# 将容器连接到网络
docker network connect <网络名称> <容器ID或名称>

# 断开容器与网络的连接
docker network disconnect <网络名称> <容器ID或名称>

# 示例
docker network connect my-network my-nginx
docker network disconnect my-network my-nginx
```

### 删除网络

```bash
# 删除网络
docker network rm <网络名称>

# 删除所有未使用的网络
docker network prune

# 示例
docker network rm my-network
docker network prune
```

## 系统命令

### 查看 Docker 信息

```bash
# 查看 Docker 系统信息
docker info

# 查看 Docker 版本
docker version

# 查看 Docker 详细版本
docker version --format '{{.Server.Version}}'

# 示例
docker info
docker version
```

### 清理系统

```bash
# 清理所有未使用的资源
docker system prune

# 清理所有未使用的资源（包括未使用的镜像）
docker system prune -a

# 清理所有未使用的资源（包括卷）
docker system prune -a --volumes

# 示例
docker system prune
docker system prune -a --volumes
```

### 查看磁盘使用情况

```bash
# 查看 Docker 磁盘使用情况
docker system df

# 查看详细磁盘使用情况
docker system df -v

# 示例
docker system df
```

## 实用技巧

### 批量操作

```bash
# 停止所有运行中的容器
docker stop $(docker ps -q)

# 删除所有已停止的容器
docker rm $(docker ps -aq)

# 删除所有未使用的镜像
docker rmi $(docker images -f "dangling=true" -q)

# 删除所有未使用的资源
docker system prune -a --volumes
```

### 命令别名

在 `~/.bashrc` 或 `~/.zshrc` 中添加别名：

```bash
# Docker 别名
alias dps='docker ps'
alias dpsa='docker ps -a'
alias di='docker images'
alias dex='docker exec -it'
alias dlog='docker logs -f'
alias dstop='docker stop'
alias drm='docker rm'
alias drmi='docker rmi'
```

### 命令补全

```bash
# Bash
echo 'source /usr/share/bash-completion/completions/docker' >> ~/.bashrc

# Zsh
echo 'source /usr/share/zsh/vendor-completions/_docker' >> ~/.zshrc
```
