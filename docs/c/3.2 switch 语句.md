# 3.2 switch 语句

switch 语句是 C 语言中用于多分支选择的结构，比多个 if-else if 语句更清晰，特别适合处理离散值的判断。

## 基本 switch 语句

### 语法结构

```c
#include <stdio.h>

int main(void) {
 int day = 3;
 
 /* switch 语句基本结构 */
 switch (day) {
 case 1:
 printf("星期一\n");
 break;
 case 2:
 printf("星期二\n");
 break;
 case 3:
 printf("星期三\n");
 break;
 case 4:
 printf("星期四\n");
 break;
 case 5:
 printf("星期五\n");
 break;
 case 6:
 printf("星期六\n");
 break;
 case 7:
 printf("星期日\n");
 break;
 default:
 printf("无效的日期\n");
 break;
 }
 
 return 0;
}
```

### switch 执行流程

```c
#include <stdio.h>

int main(void) {
 int num = 2;
 
 printf("演示 switch 执行流程:\n");
 
 switch (num) {
 case 1:
 printf("case 1\n");
 break;
 case 2:
 printf("case 2\n");
 /* 没有 break，会继续执行下一个 case */
 case 3:
 printf("case 3\n");
 break;
 default:
 printf("default\n");
 break;
 }
 
 return 0;
}
```

## break 语句

### break 的作用

```c
#include <stdio.h>

int main(void) {
 int choice = 2;
 
 printf("有 break 的情况:\n");
 switch (choice) {
 case 1:
 printf("选项 1\n");
 break;
 case 2:
 printf("选项 2\n");
 break;
 case 3:
 printf("选项 3\n");
 break;
 }
 
 printf("\n没有 break 的情况:\n");
 switch (choice) {
 case 1:
 printf("选项 1\n");
 case 2:
 printf("选项 2\n");
 case 3:
 printf("选项 3\n");
 default:
 printf("默认选项\n");
 }
 
 return 0;
}
```

### 利用穿透特性

```c
#include <stdio.h>

int main(void) {
 char grade = 'B';
 
 /* 利用 case 穿透特性 */
 switch (grade) {
 case 'A':
 case 'a':
 printf("优秀\n");
 break;
 case 'B':
 case 'b':
 printf("良好\n");
 break;
 case 'C':
 case 'c':
 printf("及格\n");
 break;
 case 'D':
 case 'd':
 case 'F':
 case 'f':
 printf("不及格\n");
 break;
 default:
 printf("无效等级\n");
 break;
 }
 
 return 0;
}
```

## default 分支

```c
#include <stdio.h>

int main(void) {
 int month = 13;
 int days;
 
 /* default 处理未匹配的情况 */
 switch (month) {
 case 1: case 3: case 5: case 7:
 case 8: case 10: case 12:
 days = 31;
 break;
 case 4: case 6: case 9: case 11:
 days = 30;
 break;
 case 2:
 days = 28; /* 简化处理，不考虑闰年 */
 break;
 default:
 printf("错误：无效的月份 %d\n", month);
 return 1;
 }
 
 printf("%d 月有 %d 天\n", month, days);
 
 return 0;
}
```

## switch 支持的类型

```c
#include <stdio.h>

int main(void) {
 /* switch 支持：int、char、枚举类型 */
 
 /* 整数类型 */
 int int_val = 1;
 switch (int_val) {
 case 0: printf("零\n"); break;
 case 1: printf("一\n"); break;
 default: printf("其他\n"); break;
 }
 
 /* 字符类型 */
 char char_val = 'y';
 switch (char_val) {
 case 'y':
 case 'Y':
 printf("是\n");
 break;
 case 'n':
 case 'N':
 printf("否\n");
 break;
 default:
 printf("无效输入\n");
 break;
 }
 
 /* 枚举类型 */
 enum Color { RED, GREEN, BLUE };
 enum Color color = GREEN;
 
 switch (color) {
 case RED: printf("红色\n"); break;
 case GREEN: printf("绿色\n"); break;
 case BLUE: printf("蓝色\n"); break;
 }
 
 return 0;
}
```

## 实际应用示例

### 简易计算器

```c
#include <stdio.h>

int main(void) {
 double num1, num2, result;
 char op;
 
 printf("简易计算器\n");
 printf("输入表达式 (如: 5 + 3): ");
 scanf("%lf %c %lf", &num1, &op, &num2);
 
 switch (op) {
 case '+':
 result = num1 + num2;
 break;
 case '-':
 result = num1 - num2;
 break;
 case '*':
 result = num1 * num2;
 break;
 case '/':
 if (num2 == 0) {
 printf("错误：除数不能为零\n");
 return 1;
 }
 result = num1 / num2;
 break;
 case '%':
 if (num2 == 0) {
 printf("错误：除数不能为零\n");
 return 1;
 }
 result = (int)num1 % (int)num2;
 break;
 default:
 printf("错误：不支持的运算符 '%c'\n", op);
 return 1;
 }
 
 printf("结果: %.2f %c %.2f = %.2f\n", num1, op, num2, result);
 
 return 0;
}
```

## switch vs if-else

```c
#include <stdio.h>

/* 使用 if-else 实现 */
void check_grade_if(int score) {
 if (score >= 90) {
 printf("优秀\n");
 } else if (score >= 80) {
 printf("良好\n");
 } else if (score >= 70) {
 printf("中等\n");
 } else if (score >= 60) {
 printf("及格\n");
 } else {
 printf("不及格\n");
 }
}

/* 使用 switch 实现（需要转换） */
void check_grade_switch(int score) {
 switch (score / 10) {
 case 10:
 case 9:
 printf("优秀\n");
 break;
 case 8:
 printf("良好\n");
 break;
 case 7:
 printf("中等\n");
 break;
 case 6:
 printf("及格\n");
 break;
 default:
 printf("不及格\n");
 break;
 }
}

int main(void) {
 int scores[] = {95, 85, 75, 65, 55};
 
 printf("if-else 实现:\n");
 for (int i = 0; i < 5; i++) {
 printf("%d 分: ", scores[i]);
 check_grade_if(scores[i]);
 }
 
 printf("\nswitch 实现:\n");
 for (int i = 0; i < 5; i++) {
 printf("%d 分: ", scores[i]);
 check_grade_switch(scores[i]);
 }
 
 return 0;
}
```

### 选择建议

| 场景 | 推荐使用 |
|------|----------|
| 离散值判断（整数、字符、枚举） | switch |
| 范围判断 | if-else |
| 浮点数判断 | if-else |
| 多个固定选项 | switch |
| 复杂逻辑条件 | if-else |

## 常见错误

### 1. 忘记 break

```c
#include <stdio.h>

int main(void) {
 int num = 2;
 
 /* 错误：忘记 break 导致穿透 */
 printf("忘记 break:\n");
 switch (num) {
 case 1:
 printf("一\n");
 case 2:
 printf("二\n");
 case 3:
 printf("三\n");
 default:
 printf("其他\n");
 }
 
 /* 正确：每个 case 后加 break */
 printf("\n正确使用 break:\n");
 switch (num) {
 case 1:
 printf("一\n");
 break;
 case 2:
 printf("二\n");
 break;
 case 3:
 printf("三\n");
 break;
 default:
 printf("其他\n");
 break;
 }
 
 return 0;
}
```

### 2. 使用浮点数

```c
#include <stdio.h>

int main(void) {
 float f = 1.5f;
 
 /* 错误：switch 不支持浮点数 */
 /* switch (f) { // 编译错误
 case 1.0:
 printf("一\n");
 break;
 } */
 
 /* 正确：使用 if-else */
 if (f == 1.0f) {
 printf("一\n");
 } else if (f == 1.5f) {
 printf("一点五\n");
 }
 
 return 0;
}
```

### 3. 变量声明问题

```c
#include <stdio.h>

int main(void) {
 int x = 1;
 
 switch (x) {
 case 1:
 /* 错误：不能在 case 中直接声明变量 */
 /* int y = 10; // 可能导致问题 */
 printf("case 1\n");
 break;
 case 2:
 {
 /* 正确：使用大括号创建作用域 */
 int y = 20;
 printf("case 2: y = %d\n", y);
 }
 break;
 }
 
 return 0;
}
```

## 小结

switch 语句要点：

1. **基本语法**：switch、case、break、default
2. **穿透特性**：没有 break 会继续执行下一个 case
3. **支持类型**：int、char、枚举类型
4. **default**：处理未匹配的情况
5. **适用场景**：离散值的多分支选择

> **提示**：养成每个 case 后都加 break 的习惯，即使看起来不需要。如果确实需要穿透，请添加注释说明。default 分支应该始终存在，用于处理意外情况。
