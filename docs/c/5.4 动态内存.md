# 5.4 动态内存

C 语言允许程序在运行时动态分配内存。动态内存管理是 C 语言的重要特性，也是最容易出错的地方。本章将介绍动态内存的分配、使用和释放。

## 内存区域

### 程序的内存布局

```c
#include <stdio.h>
#include <stdlib.h>

int global_var = 100;        /* 全局/静态区 */
static int static_var = 200; /* 全局/静态区 */

void memory_layout(void) {
    int local_var = 10;              /* 栈区 */
    static int func_static = 300;    /* 全局/静态区 */
    int *heap_var = malloc(sizeof(int));  /* 堆区 */
    
    printf("=== 内存区域地址 ===\n");
    printf("全局变量: %p\n", (void *)&global_var);
    printf("静态变量: %p\n", (void *)&static_var);
    printf("函数静态变量: %p\n", (void *)&func_static);
    printf("局部变量（栈）: %p\n", (void *)&local_var);
    printf("堆变量: %p\n", (void *)heap_var);
    
    free(heap_var);
}

int main(void) {
    memory_layout();
    return 0;
}
```

### 内存区域说明

```c
/*
 * C 程序的内存区域：
 * 
 * 1. 代码区（Text Segment）
 *    - 存储程序代码
 *    - 只读
 * 
 * 2. 全局/静态区（Data Segment）
 *    - 已初始化的全局变量和静态变量
 *    - 程序开始时分配，结束时释放
 * 
 * 3. 未初始化区（BSS Segment）
 *    - 未初始化的全局变量和静态变量
 *    - 自动初始化为 0
 * 
 * 4. 栈区（Stack）
 *    - 局部变量、函数参数、返回地址
 *    - 自动分配和释放
 *    - 大小有限
 * 
 * 5. 堆区（Heap）
 *    - 动态分配的内存
 *    - 手动分配和释放
 *    - 空间较大
 */
```

## malloc 函数

### 基本用法

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    /* malloc: 分配指定字节数的内存 */
    int *ptr = (int *)malloc(sizeof(int));
    
    /* 检查分配是否成功 */
    if (ptr == NULL) {
        printf("内存分配失败\n");
        return 1;
    }
    
    /* 使用分配的内存 */
    *ptr = 42;
    printf("分配的值: %d\n", *ptr);
    
    /* 释放内存 */
    free(ptr);
    ptr = NULL;  /* 避免野指针 */
    
    return 0;
}
```

### 分配数组

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int n;
    
    printf("请输入数组大小: ");
    scanf("%d", &n);
    
    /* 动态分配数组 */
    int *arr = (int *)malloc(n * sizeof(int));
    
    if (arr == NULL) {
        printf("内存分配失败\n");
        return 1;
    }
    
    /* 初始化数组 */
    for (int i = 0; i < n; i++) {
        arr[i] = i + 1;
    }
    
    /* 打印数组 */
    printf("数组元素: ");
    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
    
    /* 计算数组元素和 */
    int sum = 0;
    for (int i = 0; i < n; i++) {
        sum += arr[i];
    }
    printf("元素和: %d\n", sum);
    
    /* 释放内存 */
    free(arr);
    arr = NULL;
    
    return 0;
}
```

## calloc 函数

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int n = 5;
    
    /* calloc: 分配并初始化为 0 */
    /* void *calloc(size_t num, size_t size) */
    int *arr = (int *)calloc(n, sizeof(int));
    
    if (arr == NULL) {
        printf("内存分配失败\n");
        return 1;
    }
    
    /* calloc 自动初始化为 0 */
    printf("calloc 初始化后的值:\n");
    for (int i = 0; i < n; i++) {
        printf("arr[%d] = %d\n", i, arr[i]);
    }
    
    /* 修改值 */
    for (int i = 0; i < n; i++) {
        arr[i] = (i + 1) * 10;
    }
    
    printf("\n修改后的值:\n");
    for (int i = 0; i < n; i++) {
        printf("arr[%d] = %d\n", i, arr[i]);
    }
    
    free(arr);
    
    return 0;
}
```

### malloc vs calloc

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int n = 5;
    
    /* malloc: 只分配内存，不初始化 */
    int *arr1 = (int *)malloc(n * sizeof(int));
    printf("malloc 分配后的值（未定义）:\n");
    for (int i = 0; i < n; i++) {
        printf("arr1[%d] = %d\n", i, arr1[i]);
    }
    
    /* calloc: 分配内存并初始化为 0 */
    int *arr2 = (int *)calloc(n, sizeof(int));
    printf("\ncalloc 分配后的值（初始化为 0）:\n");
    for (int i = 0; i < n; i++) {
        printf("arr2[%d] = %d\n", i, arr2[i]);
    }
    
    free(arr1);
    free(arr2);
    
    return 0;
}
```

## realloc 函数

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int n = 5;
    
    /* 初始分配 */
    int *arr = (int *)malloc(n * sizeof(int));
    if (arr == NULL) {
        printf("初始分配失败\n");
        return 1;
    }
    
    /* 初始化 */
    for (int i = 0; i < n; i++) {
        arr[i] = i + 1;
    }
    
    printf("初始数组 (%d 个元素):\n", n);
    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
    
    /* 扩展数组 */
    int new_n = 10;
    int *temp = (int *)realloc(arr, new_n * sizeof(int));
    
    if (temp == NULL) {
        printf("扩展失败\n");
        free(arr);
        return 1;
    }
    
    arr = temp;  /* 使用新地址 */
    
    /* 初始化新元素 */
    for (int i = n; i < new_n; i++) {
        arr[i] = i + 1;
    }
    
    printf("\n扩展后的数组 (%d 个元素):\n", new_n);
    for (int i = 0; i < new_n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
    
    /* 缩小数组 */
    new_n = 3;
    temp = (int *)realloc(arr, new_n * sizeof(int));
    
    if (temp != NULL) {
        arr = temp;
    }
    
    printf("\n缩小后的数组 (%d 个元素):\n", new_n);
    for (int i = 0; i < new_n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
    
    free(arr);
    
    return 0;
}
```

## free 函数

### 正确释放内存

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    /* 分配内存 */
    int *ptr = (int *)malloc(sizeof(int));
    
    if (ptr != NULL) {
        *ptr = 100;
        printf("值: %d\n", *ptr);
        
        /* 释放内存 */
        free(ptr);
        
        /* 释放后置 NULL */
        ptr = NULL;
    }
    
    /* 检查后再使用 */
    if (ptr != NULL) {
        printf("ptr 不为空\n");
    } else {
        printf("ptr 已置空\n");
    }
    
    return 0;
}
```

### 常见错误

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    /* 错误 1：忘记释放内存（内存泄漏） */
    int *leak = (int *)malloc(sizeof(int));
    *leak = 100;
    /* 忘记 free(leak); */
    
    /* 错误 2：重复释放 */
    int *ptr = (int *)malloc(sizeof(int));
    free(ptr);
    /* free(ptr);  // 危险！重复释放 */
    
    /* 错误 3：释放非动态内存 */
    int a = 10;
    /* free(&a);  // 危险！不是动态分配的 */
    
    /* 错误 4：释放部分内存 */
    int *arr = (int *)malloc(5 * sizeof(int));
    /* free(arr + 2);  // 危险！必须释放起始地址 */
    free(arr);  /* 正确 */
    
    /* 错误 5：使用已释放的内存 */
    int *dangling = (int *)malloc(sizeof(int));
    *dangling = 50;
    free(dangling);
    /* printf("%d\n", *dangling);  // 危险！野指针 */
    
    printf("常见错误演示完成\n");
    
    return 0;
}
```

## 动态二维数组

### 方法一：指针数组

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int rows = 3, cols = 4;
    
    /* 分配行指针数组 */
    int **matrix = (int **)malloc(rows * sizeof(int *));
    
    if (matrix == NULL) {
        printf("分配失败\n");
        return 1;
    }
    
    /* 分配每行 */
    for (int i = 0; i < rows; i++) {
        matrix[i] = (int *)malloc(cols * sizeof(int));
        if (matrix[i] == NULL) {
            /* 分配失败，释放已分配的内存 */
            for (int j = 0; j < i; j++) {
                free(matrix[j]);
            }
            free(matrix);
            return 1;
        }
    }
    
    /* 初始化 */
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            matrix[i][j] = i * cols + j + 1;
        }
    }
    
    /* 打印 */
    printf("动态二维数组:\n");
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            printf("%3d ", matrix[i][j]);
        }
        printf("\n");
    }
    
    /* 释放：先释放行，再释放行指针数组 */
    for (int i = 0; i < rows; i++) {
        free(matrix[i]);
    }
    free(matrix);
    
    return 0;
}
```

### 方法二：连续内存

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int rows = 3, cols = 4;
    
    /* 分配连续内存 */
    int *data = (int *)malloc(rows * cols * sizeof(int));
    
    /* 分配行指针数组 */
    int **matrix = (int **)malloc(rows * sizeof(int *));
    
    if (data == NULL || matrix == NULL) {
        free(data);
        free(matrix);
        return 1;
    }
    
    /* 设置行指针 */
    for (int i = 0; i < rows; i++) {
        matrix[i] = data + i * cols;
    }
    
    /* 初始化 */
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            matrix[i][j] = i * cols + j + 1;
        }
    }
    
    /* 打印 */
    printf("连续内存的二维数组:\n");
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            printf("%3d ", matrix[i][j]);
        }
        printf("\n");
    }
    
    /* 释放 */
    free(data);
    free(matrix);
    
    return 0;
}
```

## 内存管理最佳实践

```c
#include <stdio.h>
#include <stdlib.h>

/* 安全的内存分配宏 */
#define SAFE_MALLOC(ptr, size) \
    do { \
        ptr = malloc(size); \
        if (ptr == NULL) { \
            fprintf(stderr, "内存分配失败: %s:%d\n", __FILE__, __LINE__); \
            exit(EXIT_FAILURE); \
        } \
    } while (0)

/* 安全的内存释放宏 */
#define SAFE_FREE(ptr) \
    do { \
        if (ptr != NULL) { \
            free(ptr); \
            ptr = NULL; \
        } \
    } while (0)

/* 动态数组结构 */
typedef struct {
    int *data;
    size_t size;
    size_t capacity;
} DynamicArray;

/* 初始化动态数组 */
int array_init(DynamicArray *arr, size_t capacity) {
    arr->data = (int *)malloc(capacity * sizeof(int));
    if (arr->data == NULL) {
        return -1;
    }
    arr->size = 0;
    arr->capacity = capacity;
    return 0;
}

/* 添加元素 */
int array_push(DynamicArray *arr, int value) {
    if (arr->size >= arr->capacity) {
        /* 扩容 */
        size_t new_capacity = arr->capacity * 2;
        int *new_data = (int *)realloc(arr->data, new_capacity * sizeof(int));
        if (new_data == NULL) {
            return -1;
        }
        arr->data = new_data;
        arr->capacity = new_capacity;
    }
    arr->data[arr->size++] = value;
    return 0;
}

/* 释放动态数组 */
void array_free(DynamicArray *arr) {
    SAFE_FREE(arr->data);
    arr->size = 0;
    arr->capacity = 0;
}

int main(void) {
    DynamicArray arr;
    
    if (array_init(&arr, 4) != 0) {
        printf("初始化失败\n");
        return 1;
    }
    
    /* 添加元素 */
    for (int i = 0; i < 10; i++) {
        array_push(&arr, i + 1);
    }
    
    printf("动态数组内容:\n");
    for (size_t i = 0; i < arr.size; i++) {
        printf("%d ", arr.data[i]);
    }
    printf("\n大小: %zu, 容量: %zu\n", arr.size, arr.capacity);
    
    array_free(&arr);
    
    return 0;
}
```

## 小结

动态内存要点：

1. **malloc**：分配指定字节数，不初始化
2. **calloc**：分配并初始化为 0
3. **realloc**：调整已分配内存大小
4. **free**：释放动态分配的内存

| 函数 | 功能 | 初始化 |
|------|------|--------|
| malloc(size) | 分配 size 字节 | 不初始化 |
| calloc(n, size) | 分配 n * size 字节 | 初始化为 0 |
| realloc(ptr, size) | 调整内存大小 | 保留原数据 |
| free(ptr) | 释放内存 | - |

**内存管理规则**：
- 每次 malloc/calloc 必须对应一次 free
- 不要重复释放同一块内存
- 释放后立即将指针置为 NULL
- 检查分配是否成功

> 💡 **提示**：动态内存管理是 C 语言最容易出错的地方。始终检查分配是否成功，及时释放不再使用的内存，避免内存泄漏。使用工具如 Valgrind 可以帮助检测内存问题。
