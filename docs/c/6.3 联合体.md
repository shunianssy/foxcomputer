# 6.3 联合体

联合体（Union）是 C 语言中一种特殊的数据类型，它允许在同一内存空间中存储不同类型的数据。与结构体不同，联合体的所有成员共享同一块内存。

## 联合体的定义

### 基本语法

```c
#include <stdio.h>

/* 定义联合体 */
union Data {
    int i;
    float f;
    char c;
};

int main(void) {
    union Data data;
    
    printf("联合体大小: %zu 字节\n", sizeof(union Data));
    printf("int 大小: %zu\n", sizeof(int));
    printf("float 大小: %zu\n", sizeof(float));
    printf("char 大小: %zu\n", sizeof(char));
    
    /* 联合体大小等于最大成员的大小 */
    
    return 0;
}
```

### 成员共享内存

```c
#include <stdio.h>

union Data {
    int i;
    float f;
    char c;
    char bytes[4];
};

int main(void) {
    union Data data;
    
    /* 存储整数 */
    data.i = 0x41424344;
    
    printf("存储整数 0x41424344:\n");
    printf("i = %d (0x%x)\n", data.i, data.i);
    printf("f = %f\n", data.f);  /* 无意义的浮点数 */
    printf("c = %c (0x%x)\n", data.c, data.c);
    
    printf("\n字节表示: ");
    for (int i = 0; i < 4; i++) {
        printf("0x%02x ", (unsigned char)data.bytes[i]);
    }
    printf("\n");
    
    /* 存储浮点数会覆盖整数 */
    data.f = 3.14f;
    printf("\n存储浮点数 3.14 后:\n");
    printf("i = %d\n", data.i);
    printf("f = %f\n", data.f);
    
    return 0;
}
```

## 联合体与结构体的区别

```c
#include <stdio.h>

struct StructData {
    int i;
    float f;
    char c;
};

union UnionData {
    int i;
    float f;
    char c;
};

int main(void) {
    printf("=== 结构体 vs 联合体 ===\n\n");
    
    printf("结构体:\n");
    printf("  大小: %zu 字节\n", sizeof(struct StructData));
    printf("  每个成员有独立内存\n\n");
    
    printf("联合体:\n");
    printf("  大小: %zu 字节\n", sizeof(union UnionData));
    printf("  所有成员共享内存\n\n");
    
    struct StructData s = {10, 3.14f, 'A'};
    printf("结构体可以同时存储所有值:\n");
    printf("  s.i = %d, s.f = %.2f, s.c = %c\n", s.i, s.f, s.c);
    
    union UnionData u;
    u.i = 10;
    printf("\n联合体只能存储一个值:\n");
    printf("  u.i = %d\n", u.i);
    u.f = 3.14f;
    printf("  存储浮点数后 u.i = %d (被覆盖)\n", u.i);
    
    return 0;
}
```

## 联合体的应用

### 类型标记

```c
#include <stdio.h>
#include <string.h>

/* 使用联合体实现变体类型 */
typedef enum {
    TYPE_INT,
    TYPE_FLOAT,
    TYPE_STRING
} DataType;

typedef struct {
    DataType type;
    union {
        int int_val;
        float float_val;
        char string_val[50];
    } value;
} Variant;

void print_variant(const Variant *v) {
    switch (v->type) {
        case TYPE_INT:
            printf("整数: %d\n", v->value.int_val);
            break;
        case TYPE_FLOAT:
            printf("浮点数: %.2f\n", v->value.float_val);
            break;
        case TYPE_STRING:
            printf("字符串: %s\n", v->value.string_val);
            break;
    }
}

int main(void) {
    Variant v;
    
    v.type = TYPE_INT;
    v.value.int_val = 42;
    print_variant(&v);
    
    v.type = TYPE_FLOAT;
    v.value.float_val = 3.14159f;
    print_variant(&v);
    
    v.type = TYPE_STRING;
    strcpy(v.value.string_val, "Hello");
    print_variant(&v);
    
    return 0;
}
```

### 网络数据包

```c
#include <stdio.h>
#include <stdint.h>
#include <string.h>

/* IP 地址表示 */
typedef union {
    uint32_t addr;
    uint8_t bytes[4];
} IPAddress;

/* 网络数据包 */
typedef struct {
    uint8_t type;
    uint8_t length;
    union {
        uint8_t raw[8];
        uint16_t short_data[4];
        uint32_t int_data[2];
        float float_data[2];
    } data;
} Packet;

void print_ip(IPAddress ip) {
    printf("%d.%d.%d.%d\n", 
           ip.bytes[0], ip.bytes[1], ip.bytes[2], ip.bytes[3]);
}

int main(void) {
    /* IP 地址 */
    IPAddress ip;
    ip.bytes[0] = 192;
    ip.bytes[1] = 168;
    ip.bytes[2] = 1;
    ip.bytes[3] = 100;
    
    printf("IP 地址: ");
    print_ip(ip);
    
    printf("整数表示: %u\n", ip.addr);
    
    /* 数据包 */
    Packet pkt;
    pkt.type = 1;
    pkt.length = 4;
    pkt.data.int_data[0] = 0x12345678;
    
    printf("\n数据包字节: ");
    for (int i = 0; i < 8; i++) {
        printf("%02x ", pkt.data.raw[i]);
    }
    printf("\n");
    
    return 0;
}
```

### 内存布局查看

```c
#include <stdio.h>
#include <stdint.h>

/* 查看浮点数的内存表示 */
typedef union {
    float f;
    uint32_t bits;
    uint8_t bytes[4];
} FloatView;

void print_binary(uint32_t n) {
    for (int i = 31; i >= 0; i--) {
        printf("%d", (n >> i) & 1);
        if (i == 31 || i == 23) printf(" ");
    }
}

int main(void) {
    FloatView fv;
    
    fv.f = 3.14f;
    
    printf("浮点数 %.6f 的内存表示:\n", fv.f);
    printf("十六进制: 0x%08x\n", fv.bits);
    printf("二进制: ");
    print_binary(fv.bits);
    printf("\n");
    
    printf("字节序: ");
    for (int i = 0; i < 4; i++) {
        printf("%02x ", fv.bytes[i]);
    }
    printf("\n");
    
    /* 解释浮点数格式 */
    uint32_t sign = (fv.bits >> 31) & 1;
    uint32_t exponent = (fv.bits >> 23) & 0xFF;
    uint32_t mantissa = fv.bits & 0x7FFFFF;
    
    printf("\n浮点数格式解析:\n");
    printf("符号位: %d (%s)\n", sign, sign ? "负" : "正");
    printf("指数: %d (偏移后: %d)\n", exponent, exponent - 127);
    printf("尾数: 0x%06x\n", mantissa);
    
    return 0;
}
```

## 匿名联合体

```c
#include <stdio.h>
#include <string.h>

/* C11 匿名联合体 */
typedef struct {
    char name[50];
    int type;  /* 0: 学生, 1: 教师 */
    union {
        struct {
            float score;
            int grade;
        } student;
        struct {
            char subject[30];
            int years;
        } teacher;
    };
} Person;

void print_person(const Person *p) {
    printf("姓名: %s\n", p->name);
    
    if (p->type == 0) {
        printf("类型: 学生\n");
        printf("成绩: %.1f\n", p->student.score);
        printf("年级: %d\n", p->student.grade);
    } else {
        printf("类型: 教师\n");
        printf("科目: %s\n", p->teacher.subject);
        printf("教龄: %d年\n", p->teacher.years);
    }
}

int main(void) {
    Person p1, p2;
    
    /* 学生 */
    strcpy(p1.name, "张三");
    p1.type = 0;
    p1.student.score = 85.5f;
    p1.student.grade = 3;
    
    /* 教师 */
    strcpy(p2.name, "李老师");
    p2.type = 1;
    strcpy(p2.teacher.subject, "数学");
    p2.teacher.years = 10;
    
    print_person(&p1);
    printf("\n");
    print_person(&p2);
    
    return 0;
}
```

## 联合体指针

```c
#include <stdio.h>
#include <stdlib.h>

union Value {
    int i;
    float f;
    char c;
};

void set_int(union Value *v, int val) {
    v->i = val;
}

void set_float(union Value *v, float val) {
    v->f = val;
}

void set_char(union Value *v, char val) {
    v->c = val;
}

int main(void) {
    union Value *v = (union Value *)malloc(sizeof(union Value));
    
    if (v == NULL) {
        printf("分配失败\n");
        return 1;
    }
    
    set_int(v, 42);
    printf("整数: %d\n", v->i);
    
    set_float(v, 3.14f);
    printf("浮点数: %.2f\n", v->f);
    
    set_char(v, 'A');
    printf("字符: %c\n", v->c);
    
    free(v);
    
    return 0;
}
```

## 小结

联合体要点：

1. **定义**：使用 `union` 关键字定义
2. **内存共享**：所有成员共享同一块内存
3. **大小**：等于最大成员的大小
4. **应用场景**：变体类型、网络协议、内存布局查看

| 特性 | 结构体 | 联合体 |
|------|--------|--------|
| 内存分配 | 每个成员独立 | 所有成员共享 |
| 大小 | 所有成员之和 | 最大成员大小 |
| 同时存储 | 可以 | 不可以 |
| 用途 | 组织相关数据 | 节省内存、类型转换 |

> 💡 **提示**：联合体适合用于需要在不同时间存储不同类型数据的场景。使用时要注意当前存储的是哪种类型，通常配合枚举类型标记使用。
