# 6.2 结构体指针

结构体指针是指向结构体变量的指针。通过结构体指针可以高效地访问和修改结构体成员，是 C 语言中常用的技术。

## 结构体指针基础

### 声明与初始化

```c
#include <stdio.h>
#include <string.h>

typedef struct {
 int id;
 char name[50];
 float score;
} Student;

int main(void) {
 Student stu = {1, "张三", 85.5f};
 
 /* 声明结构体指针并初始化 */
 Student *ptr = &stu;
 
 printf("通过指针访问:\n");
 printf("学号: %d\n", ptr->id);
 printf("姓名: %s\n", ptr->name);
 printf("成绩: %.1f\n", ptr->score);
 
 return 0;
}
```

### 成员访问方式

```c
#include <stdio.h>

typedef struct {
 int x;
 int y;
} Point;

int main(void) {
 Point p = {10, 20};
 Point *ptr = &p;
 
 /* 方式一：箭头运算符（推荐） */
 printf("ptr->x = %d\n", ptr->x);
 printf("ptr->y = %d\n", ptr->y);
 
 /* 方式二：解引用 + 点运算符 */
 printf("(*ptr).x = %d\n", (*ptr).x);
 printf("(*ptr).y = %d\n", (*ptr).y);
 
 /* 方式三：数组下标语法（不推荐） */
 printf("ptr[0].x = %d\n", ptr[0].x);
 
 return 0;
}
```

## 结构体指针与函数

### 指针作为参数

```c
#include <stdio.h>
#include <string.h>

typedef struct {
 char name[50];
 int age;
 float score;
} Student;

/* 值传递：复制整个结构体 */
void print_value(Student s) {
 printf("值传递: %s, %d岁, %.1f分\n", s.name, s.age, s.score);
}

/* 指针传递：只传递地址 */
void print_ptr(const Student *s) {
 printf("指针传递: %s, %d岁, %.1f分\n", s->name, s->age, s->score);
}

/* 通过指针修改 */
void update_student(Student *s, const char *name, int age, float score) {
 strcpy(s->name, name);
 s->age = age;
 s->score = score;
}

/* 初始化函数 */
void init_student(Student *s, const char *name, int age, float score) {
 if (s == NULL) return;
 strncpy(s->name, name, sizeof(s->name) - 1);
 s->name[sizeof(s->name) - 1] = '\0';
 s->age = age;
 s->score = score;
}

int main(void) {
 Student stu;
 init_student(&stu, "张三", 20, 85.5f);
 
 print_value(stu);
 print_ptr(&stu);
 
 update_student(&stu, "李四", 21, 92.0f);
 printf("\n更新后:\n");
 print_ptr(&stu);
 
 return 0;
}
```

### 返回结构体指针

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
 int id;
 char name[50];
} Person;

/* 返回静态结构体指针 */
Person *get_static_person(void) {
 static Person p = {1, "静态人员"};
 return &p;
}

/* 返回动态分配的结构体指针 */
Person *create_person(int id, const char *name) {
 Person *p = (Person *)malloc(sizeof(Person));
 if (p != NULL) {
 p->id = id;
 strncpy(p->name, name, sizeof(p->name) - 1);
 p->name[sizeof(p->name) - 1] = '\0';
 }
 return p;
}

/* 工厂函数 */
Person *person_clone(const Person *src) {
 if (src == NULL) return NULL;
 
 Person *clone = (Person *)malloc(sizeof(Person));
 if (clone != NULL) {
 *clone = *src; /* 结构体整体复制 */
 }
 return clone;
}

int main(void) {
 /* 静态结构体指针 */
 Person *p1 = get_static_person();
 printf("静态: %d - %s\n", p1->id, p1->name);
 
 /* 动态分配 */
 Person *p2 = create_person(2, "动态人员");
 if (p2 != NULL) {
 printf("动态: %d - %s\n", p2->id, p2->name);
 }
 
 /* 克隆 */
 Person *p3 = person_clone(p2);
 if (p3 != NULL) {
 p3->id = 3;
 printf("克隆: %d - %s\n", p3->id, p3->name);
 }
 
 /* 释放动态内存 */
 free(p2);
 free(p3);
 
 return 0;
}
```

## 结构体指针数组

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
 int id;
 char name[50];
 float price;
} Product;

int main(void) {
 /* 结构体指针数组 */
 Product *products[5];
 
 /* 创建并初始化 */
 for (int i = 0; i < 5; i++) {
 products[i] = (Product *)malloc(sizeof(Product));
 if (products[i] != NULL) {
 products[i]->id = i + 1;
 sprintf(products[i]->name, "商品%d", i + 1);
 products[i]->price = (i + 1) * 10.0f;
 }
 }
 
 /* 打印 */
 printf("商品列表:\n");
 printf("ID\t名称\t价格\n");
 for (int i = 0; i < 5; i++) {
 if (products[i] != NULL) {
 printf("%d\t%s\t%.1f\n", 
 products[i]->id, 
 products[i]->name, 
 products[i]->price);
 }
 }
 
 /* 释放内存 */
 for (int i = 0; i < 5; i++) {
 free(products[i]);
 }
 
 return 0;
}
```

## 链表实现

### 单向链表

```c
#include <stdio.h>
#include <stdlib.h>

/* 链表节点 */
typedef struct Node {
 int data;
 struct Node *next;
} Node;

/* 创建节点 */
Node *create_node(int data) {
 Node *node = (Node *)malloc(sizeof(Node));
 if (node != NULL) {
 node->data = data;
 node->next = NULL;
 }
 return node;
}

/* 头插法 */
void push_front(Node **head, int data) {
 Node *new_node = create_node(data);
 if (new_node == NULL) return;
 
 new_node->next = *head;
 *head = new_node;
}

/* 尾插法 */
void push_back(Node **head, int data) {
 Node *new_node = create_node(data);
 if (new_node == NULL) return;
 
 if (*head == NULL) {
 *head = new_node;
 return;
 }
 
 Node *current = *head;
 while (current->next != NULL) {
 current = current->next;
 }
 current->next = new_node;
}

/* 删除节点 */
void delete_node(Node **head, int data) {
 if (*head == NULL) return;
 
 Node *current = *head;
 Node *prev = NULL;
 
 while (current != NULL && current->data != data) {
 prev = current;
 current = current->next;
 }
 
 if (current == NULL) return; /* 未找到 */
 
 if (prev == NULL) {
 *head = current->next;
 } else {
 prev->next = current->next;
 }
 
 free(current);
}

/* 打印链表 */
void print_list(const Node *head) {
 const Node *current = head;
 while (current != NULL) {
 printf("%d -> ", current->data);
 current = current->next;
 }
 printf("NULL\n");
}

/* 释放链表 */
void free_list(Node *head) {
 Node *current = head;
 while (current != NULL) {
 Node *next = current->next;
 free(current);
 current = next;
 }
}

int main(void) {
 Node *head = NULL;
 
 /* 添加节点 */
 push_back(&head, 10);
 push_back(&head, 20);
 push_front(&head, 5);
 push_back(&head, 30);
 
 printf("链表: ");
 print_list(head);
 
 /* 删除节点 */
 delete_node(&head, 20);
 printf("删除 20 后: ");
 print_list(head);
 
 /* 释放 */
 free_list(head);
 
 return 0;
}
```

### 双向链表

```c
#include <stdio.h>
#include <stdlib.h>

/* 双向链表节点 */
typedef struct DNode {
 int data;
 struct DNode *prev;
 struct DNode *next;
} DNode;

/* 创建节点 */
DNode *create_dnode(int data) {
 DNode *node = (DNode *)malloc(sizeof(DNode));
 if (node != NULL) {
 node->data = data;
 node->prev = NULL;
 node->next = NULL;
 }
 return node;
}

/* 头插法 */
void dpush_front(DNode **head, int data) {
 DNode *new_node = create_dnode(data);
 if (new_node == NULL) return;
 
 new_node->next = *head;
 if (*head != NULL) {
 (*head)->prev = new_node;
 }
 *head = new_node;
}

/* 正向遍历 */
void dprint_forward(const DNode *head) {
 const DNode *current = head;
 printf("正向: ");
 while (current != NULL) {
 printf("%d ", current->data);
 current = current->next;
 }
 printf("\n");
}

/* 反向遍历 */
void dprint_backward(const DNode *tail) {
 const DNode *current = tail;
 printf("反向: ");
 while (current != NULL) {
 printf("%d ", current->data);
 current = current->prev;
 }
 printf("\n");
}

/* 获取尾节点 */
DNode *get_tail(DNode *head) {
 if (head == NULL) return NULL;
 
 DNode *current = head;
 while (current->next != NULL) {
 current = current->next;
 }
 return current;
}

/* 释放链表 */
void dfree_list(DNode *head) {
 DNode *current = head;
 while (current != NULL) {
 DNode *next = current->next;
 free(current);
 current = next;
 }
}

int main(void) {
 DNode *head = NULL;
 
 dpush_front(&head, 30);
 dpush_front(&head, 20);
 dpush_front(&head, 10);
 
 dprint_forward(head);
 
 DNode *tail = get_tail(head);
 dprint_backward(tail);
 
 dfree_list(head);
 
 return 0;
}
```

## 结构体指针与 const

```c
#include <stdio.h>

typedef struct {
 int x;
 int y;
} Point;

/* 指向常量结构体的指针 */
void print_point(const Point *p) {
 printf("(%d, %d)\n", p->x, p->y);
 /* p->x = 100; // 错误：不能修改 */
}

/* 常量指针 */
void modify_point(Point *const p, int x, int y) {
 p->x = x;
 p->y = y;
 /* p = NULL; // 错误：不能修改指针 */
}

/* 指向常量结构体的常量指针 */
void read_only(const Point *const p) {
 printf("(%d, %d)\n", p->x, p->y);
 /* p->x = 100; // 错误 */
 /* p = NULL; // 错误 */
}

int main(void) {
 Point p = {10, 20};
 
 print_point(&p);
 
 modify_point(&p, 30, 40);
 printf("修改后: ");
 print_point(&p);
 
 read_only(&p);
 
 return 0;
}
```

## 小结

结构体指针要点：

1. **声明**：`结构体类型 *指针名 = &结构体变量`
2. **访问成员**：`指针->成员` 或 `(*指针).成员`
3. **函数参数**：指针传递更高效，可以修改原结构体
4. **动态分配**：使用 malloc 分配结构体内存
5. **链表**：结构体指针是实现链表的基础

| 运算符 | 用法 | 说明 |
|--------|------|------|
| `.` | `变量.成员` | 结构体变量访问成员 |
| `->` | `指针->成员` | 结构体指针访问成员 |
| `*` | `(*指针).成员` | 解引用后访问成员 |

> **提示**：使用结构体指针传递大型结构体时效率更高。注意检查指针是否为 NULL，动态分配的结构体要及时释放。
