# 3.3 循环语句

循环语句用于重复执行一段代码。C 语言提供了三种循环结构：for、while 和 do-while。本章将详细介绍这三种循环的使用方法。

## for 循环

### 基本 for 循环

```c
#include <stdio.h>

int main(void) {
    /* for (初始化; 条件; 更新) { 循环体 } */
    
    /* 基本计数循环 */
    for (int i = 0; i < 5; i++) {
        printf("i = %d\n", i);
    }
    
    printf("\n");
    
    /* 递减循环 */
    for (int i = 5; i > 0; i--) {
        printf("倒计时: %d\n", i);
    }
    
    return 0;
}
```

### for 循环变体

```c
#include <stdio.h>

int main(void) {
    /* 步长为 2 */
    printf("偶数: ");
    for (int i = 0; i <= 10; i += 2) {
        printf("%d ", i);
    }
    printf("\n");
    
    /* 多个变量 */
    for (int i = 0, j = 10; i < j; i++, j--) {
        printf("i = %d, j = %d\n", i, j);
    }
    
    /* 省略部分表达式 */
    int k = 0;
    for (; k < 5; ) {
        printf("k = %d\n", k);
        k++;
    }
    
    /* 无限循环 */
    /* for (;;) {
        printf("无限循环\n");
    } */
    
    return 0;
}
```

### for 循环执行流程

```c
#include <stdio.h>

int main(void) {
    printf("for 循环执行流程:\n");
    
    /* 1. 初始化：int i = 0（只执行一次）
     * 2. 条件判断：i < 3
     * 3. 循环体：printf
     * 4. 更新：i++
     * 5. 回到步骤 2
     */
    for (int i = 0; i < 3; i++) {
        printf("循环第 %d 次\n", i + 1);
    }
    
    return 0;
}
```

## while 循环

### 基本 while 循环

```c
#include <stdio.h>

int main(void) {
    /* while (条件) { 循环体 } */
    
    int i = 0;
    while (i < 5) {
        printf("i = %d\n", i);
        i++;
    }
    
    printf("\n");
    
    /* 计算 1 到 100 的和 */
    int sum = 0;
    int n = 1;
    while (n <= 100) {
        sum += n;
        n++;
    }
    printf("1 到 100 的和: %d\n", sum);
    
    return 0;
}
```

### while 循环特点

```c
#include <stdio.h>

int main(void) {
    /* 先判断后执行：条件不满足时一次都不执行 */
    int x = 10;
    while (x < 5) {
        printf("这不会执行\n");
        x++;
    }
    
    /* 读取输入直到特定值 */
    int input;
    printf("输入数字（输入 0 结束）:\n");
    
    while (1) {
        scanf("%d", &input);
        if (input == 0) {
            break;
        }
        printf("你输入了: %d\n", input);
    }
    
    printf("程序结束\n");
    
    return 0;
}
```

## do-while 循环

### 基本 do-while 循环

```c
#include <stdio.h>

int main(void) {
    /* do { 循环体 } while (条件); */
    
    int i = 0;
    do {
        printf("i = %d\n", i);
        i++;
    } while (i < 5);
    
    printf("\n");
    
    /* 先执行后判断：至少执行一次 */
    int x = 10;
    do {
        printf("这会执行一次\n");
        x++;
    } while (x < 5);
    
    return 0;
}
```

### do-while 应用场景

```c
#include <stdio.h>

int main(void) {
    int choice;
    
    /* 菜单选择：至少显示一次 */
    do {
        printf("\n====== 菜单 ======\n");
        printf("1. 选项一\n");
        printf("2. 选项二\n");
        printf("3. 退出\n");
        printf("请选择: ");
        scanf("%d", &choice);
        
        switch (choice) {
            case 1:
                printf("执行选项一\n");
                break;
            case 2:
                printf("执行选项二\n");
                break;
            case 3:
                printf("退出\n");
                break;
            default:
                printf("无效选择\n");
        }
    } while (choice != 3);
    
    return 0;
}
```

## 嵌套循环

```c
#include <stdio.h>

int main(void) {
    /* 嵌套 for 循环：打印乘法表 */
    printf("九九乘法表:\n");
    for (int i = 1; i <= 9; i++) {
        for (int j = 1; j <= i; j++) {
            printf("%d×%d=%d\t", j, i, i * j);
        }
        printf("\n");
    }
    
    printf("\n");
    
    /* 打印图案 */
    printf("直角三角形:\n");
    for (int i = 1; i <= 5; i++) {
        for (int j = 1; j <= i; j++) {
            printf("* ");
        }
        printf("\n");
    }
    
    printf("\n");
    
    /* 倒三角 */
    printf("倒直角三角形:\n");
    for (int i = 5; i >= 1; i--) {
        for (int j = 1; j <= i; j++) {
            printf("* ");
        }
        printf("\n");
    }
    
    return 0;
}
```

## 循环控制语句

### break 语句

```c
#include <stdio.h>

int main(void) {
    /* break：跳出整个循环 */
    printf("break 示例:\n");
    for (int i = 0; i < 10; i++) {
        if (i == 5) {
            printf("遇到 5，跳出循环\n");
            break;
        }
        printf("i = %d\n", i);
    }
    
    printf("\n");
    
    /* 查找第一个负数 */
    int arr[] = {5, 3, 8, -2, 7, -5, 1};
    int size = sizeof(arr) / sizeof(arr[0]);
    int found = 0;
    
    for (int i = 0; i < size; i++) {
        if (arr[i] < 0) {
            printf("找到第一个负数: arr[%d] = %d\n", i, arr[i]);
            found = 1;
            break;
        }
    }
    
    if (!found) {
        printf("没有找到负数\n");
    }
    
    return 0;
}
```

### continue 语句

```c
#include <stdio.h>

int main(void) {
    /* continue：跳过本次循环，继续下一次 */
    printf("continue 示例（跳过偶数）:\n");
    for (int i = 0; i < 10; i++) {
        if (i % 2 == 0) {
            continue;  /* 跳过偶数 */
        }
        printf("奇数: %d\n", i);
    }
    
    printf("\n");
    
    /* 只处理有效数据 */
    int scores[] = {85, -1, 92, 78, -1, 95, 88};
    int size = sizeof(scores) / sizeof(scores[0]);
    int sum = 0;
    int count = 0;
    
    for (int i = 0; i < size; i++) {
        if (scores[i] < 0) {
            continue;  /* 跳过无效分数 */
        }
        sum += scores[i];
        count++;
    }
    
    printf("有效分数个数: %d\n", count);
    printf("平均分: %.2f\n", (float)sum / count);
    
    return 0;
}
```

### goto 语句

```c
#include <stdio.h>

int main(void) {
    /* goto：跳转到标签（谨慎使用） */
    
    /* 错误处理场景 */
    int a = 10, b = 0;
    int result;
    
    if (b == 0) {
        goto error;
    }
    
    result = a / b;
    printf("结果: %d\n", result);
    goto end;
    
error:
    printf("错误：除数不能为零\n");
    
end:
    printf("程序结束\n");
    
    return 0;
}
```

## 实际应用示例

### 猜数字游戏

```c
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main(void) {
    int secret, guess, attempts = 0;
    
    /* 生成随机数 */
    srand((unsigned int)time(NULL));
    secret = rand() % 100 + 1;  /* 1 到 100 */
    
    printf("猜数字游戏（1-100）\n");
    printf("我已经想好了一个数字，请猜猜看！\n");
    
    while (1) {
        printf("请输入你的猜测: ");
        scanf("%d", &guess);
        attempts++;
        
        if (guess < secret) {
            printf("太小了！再试试。\n");
        } else if (guess > secret) {
            printf("太大了！再试试。\n");
        } else {
            printf("恭喜你猜对了！\n");
            printf("你用了 %d 次猜中。\n", attempts);
            break;
        }
    }
    
    return 0;
}
```

### 斐波那契数列

```c
#include <stdio.h>

int main(void) {
    int n;
    long long a = 0, b = 1, next;

        b = next;

    return 0;
}
```

### 素数判断

```c
#include <stdio.h>
#include <stdbool.h>

bool is_prime(int n) {
    if (n <= 1) return false;
    if (n <= 3) return true;
    if (n % 2 == 0 || n % 3 == 0) return false;
    
    /* 检查 6k ± 1 的因子 */
    for (int i = 5; i * i <= n; i += 6) {
        if (n % i == 0 || n % (i + 2) == 0) {
            return false;
        }
    }
    return true;
}

int main(void) {
    int limit;
    
    printf("找出小于 n 的所有素数\n");
    printf("请输入 n: ");
    scanf("%d", &limit);
    
    printf("小于 %d 的素数:\n", limit);
    
    for (int i = 2; i < limit; i++) {
        if (is_prime(i)) {
            printf("%d ", i);
        }
    }
    printf("\n");
    
    return 0;
}
```

### 数字反转

```c
#include <stdio.h>

int reverse_number(int n) {
    int reversed = 0;
    
    while (n != 0) {
        int digit = n % 10;
        reversed = reversed * 10 + digit;
        n /= 10;
    }
    
    return reversed;
}

int main(void) {
    int num;
    
    printf("数字反转\n");
    printf("请输入一个整数: ");
    scanf("%d", &num);
    
    printf("原数字: %d\n", num);
    printf("反转后: %d\n", reverse_number(num));
    
    return 0;
}
```

## 循环选择建议

| 循环类型 | 适用场景 |
|----------|----------|
| for | 已知循环次数 |
| while | 不确定循环次数，可能一次都不执行 |
| do-while | 至少执行一次的情况 |

## 小结

循环语句要点：

1. **for 循环**：适合已知次数的循环
2. **while 循环**：先判断后执行，可能一次都不执行
3. **do-while 循环**：先执行后判断，至少执行一次
4. **break**：跳出整个循环
5. **continue**：跳过本次循环

> 💡 **提示**：选择合适的循环类型可以让代码更清晰。注意避免无限循环，确保循环条件最终会变为假。使用 break 和 continue 可以让循环逻辑更灵活。
