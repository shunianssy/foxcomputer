# 1.4 代码规范

良好的代码规范是编写高质量 C 程序的基础。本章将介绍 C 语言的编码规范和最佳实践，帮助你写出清晰、易读、易维护的代码。

## 命名规范

### 变量命名

```c
/* 推荐：使用小写字母，下划线分隔（snake_case） */
int student_count = 0;
float average_score = 0.0f;
char user_name[50];

/* 不推荐：使用拼音或无意义名称 */
int xueshengshuliang = 0; /* 拼音 */
int a = 0; /* 无意义 */
int AAA = 0; /* 全大写（应保留给常量） */

/* 布尔类型变量建议使用 is_、has_、can_ 等前缀 */
int is_valid = 1;
int has_permission = 0;
int can_read = 1;
```

### 常量命名

```c
/* 推荐：使用全大写字母，下划线分隔 */
#define MAX_BUFFER_SIZE 1024
#define PI 3.14159f
#define DEFAULT_PORT 8080

/* 使用 const 定义常量 */
const int MAX_STUDENTS = 100;
const float TAX_RATE = 0.08f;

/* 枚举常量 */
enum {
 STATUS_OK = 0,
 STATUS_ERROR = -1,
 STATUS_PENDING = 1
};
```

### 函数命名

```c
/* 推荐：使用小写字母，下划线分隔，动词开头 */
int calculate_sum(int a, int b);
void print_student_info(Student *student);
int get_user_count(void);
int is_valid_email(const char *email);

/* 不推荐 */
int Sum(int a, int b); /* 大写开头 */
int getusercount(void); /* 无分隔 */
int get_user(void); /* 名词开头 */
```

### 类型命名

```c
/* 结构体：使用大驼峰命名（PascalCase） */
typedef struct {
 char name[50];
 int age;
} StudentInfo;

/* 或使用 _t 后缀 */
typedef struct {
 int x;
 int y;
} Point_t;

/* 枚举类型 */
typedef enum {
 COLOR_RED,
 COLOR_GREEN,
 COLOR_BLUE
} Color;
```

## 代码格式

### 缩进与空格

```c
/* 推荐：使用 4 个空格缩进（不使用 Tab） */
int main(void) {
 int i;
 for (i = 0; i < 10; i++) {
 if (i % 2 == 0) {
 printf("%d is even\n", i);
 }
 }
 return 0;
}

/* 运算符两侧加空格 */
int result = a + b * c - d;

/* 逗号后加空格 */
int arr[] = {1, 2, 3, 4, 5};
func(a, b, c);

/* 关键字后加空格 */
if (x > 0) { }
for (i = 0; i < n; i++) { }
while (running) { }
```

### 大括号风格

```c
/* K&R 风格（推荐）：左大括号不换行 */
if (x > 0) {
 printf("Positive\n");
} else {
 printf("Non-positive\n");
}

/* 函数定义 */
int add(int a, int b) {
 return a + b;
}

/* Allman 风格：左大括号换行 */
if (x > 0)
{
 printf("Positive\n");
}
else
{
 printf("Non-positive\n");
}
```

### 行长度限制

```c
/* 每行不超过 80-100 字符 */

/* 长字符串可以分行 */
printf("This is a very long string that needs to be "
 "split across multiple lines for readability.\n");

/* 长参数列表可以分行 */
int result = calculate_complex_value(
 param1,
 param2,
 param3,
 param4
);

/* 长条件表达式可以分行 */
if (condition1 && condition2 && 
 condition3 && condition4) {
 /* do something */
}
```

### 函数注释

```c
/**
 * @brief 计算两个整数的和
 * * @param a 第一个整数
 * @param b 第二个整数
 * @return 两个整数的和
 * * @note 此函数不会检查整数溢出
 * * @example
 * int result = add(3, 5); // result = 8
 */
int add(int a, int b) {
 return a + b;
}

/**
 * @brief 查找数组中的最大值
 * * @param arr 整数数组
 * @param size 数组大小
 * @return 数组中的最大值
 * @retval -1 如果数组为空
 */
int find_max(const int *arr, int size) {
 if (arr == NULL || size <= 0) {
 return -1;
 }
 
 int max = arr[0];
 int i;
 for (i = 1; i < size; i++) {
 if (arr[i] > max) {
 max = arr[i];
 }
 }
 return max;
}
```

## 代码组织

### 头文件结构

```c
/* student.h */
#ifndef STUDENT_H
#define STUDENT_H

/* 标准库头文件 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* 自定义类型 */
typedef struct {
 int id;
 char name[50];
 int age;
 float score;
} Student;

/* 宏定义 */
#define MAX_STUDENTS 100
#define MAX_NAME_LEN 50

/* 函数声明 */
int student_init(Student *student, int id, const char *name, int age, float score);
void student_print(const Student *student);
int student_compare(const Student *a, const Student *b);

#endif /* STUDENT_H */
```

### 源文件结构

```c
/* student.c */

/* 本模块头文件（放在最前面） */
#include "student.h"

/* 标准库头文件 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* 其他模块头文件 */
#include "utils.h"

/* 私有宏定义 */
#define PRIVATE_CONSTANT 100

/* 私有类型定义 */
typedef struct {
 int internal_data;
} PrivateData;

/* 私有函数声明 */
static int helper_function(int x);

/* 公有函数实现 */
int student_init(Student *student, int id, const char *name, int age, float score) {
 if (student == NULL || name == NULL) {
 return -1;
 }
 
 student->id = id;
 strncpy(student->name, name, MAX_NAME_LEN - 1);
 student->name[MAX_NAME_LEN - 1] = '\0';
 student->age = age;
 student->score = score;
 
 return 0;
}

/* 私有函数实现 */
static int helper_function(int x) {
 return x * 2;
}
```

## 最佳实践

### 避免魔法数字

```c
/* 不推荐：使用魔法数字 */
int buffer[1024];
for (i = 0; i < 1024; i++) {
 buffer[i] = 0;
}

/* 推荐：使用命名常量 */
#define BUFFER_SIZE 1024

int buffer[BUFFER_SIZE];
for (i = 0; i < BUFFER_SIZE; i++) {
 buffer[i] = 0;
}
```

### 错误处理

```c
/* 推荐：检查返回值和错误 */
FILE *fp = fopen("data.txt", "r");
if (fp == NULL) {
 perror("无法打开文件");
 return -1;
}

/* 使用 goto 进行统一错误处理 */
int process_file(const char *filename) {
 FILE *fp = NULL;
 char *buffer = NULL;
 int result = -1;
 
 fp = fopen(filename, "r");
 if (fp == NULL) {
 perror("打开文件失败");
 goto cleanup;
 }
 
 buffer = (char *)malloc(1024);
 if (buffer == NULL) {
 perror("内存分配失败");
 goto cleanup;
 }
 
 /* 处理逻辑 */
 result = 0;
 
cleanup:
 if (buffer != NULL) {
 free(buffer);
 }
 if (fp != NULL) {
 fclose(fp);
 }
 return result;
}
```

### 资源管理

```c
/* 推荐：成对申请和释放 */
void example_function(void) {
 int *data = (int *)malloc(100 * sizeof(int));
 if (data == NULL) {
 return;
 }
 
 /* 使用数据 */
 /* ... */
 
 /* 及时释放 */
 free(data);
 data = NULL; /* 避免悬空指针 */
}

/* 文件操作 */
void file_example(void) {
 FILE *fp = fopen("data.txt", "w");
 if (fp == NULL) {
 return;
 }
 
 fprintf(fp, "Hello\n");
 
 fclose(fp); /* 及时关闭 */
}
```

### 防御性编程

```c
/* 推荐：检查所有输入参数 */
int safe_copy(char *dest, const char *src, size_t dest_size) {
 if (dest == NULL || src == NULL) {
 return -1;
 }
 
 if (dest_size == 0) {
 return -1;
 }
 
 size_t src_len = strlen(src);
 if (src_len >= dest_size) {
 /* 源字符串太长，截断复制 */
 strncpy(dest, src, dest_size - 1);
 dest[dest_size - 1] = '\0';
 return -2;
 }
 
 strcpy(dest, src);
 return 0;
}
```

## 代码检查工具

### GCC 警告选项

```bash
# 开启常用警告
gcc -Wall -Wextra -Werror program.c -o program

# 更严格的检查
gcc -Wall -Wextra -Werror -pedantic -std=c11 program.c -o program
```


## 小结

代码规范要点：

1. **命名规范**：变量用 snake_case，常量用 UPPER_CASE，类型用 PascalCase
2. **代码格式**：统一缩进、合理空格、控制行长度
3. **注释规范**：文件头、函数、关键逻辑都要有注释
4. **代码组织**：头文件保护、合理的包含顺序
5. **最佳实践**：避免魔法数字、做好错误处理、防御性编程

> **提示**：良好的代码规范不仅让代码更易读，也是团队协作的基础。建议在项目开始时就制定并遵守统一的编码规范。
