# 4.3 作用域与生命周期

变量的作用域决定了变量在程序中的可见范围，生命周期决定了变量存在的时间。理解这两个概念对于正确管理变量和内存至关重要。

## 作用域

### 块作用域

```c
#include <stdio.h>

int main(void) {
    /* 块作用域：从定义开始到块结束 */
    int a = 10;
    
    {
        int b = 20;
        printf("块内: a = %d, b = %d\n", a, b);
        
        /* 内层块可以定义同名变量，隐藏外层变量 */
        int a = 30;
        printf("内层块: a = %d (隐藏外层 a)\n", a);
    }
    
    /* b 在这里不可见 */
    /* printf("b = %d\n", b);  // 错误 */
    
    printf("块外: a = %d\n", a);
    
    return 0;
}
```

### 函数作用域

```c
#include <stdio.h>

/* 函数参数具有函数作用域 */
int add(int a, int b) {
    /* a 和 b 在整个函数内有效 */
    return a + b;
}

void example(void) {
    int x = 10;  /* 函数作用域 */
    
    if (x > 0) {
        int y = 20;  /* 块作用域 */
        printf("x = %d, y = %d\n", x, y);
    }
    
    /* y 在这里不可见 */
    /* printf("y = %d\n", y);  // 错误 */
    
    printf("x = %d\n", x);
}

int main(void) {
    example();
    /* x 在这里不可见 */
    /* printf("x = %d\n", x);  // 错误 */
    
    return 0;
}
```

### 文件作用域

```c
#include <stdio.h>

/* 全局变量：文件作用域 */
int global_counter = 0;

/* 静态全局变量：仅本文件可见 */
static int file_private = 100;

void increment_counter(void) {
    global_counter++;
}

void show_globals(void) {
    printf("global_counter = %d\n", global_counter);
    printf("file_private = %d\n", file_private);
}

int main(void) {
    show_globals();
    
    increment_counter();
    increment_counter();
    
    show_globals();
    
    return 0;
}
```

### 作用域规则

```c
#include <stdio.h>

int x = 100;  /* 全局变量 */

void demo_scope(void) {
    printf("1. 全局 x = %d\n", x);  /* 100 */
    
    int x = 200;  /* 局部变量隐藏全局变量 */
    printf("2. 局部 x = %d\n", x);  /* 200 */
    
    {
        int x = 300;  /* 块变量隐藏局部变量 */
        printf("3. 块内 x = %d\n", x);  /* 300 */
    }
    
    printf("4. 局部 x = %d\n", x);  /* 200 */
}

int main(void) {
    demo_scope();
    printf("5. 全局 x = %d\n", x);  /* 100 */
    
    return 0;
}
```

## 生命周期

### 自动变量

```c
#include <stdio.h>

void auto_example(void) {
    /* 自动变量：函数调用时创建，函数返回时销毁 */
    int a = 10;
    printf("a = %d\n", a);
    /* 函数返回后，a 被销毁 */
}

int main(void) {
    printf("第一次调用:\n");
    auto_example();
    
    printf("\n第二次调用:\n");
    auto_example();
    
    return 0;
}
```

### 静态变量

```c
#include <stdio.h>

void counter(void) {
    /* 静态局部变量：程序开始时初始化，程序结束时销毁 */
    static int count = 0;
    
    /* 只初始化一次 */
    count++;
    printf("调用次数: %d\n", count);
}

void show_static(void) {
    /* 静态变量默认初始化为 0 */
    static int value;
    printf("静态变量值: %d\n", value);
    value += 10;
}

int main(void) {
    counter();  /* 1 */
    counter();  /* 2 */
    counter();  /* 3 */
    
    printf("\n");
    show_static();  /* 0 */
    show_static();  /* 10 */
    show_static();  /* 20 */
    
    return 0;
}
```

### 全局变量

```c
#include <stdio.h>

/* 全局变量：程序开始时创建，程序结束时销毁 */
int global_value = 0;

void modify_global(void) {
    global_value += 10;
    printf("modify_global: global_value = %d\n", global_value);
}

void read_global(void) {
    printf("read_global: global_value = %d\n", global_value);
}

int main(void) {
    printf("初始值: global_value = %d\n", global_value);
    
    modify_global();
    read_global();
    
    modify_global();
    read_global();
    
    return 0;
}
```

## 存储类别

### auto

```c
#include <stdio.h>

void auto_demo(void) {
    /* auto 是默认的存储类别 */
    auto int a = 10;  /* 等价于 int a = 10; */
    
    {
        auto int b = 20;
        printf("a = %d, b = %d\n", a, b);
    }
    
    /* b 在这里不可见 */
}

int main(void) {
    auto_demo();
    return 0;
}
```

### static

```c
#include <stdio.h>

/* 静态全局变量：仅本文件可见 */
static int file_scope_var = 0;

void static_local(void) {
    /* 静态局部变量：保持值 */
    static int count = 0;
    count++;
    printf("count = %d\n", count);
}

void static_global(void) {
    file_scope_var++;
    printf("file_scope_var = %d\n", file_scope_var);
}

int main(void) {
    printf("静态局部变量:\n");
    static_local();
    static_local();
    static_local();
    
    printf("\n静态全局变量:\n");
    static_global();
    static_global();
    
    return 0;
}
```

### register

```c
#include <stdio.h>

void register_demo(void) {
    /* register：建议编译器将变量存储在寄存器中 */
    register int i;
    
    for (i = 0; i < 1000000; i++) {
        /* 频繁使用的变量 */
    }
    
    printf("循环结束: i = %d\n", i);
    
    /* 注意：register 变量不能取地址 */
    /* int *ptr = &i;  // 错误 */
}

int main(void) {
    register_demo();
    return 0;
}
```

### extern

```c
/* file1.c */
#include <stdio.h>

int shared_var = 100;  /* 定义 */

void print_shared(void) {
    printf("shared_var = %d\n", shared_var);
}
```

```c
/* file2.c */
#include <stdio.h>

/* extern：声明外部变量 */
extern int shared_var;

void modify_shared(void) {
    shared_var += 50;
    printf("modified shared_var = %d\n", shared_var);
}
```

```c
/* main.c - 演示 extern */
#include <stdio.h>

/* 声明外部变量 */
extern int shared_var;
extern void print_shared(void);
extern void modify_shared(void);

int main(void) {
    printf("初始值:\n");
    print_shared();
    
    modify_shared();
    
    printf("修改后:\n");
    print_shared();
    
    return 0;
}
```

## 变量初始化

```c
#include <stdio.h>

/* 全局变量和静态变量：自动初始化为 0 */
int global_int;
static int static_int;

void init_demo(void) {
    /* 局部自动变量：未初始化，值不确定 */
    int local_int;
    
    static int local_static;  /* 自动初始化为 0 */
    
    printf("global_int = %d\n", global_int);
    printf("static_int = %d\n", static_int);
    printf("local_int = %d (未初始化，值不确定)\n", local_int);
    printf("local_static = %d\n", local_static);
}

int main(void) {
    init_demo();
    return 0;
}
```

## 作用域与生命周期对比

```c
#include <stdio.h>

int global = 100;  /* 文件作用域，静态生存期 */

void demo(void) {
    int local = 10;           /* 块作用域，自动生存期 */
    static int static_local = 20;  /* 块作用域，静态生存期 */
    
    printf("local = %d, static_local = %d, global = %d\n", 
           local, static_local, global);
    
    local++;
    static_local++;
}

int main(void) {
    printf("=== 作用域与生命周期对比 ===\n\n");
    
    printf("第一次调用:\n");
    demo();
    
    printf("\n第二次调用:\n");
    demo();
    
    printf("\n第三次调用:\n");
    demo();
    
    return 0;
}
```

## 实际应用示例

### 单例模式

```c
#include <stdio.h>

typedef struct {
    int id;
    char name[50];
} Config;

/* 使用静态变量实现单例 */
Config *get_config(void) {
    static Config instance = {1, "Default"};
    return &instance;
}

int main(void) {
    Config *cfg1 = get_config();
    Config *cfg2 = get_config();
    
    printf("cfg1 地址: %p\n", (void *)cfg1);
    printf("cfg2 地址: %p\n", (void *)cfg2);
    printf("是同一个实例: %s\n", cfg1 == cfg2 ? "是" : "否");
    
    return 0;
}
```

### 计数器模块

```c
#include <stdio.h>

/* counter.h */
int counter_get(void);
void counter_increment(void);
void counter_reset(void);

/* counter.c */
static int count = 0;  /* 文件私有变量 */

int counter_get(void) {
    return count;
}

void counter_increment(void) {
    count++;
}

void counter_reset(void) {
    count = 0;
}

/* main.c */
int main(void) {
    printf("初始计数: %d\n", counter_get());
    
    counter_increment();
    counter_increment();
    counter_increment();
    printf("增加后: %d\n", counter_get());
    
    counter_reset();
    printf("重置后: %d\n", counter_get());
    
    return 0;
}
```

## 小结

作用域与生命周期要点：

1. **作用域**：变量可见的范围
   - 块作用域：从定义到块结束
   - 函数作用域：整个函数内
   - 文件作用域：整个文件内

2. **生命周期**：变量存在的时间
   - 自动：块结束时销毁
   - 静态：程序结束时销毁

3. **存储类别**：
   - auto：默认，自动变量
   - static：静态变量
   - register：寄存器变量
   - extern：外部变量

| 存储类别 | 作用域 | 生命周期 | 初始值 |
|----------|--------|----------|--------|
| auto | 块 | 自动 | 未定义 |
| static（局部） | 块 | 静态 | 0 |
| static（全局） | 文件 | 静态 | 0 |
| extern | 文件 | 静态 | 0 |

> 💡 **提示**：尽量减少全局变量的使用，使用局部变量和参数传递可以提高代码的可维护性。静态变量适合用于需要保持状态的场景，如计数器、单例模式等。
