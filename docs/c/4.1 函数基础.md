# 4.1 函数基础

函数是 C 语言程序的基本模块，用于将代码组织成可重用的单元。本章将介绍函数的定义、声明、调用等基础知识。

## 函数的定义

### 基本语法

```c
#include <stdio.h>

/* 函数定义语法：
 * 返回类型 函数名(参数列表) {
 *     函数体
 *     return 返回值;
 * }
 */

/* 无参数无返回值的函数 */
void say_hello(void) {
    printf("Hello, World!\n");
}

/* 有参数无返回值的函数 */
void print_sum(int a, int b) {
    printf("%d + %d = %d\n", a, b, a + b);
}

/* 有参数有返回值的函数 */
int add(int a, int b) {
    return a + b;
}

/* 无参数有返回值的函数 */
int get_pi(void) {
    return 3;
}

int main(void) {
    say_hello();
    print_sum(3, 5);
    printf("add(10, 20) = %d\n", add(10, 20));
    printf("PI ≈ %d\n", get_pi());
    
    return 0;
}
```

### 函数的组成部分

```c
#include <stdio.h>

/*
 * 函数组成部分：
 * 1. 返回类型：函数返回值的数据类型
 * 2. 函数名：函数的标识符
 * 3. 参数列表：函数接收的输入
 * 4. 函数体：函数执行的代码
 * 5. 返回值：函数的输出
 */

/* 示例：计算圆的面积 */
double circle_area(double radius) {
    /* 参数：radius - 半径 */
    /* 返回值：面积 */
    
    const double PI = 3.14159;
    double area = PI * radius * radius;
    return area;
}

int main(void) {
    double r = 5.0;
    double area = circle_area(r);
    printf("半径 %.2f 的圆面积: %.2f\n", r, area);
    
    return 0;
}
```

## 函数声明与定义

### 函数声明（原型）

```c
#include <stdio.h>

/* 函数声明（原型）：告诉编译器函数的存在 */
int add(int a, int b);           /* 完整声明 */
int subtract(int, int);          /* 可以省略参数名 */
double multiply(double, double); /* 不同类型 */

int main(void) {
    /* 在 main 之后定义的函数，需要先声明 */
    printf("add(3, 5) = %d\n", add(3, 5));
    printf("subtract(10, 4) = %d\n", subtract(10, 4));
    printf("multiply(2.5, 4.0) = %.2f\n", multiply(2.5, 4.0));
    
    return 0;
}

/* 函数定义 */
int add(int a, int b) {
    return a + b;
}

int subtract(int a, int b) {
    return a - b;
}

double multiply(double a, double b) {
    return a * b;
}
```

#include "math_utils.h"

    return a - b;

## 函数调用

### 基本调用

```c
#include <stdio.h>

int square(int n) {
    return n * n;
}

int main(void) {
    /* 直接调用 */
    int result = square(5);
    printf("5² = %d\n", result);
    
    /* 在表达式中调用 */
    printf("3² + 4² = %d\n", square(3) + square(4));
    
    /* 嵌套调用 */
    printf("(2²)² = %d\n", square(square(2)));
    
    /* 作为参数调用 */
    printf("square(square(3)) = %d\n", square(square(3)));
    
    return 0;
}
```

### 函数调用的执行过程

```c
#include <stdio.h>

void func_b(void) {
    printf("进入 func_b\n");
    printf("离开 func_b\n");
}

void func_a(void) {
    printf("进入 func_a\n");
    func_b();  /* 调用 func_b */
    printf("离开 func_a\n");
}

int main(void) {
    printf("程序开始\n");
    func_a();  /* 调用 func_a */
    printf("程序结束\n");
    
    return 0;
}

/* 输出：
 * 程序开始
 * 进入 func_a
 * 进入 func_b
 * 离开 func_b
 * 离开 func_a
 * 程序结束
 */
```

## 返回值

### return 语句

```c
#include <stdio.h>

/* 返回单个值 */
int max(int a, int b) {
    if (a > b) {
        return a;
    }
    return b;
}

/* 提前返回 */
int is_positive(int n) {
    if (n > 0) {
        return 1;
    }
    return 0;
}

/* void 函数可以没有 return */
void greet(const char *name) {
    if (name == NULL) {
        printf("Hello, Guest!\n");
        return;  /* 提前返回 */
    }
    printf("Hello, %s!\n", name);
    /* 隐式 return */
}

int main(void) {
    printf("max(10, 20) = %d\n", max(10, 20));
    printf("is_positive(5) = %d\n", is_positive(5));
    printf("is_positive(-3) = %d\n", is_positive(-3));
    
    greet("Alice");
    greet(NULL);
    
    return 0;
}
```

### 返回指针

```c
#include <stdio.h>

/* 返回指针：返回静态变量的地址 */
int *get_static_pointer(void) {
    static int value = 100;
    return &value;
}

/* 返回指针：返回参数地址 */
int *find_max(int *arr, int size) {
    if (arr == NULL || size <= 0) {
        return NULL;
    }
    
    int *max_ptr = &arr[0];
    for (int i = 1; i < size; i++) {
        if (arr[i] > *max_ptr) {
            max_ptr = &arr[i];
        }
    }
    return max_ptr;
}

int main(void) {
    int *ptr = get_static_pointer();
    printf("静态变量值: %d\n", *ptr);
    
    int arr[] = {3, 7, 2, 9, 1};
    int *max = find_max(arr, 5);
    if (max != NULL) {
        printf("最大值: %d\n", *max);
    }
    
    return 0;
}
```

## 参数

### 形参与实参

```c
#include <stdio.h>

/* 形参：函数定义时的参数 */
void swap_wrong(int a, int b) {
    /* 值传递：形参是实参的副本 */
    int temp = a;
    a = b;
    b = temp;
    printf("函数内: a = %d, b = %d\n", a, b);
}

void swap_right(int *a, int *b) {
    /* 指针传递：通过地址修改原值 */
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main(void) {
    int x = 10, y = 20;
    
    printf("原始值: x = %d, y = %d\n", x, y);
    
    swap_wrong(x, y);
    printf("swap_wrong 后: x = %d, y = %d\n", x, y);
    
    swap_right(&x, &y);
    printf("swap_right 后: x = %d, y = %d\n", x, y);
    
    return 0;
}
```

### 数组作为参数

```c
#include <stdio.h>

/* 数组参数退化为指针 */
void print_array(int arr[], int size) {
    /* sizeof(arr) 是指针大小，不是数组大小 */
    for (int i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

/* 等价写法 */
void print_array_ptr(int *arr, int size) {
    for (int i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

/* 计算数组元素和 */
int array_sum(int arr[], int size) {
    int sum = 0;
    for (int i = 0; i < size; i++) {
        sum += arr[i];
    }
    return sum;
}

int main(void) {
    int arr[] = {1, 2, 3, 4, 5};
    int size = sizeof(arr) / sizeof(arr[0]);
    
    print_array(arr, size);
    printf("数组元素和: %d\n", array_sum(arr, size));
    
    return 0;
}
```

## 实际应用示例

### 数学工具函数

```c
#include <stdio.h>
#include <stdbool.h>

/* 判断素数 */
bool is_prime(int n) {
    if (n <= 1) return false;
    if (n <= 3) return true;
    if (n % 2 == 0 || n % 3 == 0) return false;
    
    for (int i = 5; i * i <= n; i += 6) {
        if (n % i == 0 || n % (i + 2) == 0) {
            return false;
        }
    }
    return true;
}

/* 计算阶乘 */
long long factorial(int n) {
    if (n < 0) return -1;  /* 错误 */
    if (n <= 1) return 1;
    
    long long result = 1;
    for (int i = 2; i <= n; i++) {
        result *= i;
    }
    return result;
}

/* 计算幂 */
double power(double base, int exponent) {
    if (exponent == 0) return 1.0;
    
    double result = 1.0;
    int abs_exp = exponent < 0 ? -exponent : exponent;
    
    for (int i = 0; i < abs_exp; i++) {
        result *= base;
    }
    
    return exponent < 0 ? 1.0 / result : result;
}

int main(void) {
    printf("素数判断:\n");
    for (int i = 1; i <= 20; i++) {
        if (is_prime(i)) {
            printf("%d ", i);
        }
    }
    printf("\n");
    
    printf("\n阶乘:\n");
    for (int i = 0; i <= 10; i++) {
        printf("%d! = %lld\n", i, factorial(i));
    }
    
    printf("\n幂运算:\n");
    printf("2^10 = %.0f\n", power(2, 10));
    printf("2^-3 = %.6f\n", power(2, -3));
    
    return 0;
}
```

### 字符串处理函数

```c
#include <stdio.h>

/* 计算字符串长度 */
int my_strlen(const char *str) {
    int len = 0;
    while (str[len] != '\0') {
        len++;
    }
    return len;
}

/* 字符串复制 */
void my_strcpy(char *dest, const char *src) {
    int i = 0;
    while (src[i] != '\0') {
        dest[i] = src[i];
        i++;
    }
    dest[i] = '\0';
}

/* 字符串连接 */
void my_strcat(char *dest, const char *src) {
    int i = 0;
    while (dest[i] != '\0') {
        i++;
    }
    
    int j = 0;
    while (src[j] != '\0') {
        dest[i] = src[j];
        i++;
        j++;
    }
    dest[i] = '\0';
}

int main(void) {
    char str1[50] = "Hello";
    char str2[50] = "World";
    
    printf("str1 长度: %d\n", my_strlen(str1));
    
    char copy[50];
    my_strcpy(copy, str1);
    printf("复制: %s\n", copy);
    
    my_strcat(str1, " ");
    my_strcat(str1, str2);
    printf("连接: %s\n", str1);
    
    return 0;
}
```

## 小结

函数基础要点：

1. **函数定义**：返回类型、函数名、参数列表、函数体
2. **函数声明**：在使用前声明函数原型
3. **函数调用**：传递实参，接收返回值
4. **返回值**：return 语句返回结果
5. **参数传递**：值传递（默认）和指针传递

> 💡 **提示**：函数应该保持单一职责，每个函数只做一件事。函数名应该清晰表达其功能，参数和返回值要有明确的含义。
