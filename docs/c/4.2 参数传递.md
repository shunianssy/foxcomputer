# 4.2 参数传递

C 语言函数的参数传递方式主要有值传递和指针传递（地址传递）。理解这两种传递方式对于正确使用函数至关重要。

## 值传递

### 基本概念

```c
#include <stdio.h>

/* 值传递：函数内部修改不影响原值 */
void modify_value(int x) {
    printf("函数内修改前: x = %d\n", x);
    x = 100;
    printf("函数内修改后: x = %d\n", x);
}

int main(void) {
    int a = 10;
    
    printf("调用前: a = %d\n", a);
    modify_value(a);
    printf("调用后: a = %d\n", a);  /* a 仍然是 10 */
    
    return 0;
}
```

### 值传递的工作原理

```c
#include <stdio.h>

void show_value_pass(int x) {
    printf("形参地址: %p\n", (void *)&x);
    printf("形参值: %d\n", x);
}

int main(void) {
    int a = 42;
    
    printf("实参地址: %p\n", (void *)&a);
    printf("实参值: %d\n", a);
    
    show_value_pass(a);
    
    /* 形参和实参地址不同，是两个独立的变量 */
    
    return 0;
}
```

### 结构体的值传递

```c
#include <stdio.h>
#include <string.h>

typedef struct {
    char name[50];
    int age;
} Person;

/* 结构体值传递：复制整个结构体 */
void print_person(Person p) {
    printf("姓名: %s, 年龄: %d\n", p.name, p.age);
    /* 修改不影响原结构体 */
    p.age = 0;
}

int main(void) {
    Person alice = {"Alice", 25};
    
    printf("调用前: 年龄 = %d\n", alice.age);
    print_person(alice);
    printf("调用后: 年龄 = %d\n", alice.age);
    
    return 0;
}
```

## 指针传递

### 基本用法

```c
#include <stdio.h>

/* 指针传递：通过地址修改原值 */
void modify_by_pointer(int *x) {
    printf("函数内修改前: *x = %d\n", *x);
    *x = 100;
    printf("函数内修改后: *x = %d\n", *x);
}

int main(void) {
    int a = 10;
    
    printf("调用前: a = %d\n", a);
    modify_by_pointer(&a);  /* 传递地址 */
    printf("调用后: a = %d\n", a);  /* a 被修改为 100 */
    
    return 0;
}
```

### 交换两个变量

```c
#include <stdio.h>

/* 错误：值传递无法交换 */
void swap_wrong(int a, int b) {
    int temp = a;
    a = b;
    b = temp;
}

/* 正确：指针传递可以交换 */
void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main(void) {
    int x = 10, y = 20;
    
    printf("原始值: x = %d, y = %d\n", x, y);
    
    swap_wrong(x, y);
    printf("swap_wrong 后: x = %d, y = %d\n", x, y);
    
    swap(&x, &y);
    printf("swap 后: x = %d, y = %d\n", x, y);
    
    return 0;
}
```

### 通过指针返回多个值

```c
#include <stdio.h>

/* 通过指针参数返回多个值 */
void divide(int a, int b, int *quotient, int *remainder) {
    if (b == 0) {
        *quotient = 0;
        *remainder = 0;
        return;
    }
    *quotient = a / b;
    *remainder = a % b;
}

/* 计算数组的最小值和最大值 */
void find_min_max(int arr[], int size, int *min, int *max) {
    if (size <= 0) {
        return;
    }
    
    *min = arr[0];
    *max = arr[0];
    
    for (int i = 1; i < size; i++) {
        if (arr[i] < *min) {
            *min = arr[i];
        }
        if (arr[i] > *max) {
            *max = arr[i];
        }
    }
}

int main(void) {
    /* 除法示例 */
    int quot, rem;
    divide(17, 5, &quot, &rem);
    printf("17 / 5 = %d 余 %d\n", quot, rem);
    
    /* 最小最大值示例 */
    int arr[] = {3, 7, 2, 9, 1, 8, 4};
    int min_val, max_val;
    find_min_max(arr, 7, &min_val, &max_val);
    printf("最小值: %d, 最大值: %d\n", min_val, max_val);
    
    return 0;
}
```

## 数组参数

### 数组退化为指针

```c
#include <stdio.h>

/* 数组参数等价于指针参数 */
void print_size(int arr[]) {
    /* 数组参数在函数内退化为指针 */
    printf("sizeof(arr) = %zu (指针大小)\n", sizeof(arr));
}

void print_size_ptr(int *arr) {
    printf("sizeof(arr) = %zu (指针大小)\n", sizeof(arr));
}

int main(void) {
    int arr[10];
    
    printf("sizeof(arr) = %zu (数组大小)\n", sizeof(arr));
    print_size(arr);
    print_size_ptr(arr);
    
    return 0;
}
```

### 正确传递数组

```c
#include <stdio.h>

/* 必须传递数组大小 */
void print_array(int arr[], int size) {
    printf("数组元素: ");
    for (int i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

/* 修改数组元素 */
void double_array(int arr[], int size) {
    for (int i = 0; i < size; i++) {
        arr[i] *= 2;
    }
}

/* 使用指针语法 */
int sum_array(int *arr, int size) {
    int sum = 0;
    for (int i = 0; i < size; i++) {
        sum += arr[i];
    }
    return sum;
}

int main(void) {
    int arr[] = {1, 2, 3, 4, 5};
    int size = sizeof(arr) / sizeof(arr[0]);
    
    print_array(arr, size);
    
    double_array(arr, size);
    printf("翻倍后: ");
    print_array(arr, size);
    
    printf("数组元素和: %d\n", sum_array(arr, size));
    
    return 0;
}
```

## const 参数

### 保护参数不被修改

```c
#include <stdio.h>

/* const 保护数组不被修改 */
void print_array_safe(const int arr[], int size) {
    for (int i = 0; i < size; i++) {
        printf("%d ", arr[i]);
        /* arr[i] = 0;  // 编译错误 */
    }
    printf("\n");
}

/* const 保护指针指向的内容不被修改 */
int find_max(const int *arr, int size) {
    int max = arr[0];
    for (int i = 1; i < size; i++) {
        if (arr[i] > max) {
            max = arr[i];
        }
    }
    return max;
    /* *arr = 0;  // 编译错误 */
}

/* const 保护指针本身不被修改 */
void process(int *const ptr) {
    *ptr = 100;  /* 可以修改指向的内容 */
    /* ptr = NULL;  // 编译错误：不能修改指针 */
}

/* 双重 const */
void process_both(const int *const ptr) {
    /* 既不能修改指向的内容，也不能修改指针 */
    printf("值: %d\n", *ptr);
}

int main(void) {
    int arr[] = {1, 2, 3, 4, 5};
    int size = sizeof(arr) / sizeof(arr[0]);
    
    print_array_safe(arr, size);
    printf("最大值: %d\n", find_max(arr, size));
    
    int x = 10;
    process(&x);
    printf("x = %d\n", x);
    
    return 0;
}
```

## 结构体指针参数

```c
#include <stdio.h>
#include <string.h>

typedef struct {
    char name[50];
    int age;
    float score;
} Student;

/* 值传递：复制整个结构体 */
void print_student_value(Student s) {
    printf("姓名: %s, 年龄: %d, 分数: %.1f\n", s.name, s.age, s.score);
}

/* 指针传递：只传递地址 */
void print_student_ptr(const Student *s) {
    printf("姓名: %s, 年龄: %d, 分数: %.1f\n", s->name, s->age, s->score);
}

/* 通过指针修改结构体 */
void update_student(Student *s, int age, float score) {
    s->age = age;
    s->score = score;
}

/* 初始化结构体 */
void init_student(Student *s, const char *name, int age, float score) {
    strncpy(s->name, name, sizeof(s->name) - 1);
    s->name[sizeof(s->name) - 1] = '\0';
    s->age = age;
    s->score = score;
}

int main(void) {
    Student alice;
    
    init_student(&alice, "Alice", 20, 85.5f);
    
    printf("值传递:\n");
    print_student_value(alice);
    
    printf("\n指针传递:\n");
    print_student_ptr(&alice);
    
    printf("\n更新后:\n");
    update_student(&alice, 21, 90.0f);
    print_student_ptr(&alice);
    
    return 0;
}
```

## 函数指针作为参数

```c
#include <stdio.h>

/* 定义函数指针类型 */
typedef int (*Operation)(int, int);

/* 基本运算函数 */
int add(int a, int b) { return a + b; }
int subtract(int a, int b) { return a - b; }
int multiply(int a, int b) { return a * b; }

/* 使用函数指针作为参数 */
int calculate(int a, int b, Operation op) {
    return op(a, b);
}

/* 回调函数示例 */
void process_array(int arr[], int size, int (*transform)(int)) {
    for (int i = 0; i < size; i++) {
        arr[i] = transform(arr[i]);
    }
}

int double_value(int x) { return x * 2; }
int square_value(int x) { return x * x; }

int main(void) {
    /* 使用函数指针 */
    printf("10 + 5 = %d\n", calculate(10, 5, add));
    printf("10 - 5 = %d\n", calculate(10, 5, subtract));
    printf("10 * 5 = %d\n", calculate(10, 5, multiply));
    
    /* 回调函数处理数组 */
    int arr[] = {1, 2, 3, 4, 5};
    int size = sizeof(arr) / sizeof(arr[0]);
    
    printf("\n原数组: ");
    for (int i = 0; i < size; i++) printf("%d ", arr[i]);
    
    process_array(arr, size, double_value);
    printf("\n翻倍后: ");
    for (int i = 0; i < size; i++) printf("%d ", arr[i]);
    
    process_array(arr, size, square_value);
    printf("\n平方后: ");
    for (int i = 0; i < size; i++) printf("%d ", arr[i]);
    printf("\n");
    
    return 0;
}
```

## 参数传递最佳实践

```c
#include <stdio.h>
#include <string.h>

/* 1. 简单类型用值传递 */
int square(int n) {
    return n * n;
}

/* 2. 需要修改原值用指针传递 */
void increment(int *n) {
    (*n)++;
}

/* 3. 大结构体用 const 指针传递 */
typedef struct {
    char data[1000];
    int size;
} LargeStruct;

void process_large(const LargeStruct *ls) {
    printf("大小: %d\n", ls->size);
}

/* 4. 数组总是传递指针和大小 */
int sum_array(const int *arr, int size) {
    int sum = 0;
    for (int i = 0; i < size; i++) {
        sum += arr[i];
    }
    return sum;
}

/* 5. 输出参数放在最后 */
void divide_safe(int a, int b, int *quotient, int *remainder) {
    *quotient = b != 0 ? a / b : 0;
    *remainder = b != 0 ? a % b : 0;
}

int main(void) {
    int x = 5;
    printf("square(%d) = %d\n", x, square(x));
    
    increment(&x);
    printf("increment 后: %d\n", x);
    
    int arr[] = {1, 2, 3, 4, 5};
    printf("数组元素和: %d\n", sum_array(arr, 5));
    
    int q, r;
    divide_safe(17, 5, &q, &r);
    printf("17 / 5 = %d 余 %d\n", q, r);
    
    return 0;
}
```

## 小结

参数传递要点：

1. **值传递**：复制值，不影响原变量
2. **指针传递**：传递地址，可以修改原变量
3. **数组参数**：退化为指针，需要传递大小
4. **const 参数**：保护参数不被修改
5. **结构体参数**：大结构体优先使用指针传递

| 场景 | 推荐方式 |
|------|----------|
| 简单类型，不需要修改 | 值传递 |
| 需要修改原值 | 指针传递 |
| 大结构体 | const 指针传递 |
| 数组 | 指针 + 大小 |
| 字符串 | const char * |

> 💡 **提示**：选择合适的参数传递方式可以提高程序的效率和安全性。对于大结构体，使用指针传递可以避免复制开销；对于不应该被修改的参数，使用 const 保护。
