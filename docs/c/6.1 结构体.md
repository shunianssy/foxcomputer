# 6.1 结构体

结构体是 C 语言中用于组合不同类型数据的自定义数据类型。它允许将相关的数据组织在一起，形成更复杂的数据结构。

## 结构体的定义

### 基本语法

```c
#include <stdio.h>
#include <string.h>

/* 定义结构体类型 */
struct Student {
 int id;
 char name[50];
 int age;
 float score;
};

int main(void) {
 /* 声明结构体变量 */
 struct Student stu1;
 
 /* 初始化成员 */
 stu1.id = 1001;
 strcpy(stu1.name, "张三");
 stu1.age = 20;
 stu1.score = 85.5f;
 
 /* 访问成员 */
 printf("学号: %d\n", stu1.id);
 printf("姓名: %s\n", stu1.name);
 printf("年龄: %d\n", stu1.age);
 printf("成绩: %.1f\n", stu1.score);
 
 return 0;
}
```

### 结构体的初始化

```c
#include <stdio.h>
#include <string.h>

struct Point {
 int x;
 int y;
};

struct Student {
 int id;
 char name[50];
 int age;
 float score;
};

int main(void) {
 /* 方式一：顺序初始化 */
 struct Point p1 = {10, 20};
 
 /* 方式二：指定成员初始化（C99） */
 struct Point p2 = {.x = 30, .y = 40};
 
 /* 方式三：部分初始化 */
 struct Point p3 = {.x = 50}; /* y 自动为 0 */
 
 /* 结构体初始化 */
 struct Student stu1 = {1001, "李四", 21, 90.0f};
 struct Student stu2 = {.id = 1002, .name = "王五", .score = 88.5f};
 
 printf("p1: (%d, %d)\n", p1.x, p1.y);
 printf("p2: (%d, %d)\n", p2.x, p2.y);
 printf("p3: (%d, %d)\n", p3.x, p3.y);
 
 printf("\nstu1: %d, %s, %d, %.1f\n", stu1.id, stu1.name, stu1.age, stu1.score);
 printf("stu2: %d, %s, %d, %.1f\n", stu2.id, stu2.name, stu2.age, stu2.score);
 
 return 0;
}
```

### typedef 与结构体

```c
#include <stdio.h>
#include <string.h>

/* 使用 typedef 定义结构体别名 */
typedef struct {
 int id;
 char name[50];
 float price;
} Product;

/* 命名结构体 + typedef */
struct Person {
 char name[50];
 int age;
};
typedef struct Person Person;

int main(void) {
 /* 使用别名声明变量 */
 Product prod1 = {1, "笔记本电脑", 5999.0f};
 Product prod2;
 
 prod2.id = 2;
 strcpy(prod2.name, "鼠标");
 prod2.price = 99.0f;
 
 printf("商品1: %d - %s - %.2f元\n", prod1.id, prod1.name, prod1.price);
 printf("商品2: %d - %s - %.2f元\n", prod2.id, prod2.name, prod2.price);
 
 Person p = {"张三", 25};
 printf("\n人员: %s, %d岁\n", p.name, p.age);
 
 return 0;
}
```

## 结构体成员

### 成员访问

```c
#include <stdio.h>
#include <string.h>

typedef struct {
 char title[100];
 char author[50];
 int year;
 float price;
} Book;

int main(void) {
 Book book;
 
 /* 使用点运算符访问成员 */
 strcpy(book.title, "C程序设计语言");
 strcpy(book.author, "Brian Kernighan");
 book.year = 1978;
 book.price = 45.0f;
 
 printf("书名: %s\n", book.title);
 printf("作者: %s\n", book.author);
 printf("年份: %d\n", book.year);
 printf("价格: %.2f元\n", book.price);
 
 return 0;
}
```

### 嵌套结构体

```c
#include <stdio.h>
#include <string.h>

typedef struct {
 int year;
 int month;
 int day;
} Date;

typedef struct {
 char name[50];
 Date birthday;
 char address[100];
} Person;

int main(void) {
 Person p = {
 .name = "张三",
 .birthday = {1990, 5, 15},
 .address = "北京市朝阳区"
 };
 
 printf("姓名: %s\n", p.name);
 printf("生日: %d年%d月%d日\n", 
 p.birthday.year, p.birthday.month, p.birthday.day);
 printf("地址: %s\n", p.address);
 
 /* 修改嵌套成员 */
 p.birthday.year = 1995;
 printf("\n修改后的生日: %d年\n", p.birthday.year);
 
 return 0;
}
```

### 结构体数组

```c
#include <stdio.h>
#include <string.h>

typedef struct {
 int id;
 char name[50];
 float score;
} Student;

void print_student(const Student *s) {
 printf("%d\t%s\t%.1f\n", s->id, s->name, s->score);
}

int main(void) {
 /* 结构体数组 */
 Student class[3] = {
 {1, "张三", 85.5f},
 {2, "李四", 92.0f},
 {3, "王五", 78.5f}
 };
 
 printf("学号\t姓名\t成绩\n");
 printf("----\t----\t----\n");
 
 for (int i = 0; i < 3; i++) {
 print_student(&class[i]);
 }
 
 /* 计算平均分 */
 float total = 0;
 for (int i = 0; i < 3; i++) {
 total += class[i].score;
 }
 printf("\n平均分: %.2f\n", total / 3);
 
 return 0;
}
```

## 结构体与函数

### 结构体作为参数

```c
#include <stdio.h>
#include <string.h>

typedef struct {
 int x;
 int y;
} Point;

/* 值传递：复制整个结构体 */
void print_point_value(Point p) {
 printf("点: (%d, %d)\n", p.x, p.y);
 p.x = 100; /* 不影响原结构体 */
}

/* 指针传递：只传递地址 */
void print_point_ptr(const Point *p) {
 printf("点: (%d, %d)\n", p->x, p->y);
}

/* 通过指针修改 */
void move_point(Point *p, int dx, int dy) {
 p->x += dx;
 p->y += dy;
}

/* 返回结构体 */
Point create_point(int x, int y) {
 Point p = {x, y};
 return p;
}

int main(void) {
 Point p1 = {10, 20};
 
 print_point_value(p1);
 printf("调用后 p1: (%d, %d)\n", p1.x, p1.y);
 
 print_point_ptr(&p1);
 
 move_point(&p1, 5, 5);
 printf("移动后 p1: (%d, %d)\n", p1.x, p1.y);
 
 Point p2 = create_point(30, 40);
 printf("新点 p2: (%d, %d)\n", p2.x, p2.y);
 
 return 0;
}
```

### 结构体指针

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

typedef struct {
 int id;
 char name[50];
 float score;
} Student;

int main(void) {
 /* 结构体指针 */
 Student stu = {1, "张三", 85.5f};
 Student *ptr = &stu;
 
 /* 使用 -> 访问成员 */
 printf("学号: %d\n", ptr->id);
 printf("姓名: %s\n", ptr->name);
 printf("成绩: %.1f\n", ptr->score);
 
 /* 等价写法 */
 printf("\n等价写法:\n");
 printf("学号: %d\n", (*ptr).id);
 printf("姓名: %s\n", (*ptr).name);
 printf("成绩: %.1f\n", (*ptr).score);
 
 /* 动态分配结构体 */
 Student *dynamic = (Student *)malloc(sizeof(Student));
 if (dynamic != NULL) {
 dynamic->id = 2;
 strcpy(dynamic->name, "李四");
 dynamic->score = 92.0f;
 
 printf("\n动态分配:\n");
 printf("%d - %s - %.1f\n", dynamic->id, dynamic->name, dynamic->score);
 
 free(dynamic);
 }
 
 return 0;
}
```

## 结构体的大小

```c
#include <stdio.h>
#include <stddef.h>

typedef struct {
 char a;
 int b;
 char c;
} Unaligned;

typedef struct {
 char a;
 char c;
 int b;
} Aligned;

#pragma pack(push, 1)
typedef struct {
 char a;
 int b;
 char c;
} Packed;
#pragma pack(pop)

int main(void) {
 printf("结构体大小与内存对齐:\n");
 printf("sizeof(Unaligned) = %zu\n", sizeof(Unaligned));
 printf("sizeof(Aligned) = %zu\n", sizeof(Aligned));
 printf("sizeof(Packed) = %zu\n", sizeof(Packed));
 
 /* 成员偏移量 */
 printf("\n成员偏移量:\n");
 printf("Unaligned.a 偏移: %zu\n", offsetof(Unaligned, a));
 printf("Unaligned.b 偏移: %zu\n", offsetof(Unaligned, b));
 printf("Unaligned.c 偏移: %zu\n", offsetof(Unaligned, c));
 
 return 0;
}
```

## 实际应用示例

### 学生管理系统

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_STUDENTS 100

typedef struct {
 int id;
 char name[50];
 int age;
 float scores[3];
 float average;
} Student;

/* 计算平均分 */
void calc_average(Student *s) {
 float sum = 0;
 for (int i = 0; i < 3; i++) {
 sum += s->scores[i];
 }
 s->average = sum / 3;
}

/* 打印学生信息 */
void print_student(const Student *s) {
 printf("%d\t%s\t%d\t%.1f\t%.1f\t%.1f\t%.2f\n",
 s->id, s->name, s->age,
 s->scores[0], s->scores[1], s->scores[2],
 s->average);
}

/* 输入学生信息 */
void input_student(Student *s) {
 printf("学号: ");
 scanf("%d", &s->id);
 printf("姓名: ");
 scanf("%s", s->name);
 printf("年龄: ");
 scanf("%d", &s->age);
 printf("三门课程成绩: ");
 for (int i = 0; i < 3; i++) {
 scanf("%f", &s->scores[i]);
 }
 calc_average(s);
}

int main(void) {
 Student students[MAX_STUDENTS];
 int count = 0;
 int choice;
 
 while (1) {
 printf("\n=== 学生管理系统 ===\n");
 printf("1. 添加学生\n");
 printf("2. 显示所有学生\n");
 printf("3. 退出\n");
 printf("请选择: ");
 scanf("%d", &choice);
 
 switch (choice) {
 case 1:
 if (count < MAX_STUDENTS) {
 printf("\n输入第 %d 个学生信息:\n", count + 1);
 input_student(&students[count]);
 count++;
 } else {
 printf("学生数量已达上限\n");
 }
 break;
 
 case 2:
 printf("\n学号\t姓名\t年龄\t课程1\t课程2\t课程3\t平均分\n");
 printf("----\t----\t----\t-----\t-----\t-----\t------\n");
 for (int i = 0; i < count; i++) {
 print_student(&students[i]);
 }
 break;
 
 case 3:
 printf("再见！\n");
 return 0;
 
 default:
 printf("无效选择\n");
 }
 }
 
 return 0;
}
```

### 链表节点

```c
#include <stdio.h>
#include <stdlib.h>

/* 链表节点结构体 */
typedef struct Node {
 int data;
 struct Node *next;
} Node;

/* 创建节点 */
Node *create_node(int data) {
 Node *node = (Node *)malloc(sizeof(Node));
 if (node != NULL) {
 node->data = data;
 node->next = NULL;
 }
 return node;
}

/* 在链表末尾添加节点 */
void append_node(Node **head, int data) {
 Node *new_node = create_node(data);
 if (new_node == NULL) return;
 
 if (*head == NULL) {
 *head = new_node;
 return;
 }
 
 Node *current = *head;
 while (current->next != NULL) {
 current = current->next;
 }
 current->next = new_node;
}

/* 打印链表 */
void print_list(const Node *head) {
 const Node *current = head;
 while (current != NULL) {
 printf("%d -> ", current->data);
 current = current->next;
 }
 printf("NULL\n");
}

/* 释放链表 */
void free_list(Node *head) {
 Node *current = head;
 while (current != NULL) {
 Node *next = current->next;
 free(current);
 current = next;
 }
}

int main(void) {
 Node *head = NULL;
 
 /* 添加节点 */
 for (int i = 1; i <= 5; i++) {
 append_node(&head, i * 10);
 }
 
 printf("链表内容:\n");
 print_list(head);
 
 /* 释放内存 */
 free_list(head);
 
 return 0;
}
```

## 小结

结构体要点：

1. **定义**：使用 `struct` 关键字定义自定义类型
2. **初始化**：顺序初始化或指定成员初始化
3. **成员访问**：`.` 运算符（变量）或 `->` 运算符（指针）
4. **嵌套**：结构体可以包含其他结构体
5. **内存对齐**：结构体大小可能大于成员大小之和

> **提示**：使用 typedef 可以简化结构体类型的声明。传递结构体给函数时，对于大型结构体，使用指针传递更高效。注意内存对齐可能影响结构体大小。
