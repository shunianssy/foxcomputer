# 3.1 CommonJS 模块

CommonJS 是 Node.js 最初采用的模块规范，本节将详细介绍 CommonJS 模块系统的使用方法。

## 模块概述

### 什么是模块？

模块是将代码组织成独立单元的方式，具有以下优点：

1. **封装性**：模块内部变量不会污染全局作用域
2. **复用性**：模块可以在不同项目中重复使用
3. **可维护性**：代码结构清晰，便于维护
4. **依赖管理**：明确模块之间的依赖关系

### CommonJS 规范

CommonJS 规范定义了：

- `require()`：导入模块
- `module.exports` 或 `exports`：导出模块
- `__dirname`：当前模块所在目录
- `__filename`：当前模块文件路径

## 导出模块

### module.exports

```javascript
// math.js - 导出单个值

// 方式一：导出对象
module.exports = {
    add: (a, b) => a + b,
    subtract: (a, b) => a - b,
    multiply: (a, b) => a * b,
    divide: (a, b) => a / b
};

// 方式二：导出函数
module.exports = function(a, b) {
    return a + b;
};

// 方式三：导出类
class Calculator {
    add(a, b) { return a + b; }
    subtract(a, b) { return a - b; }
}

module.exports = Calculator;
```

### exports 快捷方式

```javascript
// utils.js - 使用 exports 导出多个属性

// exports 是 module.exports 的引用
exports.add = (a, b) => a + b;
exports.subtract = (a, b) => a - b;
exports.PI = 3.14159;

// 等价于：
// module.exports.add = (a, b) => a + b;
// module.exports.subtract = (a, b) => a - b;
// module.exports.PI = 3.14159;

// 注意：不能直接给 exports 赋值
// exports = { add, subtract };  // 错误！会断开与 module.exports 的引用
```

### 导出最佳实践

```javascript
// 推荐：在文件末尾统一导出
const add = (a, b) => a + b;
const subtract = (a, b) => a - b;
const multiply = (a, b) => a * b;
const divide = (a, b) => a / b;

module.exports = {
    add,
    subtract,
    multiply,
    divide
};

// 或者导出时重命名
module.exports = {
    add: add,
    sub: subtract,
    mul: multiply,
    div: divide
};
```

## 导入模块

### 基本用法

```javascript
// 导入自定义模块
const math = require('./math');
console.log(math.add(1, 2));  // 3

// 解构导入
const { add, subtract } = require('./math');
console.log(add(1, 2));       // 3
console.log(subtract(5, 3));  // 2

// 导入并重命名
const { add: addNumbers, PI: pi } = require('./math');
console.log(addNumbers(1, 2));  // 3
console.log(pi);                // 3.14159
```

### 模块路径

```javascript
// 1. 相对路径（推荐使用 ./ 或 ../）
const local = require('./local-module');
const parent = require('../parent-module');

// 2. 绝对路径
const absolute = require('/home/user/project/module');

// 3. node_modules 模块
const express = require('express');
const lodash = require('lodash');

// 4. Node.js 内置模块
const fs = require('fs');
const path = require('path');
const http = require('http');
```

### 模块查找顺序

```javascript
// require() 查找模块的顺序：

// 1. 如果是内置模块（如 fs、http），直接返回

// 2. 如果路径以 ./ 或 ../ 开头：
//    - 查找精确匹配的文件
//    - 查找 .js 文件
//    - 查找 .json 文件
//    - 查找 .node 文件（C++ 扩展）
//    - 查找目录下的 index.js

// 3. 如果是第三方模块：
//    - 当前目录的 node_modules

//    - ...
//    - 全局 node_modules

// 示例：查找 './utils'
// ./utils.js
// ./utils.json
// ./utils.node
// ./utils/index.js
// ./utils/index.json
```

## 模块缓存

### 缓存机制

```javascript
// counter.js
let count = 0;

module.exports = {
    increment() {
        return ++count;
    },
    getCount() {
        return count;
    }
};

// main.js
const counter1 = require('./counter');
const counter2 = require('./counter');  // 返回缓存的模块

console.log(counter1.increment());  // 1
console.log(counter2.increment());  // 2（同一个实例）
console.log(counter1.getCount());   // 2
console.log(counter2.getCount());   // 2

// 检查缓存
console.log(require.cache);
```

### 清除缓存

```javascript
// 清除单个模块缓存
delete require.cache[require.resolve('./counter')];

// 清除所有缓存
Object.keys(require.cache).forEach(key => {
    delete require.cache[key];
});

// 热重载示例
function hotReload(modulePath) {
    delete require.cache[require.resolve(modulePath)];
    return require(modulePath);
}

// 使用
let config = require('./config');
config = hotReload('./config');  // 重新加载
```

## 模块作用域

### 模块级变量

```javascript
// 每个模块都有独立的作用域

// __filename - 当前模块的绝对路径
console.log(__filename);
// /home/user/project/app.js

// __dirname - 当前模块所在目录
console.log(__dirname);
// /home/user/project

// module - 当前模块对象
console.log(module.id);        // 模块标识符
console.log(module.filename);  // 模块文件名
console.log(module.loaded);    // 是否已加载
console.log(module.parent);    // 父模块
console.log(module.children);  // 子模块
console.log(module.paths);     // 模块搜索路径

// exports - module.exports 的引用
console.log(exports === module.exports);  // true

// require - 导入函数
console.log(require.resolve('./utils'));  // 解析模块路径
console.log(require.main);                // 主模块
```

### 模块包装

```javascript
// Node.js 在执行模块代码前会进行包装
// 实际执行的代码类似：

(function(exports, require, module, __filename, __dirname) {
    // 模块代码在这里执行
    const x = 1;
    module.exports = { x };
});

// 这就是为什么模块内的变量不会污染全局作用域
```

## 循环依赖

### 循环依赖问题

```javascript
// a.js
const b = require('./b');
console.log('a.js: b.value =', b.value);
module.exports = { value: 'from a' };

// b.js
const a = require('./a');
console.log('b.js: a.value =', a.value);  // undefined
module.exports = { value: 'from b' };

// main.js
const a = require('./a');
// 输出：
// b.js: a.value = undefined
// a.js: b.value = from b
```

### 解决循环依赖

```javascript
// 方式一：延迟导入
// a.js
module.exports = {
    getValue() {
        const b = require('./b');  // 延迟到函数调用时导入
        return b.value;
    }
};

// 方式二：重构代码结构
// 将共享逻辑提取到第三个模块
// shared.js
module.exports = { shared: 'data' };

// a.js
const shared = require('./shared');
// ...

// b.js
const shared = require('./shared');
// ...

// 方式三：使用依赖注入
// a.js
module.exports = function(b) {
    return {
        getValue() {
            return b.value;
        }
    };
};

// main.js
const b = require('./b');
const a = require('./a')(b);
```

### 创建目录模块

## 实际应用示例

### 配置模块

```javascript
// config/index.js
const fs = require('fs');
const path = require('path');

const env = process.env.NODE_ENV || 'development';
const configPath = path.join(__dirname, 'environments', `${env}.json`);

let config;
try {
    const content = fs.readFileSync(configPath, 'utf8');
    config = JSON.parse(content);
} catch (error) {
    console.warn(`配置文件 ${configPath} 不存在，使用默认配置`);
    config = {
        port: 3000,
        database: {
            host: 'localhost',
            port: 3306
        }
    };
}

// 支持环境变量覆盖
config.port = parseInt(process.env.PORT) || config.port;
config.database.host = process.env.DB_HOST || config.database.host;

module.exports = config;
```

### 数据库连接模块

```javascript
// database/connection.js
const mysql = require('mysql2/promise');

let pool = null;

function getPool() {
    if (!pool) {
        pool = mysql.createPool({
            host: process.env.DB_HOST || 'localhost',
            port: parseInt(process.env.DB_PORT) || 3306,
            user: process.env.DB_USER || 'root',
            password: process.env.DB_PASS || '',
            database: process.env.DB_NAME || 'test',
            waitForConnections: true,
            connectionLimit: 10,
            queueLimit: 0
        });
    }
    return pool;
}

async function query(sql, params) {
    const [rows] = await getPool().execute(sql, params);
    return rows;
}

async function transaction(callback) {
    const connection = await getPool().getConnection();
    try {
        await connection.beginTransaction();
        const result = await callback(connection);
        await connection.commit();
        return result;
    } catch (error) {
        await connection.rollback();
        throw error;
    } finally {
        connection.release();
    }
}

module.exports = {
    getPool,
    query,
    transaction
};
```

### 日志模块

```javascript
// logger/index.js
const fs = require('fs');
const path = require('path');

class Logger {
    constructor(options = {}) {
        this.level = options.level || 'info';
        this.logDir = options.logDir || path.join(__dirname, 'logs');
        
        // 确保日志目录存在
        if (!fs.existsSync(this.logDir)) {
            fs.mkdirSync(this.logDir, { recursive: true });
        }
    }
    
    _formatMessage(level, message, meta = {}) {
        const timestamp = new Date().toISOString();
        return JSON.stringify({
            timestamp,
            level,
            message,
            ...meta
        }) + '\n';
    }
    
    _writeToFile(filename, content) {
        const filePath = path.join(this.logDir, filename);
        fs.appendFileSync(filePath, content);
    }
    
    log(level, message, meta) {
        const levels = ['debug', 'info', 'warn', 'error'];
        const currentLevelIndex = levels.indexOf(this.level);
        const messageLevelIndex = levels.indexOf(level);
        
        if (messageLevelIndex >= currentLevelIndex) {
            const formatted = this._formatMessage(level, message, meta);
            console.log(formatted.trim());
            this._writeToFile(`${level}.log`, formatted);
        }
    }
    
    debug(message, meta) { this.log('debug', message, meta); }
    info(message, meta) { this.log('info', message, meta); }
    warn(message, meta) { this.log('warn', message, meta); }
    error(message, meta) { this.log('error', message, meta); }
}

module.exports = new Logger();
module.exports.Logger = Logger;
```

## 小结

本节学习了：

1. **模块导出**：`module.exports` 和 `exports` 的区别
2. **模块导入**：`require()` 的使用和路径解析
3. **模块缓存**：单例模式和热重载
4. **模块作用域**：`__dirname`、`__filename`、`module` 等
5. **循环依赖**：问题和解决方案
6. **目录模块**：组织模块结构

> **提示**：CommonJS 是 Node.js 的默认模块系统，但现代 Node.js 也支持 ES Modules。在开发 npm 包时，建议同时提供两种模块格式的支持。
