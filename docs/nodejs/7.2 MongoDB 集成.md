# 7.2 MongoDB 集成

MongoDB 是流行的 NoSQL 数据库。本节将介绍如何在 Node.js 中使用 MongoDB。

## 安装配置

### 安装 mongoose

```bash
npm install mongoose
```

### 连接数据库

```javascript
const mongoose = require('mongoose');

// 连接 MongoDB
mongoose.connect('mongodb://localhost:27017/mydb', {
    useNewUrlParser: true,
    useUnifiedTopology: true
})
.then(() => console.log('MongoDB 连接成功'))
.catch(err => console.error('MongoDB 连接失败:', err));

// 连接事件
mongoose.connection.on('connected', () => {
    console.log('MongoDB 连接成功');
});

mongoose.connection.on('error', (err) => {
    console.error('MongoDB 错误:', err);
});

mongoose.connection.on('disconnected', () => {
    console.log('MongoDB 断开连接');
});

// 关闭连接
process.on('SIGINT', async () => {
    await mongoose.connection.close();
    process.exit(0);
});
```

## Schema 和 Model

### 定义 Schema

```javascript
const mongoose = require('mongoose');
const Schema = mongoose.Schema;

// 用户 Schema
const userSchema = new Schema({
    name: {
        type: String,
        required: [true, '姓名必填'],
        trim: true,
        minlength: [2, '姓名至少2个字符'],
        maxlength: [50, '姓名最多50个字符']
    },
    email: {
        type: String,
        required: true,
        unique: true,
        lowercase: true,
        match: [/^\S+@\S+\.\S+$/, '邮箱格式不正确']
    },
    password: {
        type: String,
        required: true,
        select: false  // 查询时不返回
    },
    age: {
        type: Number,
        min: 0,
        max: 150
    },
    role: {
        type: String,
        enum: ['user', 'admin'],
        default: 'user'
    },
    avatar: String,
    isActive: {
        type: Boolean,
        default: true
    },
    lastLogin: Date
}, {
    timestamps: true,  // 自动添加 createdAt, updatedAt
    versionKey: false   // 不添加 __v 字段
});

// 索引
userSchema.index({ email: 1 });
userSchema.index({ name: 'text' });

// 虚拟属性
userSchema.virtual('profile').get(function() {
    return `${this.name} (${this.email})`;
});

// 实例方法
userSchema.methods.toJSON = function() {
    const user = this.toObject();
    delete user.password;
    return user;
};

// 静态方法
userSchema.statics.findByEmail = function(email) {
    return this.findOne({ email });
};

// 中间件
userSchema.pre('save', function(next) {
    if (this.isModified('password')) {
        this.password = hashPassword(this.password);
    }
    next();
});

// 创建 Model
const User = mongoose.model('User', userSchema);

module.exports = User;
```

## CRUD 操作

### 创建文档

```javascript
const User = require('./models/User');

// 创建单个文档
async function createUser() {
    const user = await User.create({
        name: '张三',
        email: 'zhangsan@example.com',
        password: 'password123',
        age: 25
    });
    
    console.log(user);
    return user;

// 使用 new 和 save
async function createUserWithSave() {
    const user = new User({
        name: '李四',
        email: 'lisi@example.com',
        password: 'password123'
    });
    
    await user.save();
    return user;

async function createManyUsers() {
    const users = await User.insertMany([
        { name: '用户1', email: 'user1@example.com', password: 'pass' },
        { name: '用户2', email: 'user2@example.com', password: 'pass' }

    return users;

```

### 查询文档

```javascript
// 查询所有
async function findAll() {
    const users = await User.find();
    return users;
}

// 条件查询
async function findActive() {
    const users = await User.find({
        isActive: true,
        age: { $gte: 18 }
    });
    return users;
}

// 单个查询
async function findById(id) {
    const user = await User.findById(id);
    return user;
}

async function findByEmail(email) {
    const user = await User.findOne({ email });
    return user;
}

// 选择字段
async function findWithFields() {
    const users = await User.find()
        .select('name email age')
        .sort({ createdAt: -1 })
        .limit(10)
        .skip(0);
    
    return users;
}

// 分页查询
async function findWithPagination(page = 1, limit = 10) {
    const skip = (page - 1) * limit;
    
    const [users, total] = await Promise.all([
        User.find()
            .sort({ createdAt: -1 })
            .limit(limit)
            .skip(skip),
        User.countDocuments()
    ]);
    
    return {
        data: users,
        pagination: {
            page,
            limit,
            total,
            totalPages: Math.ceil(total / limit)
        }
    };
}

// 聚合查询
async function aggregateByRole() {
    const result = await User.aggregate([
        { $match: { isActive: true } },
        { $group: { _id: '$role', count: { $sum: 1 } } },
        { $sort: { count: -1 } }
    ]);
    
    return result;
}
```

### 更新文档

```javascript
// findByIdAndUpdate
async function updateUser(id, data) {
    const user = await User.findByIdAndUpdate(
        id,
        { $set: data },
        { new: true, runValidators: true }
    );
    
    return user;
}

// updateOne
async function updateByEmail(email, data) {
    const result = await User.updateOne(
        { email },
        { $set: data }
    );
    
    return result.modifiedCount > 0;
}

// updateMany
async function deactivateInactive() {
    const result = await User.updateMany(
        { lastLogin: { $lt: new Date(Date.now() - 30 * 24 * 60 * 60 * 1000) } },
        { isActive: false }
    );
    
    return result.modifiedCount;
}

// findOneAndUpdate
async function findAndUpdate(id, data) {
    const user = await User.findOneAndUpdate(
        { _id: id },
        data,
        { new: true }
    );
    
    return user;
}
```

### 删除文档

```javascript
// findByIdAndDelete
async function deleteUser(id) {
    const user = await User.findByIdAndDelete(id);
    return user;
}

// deleteOne
async function deleteByEmail(email) {
    const result = await User.deleteOne({ email });
    return result.deletedCount > 0;
}

// deleteMany
async function deleteInactive() {
    const result = await User.deleteMany({ isActive: false });
    return result.deletedCount;
}
```

## 关联查询

### 定义关联

```javascript
// 文章 Schema
const postSchema = new Schema({
    title: { type: String, required: true },
    content: { type: String, required: true },
    author: { type: Schema.Types.ObjectId, ref: 'User', required: true },
    tags: [{ type: Schema.Types.ObjectId, ref: 'Tag' }],
    comments: [{ type: Schema.Types.ObjectId, ref: 'Comment' }]
}, { timestamps: true });

const Post = mongoose.model('Post', postSchema);

// 评论 Schema
const commentSchema = new Schema({
    content: { type: String, required: true },
    author: { type: Schema.Types.ObjectId, ref: 'User', required: true },
    post: { type: Schema.Types.ObjectId, ref: 'Post', required: true }
}, { timestamps: true });

const Comment = mongoose.model('Comment', commentSchema);
```

### 关联查询

```javascript
// populate 填充关联
async function getPostWithAuthor(postId) {
    const post = await Post.findById(postId)
        .populate('author', 'name email')
        .populate('tags', 'name')
        .populate({
            path: 'comments',
            populate: { path: 'author', select: 'name' }
        });
    
    return post;
}

// 聚合关联
async function getPostsWithAuthors() {
    const posts = await Post.aggregate([
        {
            $lookup: {
                from: 'users',
                localField: 'author',
                foreignField: '_id',
                as: 'author'
            }
        },
        { $unwind: '$author' },
        {
            $project: {
                title: 1,
                content: 1,
                'author.name': 1,
                'author.email': 1
            }
        }
    ]);
    
    return posts;
}
```
