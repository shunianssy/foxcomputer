# 5.4 其他核心模块

除了 fs、path、http 模块外，Node.js 还提供了许多其他核心模块。本节将介绍常用的其他核心模块。

## stream 流模块

### 流的类型

```javascript
const stream = require('stream');

// 四种流类型：
// 1. Readable - 可读流
// 2. Writable - 可写流
// 3. Duplex - 双工流（可读可写）
// 4. Transform - 转换流（可修改数据）
```

### 可读流

```javascript
const { Readable } = require('stream');

// 创建可读流
const readable = new Readable({
    read() {}
});

// 推送数据
readable.push('Hello ');
readable.push('World');
readable.push(null);  // 结束

// 消费数据
readable.on('data', (chunk) => {
    console.log('收到:', chunk.toString());
});

readable.on('end', () => {
    console.log('读取完成');
});

// 从数组创建可读流
function createReadStreamFromArray(arr) {
    return new Readable({
        objectMode: true,  // 对象模式
        read() {
            if (arr.length === 0) {
                this.push(null);
            } else {
                this.push(arr.shift());
            }
        }
    });
}

const objStream = createReadStreamFromArray([1, 2, 3, 4, 5]);
objStream.on('data', (data) => console.log(data));
```

### 可写流

```javascript
const { Writable } = require('stream');

// 创建可写流
const writable = new Writable({
    write(chunk, encoding, callback) {
        console.log('写入:', chunk.toString());
        callback();
    }
});

// 写入数据
writable.write('Hello ');
writable.write('World');
writable.end();

// 监听完成事件
writable.on('finish', () => {
    console.log('写入完成');
});

// 文件可写流
const fs = require('fs');
const fileStream = fs.createWriteStream('output.txt');

fileStream.write('第一行\n');
fileStream.write('第二行\n');
fileStream.end();
```

### 转换流

```javascript
const { Transform } = require('stream');

// 创建转换流
const upperCase = new Transform({
    transform(chunk, encoding, callback) {
        this.push(chunk.toString().toUpperCase());
        callback();
    }
});

// 使用管道
process.stdin
    .pipe(upperCase)
    .pipe(process.stdout);

// JSON 解析转换流
const jsonParser = new Transform({
    objectMode: true,
    transform(chunk, encoding, callback) {
        try {
            const data = JSON.parse(chunk.toString());
            this.push(data);
            callback();
        } catch (error) {
            callback(error);
        }
    }
});

// 加密转换流
const crypto = require('crypto');
const encrypt = new Transform({
    transform(chunk, encoding, callback) {
        const cipher = crypto.createCipher('aes-256-cbc', 'secret');
        const encrypted = cipher.update(chunk);
        this.push(Buffer.concat([encrypted, cipher.final()]));
        callback();
    }
});
```

### 管道操作

```javascript
const fs = require('fs');
const zlib = require('zlib');
const crypto = require('crypto');

// 链式管道
fs.createReadStream('file.txt')
    .pipe(zlib.createGzip())
    .pipe(crypto.createCipher('aes-256-cbc', 'secret'))
    .pipe(fs.createWriteStream('file.txt.enc.gz'));

// pipeline（推荐）
const { pipeline } = require('stream/promises');

async function compressAndEncrypt() {
    await pipeline(
        fs.createReadStream('file.txt'),
        zlib.createGzip(),
        crypto.createCipher('aes-256-cbc', 'secret'),
        fs.createWriteStream('file.txt.enc.gz')
    );
    console.log('处理完成');
}
```

## zlib 压缩模块

```javascript
const zlib = require('zlib');
const fs = require('fs');

// 同步压缩
const input = Buffer.from('Hello World');
const compressed = zlib.deflateSync(input);
const decompressed = zlib.inflateSync(compressed);

console.log('原始大小:', input.length);
console.log('压缩后:', compressed.length);
console.log('解压后:', decompressed.toString());

// Gzip 压缩
const gzipped = zlib.gzipSync(input);
const ungzipped = zlib.gunzipSync(gzipped);

// 文件压缩
async function compressFile(inputPath, outputPath) {
    const { pipeline } = require('stream/promises');
    
    await pipeline(
        fs.createReadStream(inputPath),
        zlib.createGzip(),
        fs.createWriteStream(outputPath)
    );
}

// HTTP 响应压缩
const http = require('http');

const server = http.createServer((req, res) => {
    const acceptEncoding = req.headers['accept-encoding'] || '';
    
    if (acceptEncoding.includes('gzip')) {
        res.writeHead(200, {
            'Content-Type': 'text/plain',
            'Content-Encoding': 'gzip'
        });
        zlib.gzip('Hello World', (err, buffer) => {
            res.end(buffer);
        });
    } else {
        res.writeHead(200, { 'Content-Type': 'text/plain' });
        res.end('Hello World');
    }
});
```

## child_process 子进程模块

### exec

```javascript
const { exec } = require('child_process');

// 执行命令
exec('ls -la', (error, stdout, stderr) => {
    if (error) {
        console.error('错误:', error);
        return;
    }
    console.log('输出:', stdout);
    console.error('错误输出:', stderr);
});

// Promise 形式
const util = require('util');
const execAsync = util.promisify(exec);

async function runCommand() {
    try {
        const { stdout, stderr } = await execAsync('ls -la');
        console.log('输出:', stdout);
    } catch (error) {
        console.error('错误:', error);
    }
}
```

### spawn

```javascript
const { spawn } = require('child_process');

// 启动进程
const child = spawn('ls', ['-la']);

// 监听输出
child.stdout.on('data', (data) => {
    console.log('输出:', data.toString());
});

child.stderr.on('data', (data) => {
    console.error('错误:', data.toString());
});

child.on('close', (code) => {
    console.log('进程退出码:', code);
});

// 交互式进程
const node = spawn('node', ['-i']);

node.stdout.on('data', (data) => {
    console.log('Node 输出:', data.toString());
});

node.stdin.write('1 + 1\n');

// 流式处理
const grep = spawn('grep', ['pattern']);
const cat = spawn('cat', ['file.txt']);

cat.stdout.pipe(grep.stdin);
grep.stdout.on('data', (data) => {
    console.log('匹配:', data.toString());
});
```

### fork

```javascript
const { fork } = require('child_process');

// 创建子进程运行 Node.js 模块
const child = fork('./worker.js');

// 发送消息
child.send({ type: 'task', data: { value: 42 } });

// 接收消息
child.on('message', (message) => {
    console.log('收到消息:', message);
});

child.on('exit', (code) => {
    console.log('子进程退出:', code);
});

// worker.js
process.on('message', (message) => {
    if (message.type === 'task') {
        // 处理任务
        const result = message.data.value * 2;
        process.send({ type: 'result', data: result });
    }
});
```

## cluster 集群模块

```javascript
const cluster = require('cluster');
const http = require('http');
const os = require('os');

if (cluster.isMaster) {
    // 主进程
    const numCPUs = os.cpus().length;
    
    console.log(`主进程 ${process.pid} 运行中`);
    
    // Fork 工作进程
    for (let i = 0; i < numCPUs; i++) {
        cluster.fork();
    }
    
    // 工作进程退出时重启
    cluster.on('exit', (worker, code, signal) => {
        console.log(`工作进程 ${worker.process.pid} 退出`);
        cluster.fork();
    });
    
} else {
    // 工作进程
    http.createServer((req, res) => {
        res.writeHead(200);
        res.end(`工作进程 ${process.pid} 处理请求`);
    }).listen(3000);
    
    console.log(`工作进程 ${process.pid} 启动`);
}
```

## net 网络模块

```javascript
const net = require('net');

// 创建 TCP 服务器
const server = net.createServer((socket) => {
    console.log('客户端连接');
    
    socket.on('data', (data) => {
        console.log('收到数据:', data.toString());
        socket.write('Echo: ' + data);
    });
    
    socket.on('end', () => {
        console.log('客户端断开');
    });
    
    socket.on('error', (err) => {
        console.error('错误:', err);
    });
});

server.listen(3000, () => {
    console.log('TCP 服务器运行在端口 3000');
});

// TCP 客户端
const client = net.createConnection({ port: 3000 }, () => {
    console.log('已连接到服务器');
    client.write('Hello Server');
});

client.on('data', (data) => {
    console.log('收到:', data.toString());
    client.end();
});

client.on('end', () => {
    console.log('断开连接');
});
```

## dns 域名解析模块

```javascript
const dns = require('dns');

// 解析域名
dns.lookup('example.com', (err, address, family) => {
    console.log('IP 地址:', address);
    console.log('IP 版本:', family);  // 4 或 6
});

// 解析所有地址
dns.resolve4('example.com', (err, addresses) => {
    console.log('IPv4 地址:', addresses);
});

dns.resolve6('example.com', (err, addresses) => {
    console.log('IPv6 地址:', addresses);
});

// 反向解析
dns.reverse('93.184.216.34', (err, hostnames) => {
    console.log('主机名:', hostnames);
});

// Promise 形式
const dnsPromises = dns.promises;

async function resolveDomain(domain) {
    try {
        const addresses = await dnsPromises.resolve4(domain);
        console.log('IPv4 地址:', addresses);
        
        const mxRecords = await dnsPromises.resolveMx(domain);
        console.log('MX 记录:', mxRecords);
        
        const txtRecords = await dnsPromises.resolveTxt(domain);
        console.log('TXT 记录:', txtRecords);
    } catch (error) {
        console.error('解析失败:', error);
    }
}
```

## readline 逐行读取模块

```javascript
const readline = require('readline');
const fs = require('fs');

// 从文件逐行读取
async function processFile() {
    const fileStream = fs.createReadStream('file.txt');
    
    const rl = readline.createInterface({
        input: fileStream,
        crlfDelay: Infinity
    });
    
    let lineCount = 0;
    
    for await (const line of rl) {
        lineCount++;
        console.log(`第 ${lineCount} 行: ${line}`);
    }
    
    console.log(`总共 ${lineCount} 行`);
}

// 命令行交互
const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout
});

rl.question('请输入你的名字: ', (answer) => {
    console.log(`你好, ${answer}!`);
    rl.close();
});

// 多次交互
async function askQuestions() {
    const answers = {};
    
    answers.name = await new Promise(resolve => {
        rl.question('姓名: ', resolve);
    });
    
    answers.age = await new Promise(resolve => {
        rl.question('年龄: ', resolve);
    });
    
    console.log('用户信息:', answers);
    rl.close();
}
```

## 小结

本节学习了：

1. **stream 流模块**：可读流、可写流、转换流、管道
2. **zlib 压缩模块**：gzip、deflate 压缩解压
3. **child_process 子进程模块**：exec、spawn、fork
4. **cluster 集群模块**：多进程集群
5. **net 网络模块**：TCP 服务器和客户端
6. **dns 域名解析模块**：域名解析
7. **readline 逐行读取模块**：文件和命令行交互

> **提示**：这些核心模块是 Node.js 开发的基础工具。stream 模块在处理大文件时特别有用，cluster 模块可以充分利用多核 CPU，child_process 模块可以执行系统命令和运行外部程序。
