# 2.3 对象与数组

对象和数组是 JavaScript 中最常用的数据结构。本节将详细介绍它们的创建、操作和常用方法。

## 对象

### 创建对象

```javascript
// 对象字面量
const person = {
    name: '张三',
    age: 25,
    email: 'zhangsan@example.com',
    greet() {
        return `你好，我是${this.name}`;
    }
};

// 构造函数
function Person(name, age) {
    this.name = name;
    this.age = age;
}

const person1 = new Person('张三', 25);

// Object.create()
const animal = {
    type: '动物',
    speak() {
        console.log('发出声音');
    }
};

const dog = Object.create(animal);
dog.type = '狗';
dog.speak();  // 发出声音

// 类语法
class User {
    constructor(name, email) {
        this.name = name;
        this.email = email;
    }
    
    getInfo() {
        return `${this.name} (${this.email})`;
    }
}

const user = new User('张三', 'zhangsan@example.com');
```

### 属性操作

```javascript
const obj = {};

// 添加属性
obj.name = '张三';
obj['age'] = 25;

// 读取属性
console.log(obj.name);      // 张三
console.log(obj['age']);    // 25
console.log(obj.email);     // undefined

// 删除属性
delete obj.age;
console.log(obj.age);       // undefined

// 检查属性是否存在
console.log('name' in obj);           // true
console.log(obj.hasOwnProperty('name')); // true

// 计算属性名
const key = 'dynamic';
const dynamicObj = {
    [key]: '动态属性',
    [`${key}Key`]: '拼接属性名'
};
console.log(dynamicObj.dynamic);      // 动态属性
console.log(dynamicObj.dynamicKey);   // 拼接属性名
```

### 属性描述符

```javascript
const obj = {};

// 定义属性描述符
Object.defineProperty(obj, 'name', {
    value: '张三',
    writable: false,        // 不可写
    enumerable: true,       // 可枚举
    configurable: false     // 不可配置
});

obj.name = '李四';  // 静默失败（严格模式会报错）
console.log(obj.name);  // 张三

// 定义多个属性
Object.defineProperties(obj, {
    age: {
        value: 25,
        writable: true,
        enumerable: true
    },
    email: {
        value: 'test@example.com',
        enumerable: false
    }
});

// 获取属性描述符
console.log(Object.getOwnPropertyDescriptor(obj, 'name'));
// { value: '张三', writable: false, enumerable: true, configurable: false }
```

### Getter 和 Setter

```javascript
const person = {
    firstName: '张',
    lastName: '三',
    
    // Getter
    get fullName() {
        return `${this.firstName}${this.lastName}`;
    },
    
    // Setter
    set fullName(value) {
        [this.firstName, this.lastName] = value.split('');
    }
};

console.log(person.fullName);  // 张三
person.fullName = '李四';
console.log(person.firstName);  // 李
console.log(person.lastName);   // 四

// 使用 defineProperty 定义
const user = {
    _age: 0
};

Object.defineProperty(user, 'age', {
    get() {
        return this._age;
    },
    set(value) {
        if (value >= 0 && value <= 150) {
            this._age = value;
        } else {
            console.log('年龄无效');
        }
    }
});

user.age = 25;
console.log(user.age);  // 25
user.age = -1;  // 年龄无效
```

### 对象方法

```javascript
const obj = { a: 1, b: 2, c: 3 };

// 获取所有键
console.log(Object.keys(obj));     // ['a', 'b', 'c']

// 获取所有值
console.log(Object.values(obj));   // [1, 2, 3]

// 获取键值对
console.log(Object.entries(obj));  // [['a', 1], ['b', 2], ['c', 3]]

// 从键值对创建对象
const entries = [['x', 1], ['y', 2]];
console.log(Object.fromEntries(entries));  // { x: 1, y: 2 }

// 合并对象
const obj1 = { a: 1 };
const obj2 = { b: 2 };
const obj3 = { c: 3 };

const merged = Object.assign({}, obj1, obj2, obj3);
console.log(merged);  // { a: 1, b: 2, c: 3 }

// 展开运算符
const spread = { ...obj1, ...obj2, ...obj3 };
console.log(spread);  // { a: 1, b: 2, c: 3 }

// 冻结对象（不可修改）
const frozen = Object.freeze({ x: 1 });
// frozen.x = 2;  // 静默失败
console.log(Object.isFrozen(frozen));  // true

// 密封对象（不能添加/删除属性）
const sealed = Object.seal({ x: 1 });
sealed.x = 2;  // 可以修改
// sealed.y = 3;  // 不能添加
// delete sealed.x;  // 不能删除
console.log(Object.isSealed(sealed));  // true
```

### 对象遍历

```javascript
const obj = { a: 1, b: 2, c: 3 };

// for...in（遍历可枚举属性，包括原型链）
for (const key in obj) {
    if (obj.hasOwnProperty(key)) {
        console.log(`${key}: ${obj[key]}`);
    }
}

// Object.keys() + forEach
Object.keys(obj).forEach(key => {
    console.log(`${key}: ${obj[key]}`);
});

// Object.entries() + for...of
for (const [key, value] of Object.entries(obj)) {
    console.log(`${key}: ${value}`);
}

// Object.values()
for (const value of Object.values(obj)) {
    console.log(value);
}
```

## 数组

### 创建数组

```javascript
// 数组字面量
const arr1 = [1, 2, 3, 4, 5];

// Array 构造函数
const arr2 = new Array(5);        // [empty x 5]
const arr3 = new Array(1, 2, 3);  // [1, 2, 3]

// Array.of()
const arr4 = Array.of(1, 2, 3);   // [1, 2, 3]
const arr5 = Array.of(5);         // [5]（不是空数组）

// Array.from()
const arr6 = Array.from('hello');  // ['h', 'e', 'l', 'l', 'o']
const arr7 = Array.from({ length: 5 }, (_, i) => i);  // [0, 1, 2, 3, 4]

// 填充数组
const arr8 = new Array(5).fill(0);  // [0, 0, 0, 0, 0]
const arr9 = [1, 2, 3, 4, 5].fill(0, 1, 3);  // [1, 0, 0, 4, 5]
```

### 数组访问

```javascript
const arr = ['a', 'b', 'c', 'd', 'e'];

// 索引访问
console.log(arr[0]);      // 'a'
console.log(arr[-1]);     // undefined（不支持负索引）
console.log(arr.at(-1));  // 'e'（ES2022 at() 方法）

// 长度
console.log(arr.length);  // 5

// 修改元素
arr[0] = 'A';
console.log(arr);  // ['A', 'b', 'c', 'd', 'e']

// 添加元素（不推荐）
arr[10] = 'j';
console.log(arr.length);  // 11（中间是空位）
```

### 数组方法 - 添加/删除

```javascript
const arr = [1, 2, 3];

// push: 末尾添加
arr.push(4, 5);
console.log(arr);  // [1, 2, 3, 4, 5]

// pop: 末尾删除
const last = arr.pop();
console.log(last);  // 5
console.log(arr);   // [1, 2, 3, 4]

// unshift: 开头添加
arr.unshift(0);
console.log(arr);  // [0, 1, 2, 3, 4]

// shift: 开头删除
const first = arr.shift();
console.log(first);  // 0
console.log(arr);    // [1, 2, 3, 4]

// splice: 任意位置添加/删除
const arr2 = [1, 2, 3, 4, 5];

// 删除元素
const removed = arr2.splice(1, 2);  // 从索引1开始删除2个
console.log(removed);  // [2, 3]
console.log(arr2);     // [1, 4, 5]

// 插入元素
arr2.splice(1, 0, 2, 3);  // 从索引1开始删除0个，插入2和3
console.log(arr2);  // [1, 2, 3, 4, 5]

// 替换元素
arr2.splice(1, 2, 'a', 'b');  // 从索引1开始删除2个，插入'a'和'b'
console.log(arr2);  // [1, 'a', 'b', 4, 5]

// toSpliced: 不修改原数组（ES2023）
const arr3 = [1, 2, 3, 4, 5];
const newArr = arr3.toSpliced(1, 2, 'a', 'b');
console.log(arr3);    // [1, 2, 3, 4, 5]
console.log(newArr);  // [1, 'a', 'b', 4, 5]
```

### 数组方法 - 查找

```javascript
const arr = [1, 2, 3, 4, 5, 3];

// indexOf: 查找索引
console.log(arr.indexOf(3));      // 2
console.log(arr.indexOf(6));      // -1
console.log(arr.indexOf(3, 3));   // 5（从索引3开始查找）

// lastIndexOf: 从后往前查找
console.log(arr.lastIndexOf(3));  // 5

// includes: 是否包含
console.log(arr.includes(3));     // true
console.log(arr.includes(6));     // false

// find: 查找元素
const found = arr.find(n => n > 3);
console.log(found);  // 4

// findIndex: 查找索引
const index = arr.findIndex(n => n > 3);
console.log(index);  // 3

// findLast / findLastIndex（ES2023）
const lastFound = arr.findLast(n => n > 3);
console.log(lastFound);  // 5

const lastIndex = arr.findLastIndex(n => n > 3);
console.log(lastIndex);  // 5

// 对象数组查找
const users = [
    { id: 1, name: '张三' },
    { id: 2, name: '李四' },
    { id: 3, name: '王五' }
];

const user = users.find(u => u.id === 2);
console.log(user);  // { id: 2, name: '李四' }
```

### 数组方法 - 遍历

```javascript
const arr = [1, 2, 3, 4, 5];

// forEach: 遍历
arr.forEach((value, index, array) => {
    console.log(`索引 ${index}: ${value}`);
});

// map: 映射
const doubled = arr.map(n => n * 2);
console.log(doubled);  // [2, 4, 6, 8, 10]

// filter: 过滤
const evens = arr.filter(n => n % 2 === 0);
console.log(evens);  // [2, 4]

// reduce: 归约
const sum = arr.reduce((acc, n) => acc + n, 0);
console.log(sum);  // 15

// reduceRight: 从右向左归约
const reversed = arr.reduceRight((acc, n) => [...acc, n], []);
console.log(reversed);  // [5, 4, 3, 2, 1]

// some: 存在判断
const hasEven = arr.some(n => n % 2 === 0);
console.log(hasEven);  // true

// every: 全部判断
const allPositive = arr.every(n => n > 0);
console.log(allPositive);  // true
```

### 数组方法 - 排序和反转

```javascript
const arr = [3, 1, 4, 1, 5, 9, 2, 6];

// sort: 排序（修改原数组）
arr.sort();
console.log(arr);  // [1, 1, 2, 3, 4, 5, 6, 9]

// 自定义排序
const nums = [10, 2, 30, 4, 50];
nums.sort((a, b) => a - b);  // 升序
console.log(nums);  // [2, 4, 10, 30, 50]

nums.sort((a, b) => b - a);  // 降序
console.log(nums);  // [50, 30, 10, 4, 2]

// 对象数组排序
const users = [
    { name: '张三', age: 25 },
    { name: '李四', age: 20 },
    { name: '王五', age: 30 }
];

users.sort((a, b) => a.age - b.age);
console.log(users);
// [{ name: '李四', age: 20 }, { name: '张三', age: 25 }, { name: '王五', age: 30 }]

// toSorted: 不修改原数组（ES2023）
const arr2 = [3, 1, 4, 1, 5];
const sorted = arr2.toSorted((a, b) => a - b);
console.log(arr2);    // [3, 1, 4, 1, 5]
console.log(sorted);  // [1, 1, 3, 4, 5]

// reverse: 反转（修改原数组）
const arr3 = [1, 2, 3, 4, 5];
arr3.reverse();
console.log(arr3);  // [5, 4, 3, 2, 1]

// toReversed: 不修改原数组（ES2023）
const arr4 = [1, 2, 3, 4, 5];
const reversed = arr4.toReversed();
console.log(arr4);      // [1, 2, 3, 4, 5]
console.log(reversed);  // [5, 4, 3, 2, 1]
```

### 数组方法 - 其他

```javascript
const arr = [1, 2, 3, 4, 5];

// concat: 合并
const merged = arr.concat([6, 7], [8, 9]);
console.log(merged);  // [1, 2, 3, 4, 5, 6, 7, 8, 9]

// slice: 截取
const sliced = arr.slice(1, 4);
console.log(sliced);  // [2, 3, 4]

// join: 转字符串
console.log(arr.join('-'));  // '1-2-3-4-5'

// flat: 扁平化
const nested = [1, [2, 3], [4, [5, 6]]];
console.log(nested.flat());     // [1, 2, 3, 4, [5, 6]]
console.log(nested.flat(2));    // [1, 2, 3, 4, 5, 6]
console.log(nested.flat(Infinity));  // 完全扁平化

// flatMap: 映射后扁平化
const sentences = ['Hello World', 'Good Morning'];
const words = sentences.flatMap(s => s.split(' '));
console.log(words);  // ['Hello', 'World', 'Good', 'Morning']

// copyWithin: 内部复制
const arr2 = [1, 2, 3, 4, 5];
arr2.copyWithin(0, 3);  // 从索引3开始复制到索引0
console.log(arr2);  // [4, 5, 3, 4, 5]

// with: 修改指定索引元素（ES2023）
const arr3 = [1, 2, 3, 4, 5];
const newArr = arr3.with(2, 'three');
console.log(arr3);    // [1, 2, 3, 4, 5]
console.log(newArr);  // [1, 2, 'three', 4, 5]
```

### 数组遍历

```javascript
const arr = [1, 2, 3, 4, 5];

// for 循环
for (let i = 0; i < arr.length; i++) {
    console.log(arr[i]);
}

// for...of（推荐）
for (const value of arr) {
    console.log(value);
}

// for...of 获取索引
for (const [index, value] of arr.entries()) {
    console.log(`${index}: ${value}`);
}

// while
let i = 0;
while (i < arr.length) {
    console.log(arr[i]);
    i++;
}

// 迭代器
const iterator = arr[Symbol.iterator]();
let result;
while (!(result = iterator.next()).done) {
    console.log(result.value);
}
```

## 小结

本节学习了：

1. **对象操作**：创建、属性操作、getter/setter、遍历
2. **数组操作**：创建、访问、添加/删除、查找、遍历
3. **常用方法**：map、filter、reduce、sort、find 等
4. **ES2023 新方法**：toSorted、toReversed、toSpliced、with

> **提示**：数组方法中，`push`、`pop`、`shift`、`unshift`、`splice`、`sort`、`reverse` 会修改原数组，而 `map`、`filter`、`slice`、`concat` 等会返回新数组。ES2023 新增的 `toSorted`、`toReversed`、`toSpliced`、`with` 方法不会修改原数组，推荐使用。
