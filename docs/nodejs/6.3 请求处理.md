# 6.3 请求处理

Express 提供了丰富的请求处理功能。本节将详细介绍请求对象、响应对象和请求处理最佳实践。

## 请求对象 (req)

### 请求属性

```javascript
app.get('/info', (req, res) => {
    // 基本信息
    console.log('方法:', req.method);          // GET
    console.log('URL:', req.url);              // /info?name=value
    console.log('路径:', req.path);            // /info
    console.log('协议:', req.protocol);        // http
    console.log('主机名:', req.hostname);      // localhost
    console.log('IP:', req.ip);                // ::1
    
    // 完整 URL
    console.log('完整 URL:', req.originalUrl); // /info?name=value
    
    // 查询参数
    console.log('查询参数:', req.query);       // { name: 'value' }
    
    // 请求头
    console.log('所有请求头:', req.headers);
    console.log('User-Agent:', req.get('User-Agent'));
    console.log('Content-Type:', req.get('Content-Type'));
    
    // Cookies（需要 cookie-parser）
    console.log('Cookies:', req.cookies);
    
    res.json({ message: '请求信息已打印' });
});
```

### 路由参数

```javascript
// 路径参数
app.get('/users/:id', (req, res) => {
    console.log(req.params);  // { id: '123' }
    res.json({ userId: req.params.id });
});

// 多个参数
app.get('/posts/:postId/comments/:commentId', (req, res) => {
    const { postId, commentId } = req.params;
    res.json({ postId, commentId });
});
```

### 请求体

```javascript
// 解析中间件
app.use(express.json());                           // JSON
app.use(express.urlencoded({ extended: true }));   // URL 编码
app.use(express.raw({ type: 'application/octet-stream', limit: '10mb' }));  // 原始数据

// 处理请求体
app.post('/users', (req, res) => {
    console.log('请求体:', req.body);
    res.json({ received: req.body });
});

// 原始数据
app.post('/upload', (req, res) => {
    console.log('原始数据:', req.body);
    res.send('收到原始数据');
});
```

### 文件上传

```javascript
const multer = require('multer');
const path = require('path');

// 配置存储
const storage = multer.diskStorage({
    destination: (req, file, cb) => {
        cb(null, 'uploads/');
    },
    filename: (req, file, cb) => {
        const uniqueSuffix = Date.now() + '-' + Math.round(Math.random() * 1E9);
        cb(null, file.fieldname + '-' + uniqueSuffix + path.extname(file.originalFilename));
    }
});

const upload = multer({
    storage: storage,
    limits: {
        fileSize: 5 * 1024 * 1024  // 5MB
    },
    fileFilter: (req, file, cb) => {
        const allowedTypes = /jpeg|jpg|png|gif/;
        const extname = allowedTypes.test(path.extname(file.originalname).toLowerCase());
        const mimetype = allowedTypes.test(file.mimetype);
        
        if (extname && mimetype) {
            cb(null, true);
        } else {
            cb(new Error('只允许图片文件'));
        }
    }
});

// 单文件上传
app.post('/upload/single', upload.single('file'), (req, res) => {
    console.log(req.file);
    res.json({ file: req.file });
});

// 多文件上传
app.post('/upload/multiple', upload.array('files', 10), (req, res) => {
    console.log(req.files);
    res.json({ files: req.files });
});

// 多字段上传
app.post('/upload/fields', upload.fields([
    { name: 'avatar', maxCount: 1 },
    { name: 'documents', maxCount: 5 }
]), (req, res) => {
    console.log(req.files);
    res.json({ files: req.files });
});
```

## 响应对象 (res)

### 响应方法

```javascript
// 发送字符串
app.get('/send', (req, res) => {
    res.send('Hello World');
});

// 发送 JSON
app.get('/json', (req, res) => {
    res.json({ message: 'JSON 响应', data: [] });
});

// 发送状态码
app.get('/status', (req, res) => {
    res.status(201).json({ created: true });
});

// 发送文件
app.get('/file', (req, res) => {
    const filePath = path.join(__dirname, 'public', 'document.pdf');
    res.sendFile(filePath);
});

// 下载文件
app.get('/download', (req, res) => {
    const filePath = path.join(__dirname, 'files', 'report.pdf');
    res.download(filePath, 'report.pdf', (err) => {
        if (err) {
            console.error('下载错误:', err);
        }
    });
});

// 重定向
app.get('/redirect', (req, res) => {
    res.redirect('/new-location');
    res.redirect(301, '/permanent');  // 永久重定向
    res.redirect('https://example.com');  // 外部重定向
});

// 渲染模板
app.get('/render', (req, res) => {
    res.render('index', { title: '首页', user: { name: '张三' } });
});
```

### 设置响应头

```javascript
app.get('/headers', (req, res) => {
    // 设置单个响应头
    res.set('X-Custom-Header', 'value');
    
    // 设置多个响应头
    res.set({
        'Content-Type': 'application/json',
        'Cache-Control': 'no-cache, no-store, must-revalidate',
        'Pragma': 'no-cache',
        'Expires': '0'
    });
    
    // CORS 头
    res.set({
        'Access-Control-Allow-Origin': '*',
        'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE',
        'Access-Control-Allow-Headers': 'Content-Type, Authorization'
    });
    
    res.json({ headers: '已设置' });
});

// Content-Type 快捷方法
app.get('/type', (req, res) => {
    res.type('json');  // application/json
    res.type('html');  // text/html
    res.type('png');   // image/png
    res.send('响应');
});
```

### Cookie 操作

```javascript
const cookieParser = require('cookie-parser');
app.use(cookieParser());

app.get('/set-cookie', (req, res) => {
    res.cookie('token', 'abc123', {
        httpOnly: true,
        secure: true,      // 仅 HTTPS
        sameSite: 'strict',
        maxAge: 7 * 24 * 60 * 60 * 1000  // 7 天
    });
    
    res.cookie('preferences', { theme: 'dark', lang: 'zh' }, {
        encode: JSON.stringify
    });
    
    res.send('Cookie 已设置');
});

app.get('/get-cookie', (req, res) => {
    console.log(req.cookies);
    res.json(req.cookies);
});

app.get('/clear-cookie', (req, res) => {
    res.clearCookie('token');
    res.send('Cookie 已清除');
});
```

## 中间件处理

### 请求处理链

```javascript
// 日志中间件
app.use((req, res, next) => {
    req.startTime = Date.now();
    console.log(`[${new Date().toISOString()}] ${req.method} ${req.url}`);
    next();
});

// 解析中间件
app.use(express.json());

// 认证中间件
app.use((req, res, next) => {
    const token = req.headers.authorization;
    if (token) {
        req.user = { id: 1, name: '张三' };  // 模拟解析
    }
    next();
});

// 路由处理
app.get('/profile', (req, res) => {
    if (!req.user) {
        return res.status(401).json({ error: '未登录' });
    }
    res.json(req.user);
});

// 响应时间中间件
app.use((req, res, next) => {
    const originalEnd = res.end;
    res.end = function(...args) {
        const duration = Date.now() - req.startTime;
        res.set('X-Response-Time', `${duration}ms`);
        originalEnd.apply(res, args);
    };
    next();
});
```

### 错误处理

```javascript
// 异步错误包装器
const asyncHandler = (fn) => (req, res, next) => {
    Promise.resolve(fn(req, res, next)).catch(next);
};

// 使用
app.get('/users/:id', asyncHandler(async (req, res) => {
    const user = await User.findById(req.params.id);
    if (!user) {
        throw new AppError('用户不存在', 404);
    }
    res.json(user);
}));

// 自定义错误类
class AppError extends Error {
    constructor(message, statusCode = 500) {
        super(message);
        this.statusCode = statusCode;
        this.isOperational = true;
        Error.captureStackTrace(this, this.constructor);
    }
}

// 全局错误处理
app.use((err, req, res, next) => {
    err.statusCode = err.statusCode || 500;
    err.status = err.status || 'error';
    
    if (process.env.NODE_ENV === 'development') {
        res.status(err.statusCode).json({
            status: err.status,
            message: err.message,
            stack: err.stack
        });
    } else {
        // 生产环境不暴露错误详情
        res.status(err.statusCode).json({
            status: err.status,
            message: err.isOperational ? err.message : '服务器错误'
        });
    }
});
```

## 数据验证

### 使用 Joi 验证

```javascript
const Joi = require('joi');

// 验证中间件
function validateBody(schema) {
    return (req, res, next) => {
        const { error, value } = schema.validate(req.body);
        if (error) {
            return res.status(400).json({
                error: '验证失败',
                details: error.details.map(d => d.message)
            });
        }
        req.body = value;
        next();
    };
}

function validateParams(schema) {
    return (req, res, next) => {
        const { error, value } = schema.validate(req.params);
        if (error) {
            return res.status(400).json({
                error: '参数无效',
                details: error.details.map(d => d.message)
            });
        }
        req.params = value;
        next();
    };
}

function validateQuery(schema) {
    return (req, res, next) => {
        const { error, value } = schema.validate(req.query);
        if (error) {
            return res.status(400).json({
                error: '查询参数无效',
                details: error.details.map(d => d.message)
            });
        }
        req.query = value;
        next();
    };
}

// 定义验证规则
const userSchemas = {
    create: Joi.object({
        name: Joi.string().min(2).max(50).required(),
        email: Joi.string().email().required(),
        password: Joi.string().min(8).pattern(/^(?=.*[A-Za-z])(?=.*\d)/).required(),
        age: Joi.number().integer().min(0).max(150)
    }),
    update: Joi.object({
        name: Joi.string().min(2).max(50),
        email: Joi.string().email(),
        age: Joi.number().integer().min(0).max(150)
    }).min(1),
    id: Joi.object({
        id: Joi.string().pattern(/^[0-9a-fA-F]{24}$/).required()
    })
};

// 使用验证
router.post('/',
    validateBody(userSchemas.create),
    userController.create
);

router.put('/:id',
    validateParams(userSchemas.id),
    validateBody(userSchemas.update),
    userController.update
);
```

## 小结

本节学习了：

1. **请求对象**：属性、参数、请求体、文件上传
2. **响应对象**：send、json、status、redirect、文件下载
3. **响应头设置**：set、type、CORS
4. **Cookie 操作**：设置、获取、清除
5. **中间件处理**：请求链、错误处理
6. **数据验证**：Joi 验证中间件

> **提示**：良好的请求处理需要考虑安全性（验证、过滤）、用户体验（错误信息）和性能（响应时间）。推荐使用验证中间件统一处理数据验证。
