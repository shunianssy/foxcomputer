# 1.1 为什么选择 Node.js？

Node.js 是一个基于 Chrome V8 引擎的 JavaScript 运行时环境，让 JavaScript 可以脱离浏览器在服务端运行。它由 Ryan Dahl 于 2009 年创建，如今已成为最受欢迎的服务端技术之一。

## Node.js 的核心特点

### 事件驱动与非阻塞 I/O

Node.js 采用事件驱动、非阻塞 I/O 模型，使其轻量且高效：

```javascript
// 非阻塞文件读取示例
const fs = require('fs');

// 异步读取文件，不会阻塞后续代码执行
fs.readFile('example.txt', 'utf8', (err, data) => {
    if (err) {
        console.error('读取文件出错:', err);
        return;
    }
    console.log('文件内容:', data);
});

// 这行代码会立即执行，不需要等待文件读取完成
console.log('这条消息会先显示');
```

### 单线程事件循环

Node.js 采用单线程模型，通过事件循环处理并发请求：

```javascript
// 事件循环示例
console.log('1. 开始');

// 设置一个定时器（异步操作）
setTimeout(() => {
    console.log('2. 定时器回调');
}, 0);

// Promise 也是异步操作
Promise.resolve().then(() => {
    console.log('3. Promise 回调');
});

console.log('4. 结束');

// 输出顺序: 1 -> 4 -> 3 -> 2
```

### 跨平台支持

Node.js 支持 Windows、macOS、Linux 等主流操作系统，一次编写，处处运行。

### 丰富的 npm 生态系统

npm（Node Package Manager）是世界上最大的软件注册表，拥有超过 200 万个包：

```bash
# 查看当前项目依赖
npm list

# 查看全局安装的包
npm list -g --depth=0

# 搜索包
npm search express

# 查看包信息
npm view express
```

## Node.js 的应用领域

### Web 服务器与 API 服务

```javascript
// 创建一个简单的 HTTP 服务器
const http = require('http');

const server = http.createServer((req, res) => {
    // 设置响应头
    res.writeHead(200, { 'Content-Type': 'application/json' });
    
    // 返回 JSON 数据
    res.end(JSON.stringify({
        message: 'Hello, Node.js!',
        timestamp: new Date().toISOString()
    }));
});

server.listen(3000, () => {
    console.log('服务器运行在 http://localhost:3000');
});
```

### 实时应用

```javascript
// 使用 Socket.io 创建实时聊天应用
const express = require('express');
const http = require('http');
const { Server } = require('socket.io');

const app = express();
const server = http.createServer(app);
const io = new Server(server);

// 处理客户端连接
io.on('connection', (socket) => {
    console.log('用户已连接');
    
    // 监听聊天消息
    socket.on('chat message', (msg) => {
        // 广播消息给所有客户端
        io.emit('chat message', msg);
    });
    
    // 监听断开连接
    socket.on('disconnect', () => {
        console.log('用户已断开');
    });
});

server.listen(3000, () => {
    console.log('聊天服务器运行在 http://localhost:3000');
});
```

### 命令行工具

```javascript
// 创建命令行工具示例
#!/usr/bin/env node

const fs = require('fs');
const path = require('path');

// 获取命令行参数
const args = process.argv.slice(2);
const command = args[0];

switch (command) {
    case 'list':
        // 列出当前目录文件
        const files = fs.readdirSync(process.cwd());
        console.log('当前目录文件:');
        files.forEach(file => console.log(`  - ${file}`));
        break;
    case 'info':
        // 显示文件信息
        const filePath = args[1];
        if (filePath && fs.existsSync(filePath)) {
            const stats = fs.statSync(filePath);
            console.log(`文件大小: ${stats.size} 字节`);
            console.log(`创建时间: ${stats.birthtime}`);
        }
        break;
    default:
        console.log('用法: mycli [list|info] [文件路径]');
}
```

### 微服务架构

```javascript
// 微服务示例：用户服务
const express = require('express');
const app = express();

app.use(express.json());

// 模拟用户数据库
const users = [
    { id: 1, name: '张三', email: 'zhangsan@example.com' },
    { id: 2, name: '李四', email: 'lisi@example.com' }
];

// 获取所有用户
app.get('/users', (req, res) => {
    res.json(users);
});

// 获取单个用户
app.get('/users/:id', (req, res) => {
    const user = users.find(u => u.id === parseInt(req.params.id));
    if (!user) {
        return res.status(404).json({ error: '用户不存在' });
    }
    res.json(user);
});

// 创建用户
app.post('/users', (req, res) => {
    const newUser = {
        id: users.length + 1,
        ...req.body
    };
    users.push(newUser);
    res.status(201).json(newUser);
});

app.listen(3001, () => {
    console.log('用户服务运行在端口 3001');
});
```

## Node.js 的优势

### 前后端统一语言

使用 JavaScript 可以同时开发前端和后端：

```javascript
// 前端代码（浏览器）
fetch('/api/users')
    .then(res => res.json())
    .then(data => console.log(data));

// 后端代码（Node.js）
app.get('/api/users', async (req, res) => {
    const users = await User.findAll();
    res.json(users);
});
```

### 高并发处理能力

Node.js 非常适合处理大量并发连接：

```javascript
// 处理并发请求示例
const express = require('express');
const app = express();

// 模拟异步数据库查询
async function queryDatabase(id) {
    return new Promise((resolve) => {
        setTimeout(() => {
            resolve({ id, data: `数据 ${id}` });
        }, 100);
    });
}

// 可以同时处理大量请求
app.get('/data/:id', async (req, res) => {
    const result = await queryDatabase(req.params.id);
    res.json(result);
});

app.listen(3000);
```

### JSON 原生支持

JavaScript 对 JSON 有原生支持，非常适合构建 RESTful API：

```javascript
// JSON 处理示例
const data = {
    name: 'Node.js 教程',
    version: '1.0.0',
    tags: ['javascript', 'nodejs', 'backend']
};

// 序列化
const jsonString = JSON.stringify(data, null, 2);
console.log(jsonString);

// 反序列化
const parsed = JSON.parse(jsonString);
console.log(parsed.name);
```

## 小结

选择 Node.js 的理由：

1. **高性能**：基于 V8 引擎，执行速度快
2. **高并发**：事件驱动模型，适合 I/O 密集型应用
3. **全栈开发**：前后端使用同一语言，降低学习成本
4. **生态丰富**：npm 拥有海量第三方包
5. **社区活跃**：文档丰富，问题易解决
6. **就业前景好**：市场需求大，薪资可观

> **提示**：Node.js 特别适合构建实时应用、API 服务、单页应用后端、微服务等场景。如果你的项目涉及大量 I/O 操作或需要高并发处理，Node.js 是一个绝佳的选择。
