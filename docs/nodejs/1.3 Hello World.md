# 1.3 Hello World

本节将带你创建第一个 Node.js 程序，并介绍 Node.js 程序的基本运行方式。

## 第一个 Node.js 程序

### 方式一：直接运行

创建一个名为 `hello.js` 的文件：

```javascript
// hello.js - 第一个 Node.js 程序

// 使用 console.log 输出内容到控制台
console.log('Hello, World!');
console.log('欢迎来到 Node.js 世界！');

// 输出当前时间
console.log('当前时间:', new Date().toLocaleString('zh-CN'));

// 输出 Node.js 版本信息
console.log('Node.js 版本:', process.version);
```

在终端中运行：

```bash
node hello.js
```

输出结果：

```
Hello, World!
欢迎来到 Node.js 世界！
当前时间: 2024/1/15 10:30:00
Node.js 版本: v20.10.0
```

### 方式二：交互式运行（REPL）

REPL（Read-Eval-Print-Loop）是 Node.js 提供的交互式运行环境：

```bash
# 启动 REPL
node
```

进入 REPL 后，可以直接输入 JavaScript 代码：

```javascript
// REPL 交互示例
> console.log('Hello, World!')
Hello, World!
undefined

> 1 + 2
3

> const name = 'Node.js'
undefined

> `Hello, ${name}`
'Hello, Node.js'

> .exit  // 退出 REPL
```

REPL 常用快捷键：

| 快捷键 | 功能 |
|--------|------|
| Ctrl + C (两次) | 退出 REPL |
| Ctrl + D | 退出 REPL |
| Tab | 自动补全 |
| 上/下箭头 | 历史命令 |
| .help | 显示帮助 |
| .clear | 清除上下文 |
| .save filename | 保存会话到文件 |
| .load filename | 加载文件到会话 |

## 创建 HTTP 服务器

Node.js 最常见的用途是创建 Web 服务器：

```javascript
// server.js - 创建 HTTP 服务器
const http = require('http');

// 创建服务器
const server = http.createServer((request, response) => {
    // 输出请求信息
    console.log(`收到请求: ${request.method} ${request.url}`);
    
    // 设置响应头
    response.writeHead(200, {
        'Content-Type': 'text/html; charset=utf-8'
    });
    
    // 发送响应内容
    response.end(`
        <!DOCTYPE html>
        <html lang="zh-CN">
        <head>
            <meta charset="UTF-8">
            <title>Node.js 服务器</title>
        </head>
        <body>
            <h1>Hello, World!</h1>
            <p>这是我的第一个 Node.js 服务器</p>
            <p>当前时间: ${new Date().toLocaleString('zh-CN')}</p>
        </body>
        </html>
    `);
});

// 设置端口和主机
const PORT = 3000;
const HOST = 'localhost';

// 启动服务器
server.listen(PORT, HOST, () => {
    console.log(`服务器运行在 http://${HOST}:${PORT}`);
    console.log('按 Ctrl+C 停止服务器');
});
```

运行服务器：

```bash
node server.js
```

然后在浏览器中访问 `http://localhost:3000`。

## 命令行参数

Node.js 程序可以接收命令行参数：

```javascript
// args.js - 命令行参数处理
// process.argv 包含所有命令行参数

// 打印所有参数
console.log('所有参数:', process.argv);

// 第一个参数是 Node.js 可执行文件路径
// 第二个参数是脚本文件路径
// 后面是用户传入的参数

// 获取用户参数（从索引 2 开始）
const userArgs = process.argv.slice(2);

console.log('用户参数:', userArgs);

// 解析命名参数
function parseArgs(args) {
    const result = {};
    
    for (let i = 0; i < args.length; i++) {
        const arg = args[i];
        
        // 检查是否是命名参数（以 -- 开头）
        if (arg.startsWith('--')) {
            const key = arg.slice(2);
            const value = args[i + 1] && !args[i + 1].startsWith('--') 
                ? args[i + 1] 
                : true;
            result[key] = value;
        }
    }
    
    return result;
}

const params = parseArgs(userArgs);
console.log('解析后的参数:', params);

// 使用示例: node args.js --name 张三 --age 25 --verbose
```

运行示例：

```bash
node args.js --name 张三 --age 25 --verbose
```

## 使用 npm 脚本运行

创建 `package.json` 文件来管理项目：

```bash
# 初始化项目
npm init -y
```

修改 `package.json`：

```json
{
    "name": "hello-nodejs",
    "version": "1.0.0",
    "description": "Node.js 入门示例",
    "main": "hello.js",
    "scripts": {
        "start": "node hello.js",
        "server": "node server.js",
        "dev": "nodemon server.js"
    },
    "keywords": ["nodejs", "tutorial"],
    "author": "",
    "license": "MIT"
}
```

运行 npm 脚本：

```bash
# 运行 start 脚本
npm start

# 运行 server 脚本
npm run server

# 安装 nodemon 并运行开发模式
npm install -D nodemon
npm run dev
```

## 环境变量

使用环境变量配置应用程序：

```javascript
// env.js - 环境变量使用示例

// 访问所有环境变量
console.log('所有环境变量:', process.env);

// 访问特定环境变量
console.log('系统路径 (PATH):', process.env.PATH);
console.log('用户目录 (HOME):', process.env.HOME || process.env.USERPROFILE);
console.log('系统平台:', process.env.OS);

// 自定义环境变量
// 可以通过命令行设置: set PORT=8080 && node env.js (Windows)
// 或: PORT=8080 node env.js (macOS/Linux)

const PORT = process.env.PORT || 3000;
const NODE_ENV = process.env.NODE_ENV || 'development';

console.log(`服务器端口: ${PORT}`);
console.log(`运行环境: ${NODE_ENV}`);
```

使用 `.env` 文件管理环境变量：

```bash
# 安装 dotenv 包
npm install dotenv
```

创建 `.env` 文件：

```ini
# .env 文件
PORT=8080
NODE_ENV=development
DB_HOST=localhost
DB_PORT=3306
DB_USER=root
DB_PASS=secret
```

在代码中使用：

```javascript
// config.js - 使用 dotenv 加载环境变量
require('dotenv').config();

const config = {
    port: parseInt(process.env.PORT) || 3000,
    env: process.env.NODE_ENV || 'development',
    database: {
        host: process.env.DB_HOST || 'localhost',
        port: parseInt(process.env.DB_PORT) || 3306,
        user: process.env.DB_USER || 'root',
        password: process.env.DB_PASS || ''
    }
};

console.log('配置信息:', config);

module.exports = config;
```

## 读取用户输入

从命令行读取用户输入：

```javascript
// input.js - 读取用户输入
const readline = require('readline');

// 创建 readline 接口
const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout
});

// 提问并获取输入
rl.question('请输入你的名字: ', (name) => {
    console.log(`你好, ${name}!`);
    
    rl.question('请输入你的年龄: ', (age) => {
        console.log(`你今年 ${age} 岁`);
        
        // 关闭接口
        rl.close();
    });
});

// 监听关闭事件
rl.on('close', () => {
    console.log('程序结束');
    process.exit(0);
});
```

## 完整示例：命令行计算器

```javascript
// calculator.js - 命令行计算器
const readline = require('readline');

const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout
});

// 显示菜单
function showMenu() {
    console.log('\n=== 简易计算器 ===');
    console.log('1. 加法');
    console.log('2. 减法');
    console.log('3. 乘法');
    console.log('4. 除法');
    console.log('0. 退出');
    console.log('==================');
}

// 计算函数
function calculate(operation, a, b) {
    switch (operation) {
        case '1':
            return a + b;
        case '2':
            return a - b;
        case '3':
            return a * b;
        case '4':
            if (b === 0) {
                return '错误: 除数不能为零';
            }
            return a / b;
        default:
            return '无效的操作';
    }
}

// 主循环
function main() {
    showMenu();
    
    rl.question('请选择操作 (0-4): ', (choice) => {
        if (choice === '0') {
            console.log('感谢使用，再见！');
            rl.close();
            return;
        }
        
        if (!['1', '2', '3', '4'].includes(choice)) {
            console.log('无效的选择，请重新输入');
            main();
            return;
        }
        
        rl.question('请输入第一个数字: ', (num1) => {
            rl.question('请输入第二个数字: ', (num2) => {
                const a = parseFloat(num1);
                const b = parseFloat(num2);
                
                if (isNaN(a) || isNaN(b)) {
                    console.log('请输入有效的数字');
                } else {
                    const result = calculate(choice, a, b);
                    console.log(`结果: ${result}`);
                }
                
                main(); // 继续循环
            });
        });
    });
}

// 启动程序
main();
```

## 小结

本节学习了：

1. **运行 Node.js 程序**：使用 `node filename.js` 命令
2. **REPL 交互环境**：直接执行 JavaScript 代码
3. **创建 HTTP 服务器**：使用 `http` 模块
4. **命令行参数**：通过 `process.argv` 获取
5. **环境变量**：使用 `process.env` 和 `dotenv`
6. **用户输入**：使用 `readline` 模块

> **提示**：建议在开发时使用 `nodemon` 工具，它可以监听文件变化并自动重启服务器，大大提高开发效率。安装命令：`npm install -g nodemon`
