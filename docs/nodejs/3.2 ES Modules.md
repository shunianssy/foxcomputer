# 3.2 ES Modules

ES Modules（ESM）是 ECMAScript 标准定义的模块系统，是现代 JavaScript 的官方模块规范。本节将详细介绍 ES Modules 的使用方法。

## 启用 ES Modules

### 方式一：使用 .mjs 扩展名

```javascript
// math.mjs
export const add = (a, b) => a + b;
export const subtract = (a, b) => a - b;

// main.mjs
import { add, subtract } from './math.mjs';
console.log(add(1, 2));  // 3
```

### 方式二：在 package.json 中设置 type

```json
{
    "name": "my-project",
    "version": "1.0.0",
    "type": "module"
}
```

```javascript
// 现在 .js 文件会被当作 ES Modules 处理
// math.js
export const add = (a, b) => a + b;

// main.js
import { add } from './math.js';  // 必须包含 .js 扩展名
```

### 方式三：使用双扩展名

```text
project/
├── package.json
├── src/
│   ├── index.cjs    # CommonJS 模块
│   └── utils.mjs    # ES Module 模块
```

## 导出语法

### 命名导出

```javascript
// utils.js

// 导出单个值
export const PI = 3.14159;

export function add(a, b) {
    return a + b;
}

export function subtract(a, b) {
    return a - b;
}

export class Calculator {
    multiply(a, b) {
        return a * b;
    }
}

// 先定义后导出
const NAME = 'MyModule';
const VERSION = '1.0.0';

function greet(name) {
    return `Hello, ${name}!`;
}

export { NAME, VERSION, greet };

// 导出时重命名
export {
    add as addition,
    subtract as subtraction
};

// 重新导出
export { add, subtract } from './math.js';
export * from './constants.js';
export * as constants from './constants.js';  // 导出为命名空间
```

### 默认导出

```javascript
// 每个模块只能有一个默认导出

// 导出函数
export default function(a, b) {
    return a + b;
}

// 导出类
export default class User {
    constructor(name) {
        this.name = name;
    }
}

// 导出对象
export default {
    name: 'MyModule',
    version: '1.0.0'
};

// 先定义后导出
const config = {
    apiUrl: 'https://api.example.com',
    timeout: 5000
};

export default config;

// 同时有命名导出和默认导出
export const PI = 3.14159;
export default function calculate(r) {
    return PI * r * r;
}
```

## 导入语法

### 命名导入

```javascript
// 导入特定成员
import { add, subtract } from './math.js';

console.log(add(1, 2));       // 3
console.log(subtract(5, 3));  // 2

// 导入时重命名
import { add as addition, subtract as subtraction } from './math.js';

console.log(addition(1, 2));  // 3

// 导入所有命名导出
import * as math from './math.js';

console.log(math.add(1, 2));      // 3
console.log(math.subtract(5, 3)); // 2
console.log(math.PI);             // 3.14159
```

### 默认导入

```javascript
// 导入默认导出
import add from './math.js';

console.log(add(1, 2));  // 3

// 同时导入默认导出和命名导出
import calculate, { PI } from './math.js';

console.log(calculate(5));  // 78.53975
console.log(PI);            // 3.14159

// 重命名默认导入
import { default as add, PI } from './math.js';
```

### 动态导入

```javascript
// 动态导入返回 Promise
async function loadModule() {
    const math = await import('./math.js');
    console.log(math.add(1, 2));  // 3
}

// 条件导入
if (condition) {
    import('./feature.js')
        .then(module => module.init())
        .catch(err => console.error(err));
}

// 按需加载
document.getElementById('loadButton').addEventListener('click', async () => {
    const { default: HeavyModule } = await import('./heavy-module.js');
    const instance = new HeavyModule();
    instance.doSomething();
});
```

### 仅执行模块

```javascript
// 导入模块但不使用其导出
import './polyfills.js';
import './styles.css';

// 常用于 polyfill、CSS 导入等场景
```

## ES Modules 与 CommonJS 互操作

### 在 ES Modules 中导入 CommonJS

```javascript
// commonjs-module.cjs
module.exports = {
    add: (a, b) => a + b,
    subtract: (a, b) => a - b
};

// es-module.mjs
// CommonJS 模块的 module.exports 作为默认导出
import cjsModule from './commonjs-module.cjs';
console.log(cjsModule.add(1, 2));  // 3

// 或者使用命名空间导入
import * as cjsModule from './commonjs-module.cjs';
```

### 在 CommonJS 中导入 ES Modules

```javascript
// es-module.mjs
export const add = (a, b) => a + b;
export default function multiply(a, b) {
    return a * b;
}

// commonjs-module.cjs
// 必须使用动态导入
async function main() {
    const esModule = await import('./es-module.mjs');
    console.log(esModule.add(1, 2));         // 3
    console.log(esModule.default(2, 3));     // 6
}

main();
```

## ES Modules 特性

### 严格模式

```javascript
// ES Modules 默认启用严格模式
// 无需 'use strict'

// 以下行为会报错：
// this = undefined（不是全局对象）
// 未声明的变量赋值
// 重复的参数名
// 八进制字面量（如 010）
```

### 顶层 await

```javascript
// ES2022+ 支持顶层 await
// 无需包装在 async 函数中

const response = await fetch('/api/config');
const config = await response.json();

export default config;

// 条件加载
let module;
if (process.env.NODE_ENV === 'production') {
    module = await import('./production.js');
} else {
    module = await import('./development.js');
}

export default module;
```

### import.meta

```javascript
// import.meta 包含模块元信息

// 获取模块 URL
console.log(import.meta.url);
// file:///home/user/project/main.js

// 获取模块目录
import { fileURLToPath } from 'url';
import { dirname } from 'path';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

console.log(__filename);
console.log(__dirname);

// 检查是否是主模块
if (import.meta.url === `file://${process.argv[1]}`) {
    console.log('这是主模块');
}

// 动态导入相对路径
const modulePath = new URL('./utils.js', import.meta.url);
const utils = await import(modulePath);
```

## 模块解析规则

### 相对路径

```javascript
// 必须包含文件扩展名
import { add } from './math.js';      // 正确
import { add } from './math';         // 错误

// 目录导入需要指定入口文件
import { something } from './module/index.js';

// 或在 package.json 中指定 exports
```

### 裸模块名

```javascript
// 导入 node_modules 中的模块
import express from 'express';
import { readFile } from 'fs/promises';

// Node.js 内置模块
import fs from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';
```

### package.json exports 字段

```json
{
    "name": "my-package",
    "version": "1.0.0",
    "type": "module",
    "exports": {
        ".": "./src/index.js",
        "./utils": "./src/utils.js",
        "./constants": "./src/constants.js"
    }
}
```

```javascript
// 使用方式
import myPackage from 'my-package';
import { helper } from 'my-package/utils';
import { API_URL } from 'my-package/constants';
```

### 配置模块

```javascript
// config/index.js
import { readFile } from 'fs/promises';
import { fileURLToPath } from 'url';
import { join, dirname } from 'path';

const __dirname = dirname(fileURLToPath(import.meta.url));

async function loadConfig() {
    const env = process.env.NODE_ENV || 'development';
    const configPath = join(__dirname, 'environments', `${env}.json`);
    
    try {
        const content = await readFile(configPath, 'utf8');
        return JSON.parse(content);
    } catch {
        return getDefaultConfig();
    }
}

function getDefaultConfig() {
    return {
        port: 3000,
        database: {
            host: 'localhost',
            port: 3306
        }
    };
}

export default await loadConfig();
```

### 服务模块

```javascript
// services/user.service.js
import { query } from '../database/index.js';
import { logger } from '../logger/index.js';

export class UserService {
    static async findAll() {
        const sql = 'SELECT * FROM users';
        const users = await query(sql);
        logger.info('获取所有用户', { count: users.length });
        return users;
    }
    
    static async findById(id) {
        const sql = 'SELECT * FROM users WHERE id = ?';
        const [user] = await query(sql, [id]);
        return user || null;
    }
    
    static async create(userData) {
        const sql = 'INSERT INTO users SET ?';
        const result = await query(sql, [userData]);
        logger.info('创建用户', { id: result.insertId });
        return this.findById(result.insertId);
    }
    
    static async update(id, userData) {
        const sql = 'UPDATE users SET ? WHERE id = ?';
        await query(sql, [userData, id]);
        return this.findById(id);
    }
    
    static async delete(id) {
        const sql = 'DELETE FROM users WHERE id = ?';
        await query(sql, [id]);
        logger.info('删除用户', { id });
    }
}

export default UserService;
```

## 小结

本节学习了：

1. **启用 ES Modules**：.mjs 扩展名、package.json type 字段
2. **导出语法**：命名导出、默认导出、重新导出
3. **导入语法**：命名导入、默认导入、动态导入

> **提示**：ES Modules 是现代 JavaScript 的标准模块系统，推荐在新项目中使用。对于 npm 包，建议同时提供 ESM 和 CommonJS 两种格式，以兼容不同的使用场景。
