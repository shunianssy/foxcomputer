# 7.3 Redis 缓存

Redis 是高性能的内存数据库，常用于缓存、会话存储等场景。本节将介绍如何在 Node.js 中使用 Redis。

## 安装配置

### 安装 ioredis

```bash
npm install ioredis
```

### 连接 Redis

```javascript
const Redis = require('ioredis');

// 创建连接
const redis = new Redis({
    host: 'localhost',
    port: 6379,
    password: '',
    db: 0
});

// 连接事件
redis.on('connect', () => {
    console.log('Redis 连接成功');
});

redis.on('error', (err) => {
    console.error('Redis 错误:', err);
});

redis.on('close', () => {
    console.log('Redis 连接关闭');
});

// 集群模式
const cluster = new Redis.Cluster([
    { host: '127.0.0.1', port: 7000 },
    { host: '127.0.0.1', port: 7001 },
    { host: '127.0.0.1', port: 7002 }
]);
```

## 基本操作

### 字符串操作

```javascript
// 设置值
await redis.set('key', 'value');
await redis.set('key', 'value', 'EX', 3600);  // 过期时间 1 小时
await redis.set('key', 'value', 'PX', 3600000);  // 毫秒
await redis.set('key', 'value', 'NX');  // 不存在才设置
await redis.set('key', 'value', 'XX');  // 存在才设置

// 获取值
const value = await redis.get('key');

// 删除
await redis.del('key');

// 检查存在
const exists = await redis.exists('key');

// 设置过期时间
await redis.expire('key', 3600);  // 秒
await redis.pexpire('key', 3600000);  // 毫秒

// 获取剩余时间
const ttl = await redis.ttl('key');

// 自增/自减
await redis.incr('counter');
await redis.incrby('counter', 10);
await redis.decr('counter');
await redis.decrby('counter', 5);

// 追加
await redis.append('key', 'suffix');

// 获取长度
const len = await redis.strlen('key');
```

### 哈希操作

```javascript
// 设置字段
await redis.hset('user:1', 'name', '张三');
await redis.hset('user:1', 'age', 25);
await redis.hset('user:1', { name: '张三', age: 25, email: 'test@example.com' });

// 获取字段
const name = await redis.hget('user:1', 'name');
const user = await redis.hgetall('user:1');  // 获取所有字段

// 删除字段
await redis.hdel('user:1', 'age');

// 检查字段存在
const exists = await redis.hexists('user:1', 'name');

// 获取所有字段名
const keys = await redis.hkeys('user:1');

// 获取所有值
const values = await redis.hvals('user:1');

// 获取字段数量
const count = await redis.hlen('user:1');

// 字段自增
await redis.hincrby('user:1', 'age', 1);
```

### 列表操作

```javascript
// 左侧插入
await redis.lpush('list', 'a', 'b', 'c');

// 右侧插入
await redis.rpush('list', 'd', 'e');

// 左侧弹出
const left = await redis.lpop('list');

// 右侧弹出
const right = await redis.rpop('list');

// 获取列表长度
const len = await redis.llen('list');

// 获取范围内的元素
const items = await redis.lrange('list', 0, -1);  // 所有元素
const first10 = await redis.lrange('list', 0, 9);  // 前 10 个

// 获取指定索引的元素
const item = await redis.lindex('list', 0);

// 设置指定索引的元素
await redis.lset('list', 0, 'new value');

// 删除指定元素
await redis.lrem('list', 1, 'value');  // 删除 1 个

// 阻塞弹出
const result = await redis.blpop('list', 5);  // 等待 5 秒
```

### 集合操作

```javascript
// 添加元素
await redis.sadd('set', 'a', 'b', 'c');

// 删除元素
await redis.srem('set', 'a');

// 检查元素存在
const isMember = await redis.sismember('set', 'a');

// 获取所有元素
const members = await redis.smembers('set');

// 获取元素数量
const count = await redis.scard('set');

// 随机获取元素
const member = await redis.srandmember('set');

// 随机弹出元素
const popped = await redis.spop('set');

// 集合运算
const union = await redis.sunion('set1', 'set2');
const intersect = await redis.sinter('set1', 'set2');
const diff = await redis.sdiff('set1', 'set2');
```

### 有序集合操作

```javascript
// 添加元素
await redis.zadd('leaderboard', 100, 'user1');
await redis.zadd('leaderboard', 200, 'user2');
await redis.zadd('leaderboard', 150, 'user3');

// 获取分数
const score = await redis.zscore('leaderboard', 'user1');

// 获取排名（从 0 开始）
const rank = await redis.zrank('leaderboard', 'user1');  // 升序
const revRank = await redis.zrevrank('leaderboard', 'user1');  // 降序

// 获取范围内的元素
const top10 = await redis.zrevrange('leaderboard', 0, 9, 'WITHSCORES');

// 获取分数范围内的元素
const range = await redis.zrangebyscore('leaderboard', 100, 200);

// 增加分数
await redis.zincrby('leaderboard', 50, 'user1');

// 删除元素
await redis.zrem('leaderboard', 'user1');

// 获取元素数量
const count = await redis.zcard('leaderboard');
```

## 缓存封装

```javascript
class CacheService {
    constructor(redis) {
        this.redis = redis;
        this.defaultTTL = 3600;  // 默认 1 小时
    }
    
    // 获取缓存
    async get(key) {
        const data = await this.redis.get(key);
        if (!data) return null;
        
        try {
            return JSON.parse(data);
        } catch {
            return data;
        }
    }
    
    // 设置缓存
    async set(key, value, ttl = this.defaultTTL) {
        const data = typeof value === 'string' ? value : JSON.stringify(value);
        await this.redis.set(key, data, 'EX', ttl);
    }
    
    // 删除缓存
    async del(key) {
        await this.redis.del(key);
    }
    
    // 批量删除
    async delPattern(pattern) {
        const keys = await this.redis.keys(pattern);
        if (keys.length > 0) {
            await this.redis.del(keys);
        }
    }
    
    // 获取或设置
    async getOrSet(key, fetchFn, ttl = this.defaultTTL) {
        const cached = await this.get(key);
        
        if (cached !== null) {
            return cached;
        }
        
        const data = await fetchFn();
        await this.set(key, data, ttl);
        
        return data;
    }
    
    // 哈希缓存
    async hget(key, field) {
        const data = await this.redis.hget(key, field);
        return data ? JSON.parse(data) : null;
    }
    
    async hset(key, field, value, ttl) {
        await this.redis.hset(key, field, JSON.stringify(value));
        if (ttl) {
            await this.redis.expire(key, ttl);
        }
    }
}

module.exports = CacheService;
```

## 小结

本节学习了：

1. **连接配置**：单机和集群模式
2. **数据类型**：字符串、哈希、列表、集合、有序集合
3. **缓存封装**：getOrSet、批量删除
4. **应用场景**：会话存储、缓存、排行榜

> **提示**：Redis 是高性能缓存的首选，合理设置过期时间可以避免内存溢出。使用缓存时注意缓存穿透、缓存击穿、缓存雪崩等问题。
