# 6.4 模板引擎

Express 支持多种模板引擎，本节将介绍 EJS 模板引擎的使用方法。

## 安装配置

### 安装 EJS

```bash
npm install ejs
```

### 配置 Express

```javascript
const express = require('express');
const path = require('path');
const app = express();

// 设置模板引擎
app.set('view engine', 'ejs');

// 设置模板目录（默认是 views）
app.set('views', path.join(__dirname, 'views'));

// 开启模板缓存（生产环境）
app.set('view cache', process.env.NODE_ENV === 'production');
```

## 基本语法

### 输出内容

```ejs
<!-- views/index.ejs -->
<!DOCTYPE html>
<html>
<head>
    <title><%= title %></title>
</head>
<body>
    <!-- 转义输出 -->
    <h1><%= message %></h1>
    
    <!-- 不转义输出（注意 XSS 风险） -->
    <div><%- htmlContent %></div>
    
    <!-- 原始文本 -->
    <%_ 去除前面的空白 _%>
    <%_ 去除后面的空白 -%>
</body>
</html>
```

### 控制流程

```ejs
<!-- 条件判断 -->
<% if (user) { %>
    <p>欢迎, <%= user.name %></p>
<% } else { %>
    <p>请登录</p>
<% } %>

<!-- 循环 -->
<ul>
<% users.forEach(function(user) { %>
    <li><%= user.name %> - <%= user.email %></li>
<% }); %>
</ul>

<!-- for 循环 -->
<% for (let i = 0; i < items.length; i++) { %>
    <div><%= items[i] %></div>
<% } %>
```

### 包含文件

```ejs
<!-- 包含其他模板 -->
<%- include('header', { title: title }) %>

<main>
    <h1><%= title %></h1>
</main>

<%- include('footer') %>
```

## 渲染模板

### 基本渲染

```javascript
// 渲染模板
app.get('/', (req, res) => {
    res.render('index', {
        title: '首页',
        message: 'Hello World'
    });
});

// 渲染并回调
app.get('/with-callback', (req, res) => {
    res.render('index', { title: '首页' }, (err, html) => {
        if (err) {
            console.error(err);
            return res.status(500).send('渲染错误');
        }
        // 可以修改 html 后发送
        res.send(html);
    });
});
```

### 渲染选项

```javascript
app.get('/options', (req, res) => {
    res.render('index', {
        title: '首页',
        // 自定义分隔符
        delimiter: '?',
        // 调试模式
        debug: true,
        // 本地变量
        _locals: {
            appName: 'My App'
        }
    });
});
```

## 模板继承

### 布局模板

```ejs
<!-- views/layout.ejs -->
<!DOCTYPE html>
<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title><%= title %> - <%= siteName %></title>
    <link rel="stylesheet" href="/css/style.css">
    <%- include('partials/head') %>
</head>
<body>
    <%- include('partials/header') %>
    
    <main class="container">
        <%- body %>
    </main>
    
    <%- include('partials/footer') %>
    
    <script src="/js/main.js"></script>
    <%- include('partials/scripts') %>
</body>
</html>
```

### 使用布局

```javascript
const express = require('express');
const expressLayouts = require('express-ejs-layouts');
const app = express();

app.set('view engine', 'ejs');
app.use(expressLayouts);
app.set('layout', 'layout');  // 默认布局

// 使用布局
app.get('/', (req, res) => {
    res.render('index', { title: '首页' });
});

// 不使用布局
app.get('/no-layout', (req, res) => {
    res.render('page', { layout: false });
});

// 使用不同布局
app.get('/admin', (req, res) => {
    res.render('admin/dashboard', { layout: 'admin-layout' });
});
```

## 部分模板

### 头部组件

```ejs
<!-- views/partials/header.ejs -->
<header class="header">
    <nav class="nav">
        <a href="/" class="logo"><%= siteName %></a>
        <ul class="nav-menu">
            <li><a href="/">首页</a></li>
            <li><a href="/about">关于</a></li>
            <li><a href="/contact">联系</a></li>
        </ul>
        <div class="nav-user">
            <% if (user) { %>
                <span><%= user.name %></span>
                <a href="/logout">退出</a>
            <% } else { %>
                <a href="/login">登录</a>
            <% } %>
        </div>
    </nav>
</header>
```

### 消息提示

```ejs
<!-- views/partials/messages.ejs -->
<% if (success && success.length > 0) { %>
<div class="alert alert-success">
    <% success.forEach(msg => { %>
        <p><%= msg %></p>
    <% }) %>
</div>
<% } %>

<% if (error && error.length > 0) { %>
<div class="alert alert-error">
    <% error.forEach(msg => { %>
        <p><%= msg %></p>
    <% }) %>
</div>
<% } %>
```

### 分页组件

```ejs
<!-- views/partials/pagination.ejs -->
<% if (totalPages > 1) { %>
<div class="pagination">
    <% if (currentPage > 1) { %>
        <a href="?page=<%= currentPage - 1 %>" class="prev">上一页</a>
    <% } %>
    
    <% for (let i = 1; i <= totalPages; i++) { %>
        <% if (i === currentPage) { %>
            <span class="active"><%= i %></span>
        <% } else { %>
            <a href="?page=<%= i %>"><%= i %></a>
        <% } %>
    <% } %>
    
    <% if (currentPage < totalPages) { %>
        <a href="?page=<%= currentPage + 1 %>" class="next">下一页</a>
    <% } %>
</div>
<% } %>
```

## 自定义函数

### 添加本地变量

```javascript
// 中间件添加全局变量
app.use((req, res, next) => {
    res.locals.siteName = 'My Website';
    res.locals.currentPath = req.path;
    res.locals.user = req.user || null;
    res.locals.formatDate = (date) => {
        return new Date(date).toLocaleDateString('zh-CN');
    };
    next();
});

// 使用
app.get('/', (req, res) => {
    res.render('index', { title: '首页' });
});
```

### 模板中使用

```ejs
<!-- 使用全局变量 -->
<p>网站名称: <%= siteName %></p>
<p>当前路径: <%= currentPath %></p>

<!-- 使用自定义函数 -->
<p>日期: <%= formatDate(post.createdAt) %></p>
```

## 静态资源

```javascript
// 配置静态文件目录
app.use(express.static(path.join(__dirname, 'public')));

// 多个静态目录
app.use(express.static(path.join(__dirname, 'public')));
app.use('/uploads', express.static(path.join(__dirname, 'uploads')));

// 虚拟路径
app.use('/static', express.static(path.join(__dirname, 'public')));
```

```ejs
<!-- 引用静态资源 -->
<link rel="stylesheet" href="/css/style.css">
<script src="/js/main.js"></script>
<img src="/images/logo.png" alt="Logo">

<!-- 虚拟路径 -->
<link rel="stylesheet" href="/static/css/style.css">
```

## 完整示例

### 项目结构

```text
views/
├── layout.ejs
├── index.ejs
├── about.ejs
├── users/
│   ├── index.ejs
│   ├── show.ejs
│   └── edit.ejs
└── partials/
    ├── head.ejs
    ├── header.ejs
    ├── footer.ejs
    ├── messages.ejs
    └── pagination.ejs
```

### 应用代码

```javascript
const express = require('express');
const expressLayouts = require('express-ejs-layouts');
const path = require('path');
const app = express();

// 配置
app.set('view engine', 'ejs');
app.set('views', path.join(__dirname, 'views'));
app.use(expressLayouts);
app.set('layout', 'layout');

// 静态文件
app.use(express.static(path.join(__dirname, 'public')));

// 全局变量
app.use((req, res, next) => {
    res.locals.siteName = 'My Website';
    res.locals.formatDate = (date) => {
        return new Date(date).toLocaleDateString('zh-CN');
    };
    next();
});

// 路由
app.get('/', (req, res) => {
    res.render('index', {
        title: '首页',
        posts: [
            { id: 1, title: '文章1', createdAt: new Date() },
            { id: 2, title: '文章2', createdAt: new Date() }
        ]
    });
});

app.get('/users', async (req, res) => {
    const users = await User.findAll();
    res.render('users/index', {
        title: '用户列表',
        users,
        success: req.flash('success')
    });
});

app.get('/users/:id', async (req, res) => {
    const user = await User.findById(req.params.id);
    res.render('users/show', {
        title: user.name,
        user
    });
});

app.listen(3000);
```

## 小结

本节学习了：

1. **安装配置**：设置 EJS 模板引擎
2. **基本语法**：输出、条件、循环
3. **包含文件**：include 部分模板
4. **模板继承**：布局模板
5. **部分模板**：组件化开发
6. **自定义函数**：本地变量和辅助函数
7. **静态资源**：CSS、JS、图片

> **提示**：EJS 是简单易用的模板引擎，适合服务端渲染场景。对于更复杂的需求，可以考虑 Pug、Handlebars 或 Nunjucks 等其他模板引擎。
