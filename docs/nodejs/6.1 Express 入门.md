# 6.1 Express 入门

Express 是 Node.js 最流行的 Web 框架，提供了丰富的 HTTP 工具和中间件支持。本节将介绍 Express 的基本使用方法。

## 安装与初始化

### 安装 Express

```bash
# 创建项目
mkdir my-express-app
cd my-express-app
npm init -y

# 安装 Express
npm install express

# 安装开发依赖
npm install -D nodemon
```

### 基本应用

```javascript
const express = require('express');

// 创建应用
const app = express();

// 定义端口
const PORT = process.env.PORT || 3000;

// 定义路由
app.get('/', (req, res) => {
    res.send('Hello World');
});

// 启动服务器
app.listen(PORT, () => {
    console.log(`服务器运行在 http://localhost:${PORT}`);
});
```

## 路由基础

### 基本路由

```javascript
const express = require('express');
const app = express();

// GET 请求
app.get('/', (req, res) => {
    res.send('GET 请求');
});

// POST 请求
app.post('/', (req, res) => {
    res.send('POST 请求');
});

// PUT 请求
app.put('/', (req, res) => {
    res.send('PUT 请求');
});

// DELETE 请求
app.delete('/', (req, res) => {
    res.send('DELETE 请求');
});

// 匹配所有 HTTP 方法
app.all('/all', (req, res) => {
    res.send(`所有方法: ${req.method}`);
});

// 启动服务器
app.listen(3000);
```

### 路由参数

```javascript
// 路径参数
app.get('/users/:id', (req, res) => {
    console.log(req.params);  // { id: '123' }
    res.send(`用户 ID: ${req.params.id}`);
});

// 多个路径参数
app.get('/users/:userId/posts/:postId', (req, res) => {
    const { userId, postId } = req.params;
    res.send(`用户 ${userId} 的文章 ${postId}`);
});

// 可选参数
app.get('/books/:id?', (req, res) => {
    if (req.params.id) {
        res.send(`书籍 ID: ${req.params.id}`);
    } else {
        res.send('所有书籍');
    }
});

// 正则表达式
app.get('/users/:id(\\d+)', (req, res) => {
    res.send(`数字 ID: ${req.params.id}`);
});
```

### 查询参数

```javascript
app.get('/search', (req, res) => {
    console.log(req.query);  // { q: 'keyword', page: '1' }
    
    const { q, page = 1, limit = 10 } = req.query;
    
    res.json({
        query: q,
        page: parseInt(page),
        limit: parseInt(limit)
    });
});

// 访问: /search?q=keyword&page=1&limit=20
```

### 请求体解析

```javascript
// 解析 JSON 请求体
app.use(express.json());

// 解析 URL 编码请求体
app.use(express.urlencoded({ extended: true }));

// POST 请求处理
app.post('/users', (req, res) => {
    console.log(req.body);  // 请求体数据
    
    const { name, email } = req.body;
    
    res.status(201).json({
        message: '用户创建成功',
        user: { name, email }
    });
});
```

## 响应方法

```javascript
// 发送字符串
app.get('/string', (req, res) => {
    res.send('字符串响应');
});

// 发送 JSON
app.get('/json', (req, res) => {
    res.json({ message: 'JSON 响应' });
});

// 发送状态码
app.get('/status', (req, res) => {
    res.status(201).json({ created: true });
});

// 发送文件
app.get('/file', (req, res) => {
    res.sendFile('/path/to/file.pdf');
});

// 下载文件
app.get('/download', (req, res) => {
    res.download('/path/to/file.pdf', 'report.pdf');
});

// 重定向
app.get('/redirect', (req, res) => {
    res.redirect('/new-url');
    res.redirect(301, '/permanent-redirect');
});

// 设置响应头
app.get('/headers', (req, res) => {
    res.set('X-Custom-Header', 'value');
    res.set({
        'Cache-Control': 'no-cache',
        'Content-Type': 'application/json'
    });
    res.json({ headers: 'set' });
});

// Cookie 操作
app.get('/cookie', (req, res) => {
    // 设置 Cookie
    res.cookie('token', 'abc123', {
        httpOnly: true,
        maxAge: 7 * 24 * 60 * 60 * 1000  // 7 天
    });
    
    // 清除 Cookie
    // res.clearCookie('token');
    
    res.send('Cookie 已设置');
});
```

## 路由模块化

### 创建路由模块

```javascript
// routes/users.js
const express = require('express');
const router = express.Router();

// 获取所有用户
router.get('/', (req, res) => {
    res.json([{ id: 1, name: '张三' }]);
});

// 获取单个用户
router.get('/:id', (req, res) => {
    res.json({ id: req.params.id, name: '张三' });
});

// 创建用户
router.post('/', (req, res) => {
    const user = req.body;
    res.status(201).json(user);
});

// 更新用户
router.put('/:id', (req, res) => {
    res.json({ id: req.params.id, ...req.body });
});

// 删除用户
router.delete('/:id', (req, res) => {
    res.status(204).send();
});

module.exports = router;
```

### 使用路由模块

```javascript
// app.js
const express = require('express');
const usersRouter = require('./routes/users');

const app = express();

app.use(express.json());

// 挂载路由
app.use('/api/users', usersRouter);

app.listen(3000);
```

## 中间件

### 应用级中间件

```javascript
const express = require('express');
const app = express();

// 日志中间件
app.use((req, res, next) => {
    console.log(`${new Date().toISOString()} ${req.method} ${req.url}`);
    next();
});

// 静态文件
app.use(express.static('public'));

// 解析请求体
app.use(express.json());
app.use(express.urlencoded({ extended: true }));
```

### 路由级中间件

```javascript
// 路由级中间件
const authMiddleware = (req, res, next) => {
    const token = req.headers.authorization;
    
    if (!token) {
        return res.status(401).json({ error: '未授权' });
    }
    
    req.user = { id: 1, name: '张三' };
    next();
};

// 应用到特定路由
app.get('/profile', authMiddleware, (req, res) => {
    res.json(req.user);
});

// 应用到路由组
app.use('/admin', authMiddleware);
app.get('/admin/dashboard', (req, res) => {
    res.json({ message: '管理面板' });
});
```

### 错误处理中间件

```javascript
// 错误处理中间件（必须放在最后）
app.use((err, req, res, next) => {
    console.error('错误:', err);
    
    const statusCode = err.statusCode || 500;
    const message = err.message || '服务器内部错误';
    
    res.status(statusCode).json({
        error: {
            message,
            ...(process.env.NODE_ENV === 'development' && { stack: err.stack })
        }
    });
});

// 自定义错误类
class AppError extends Error {
    constructor(message, statusCode) {
        super(message);
        this.statusCode = statusCode;
        this.name = 'AppError';
    }
}

// 使用
app.get('/error', (req, res, next) => {
    try {
        throw new AppError('出错了', 400);
    } catch (error) {
        next(error);
    }
});

// 异步错误处理
const asyncHandler = (fn) => (req, res, next) => {
    Promise.resolve(fn(req, res, next)).catch(next);
};

app.get('/async-error', asyncHandler(async (req, res) => {
    const user = await User.findById(req.params.id);
    if (!user) {
        throw new AppError('用户不存在', 404);
    }
    res.json(user);
}));
```

## 完整示例

```javascript
const express = require('express');
const path = require('path');

const app = express();

// 配置
app.set('port', process.env.PORT || 3000);
app.set('env', process.env.NODE_ENV || 'development');

// 中间件
app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.use(express.static(path.join(__dirname, 'public')));

// 请求日志
app.use((req, res, next) => {
    const start = Date.now();
    res.on('finish', () => {
        const duration = Date.now() - start;
        console.log(`${req.method} ${req.url} ${res.statusCode} ${duration}ms`);
    });
    next();
});

// 路由
app.get('/', (req, res) => {
    res.json({ message: 'Welcome to Express API' });
});

app.get('/health', (req, res) => {
    res.json({ status: 'ok', timestamp: new Date().toISOString() });
});

// API 路由
const apiRouter = express.Router();

apiRouter.get('/users', (req, res) => {
    res.json([{ id: 1, name: '张三' }]);
});

apiRouter.post('/users', (req, res) => {
    const user = { id: Date.now(), ...req.body };
    res.status(201).json(user);
});

app.use('/api', apiRouter);

// 404 处理
app.use((req, res) => {
    res.status(404).json({ error: 'Not Found' });
});

// 错误处理
app.use((err, req, res, next) => {
    console.error(err);
    res.status(err.statusCode || 500).json({
        error: err.message || 'Internal Server Error'
    });
});

// 启动服务器
app.listen(app.get('port'), () => {
    console.log(`服务器运行在 http://localhost:${app.get('port')}`);
});
```
