# 2.1 变量与数据类型

JavaScript 是一门动态类型语言，变量可以在运行时持有任何类型的值。本节将详细介绍 JavaScript 中的变量声明方式和基本数据类型。

## 变量声明

### var、let 和 const 的区别

```javascript
// var - 函数作用域，存在变量提升
var name = '张三';
var name = '李四';  // 允许重复声明
console.log(name);  // 李四

// let - 块级作用域，不存在变量提升
let age = 25;
// let age = 30;  // 错误：不允许重复声明
age = 30;  // 允许重新赋值
console.log(age);  // 30

// const - 块级作用域，声明时必须初始化，不能重新赋值
const PI = 3.14159;
// PI = 3.14;  // 错误：不能重新赋值
// const E;   // 错误：声明时必须初始化

// const 对象的属性可以修改
const user = { name: '张三', age: 25 };
user.age = 26;  // 允许修改属性
user.email = 'zhangsan@example.com';  // 允许添加属性
// user = {};  // 错误：不能重新赋值整个对象

// const 数组的元素可以修改
const numbers = [1, 2, 3];
numbers.push(4);  // 允许添加元素
numbers[0] = 0;   // 允许修改元素
// numbers = [];  // 错误：不能重新赋值
```

### 变量提升

```javascript
// var 存在变量提升
console.log(a);  // undefined（不是错误）
var a = 10;

// 等价于：
var a;
console.log(a);
a = 10;

// let 和 const 不存在变量提升
// console.log(b);  // 错误：Cannot access 'b' before initialization
let b = 20;

// 暂时性死区（Temporal Dead Zone）
{
    // TDZ 开始
    // console.log(x);  // 错误
    let x = 10;  // TDZ 结束
    console.log(x);  // 10
}
```

### 最佳实践

```javascript
// 1. 默认使用 const
const API_URL = 'https://api.example.com';
const MAX_RETRIES = 3;

// 2. 需要重新赋值时使用 let
let count = 0;
let isLoading = false;

// 3. 避免使用 var
// var oldStyle = '不推荐';  // 不推荐

// 4. 变量命名规范
let userName = '张三';        // 小驼峰命名
let MAX_SIZE = 100;          // 常量使用大写
let _privateVar = '私有';    // 私有变量前缀下划线
let $element = document.getElementById('app');  // DOM 元素前缀 $
```

## 基本数据类型

JavaScript 有 7 种基本数据类型（原始类型）：

### 1. Number（数字）

```javascript
// 整数
let integer = 42;
let negative = -10;
let binary = 0b1010;      // 二进制：10
let octal = 0o777;        // 八进制：511
let hex = 0xFF;           // 十六进制：255

// 浮点数
let float = 3.14;
let scientific = 2.5e6;   // 科学计数法：2500000

// 特殊值
let infinity = Infinity;        // 正无穷
let negInfinity = -Infinity;    // 负无穷
let notANumber = NaN;           // 非数字

// 检查 NaN
console.log(Number.isNaN(NaN));      // true
console.log(Number.isNaN('hello'));  // false
console.log(isNaN('hello'));         // true（会先转换）

// 检查有限数
console.log(Number.isFinite(42));       // true
console.log(Number.isFinite(Infinity)); // false

// 数字转换
console.log(Number('42'));       // 42
console.log(Number('42px'));     // NaN
console.log(parseInt('42px'));   // 42
console.log(parseInt('42', 16)); // 66（十六进制）
console.log(parseFloat('3.14')); // 3.14

// 数字方法
let num = 3.14159;
console.log(num.toFixed(2));     // "3.14"（字符串）
console.log(num.toPrecision(4)); // "3.142"（字符串）
console.log(Math.round(num));    // 3（四舍五入）
console.log(Math.floor(num));    // 3（向下取整）
console.log(Math.ceil(num));     // 4（向上取整）

// 安全整数范围
console.log(Number.MAX_SAFE_INTEGER);  // 9007199254740991
console.log(Number.MIN_SAFE_INTEGER);  // -9007199254740991
console.log(Number.isSafeInteger(9007199254740991));  // true
console.log(Number.isSafeInteger(9007199254740992));  // false

// 处理大整数（ES2020+）
let bigNumber = 9007199254740993n;  // BigInt
console.log(bigNumber + 1n);        // 9007199254740994n
```

### 2. String（字符串）

```javascript
// 字符串创建
let single = '单引号字符串';
let double = "双引号字符串";
let template = `模板字符串`;

// 模板字符串
let name = '张三';
let age = 25;
let greeting = `你好，我是${name}，今年${age}岁`;
console.log(greeting);  // 你好，我是张三，今年25岁

// 多行字符串
let multiLine = `
    第一行
    第二行
    第三行
`;

// 标签模板
function tag(strings, ...values) {
    console.log(strings);  // ['Hello ', '!']
    console.log(values);   // ['World']
    return strings[0] + values[0] + strings[1];
}
let result = tag`Hello ${'World'}!`;  // Hello World!

// 字符串方法
let str = 'Hello, World!';

console.log(str.length);              // 13
console.log(str.charAt(0));           // 'H'
console.log(str[0]);                  // 'H'
console.log(str.charCodeAt(0));       // 72（ASCII 码）
console.log(str.indexOf('o'));        // 4
console.log(str.lastIndexOf('o'));    // 8
console.log(str.includes('World'));   // true
console.log(str.startsWith('Hello')); // true
console.log(str.endsWith('!'));       // true

console.log(str.slice(0, 5));         // 'Hello'
console.log(str.substring(0, 5));     // 'Hello'
console.log(str.substr(0, 5));        // 'Hello'（已废弃）

console.log(str.toLowerCase());       // 'hello, world!'
console.log(str.toUpperCase());       // 'HELLO, WORLD!'

console.log(str.trim());              // 去除首尾空格
console.log(str.trimStart());         // 去除开头空格
console.log(str.trimEnd());           // 去除结尾空格

console.log(str.split(', '));         // ['Hello', 'World!']
console.log(str.replace('World', 'Node.js'));  // 'Hello, Node.js!'
console.log(str.replaceAll('o', '0')); // 'Hell0, W0rld!'

console.log(str.repeat(2));           // 'Hello, World!Hello, World!'
console.log(str.padStart(20, '*'));   // '*******Hello, World!'
console.log(str.padEnd(20, '*'));     // 'Hello, World!*******'

// 字符串遍历
for (let char of 'Hello') {
    console.log(char);
}

// Unicode 支持
console.log('\u{1F600}');  // 😀（ES6+ Unicode 码点）
console.log('😀'.length);  // 2（代理对）
console.log([... '😀'].length);  // 1（正确计数）
```

### 3. Boolean（布尔值）

```javascript
// 布尔值
let isTrue = true;
let isFalse = false;

// 布尔转换
console.log(Boolean(1));          // true
console.log(Boolean(0));          // false
console.log(Boolean(''));         // false
console.log(Boolean('hello'));    // true
console.log(Boolean(null));       // false
console.log(Boolean(undefined));  // false
console.log(Boolean({}));         // true
console.log(Boolean([]));         // true

// 假值（Falsy Values）
const falsyValues = [false, 0, -0, 0n, '', null, undefined, NaN];

// 真值（Truthy Values）
// 除了假值以外的所有值都是真值

// 逻辑运算符
console.log(true && false);   // false（与）
console.log(true || false);   // true（或）
console.log(!true);           // false（非）

// 短路求值
let value = null;
let result = value || '默认值';  // '默认值'
let result2 = value && value.name;  // null

// 空值合并运算符（??）
let x = null;
let y = undefined;
let z = 0;
console.log(x ?? 'default');  // 'default'
console.log(y ?? 'default');  // 'default'
console.log(z ?? 'default');  // 0（与 || 的区别）

// 可选链操作符（?.）
let user = { profile: { name: '张三' } };
console.log(user?.profile?.name);    // '张三'
console.log(user?.address?.city);    // undefined（不会报错）
```

### 4. undefined 和 null

```javascript
// undefined - 未定义
let a;
console.log(a);  // undefined
console.log(typeof a);  // 'undefined'

// null - 空值
let b = null;
console.log(b);  // null
console.log(typeof b);  // 'object'（历史遗留问题）

// 区别
console.log(undefined == null);   // true
console.log(undefined === null);  // false

// 检查 undefined
let x;
if (x === undefined) {
    console.log('x 未定义');
}

// 检查 null
let y = null;
if (y === null) {
    console.log('y 为 null');
}

// 使用场景
function greet(name) {
    // 参数未传递时使用默认值
    name = name || '访客';
    console.log(`你好，${name}`);
}

function findUser(id) {
    // 找不到用户时返回 null
    if (id < 0) {
        return null;
    }
    return { id, name: '用户' + id };
}
```

### 5. Symbol（符号）

```javascript
// 创建 Symbol
let sym1 = Symbol();
let sym2 = Symbol('描述');
let sym3 = Symbol('描述');

console.log(sym2 === sym3);  // false（每个 Symbol 都是唯一的）

// 全局 Symbol 注册表
let globalSym1 = Symbol.for('key');
let globalSym2 = Symbol.for('key');
console.log(globalSym1 === globalSym2);  // true

// 获取 Symbol 描述
console.log(sym2.description);  // '描述'
console.log(Symbol.keyFor(globalSym1));  // 'key'

// 作为对象属性键
let obj = {
    [Symbol('name')]: '张三',
    [Symbol('age')]: 25
};

// Symbol 属性不会被常规方法枚举
console.log(Object.keys(obj));  // []
console.log(Object.getOwnPropertySymbols(obj));  // [Symbol(name), Symbol(age)]

// 常用内置 Symbol
let array = [1, 2, 3];
console.log(array[Symbol.iterator]);  // 迭代器方法

// 自定义迭代器
let range = {
    from: 1,
    to: 5,
    [Symbol.iterator]() {
        return {
            current: this.from,
            last: this.to,
            next() {
                if (this.current <= this.last) {
                    return { done: false, value: this.current++ };
                } else {
                    return { done: true };
                }
            }
        };
    }
};

for (let num of range) {
    console.log(num);  // 1, 2, 3, 4, 5
}
```

### 6. BigInt（大整数）

```javascript
// 创建 BigInt
let big1 = 9007199254740993n;
let big2 = BigInt(9007199254740993);
let big3 = BigInt('9007199254740993');

console.log(big1 === big2);  // true

// BigInt 运算
console.log(10n + 20n);   // 30n
console.log(10n - 5n);    // 5n
console.log(10n * 3n);    // 30n
console.log(10n / 3n);    // 3n（向下取整）
console.log(10n % 3n);    // 1n
console.log(2n ** 100n);  // 大数幂运算

// 比较运算
console.log(10n > 5);     // true
console.log(10n === 10);  // false（类型不同）
console.log(10n == 10);   // true

// 类型转换
console.log(Number(10n));   // 10
console.log(String(10n));   // '10'
console.log(Boolean(0n));   // false
console.log(Boolean(1n));   // true

// 注意：BigInt 不能与普通数字混合运算
// console.log(10n + 5);  // 错误
console.log(10n + BigInt(5));  // 15n
```

## 类型检测

### typeof 操作符

```javascript
console.log(typeof 42);           // 'number'
console.log(typeof 'hello');      // 'string'
console.log(typeof true);         // 'boolean'
console.log(typeof undefined);    // 'undefined'
console.log(typeof null);         // 'object'（历史遗留问题）
console.log(typeof {});           // 'object'
console.log(typeof []);           // 'object'
console.log(typeof function(){}); // 'function'
console.log(typeof Symbol());     // 'symbol'
console.log(typeof 10n);          // 'bigint'
```

### instanceof 操作符

```javascript
// 检查原型链
let arr = [1, 2, 3];
console.log(arr instanceof Array);   // true
console.log(arr instanceof Object);  // true

let date = new Date();
console.log(date instanceof Date);   // true
console.log(date instanceof Object); // true

// 自定义类
class Person {}
let person = new Person();
console.log(person instanceof Person);  // true
```

### 其他检测方法

```javascript
// Array.isArray()
console.log(Array.isArray([1, 2, 3]));  // true
console.log(Array.isArray({}));         // false

// Object.prototype.toString
function getType(value) {
    return Object.prototype.toString.call(value).slice(8, -1);
}

console.log(getType(42));        // 'Number'
console.log(getType('hello'));   // 'String'
console.log(getType(true));      // 'Boolean'
console.log(getType(null));      // 'Null'

console.log(getType([]));        // 'Array'
console.log(getType({}));        // 'Object'
console.log(getType(new Date())); // 'Date'
console.log(getType(/regex/));   // 'RegExp'

// Number.isNaN() vs isNaN()
console.log(Number.isNaN(NaN));       // true
console.log(Number.isNaN('hello'));   // false
console.log(isNaN('hello'));          // true（会先转换）

// Number.isFinite() vs isFinite()
console.log(Number.isFinite(42));      // true
console.log(Number.isFinite('42'));    // false
console.log(isFinite('42'));           // true（会先转换）
```

## 类型转换

### 显式转换

```javascript
// 转换为字符串
console.log(String(42));           // '42'
console.log(String(true));         // 'true'
console.log(String(null));         // 'null'
console.log(String(undefined));    // 'undefined'
console.log(String([1, 2, 3]));    // '1,2,3'
console.log(String({}));           // '[object Object]'
console.log((42).toString());      // '42'
console.log((42).toFixed(2));      // '42.00'

// 转换为数字
console.log(Number('42'));         // 42
console.log(Number('42px'));       // NaN
console.log(Number(true));         // 1
console.log(Number(false));        // 0
console.log(Number(null));         // 0
console.log(Number(undefined));    // NaN
console.log(parseInt('42px'));     // 42
console.log(parseFloat('3.14px')); // 3.14
console.log(+'42');                // 42（一元加号）

// 转换为布尔值
console.log(Boolean(1));           // true
console.log(Boolean(0));           // false
console.log(Boolean(''));          // false
console.log(Boolean('hello'));     // true
console.log(Boolean(null));        // false
console.log(Boolean(undefined));   // false
console.log(!!1);                  // true（双重否定）
```

### 隐式转换

```javascript
// 字符串拼接
console.log('1' + 2);        // '12'
console.log(1 + '2');        // '12'
console.log(1 + 2 + '3');    // '33'
console.log('1' + 2 + 3);    // '123'

// 算术运算
console.log('5' - 3);        // 2
console.log('5' * '2');      // 10
console.log('10' / '2');     // 5
console.log('10' % '3');     // 1

// 比较运算
console.log('5' == 5);       // true（类型转换后比较）
console.log('5' === 5);      // false（严格比较，不转换）
console.log(null == undefined);  // true
console.log(null === undefined); // false

// 对象转原始值
let obj = {
    valueOf() { return 42; },
    toString() { return 'hello'; }
};
console.log(obj + 8);        // 50（优先使用 valueOf）
console.log(String(obj));    // 'hello'（使用 toString）
```

## 小结

本节学习了：

1. **变量声明**：var、let、const 的区别和最佳实践
2. **基本数据类型**：Number、String、Boolean、undefined、null、Symbol、BigInt
3. **类型检测**：typeof、instanceof、Array.isArray 等
4. **类型转换**：显式转换和隐式转换

> **提示**：在实际开发中，推荐使用 `===` 严格相等运算符，避免隐式类型转换带来的意外结果。使用 `const` 声明不会改变的值，使用 `let` 声明需要重新赋值的变量。
