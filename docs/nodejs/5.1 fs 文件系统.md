# 5.1 fs 文件系统

fs 模块是 Node.js 最常用的核心模块之一，用于文件的读写操作。本节将详细介绍 fs 模块的使用方法。

## 同步与异步 API

### API 命名规则

```javascript
const fs = require('fs');

// 异步回调形式
fs.readFile('file.txt', 'utf8', (err, data) => {
    if (err) throw err;
    console.log(data);
});

// 同步形式（以 Sync 结尾）
const data = fs.readFileSync('file.txt', 'utf8');
console.log(data);

// Promise 形式（fs.promises）
const fsPromises = require('fs/promises');

async function readFile() {
    const data = await fsPromises.readFile('file.txt', 'utf8');
    console.log(data);
}
```

## 文件读取

### readFile

```javascript
const fs = require('fs');
const fsPromises = require('fs/promises');

// 异步回调
fs.readFile('file.txt', 'utf8', (err, data) => {
    if (err) {
        console.error('读取失败:', err);
        return;
    }
    console.log(data);
});

// 同步
try {
    const data = fs.readFileSync('file.txt', 'utf8');
    console.log(data);
} catch (error) {
    console.error('读取失败:', error);
}

// Promise
async function readFileAsync() {
    try {
        const data = await fsPromises.readFile('file.txt', 'utf8');
        console.log(data);
    } catch (error) {
        console.error('读取失败:', error);
    }
}

// 读取二进制文件
fs.readFile('image.png', (err, buffer) => {
    if (err) throw err;
    console.log(buffer);  // Buffer 对象
    console.log(buffer.length);  // 文件大小
});
```

### 逐行读取

```javascript
const fs = require('fs');
const readline = require('readline');

async function processLineByLine() {
    const fileStream = fs.createReadStream('large-file.txt');
    
    const rl = readline.createInterface({
        input: fileStream,
        crlfDelay: Infinity
    });
    
    for await (const line of rl) {
        console.log(`行: ${line}`);
    }
}

processLineByLine();
```

## 文件写入

### writeFile

```javascript
const fs = require('fs');
const fsPromises = require('fs/promises');

// 异步回调
fs.writeFile('output.txt', 'Hello World', (err) => {
    if (err) throw err;
    console.log('写入成功');
});

// 同步
fs.writeFileSync('output.txt', 'Hello World');

// Promise
await fsPromises.writeFile('output.txt', 'Hello World');

// 写入 JSON
const data = { name: '张三', age: 25 };
await fsPromises.writeFile('data.json', JSON.stringify(data, null, 2));

// 追加内容
await fsPromises.appendFile('log.txt', '新的日志\n');

// 写入 Buffer
const buffer = Buffer.from('Hello World');
await fsPromises.writeFile('binary.bin', buffer);
```

### 文件描述符

```javascript
const fs = require('fs/promises');

async function workWithFileDescriptor() {
    let fd;
    
    try {
        // 打开文件
        fd = await fs.open('file.txt', 'r+');  // 读写模式
        
        // 读取
        const buffer = Buffer.alloc(100);
        const { bytesRead } = await fd.read(buffer, 0, 100, 0);
        console.log(`读取 ${bytesRead} 字节`);
        console.log(buffer.toString('utf8', 0, bytesRead));
        
        // 写入
        await fd.write(Buffer.from('Hello'), 0, 5, 0);
        
        // 同步文件
        await fd.sync();
        
    } finally {
        if (fd) {
            await fd.close();
        }
    }
}

// 文件打开模式
// 'r'   - 只读
// 'r+'  - 读写
// 'w'   - 只写，创建或截断
// 'w+'  - 读写，创建或截断
// 'a'   - 追加
// 'a+'  - 读写追加
// 'ax'  - 排他创建
```

## 目录操作

```javascript
const fs = require('fs/promises');
const path = require('path');

// 创建目录
await fs.mkdir('new-dir');
await fs.mkdir('nested/dir', { recursive: true });  // 递归创建

// 读取目录
const files = await fs.readdir('directory');
console.log(files);  // ['file1.txt', 'file2.txt', 'subdir']

// 带详细信息
const filesWithDetails = await fs.readdir('directory', { withFileTypes: true });
filesWithDetails.forEach(file => {
    console.log(`${file.name}: ${file.isDirectory() ? '目录' : '文件'}`);
});

// 删除空目录
await fs.rmdir('empty-dir');

// 删除非空目录
await fs.rm('non-empty-dir', { recursive: true, force: true });

// 检查目录是否存在
try {
    await fs.access('directory');
    console.log('目录存在');
} catch {
    console.log('目录不存在');
}

// 遍历目录树
async function walkDir(dir) {
    const files = await fs.readdir(dir, { withFileTypes: true });
    
    for (const file of files) {
        const filePath = path.join(dir, file.name);
        
        if (file.isDirectory()) {
            await walkDir(filePath);
        } else {
            console.log(filePath);
        }
    }
}
```

## 文件信息

```javascript
const fs = require('fs/promises');

// 获取文件状态
async function getFileInfo() {
    const stats = await fs.stat('file.txt');
    
    console.log('是文件:', stats.isFile());
    console.log('是目录:', stats.isDirectory());
    console.log('大小:', stats.size, '字节');
    console.log('创建时间:', stats.birthtime);
    console.log('修改时间:', stats.mtime);
    console.log('访问时间:', stats.atime);
    console.log('权限:', stats.mode.toString(8));
    console.log('所有者:', stats.uid);
    console.log('组:', stats.gid);
}

// 获取符号链接信息
const linkStats = await fs.lstat('symlink');
console.log('是符号链接:', linkStats.isSymbolicLink());

// 检查文件是否存在
try {
    await fs.access('file.txt', fs.constants.F_OK);
    console.log('文件存在');
} catch {
    console.log('文件不存在');
}

// 检查读写权限
try {
    await fs.access('file.txt', fs.constants.R_OK | fs.constants.W_OK);
    console.log('文件可读写');
} catch {
    console.log('文件不可读写');
}
```

## 文件操作

```javascript
const fs = require('fs/promises');
const path = require('path');

// 复制文件
await fs.copyFile('source.txt', 'destination.txt');
await fs.copyFile('source.txt', 'destination.txt', fs.constants.COPYFILE_EXCL);  // 目标存在则失败

// 移动/重命名文件
await fs.rename('old-name.txt', 'new-name.txt');
await fs.rename('file.txt', 'subdir/file.txt');  // 移动到子目录

// 删除文件
await fs.unlink('file.txt');

// 截断文件
await fs.truncate('file.txt', 100);  // 保留前 100 字节

// 修改权限
await fs.chmod('file.txt', 0o644);  // rw-r--r--
await fs.chown('file.txt', 1000, 1000);  // uid, gid

// 创建符号链接
await fs.symlink('target.txt', 'link.txt');
await fs.symlink('target-dir', 'link-dir', 'dir');

// 读取符号链接
const target = await fs.readlink('link.txt');
console.log('链接目标:', target);

// 创建硬链接
await fs.link('source.txt', 'hardlink.txt');
```

## 文件监视

```javascript
const fs = require('fs');

// 监视文件变化
const watcher = fs.watch('file.txt', (eventType, filename) => {
    console.log(`事件类型: ${eventType}`);
    console.log(`文件名: ${filename}`);
});

// 停止监视
watcher.close();

// 监视目录
fs.watch('directory', { recursive: true }, (eventType, filename) => {
    console.log(`${filename} ${eventType}`);
});

// 使用 fs.watchFile（轮询方式，效率较低）
fs.watchFile('file.txt', (curr, prev) => {
    console.log(`当前修改时间: ${curr.mtime}`);
    console.log(`之前修改时间: ${prev.mtime}`);
});

// 停止监视
fs.unwatchFile('file.txt');
```

## 流式操作

```javascript
const fs = require('fs');
const path = require('path');

// 创建读取流
const readStream = fs.createReadStream('large-file.txt', {
    encoding: 'utf8',
    highWaterMark: 64 * 1024  // 64KB 缓冲区
});

readStream.on('data', (chunk) => {
    console.log('读取到数据:', chunk.length, '字节');
});

readStream.on('end', () => {
    console.log('读取完成');
});

readStream.on('error', (error) => {
    console.error('读取错误:', error);
});

// 创建写入流
const writeStream = fs.createWriteStream('output.txt');

writeStream.write('第一行\n');
writeStream.write('第二行\n');
writeStream.end('最后一行\n');

writeStream.on('finish', () => {
    console.log('写入完成');
});

// 管道操作
const readStream2 = fs.createReadStream('source.txt');
const writeStream2 = fs.createWriteStream('destination.txt');

readStream2.pipe(writeStream2).on('finish', () => {
    console.log('复制完成');
});

// 链式管道
const zlib = require('zlib');
const crypto = require('crypto');

fs.createReadStream('file.txt')
    .pipe(zlib.createGzip())
    .pipe(crypto.createCipher('aes-256-cbc', 'secret'))
    .pipe(fs.createWriteStream('file.txt.enc'))
    .on('finish', () => {
        console.log('加密压缩完成');
    });
```

## 实用工具函数

```javascript
const fs = require('fs/promises');
const path = require('path');

// 确保目录存在
async function ensureDir(dir) {
    try {
        await fs.mkdir(dir, { recursive: true });
    } catch (error) {
        if (error.code !== 'EEXIST') throw error;
    }
}

// 确保文件目录存在
async function ensureFileDir(filePath) {
    const dir = path.dirname(filePath);
    await ensureDir(dir);
}

// 读取 JSON 文件
async function readJson(filePath) {
    const content = await fs.readFile(filePath, 'utf8');
    return JSON.parse(content);
}

// 写入 JSON 文件
async function writeJson(filePath, data) {
    await ensureFileDir(filePath);
    await fs.writeFile(filePath, JSON.stringify(data, null, 2));
}

// 复制目录
async function copyDir(src, dest) {
    await ensureDir(dest);
    
    const entries = await fs.readdir(src, { withFileTypes: true });
    
    for (const entry of entries) {
        const srcPath = path.join(src, entry.name);
        const destPath = path.join(dest, entry.name);
        
        if (entry.isDirectory()) {
            await copyDir(srcPath, destPath);
        } else {
            await fs.copyFile(srcPath, destPath);
        }
    }
}

// 清空目录
async function emptyDir(dir) {
    const entries = await fs.readdir(dir, { withFileTypes: true });
    
    for (const entry of entries) {
        const filePath = path.join(dir, entry.name);
        
        if (entry.isDirectory()) {
            await fs.rm(filePath, { recursive: true });
        } else {
            await fs.unlink(filePath);
        }
    }
}

// 获取文件大小
async function getFileSize(filePath) {
    const stats = await fs.stat(filePath);
    return stats.size;
}

// 格式化文件大小
function formatSize(bytes) {
    const units = ['B', 'KB', 'MB', 'GB', 'TB'];
    let size = bytes;
    let unitIndex = 0;
    
    while (size >= 1024 && unitIndex < units.length - 1) {
        size /= 1024;
        unitIndex++;
    }
    
    return `${size.toFixed(2)} ${units[unitIndex]}`;
}
```

## 小结

本节学习了：

1. **同步与异步 API**：回调、Sync、Promise 形式
2. **文件读写**：readFile、writeFile、文件描述符
3. **目录操作**：创建、读取、删除目录
4. **文件信息**：stat、access
5. **文件操作**：复制、移动、删除、链接
6. **文件监视**：watch、watchFile
7. **流式操作**：ReadStream、WriteStream、pipe

> **提示**：推荐使用 fs.promises API 配合 async/await，代码更简洁易读。处理大文件时使用流式操作，避免内存问题。
