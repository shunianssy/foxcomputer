# 4.1 回调函数

回调函数是 JavaScript 异步编程的基础。本节将详细介绍回调函数的概念、使用方式和常见问题。

## 什么是回调函数

回调函数是作为参数传递给另一个函数的函数，在特定事件或条件发生时被调用。

```javascript
// 同步回调
const numbers = [1, 2, 3, 4, 5];

numbers.forEach(function(number) {
    console.log(number);
});

// 简写
numbers.forEach(n => console.log(n));

// 异步回调
setTimeout(() => {
    console.log('1秒后执行');
}, 1000);

// 事件回调
const button = document.querySelector('button');
button.addEventListener('click', function(event) {
    console.log('按钮被点击');
});
```

## 回调函数的基本使用

### 文件操作回调

```javascript
const fs = require('fs');

// 读取文件
fs.readFile('example.txt', 'utf8', (err, data) => {
    if (err) {
        console.error('读取失败:', err);
        return;
    }
    console.log('文件内容:', data);
});

// 写入文件
fs.writeFile('output.txt', 'Hello World', (err) => {
    if (err) {
        console.error('写入失败:', err);
        return;
    }
    console.log('写入成功');
});

// 追加内容
fs.appendFile('log.txt', '新的日志\n', (err) => {
    if (err) throw err;
    console.log('追加成功');
});

// 检查文件是否存在
fs.access('config.json', fs.constants.F_OK, (err) => {
    if (err) {
        console.log('文件不存在');
    } else {
        console.log('文件存在');
    }
});
```

### HTTP 请求回调

```javascript
const http = require('http');

// 创建服务器
const server = http.createServer((req, res) => {
    // 请求回调
    console.log(`收到请求: ${req.method} ${req.url}`);
    
    // 响应
    res.writeHead(200, { 'Content-Type': 'text/plain' });
    res.end('Hello World');
});

server.listen(3000, () => {
    console.log('服务器启动成功');
});

// 发起请求
http.get('http://example.com', (res) => {
    let data = '';
    
    res.on('data', (chunk) => {
        data += chunk;
    });
    
    res.on('end', () => {
        console.log('响应数据:', data);
    });
}).on('error', (err) => {
    console.error('请求错误:', err);
});
```

### 自定义回调函数

```javascript
// 定义接受回调的函数
function fetchData(url, callback) {
    console.log(`正在获取: ${url}`);
    
    // 模拟异步操作
    setTimeout(() => {
        const success = Math.random() > 0.3;
        
        if (success) {
            const data = { url, timestamp: Date.now() };
            callback(null, data);
        } else {
            callback(new Error('获取数据失败'));
        }
    }, 1000);
}

// 使用回调函数
fetchData('https://api.example.com/users', (err, data) => {
    if (err) {
        console.error('错误:', err.message);
        return;
    }
    console.log('数据:', data);
});

// 回调函数模式
function processData(data, onSuccess, onError) {
    try {
        // 处理数据
        const result = JSON.parse(data);
        
        if (result.valid) {
            onSuccess(result);
        } else {
            onError(new Error('数据无效'));
        }
    } catch (error) {
        onError(error);
    }
}

processData(
    '{"valid": true, "value": 42}',
    (result) => console.log('成功:', result),
    (error) => console.error('失败:', error)
);
```

## 错误优先回调

Node.js 采用"错误优先"的回调模式，回调函数的第一个参数是错误对象。

```javascript
// 标准的错误优先回调
function readFile(path, callback) {
    fs.readFile(path, 'utf8', (err, data) => {
        if (err) {
            callback(err);  // 第一个参数是错误
            return;
        }
        
        try {
            const json = JSON.parse(data);
            callback(null, json);  // 第二个参数是结果
        } catch (error) {
            callback(error);
        }
    });
}

// 使用
readFile('config.json', (err, config) => {
    if (err) {
        console.error('读取配置失败:', err);
        return;
    }
    console.log('配置:', config);
});

// 封装回调函数
function wrapCallback(fn) {
    return function(...args) {
        const callback = args[args.length - 1];
        
        fn.apply(this, args).then(
            result => callback(null, result),
            error => callback(error)
        );
    };
}
```

## 回调地狱

当多个异步操作需要顺序执行时，回调函数会嵌套多层，形成"回调地狱"。

```javascript
// 回调地狱示例
fs.readFile('user.json', 'utf8', (err, userData) => {
    if (err) {
        console.error(err);
        return;
    }
    
    const user = JSON.parse(userData);
    
    fs.readFile('posts.json', 'utf8', (err, postsData) => {
        if (err) {
            console.error(err);
            return;
        }
        
        const posts = JSON.parse(postsData);
        
        fs.readFile('comments.json', 'utf8', (err, commentsData) => {
            if (err) {
                console.error(err);
                return;
            }
            
            const comments = JSON.parse(commentsData);
            
            // 处理数据
            const userPosts = posts.filter(p => p.userId === user.id);
            const result = {
                user,
                posts: userPosts,
                comments
            };
            
            fs.writeFile('result.json', JSON.stringify(result), (err) => {
                if (err) {
                    console.error(err);
                    return;
                }
                console.log('处理完成');
            });
        });
    });
});
```

## 解决回调地狱

### 拆分函数

```javascript
// 将嵌套的回调拆分成独立的函数
function readUser(callback) {
    fs.readFile('user.json', 'utf8', (err, data) => {
        if (err) return callback(err);
        callback(null, JSON.parse(data));
    });
}

function readPosts(callback) {
    fs.readFile('posts.json', 'utf8', (err, data) => {
        if (err) return callback(err);
        callback(null, JSON.parse(data));
    });
}

function readComments(callback) {
    fs.readFile('comments.json', 'utf8', (err, data) => {
        if (err) return callback(err);
        callback(null, JSON.parse(data));
    });
}

function writeResult(result, callback) {
    fs.writeFile('result.json', JSON.stringify(result), callback);
}

// 顺序调用
readUser((err, user) => {
    if (err) return console.error(err);
    
    readPosts((err, posts) => {
        if (err) return console.error(err);
        
        readComments((err, comments) => {
            if (err) return console.error(err);
            
            const result = {
                user,
                posts: posts.filter(p => p.userId === user.id),
                comments
            };
            
            writeResult(result, (err) => {
                if (err) return console.error(err);
                console.log('处理完成');
            });
        });
    });
});
```

### 使用 async 库

```javascript
// 安装: npm install async
const async = require('async');

// 并行执行
async.parallel({
    user: (callback) => readUser(callback),
    posts: (callback) => readPosts(callback),
    comments: (callback) => readComments(callback)
}, (err, results) => {
    if (err) return console.error(err);
    console.log('所有文件读取完成:', results);
});

// 顺序执行
async.series({
    user: (callback) => readUser(callback),
    posts: (callback) => readPosts(callback),
    comments: (callback) => readComments(callback)
}, (err, results) => {
    if (err) return console.error(err);
    console.log('顺序执行完成:', results);
});

// 瀑布流（前一个结果传给下一个）
async.waterfall([
    (callback) => readUser(callback),
    (user, callback) => {
        readPosts((err, posts) => {
            if (err) return callback(err);
            callback(null, { user, posts });
        });
    },
    (data, callback) => {
        readComments((err, comments) => {
            if (err) return callback(err);
            callback(null, { ...data, comments });
        });
    }
], (err, result) => {
    if (err) return console.error(err);
    console.log('瀑布流完成:', result);
});
```

## 回调函数最佳实践

### 1. 始终处理错误

```javascript
// 不好的做法
fs.readFile('file.txt', 'utf8', (err, data) => {
    console.log(data);  // 忽略了错误
});

// 好的做法
fs.readFile('file.txt', 'utf8', (err, data) => {
    if (err) {
        console.error('读取失败:', err);
        return;  // 提前返回
    }
    console.log(data);
});
```

### 2. 保持回调一致

```javascript
// 不好的做法 - 回调参数顺序不一致
function badExample1(data, callback) {
    callback(data);  // 没有错误参数
}

function badExample2(data, callback) {
    callback(data, null);  // 结果在前，错误在后
}

// 好的做法 - 统一使用错误优先
function goodExample(data, callback) {
    if (!data) {
        callback(new Error('数据无效'));
        return;
    }
    callback(null, data);
}
```

### 3. 避免同步回调中抛出错误

```javascript
// 不好的做法
function badCallback(arr, callback) {
    arr.forEach(item => {
        if (item < 0) {
            throw new Error('不能有负数');  // 可能导致程序崩溃
        }
        callback(item);
    });
}

// 好的做法
function goodCallback(arr, callback) {
    arr.forEach(item => {
        if (item < 0) {
            callback(new Error('不能有负数'));
            return;
        }
        callback(null, item);
    });
}
```

### 4. 使用命名函数提高可读性

```javascript
// 不好的做法 - 匿名函数嵌套
fs.readFile('file1.txt', 'utf8', (err, data1) => {
    if (err) throw err;
    fs.readFile('file2.txt', 'utf8', (err, data2) => {
        if (err) throw err;
        console.log(data1 + data2);
    });
});

// 好的做法 - 使用命名函数
function readFirstFile(callback) {
    fs.readFile('file1.txt', 'utf8', callback);
}

function readSecondFile(data1, callback) {
    fs.readFile('file2.txt', 'utf8', (err, data2) => {
        if (err) return callback(err);
        callback(null, data1 + data2);
    });
}

readFirstFile((err, data1) => {
    if (err) return console.error(err);
    readSecondFile(data1, (err, result) => {
        if (err) return console.error(err);
        console.log(result);
    });
});
```

## 将回调转为 Promise

```javascript
// 使用 util.promisify
const util = require('util');
const fs = require('fs');

const readFile = util.promisify(fs.readFile);
const writeFile = util.promisify(fs.writeFile);

// 使用 Promise
async function processFiles() {
    try {
        const data1 = await readFile('file1.txt', 'utf8');
        const data2 = await readFile('file2.txt', 'utf8');
        await writeFile('output.txt', data1 + data2);
        console.log('处理完成');
    } catch (error) {
        console.error('处理失败:', error);
    }
}

// 手动包装
function promisify(fn) {
    return function(...args) {
        return new Promise((resolve, reject) => {
            fn.apply(this, [...args, (err, result) => {
                if (err) reject(err);
                else resolve(result);
            }]);
        });
    };
}

const readFileAsync = promisify(fs.readFile);
```

## 小结

本节学习了：

1. **回调函数概念**：作为参数传递的函数
2. **错误优先模式**：第一个参数是错误对象
3. **回调地狱**：多层嵌套的问题
4. **解决方案**：拆分函数、使用 async 库、转为 Promise
5. **最佳实践**：错误处理、一致性、可读性

> **提示**：回调函数是 JavaScript 异步编程的基础，但现代开发推荐使用 Promise 和 async/await 来处理异步操作，可以避免回调地狱问题。
