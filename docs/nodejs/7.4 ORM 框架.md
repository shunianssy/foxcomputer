# 7.4 ORM 框架

ORM（对象关系映射）框架可以简化数据库操作。本节将介绍 Sequelize 和 Prisma 两个流行的 ORM 框架。

## Sequelize

### 安装配置

```bash
npm install sequelize mysql2
```

### 初始化连接

```javascript
const { Sequelize } = require('sequelize');

// 创建连接
const sequelize = new Sequelize('database', 'username', 'password', {
    host: 'localhost',
    dialect: 'mysql',
    logging: false,  // 关闭日志
    pool: {
        max: 10,
        min: 0,
        acquire: 30000,
        idle: 10000
    }
});

// 测试连接
async function testConnection() {
    try {
        await sequelize.authenticate();
        console.log('数据库连接成功');
    } catch (error) {
        console.error('数据库连接失败:', error);
    }
}
```

### 定义模型

```javascript
const { DataTypes } = require('sequelize');

// 定义 User 模型
const User = sequelize.define('User', {
    id: {
        type: DataTypes.INTEGER,
        primaryKey: true,
        autoIncrement: true
    },
    name: {
        type: DataTypes.STRING(50),
        allowNull: false,
        validate: {
            len: [2, 50]
        }
    },
    email: {
        type: DataTypes.STRING(100),
        allowNull: false,
        unique: true,
        validate: {
            isEmail: true
        }
    },
    password: {
        type: DataTypes.STRING(255),
        allowNull: false
    },
    age: {
        type: DataTypes.INTEGER,
        validate: {
            min: 0,
            max: 150
        }
    },
    isActive: {
        type: DataTypes.BOOLEAN,
        defaultValue: true
    }
}, {
    tableName: 'users',
    timestamps: true,
    createdAt: 'created_at',
    updatedAt: 'updated_at'
});

module.exports = User;
```

### CRUD 操作

```javascript
// 创建
async function createUser(data) {
    const user = await User.create(data);
    return user;
}

// 查询
async function findUser(id) {
    const user = await User.findByPk(id);
    return user;
}

async function findAllUsers() {
    const users = await User.findAll({
        where: { isActive: true },
        order: [['created_at', 'DESC']],
        limit: 10
    });
    return users;
}

// 更新
async function updateUser(id, data) {
    const [affected] = await User.update(data, {
        where: { id }
    });
    return affected > 0;
}

// 删除
async function deleteUser(id) {
    const affected = await User.destroy({
        where: { id }
    });
    return affected > 0;
}
```

## Prisma

### 安装配置

```bash
npm install prisma --save-dev
npm install @prisma/client

# 初始化
npx prisma init
```

### 定义模型

```prisma
// prisma/schema.prisma

datasource db {
    provider = "mysql"
    url      = env("DATABASE_URL")
}

generator client {
    provider = "prisma-client-js"
}

model User {
    id        Int      @id @default(autoincrement())
    name      String   @db.VarChar(50)
    email     String   @unique @db.VarChar(100)
    password  String   @db.VarChar(255)
    age       Int?
    isActive  Boolean  @default(true) @map("is_active")
    posts     Post[]
    createdAt DateTime @default(now()) @map("created_at")
    updatedAt DateTime @updatedAt @map("updated_at")

    @@map("users")
}

model Post {
    id        Int       @id @default(autoincrement())
    title     String    @db.VarChar(200)
    content   String?   @db.Text
    authorId  Int       @map("author_id")
    author    User      @relation(fields: [authorId], references: [id])
    createdAt DateTime  @default(now()) @map("created_at")
    updatedAt DateTime  @updatedAt @map("updated_at")

    @@map("posts")
}
```

### 使用 Prisma Client

```javascript
const { PrismaClient } = require('@prisma/client');
const prisma = new PrismaClient();

// 创建
async function createUser(data) {
    const user = await prisma.user.create({
        data: {
            name: data.name,
            email: data.email,
            password: data.password,
            age: data.age
        }
    });
    return user;
}

// 查询
async function findUser(id) {
    const user = await prisma.user.findUnique({
        where: { id },
        include: { posts: true }
    });
    return user;
}

async function findAllUsers() {
    const users = await prisma.user.findMany({
        where: { isActive: true },
        orderBy: { createdAt: 'desc' },
        take: 10
    });
    return users;
}

// 分页
async function findWithPagination(page = 1, limit = 10) {
    const skip = (page - 1) * limit;
    
    const [users, total] = await Promise.all([
        prisma.user.findMany({
            skip,
            take: limit,
            orderBy: { createdAt: 'desc' }
        }),
        prisma.user.count()
    ]);
    
    return {
        data: users,
        pagination: {
            page,
            limit,
            total,
            totalPages: Math.ceil(total / limit)
        }
    };
}

// 更新
async function updateUser(id, data) {
    const user = await prisma.user.update({
        where: { id },
        data
    });
    return user;
}

// 删除
async function deleteUser(id) {
    await prisma.user.delete({
        where: { id }
    });
}

// 关闭连接
process.on('beforeExit', async () => {
    await prisma.$disconnect();
});
```

## 小结

本节学习了：

1. **Sequelize**：模型定义、CRUD 操作
2. **Prisma**：Schema 定义、Prisma Client
3. **选择建议**：Sequelize 成熟稳定，Prisma 类型安全

> **提示**：ORM 框架可以大大提高开发效率，但复杂查询可能需要使用原生 SQL。根据项目需求选择合适的 ORM。
