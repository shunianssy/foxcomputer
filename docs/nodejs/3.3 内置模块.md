# 3.3 内置模块

Node.js 提供了丰富的内置模块，无需安装即可使用。本节将介绍常用的内置模块及其功能。

## os 模块

```javascript
const os = require('os');

// 系统信息
console.log('操作系统:', os.type());           // Windows_NT / Linux / Darwin
console.log('系统平台:', os.platform());       // win32 / linux / darwin
console.log('系统架构:', os.arch());           // x64 / arm64
console.log('系统版本:', os.release());        // 版本号
console.log('主机名:', os.hostname());         // 计算机名

// CPU 信息
console.log('CPU 核心:', os.cpus().length);
console.log('CPU 型号:', os.cpus()[0].model);
console.log('CPU 负载:', os.loadavg());        // [1, 5, 15] 分钟平均负载

// 内存信息
console.log('总内存:', (os.totalmem() / 1024 / 1024 / 1024).toFixed(2), 'GB');
console.log('空闲内存:', (os.freemem() / 1024 / 1024 / 1024).toFixed(2), 'GB');
console.log('内存使用率:', ((1 - os.freemem() / os.totalmem()) * 100).toFixed(2), '%');

// 用户信息
console.log('用户信息:', os.userInfo());
console.log('用户目录:', os.homedir());        // C:\Users\用户名 或 /home/用户名
console.log('临时目录:', os.tmpdir());         // 临时文件目录

// 网络接口
console.log('网络接口:', os.networkInterfaces());

// 系统运行时间
console.log('系统运行时间:', (os.uptime() / 3600).toFixed(2), '小时');

// 换行符
console.log('换行符:', JSON.stringify(os.EOL));  // "\r\n" 或 "\n"

// 路径分隔符
console.log('路径分隔符:', os.constants);      // 系统常量
```

## util 模块

```javascript
const util = require('util');

// util.promisify - 将回调风格函数转为 Promise
const fs = require('fs');
const readFile = util.promisify(fs.readFile);
const writeFile = util.promisify(fs.writeFile);

async function readConfig() {
    try {
        const data = await readFile('config.json', 'utf8');
        return JSON.parse(data);
    } catch (error) {
        console.error('读取配置失败:', error);
        return null;
    }
}

// util.callbackify - 将 Promise 函数转为回调风格
async function asyncFunction() {
    return 'Hello';
}

const callbackFunction = util.callbackify(asyncFunction);
callbackFunction((err, result) => {
    console.log(result);  // Hello
});

// util.format - 格式化字符串
console.log(util.format('姓名: %s, 年龄: %d', '张三', 25));
// 姓名: 张三, 年龄: 25

// 格式化占位符:
// %s - 字符串
// %d - 数字
// %i - 整数
// %f - 浮点数
// %j - JSON
// %o - 对象
// %O - 详细对象

// util.inspect - 检查对象
const obj = { a: 1, b: { c: 2 } };
console.log(util.inspect(obj, { 
    depth: null,      // 深度，null 表示无限
    colors: true,     // 彩色输出
    compact: false    // 紧凑格式
}));

// util.types - 类型检查
console.log(util.types.isDate(new Date()));       // true
console.log(util.types.isMap(new Map()));          // true
console.log(util.types.isSet(new Set()));          // true
console.log(util.types.isPromise(Promise.resolve())); // true
console.log(util.types.isArrayBuffer(new ArrayBuffer(8))); // true

// util.deprecate - 标记废弃函数
function oldFunction() {
    return '旧函数';
}

const deprecatedFunction = util.deprecate(oldFunction, 'oldFunction 已废弃，请使用 newFunction');
deprecatedFunction();  // 会输出废弃警告

// util.inherits - 继承（已废弃，推荐使用 class extends）
function Animal(name) {
    this.name = name;
}

Animal.prototype.speak = function() {
    console.log(`${this.name} 发出声音`);
};

function Dog(name) {
    Animal.call(this, name);
}

util.inherits(Dog, Animal);

Dog.prototype.bark = function() {
    console.log(`${this.name} 汪汪叫`);
};
```

## events 模块

```javascript
const EventEmitter = require('events');

// 创建事件发射器
const emitter = new EventEmitter();

// 监听事件
emitter.on('message', (data) => {
    console.log('收到消息:', data);
});

// 只监听一次
emitter.once('connect', () => {
    console.log('已连接');
});

// 发射事件
emitter.emit('message', { text: 'Hello' });
emitter.emit('connect');  // 触发
emitter.emit('connect');  // 不触发（once 只触发一次）

// 移除监听器
const handler = (data) => console.log('处理数据:', data);
emitter.on('data', handler);
emitter.off('data', handler);  // 或 emitter.removeListener('data', handler)

// 获取监听器
console.log(emitter.listeners('message'));     // 监听器数组
console.log(emitter.listenerCount('message')); // 监听器数量

// 设置最大监听器数量
emitter.setMaxListeners(20);
console.log(emitter.getMaxListeners());

// 自定义事件类
class MyEmitter extends EventEmitter {
    constructor() {
        super();
        this.data = [];
    }
    
    addData(item) {
        this.data.push(item);
        this.emit('dataAdded', item);
    }
    
    getData() {
        return [...this.data];
    }
}

const myEmitter = new MyEmitter();

myEmitter.on('dataAdded', (item) => {
    console.log('新数据添加:', item);
});

myEmitter.addData('项目1');
myEmitter.addData('项目2');

// 错误事件
emitter.on('error', (err) => {
    console.error('发生错误:', err);
});

emitter.emit('error', new Error('出错了'));

// 使用 captureRejections 处理异步错误
emitter.captureRejections = true;

emitter.on('asyncEvent', async () => {
    throw new Error('异步错误');
});

emitter.on('error', (err) => {
    console.error('捕获异步错误:', err.message);
});
```

## buffer 模块

```javascript
// 创建 Buffer
const buf1 = Buffer.alloc(10);           // 分配 10 字节，初始化为 0
const buf2 = Buffer.allocUnsafe(10);     // 分配 10 字节，不初始化
const buf3 = Buffer.from('Hello');       // 从字符串创建
const buf4 = Buffer.from([1, 2, 3]);     // 从数组创建
const buf5 = Buffer.from('Hello', 'utf8'); // 指定编码

console.log(buf1);  // <Buffer 00 00 00 00 00 00 00 00 00 00>
console.log(buf3);  // <Buffer 48 65 6c 6c 6f>

// 写入数据
const buf = Buffer.alloc(10);
buf.write('Hello', 'utf8');
console.log(buf.toString());  // Hello

// 读取数据
console.log(buf.toString('utf8'));
console.log(buf.toString('hex'));    // 十六进制
console.log(buf.toString('base64')); // Base64

// 访问字节
console.log(buf[0]);  // 72 (H 的 ASCII 码)
buf[0] = 74;          // 修改为 J
console.log(buf.toString());  // Jello

// Buffer 操作
const bufA = Buffer.from('Hello ');
const bufB = Buffer.from('World');

// 拼接
const bufC = Buffer.concat([bufA, bufB]);
console.log(bufC.toString());  // Hello World

// 比较
console.log(bufA.equals(bufB));     // false
console.log(bufA.compare(bufB));    // -1（bufA < bufB）

// 查找
const bufD = Buffer.from('Hello World');
console.log(bufD.indexOf('World'));  // 6
console.log(bufD.includes('World')); // true

// 切片
const slice = bufD.slice(0, 5);
console.log(slice.toString());  // Hello

// 复制
const target = Buffer.alloc(5);
bufD.copy(target, 0, 0, 5);
console.log(target.toString());  // Hello

// Buffer 编码
const text = '你好世界';
const encoded = Buffer.from(text, 'utf8');
console.log(encoded.toString('hex'));    // 十六进制
console.log(encoded.toString('base64')); // Base64

// 解码
const decoded = Buffer.from(encoded.toString('base64'), 'base64');
console.log(decoded.toString('utf8'));   // 你好世界

// Buffer 与 JSON
const bufJson = JSON.stringify(buf3);
console.log(bufJson);  // {"type":"Buffer","data":[72,101,108,108,111]}

const parsed = JSON.parse(bufJson);
const bufFromJson = Buffer.from(parsed.data);
console.log(bufFromJson.toString());  // Hello
```

## url 模块

```javascript
const url = require('url');

// 解析 URL
const myUrl = new URL('https://user:pass@example.com:8080/path/name?query=string#hash');

console.log('协议:', myUrl.protocol);     // https:
console.log('用户名:', myUrl.username);   // user
console.log('密码:', myUrl.password);     // pass
console.log('主机名:', myUrl.hostname);   // example.com
console.log('端口:', myUrl.port);         // 8080
console.log('主机:', myUrl.host);         // example.com:8080
console.log('路径:', myUrl.pathname);     // /path/name
console.log('查询:', myUrl.search);       // ?query=string
console.log('哈希:', myUrl.hash);         // #hash
console.log('完整 URL:', myUrl.href);     // 完整 URL
console.log('来源:', myUrl.origin);       // https://example.com:8080

// 操作 URL
myUrl.pathname = '/new/path';
myUrl.searchParams.append('page', '1');
myUrl.searchParams.set('limit', '10');
console.log(myUrl.toString());

// URLSearchParams
const params = new URLSearchParams('name=张三&age=25');
console.log(params.get('name'));      // 张三
console.log(params.has('age'));       // true
console.log(params.getAll('name'));   // ['张三']

params.append('hobby', 'reading');
params.append('hobby', 'coding');
console.log(params.getAll('hobby'));  // ['reading', 'coding']

params.set('name', '李四');
params.delete('age');

console.log(params.toString());       // name=李四&hobby=reading&hobby=coding

// 从对象创建查询字符串
const queryObject = {
    name: '张三',
    age: 25,
    hobbies: ['reading', 'coding']
};

const searchParams = new URLSearchParams();
for (const [key, value] of Object.entries(queryObject)) {
    if (Array.isArray(value)) {
        value.forEach(v => searchParams.append(key, v));
    } else {
        searchParams.set(key, value);
    }
}
console.log(searchParams.toString());

// url.format（旧 API，不推荐）
const legacyUrl = url.format({
    protocol: 'https',
    hostname: 'example.com',
    pathname: '/path',
    query: { name: '张三' }
});
console.log(legacyUrl);
```

## querystring 模块

```javascript
const querystring = require('querystring');

// 解析查询字符串
const query = 'name=张三&age=25&hobby=reading&hobby=coding';
const parsed = querystring.parse(query);

console.log(parsed);
// { name: '张三', age: '25', hobby: ['reading', 'coding'] }

// 自定义分隔符
const custom = 'name:张三;age:25';
const customParsed = querystring.parse(custom, ';', ':');
console.log(customParsed);  // { name: '张三', age: '25' }

// 序列化对象
const obj = {
    name: '张三',
    age: 25,
    hobbies: ['reading', 'coding']
};

const stringified = querystring.stringify(obj);
console.log(stringified);  // name=张三&age=25&hobbies=reading&hobbies=coding

// 自定义分隔符
const customString = querystring.stringify(obj, ';', ':');
console.log(customString);  // name:张三;age:25;hobbies:reading;hobbies:coding

// URL 编码解码
console.log(querystring.escape('你好世界'));  // %E4%BD%A0%E5%A5%BD%E4%B8%96%E7%95%8C
console.log(querystring.unescape('%E4%BD%A0%E5%A5%BD%E4%B8%96%E7%95%8C'));  // 你好世界

// 注意：推荐使用 URLSearchParams
const params = new URLSearchParams(query);
console.log(Object.fromEntries(params));
```

## crypto 模块

```javascript
const crypto = require('crypto');

// 哈希算法
const hash = crypto.createHash('sha256');
hash.update('Hello World');
console.log(hash.digest('hex'));
// a591a6d40bf420404a011733cfb7b190d62c65bf0bcda32b57b277d9ad9f146e

// 常用哈希算法
const algorithms = ['md5', 'sha1', 'sha256', 'sha512'];
algorithms.forEach(algo => {
    const h = crypto.createHash(algo).update('Hello').digest('hex');
    console.log(`${algo}: ${h}`);
});

// HMAC（带密钥的哈希）
const hmac = crypto.createHmac('sha256', 'secret-key');
hmac.update('Hello World');
console.log(hmac.digest('hex'));

// 对称加密
const algorithm = 'aes-256-cbc';
const key = crypto.randomBytes(32);  // 256 位密钥
const iv = crypto.randomBytes(16);   // 初始化向量

// 加密
function encrypt(text) {
    const cipher = crypto.createCipheriv(algorithm, key, iv);
    let encrypted = cipher.update(text, 'utf8', 'hex');
    encrypted += cipher.final('hex');
    return encrypted;
}

// 解密
function decrypt(encrypted) {
    const decipher = crypto.createDecipheriv(algorithm, key, iv);
    let decrypted = decipher.update(encrypted, 'hex', 'utf8');
    decrypted += decipher.final('utf8');
    return decrypted;
}

const encrypted = encrypt('Hello World');
console.log('加密:', encrypted);
console.log('解密:', decrypt(encrypted));

// 生成随机字节
crypto.randomBytes(16, (err, buffer) => {
    if (err) throw err;
    console.log('随机字节:', buffer.toString('hex'));
});

// 同步方式
const randomBytes = crypto.randomBytes(16).toString('hex');
console.log('随机字符串:', randomBytes);

// 生成随机整数
const randomInt = crypto.randomInt(1, 100);
console.log('随机整数:', randomInt);

// 密码派生
crypto.pbkdf2('password', 'salt', 100000, 64, 'sha512', (err, derivedKey) => {
    if (err) throw err;
    console.log('派生密钥:', derivedKey.toString('hex'));
});

// 同步方式
const derivedKey = crypto.pbkdf2Sync('password', 'salt', 100000, 64, 'sha512');
console.log('派生密钥:', derivedKey.toString('hex'));

// 数字签名
const { publicKey, privateKey } = crypto.generateKeyPairSync('rsa', {
    modulusLength: 2048
});

const data = 'Hello World';
const sign = crypto.sign('sha256', Buffer.from(data), privateKey);
console.log('签名:', sign.toString('hex'));

const verify = crypto.verify('sha256', Buffer.from(data), publicKey, sign);
console.log('验证结果:', verify);  // true
```

## 小结

本节学习了：

1. **os 模块**：获取系统信息
2. **util 模块**：实用工具函数
3. **events 模块**：事件处理
