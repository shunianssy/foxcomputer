# 8.2 日志系统

日志是应用调试和监控的重要工具。本节将介绍 Node.js 中的日志管理最佳实践。

## console 日志

```javascript
// 基本日志
console.log('普通日志');
console.info('信息日志');
console.warn('警告日志');
console.error('错误日志');

// 格式化输出
console.log('用户: %s, 年龄: %d', '张三', 25);
console.log('对象: %j', { name: '张三' });

// 计时
console.time('操作');
// ... 操作
console.timeEnd('操作');  // 操作: 123.456ms

// 堆栈跟踪
console.trace('跟踪信息');

// 表格输出
console.table([
    { name: '张三', age: 25 },
    { name: '李四', age: 30 }
]);
```

## winston 日志库

### 安装配置

```bash
npm install winston
```

### 基本使用

```javascript
const winston = require('winston');

// 创建 logger
const logger = winston.createLogger({
    level: process.env.LOG_LEVEL || 'info',
    format: winston.format.combine(
        winston.format.timestamp({ format: 'YYYY-MM-DD HH:mm:ss' }),
        winston.format.errors({ stack: true }),
        winston.format.json()
    ),
    defaultMeta: { service: 'my-app' },
    transports: [
        // 错误日志
        new winston.transports.File({ 
            filename: 'logs/error.log', 
            level: 'error' 
        }),
        // 所有日志
        new winston.transports.File({ 
            filename: 'logs/combined.log' 
        })
    ]
});

// 开发环境控制台输出
if (process.env.NODE_ENV !== 'production') {
    logger.add(new winston.transports.Console({
        format: winston.format.combine(
            winston.format.colorize(),
            winston.format.printf(({ level, message, timestamp, ...meta }) => {
                return `${timestamp} [${level}]: ${message} ${
                    Object.keys(meta).length ? JSON.stringify(meta) : ''
                }`;
            })
        )
    }));
}

// 使用
logger.error('错误信息');
logger.warn('警告信息');
logger.info('普通信息');
logger.debug('调试信息');
```

### 日志级别

```javascript
// 日志级别（从低到高）
const levels = {
    error: 0,   // 错误
    warn: 1,    // 警告
    info: 2,    // 信息
    http: 3,    // HTTP 请求
    verbose: 4, // 详细
    debug: 5,   // 调试
    silly: 6    // 芜杂
};

// 设置级别
logger.level = 'debug';  // 记录 debug 及以上级别
```

### 自定义格式

```javascript
const { format } = winston;
const { combine, timestamp, printf, colorize } = format;

// 自定义格式
const customFormat = printf(({ level, message, timestamp, stack }) => {
    return `${timestamp} [${level}]: ${stack || message}`;
});

const logger = winston.createLogger({
    format: combine(
        timestamp({ format: 'YYYY-MM-DD HH:mm:ss' }),
        format.errors({ stack: true }),
        customFormat
    ),
    transports: [new winston.transports.Console()]
});
```

## Express 日志中间件

```javascript
const morgan = require('morgan');

// 安装: npm install morgan

// 基本使用
app.use(morgan('dev'));  // 开发模式
app.use(morgan('combined'));  // 完整日志
app.use(morgan('tiny'));  // 简洁日志

// 自定义格式
morgan.token('user-id', (req) => req.user?.id || '-');
app.use(morgan(':method :url :status :response-time ms - :user-id'));

// 写入文件
const fs = require('fs');
const path = require('path');

const accessLogStream = fs.createWriteStream(
    path.join(__dirname, 'logs/access.log'),
    { flags: 'a' }
);

app.use(morgan('combined', { stream: accessLogStream }));
```

## 小结

本节学习了：

1. **console 日志**：基本输出、格式化
2. **winston 日志库**：配置、级别、格式
3. **Express 日志中间件**：morgan

> **提示**：生产环境应该使用专业的日志库，并将日志写入文件或发送到日志服务。合理设置日志级别，避免日志过多影响性能。
