# 3.4 第三方模块

npm 生态系统提供了海量的第三方模块，可以快速实现各种功能。本节将介绍常用第三方模块的使用方法和最佳实践。

## 常用工具库

### lodash

```javascript
// 安装: npm install lodash

const _ = require('lodash');

// 数组操作
const arr = [1, 2, 3, 4, 5];

console.log(_.chunk(arr, 2));        // [[1, 2], [3, 4], [5]]
console.log(_.compact([0, 1, false, 2, '', 3])); // [1, 2, 3]
console.log(_.uniq([1, 1, 2, 2, 3])); // [1, 2, 3]
console.log(_.flattenDeep([1, [2, [3, [4]]]])); // [1, 2, 3, 4]
console.log(_.difference([1, 2, 3], [2, 3])); // [1]
console.log(_.intersection([1, 2, 3], [2, 3, 4])); // [2, 3]

// 对象操作
const obj = { a: 1, b: 2, c: 3 };

console.log(_.pick(obj, ['a', 'b']));     // { a: 1, b: 2 }
console.log(_.omit(obj, ['a']));          // { b: 2, c: 3 }
console.log(_.merge({}, { a: 1 }, { b: 2 })); // { a: 1, b: 2 }
console.log(_.cloneDeep({ a: { b: 1 } })); // 深拷贝

// 字符串操作
console.log(_.camelCase('hello world'));   // helloWorld
console.log(_.kebabCase('helloWorld'));    // hello-world
console.log(_.snakeCase('hello world'));   // hello_world
console.log(_.capitalize('hello'));        // Hello
console.log(_.truncate('hello world', { length: 5 })); // he...

// 函数操作
const debounceFn = _.debounce(() => {
    console.log('防抖执行');
}, 300);

const throttleFn = _.throttle(() => {
    console.log('节流执行');
}, 300);

// 实用函数
console.log(_.random(1, 100));           // 随机数
console.log(_.shuffle([1, 2, 3, 4, 5])); // 随机排序
console.log(_.sample([1, 2, 3, 4, 5]));  // 随机取一个
console.log(_.range(1, 6));              // [1, 2, 3, 4, 5]
console.log(_.times(3, String));         // ['0', '1', '2']

// 类型检查
console.log(_.isEmpty({}));              // true
console.log(_.isArray([1, 2, 3]));       // true
console.log(_.isObject({}));             // true
console.log(_.isString('hello'));        // true
console.log(_.isNumber(42));             // true
```

### dayjs

```javascript
// 安装: npm install dayjs

const dayjs = require('dayjs');

// 创建日期
const now = dayjs();
const date = dayjs('2024-01-15');
const date2 = dayjs('2024-01-15 10:30:00');

// 格式化
console.log(now.format());                    // 2024-01-15T10:30:00+08:00
console.log(now.format('YYYY-MM-DD'));        // 2024-01-15
console.log(now.format('YYYY年MM月DD日'));     // 2024年01月15日
console.log(now.format('HH:mm:ss'));          // 10:30:00

// 获取/设置
console.log(now.year());       // 年
console.log(now.month());      // 月（0-11）
console.log(now.date());       // 日
console.log(now.day());        // 星期（0-6）
console.log(now.hour());       // 时
console.log(now.minute());     // 分
console.log(now.second());     // 秒

// 操作
console.log(now.add(7, 'day').format('YYYY-MM-DD'));    // 加7天
console.log(now.subtract(1, 'month').format('YYYY-MM-DD')); // 减1月
console.log(now.startOf('month').format('YYYY-MM-DD')); // 月初
console.log(now.endOf('month').format('YYYY-MM-DD'));   // 月末

// 比较
const date1 = dayjs('2024-01-15');
const date3 = dayjs('2024-01-20');

console.log(date1.isBefore(date3));    // true
console.log(date1.isAfter(date3));     // false
console.log(date1.isSame(date3));      // false
console.log(date1.isSame(date3, 'month')); // true（同月）

// 差值
console.log(date3.diff(date1, 'day'));   // 5
console.log(date3.diff(date1, 'hour'));  // 120

// 相对时间（需要插件）
const relativeTime = require('dayjs/plugin/relativeTime');
dayjs.extend(relativeTime);

console.log(dayjs().to(dayjs('2024-01-01')));    // XX 天前
console.log(dayjs().from(dayjs('2024-12-31')));  // XX 天后

// 本地化
require('dayjs/locale/zh-cn');
dayjs.locale('zh-cn');

console.log(dayjs().format('MMMM'));  // 一月
```

### uuid

```javascript
// 安装: npm install uuid

const { v4: uuidv4, v1: uuidv1, v3: uuidv3, v5: uuidv5 } = require('uuid');

// v4 - 随机 UUID
console.log(uuidv4());
// '9b1deb4d-3b7d-4bad-9bdd-2b0d7b3dcb6d'

// v1 - 基于时间戳
console.log(uuidv1());
// '6ec0bd7f-11c0-43da-975e-2a8ad9ebae0b'

// v3 - 基于命名空间和 MD5
const NAMESPACE = '6ba7b810-9dad-11d1-80b4-00c04fd430c8';
console.log(uuidv3('hello', NAMESPACE));

// v5 - 基于命名空间和 SHA-1
console.log(uuidv5('hello', NAMESPACE));

// 验证 UUID
const { validate } = require('uuid');
console.log(validate('9b1deb4d-3b7d-4bad-9bdd-2b0d7b3dcb6d')); // true
```

## 网络请求库

### axios

```javascript
// 安装: npm install axios

const axios = require('axios');

// 创建实例
const api = axios.create({
    baseURL: 'https://api.example.com',
    timeout: 5000,
    headers: {
        'Content-Type': 'application/json'
    }
});

// 请求拦截器
api.interceptors.request.use(
    config => {
        // 添加 token
        const token = process.env.API_TOKEN;
        if (token) {
            config.headers.Authorization = `Bearer ${token}`;
        }
        console.log('请求:', config.method.toUpperCase(), config.url);
        return config;
    },
    error => Promise.reject(error)
);

// 响应拦截器
api.interceptors.response.use(
    response => {
        console.log('响应:', response.status);
        return response.data;
    },
    error => {
        if (error.response) {
            console.error('响应错误:', error.response.status);
        } else if (error.request) {
            console.error('无响应:', error.request);
        } else {
            console.error('请求错误:', error.message);
        }
        return Promise.reject(error);
    }
);

// GET 请求
async function getUsers() {
    try {
        const data = await api.get('/users', {
            params: {
                page: 1,
                limit: 10
            }
        });
        return data;
    } catch (error) {
        console.error('获取用户失败:', error);
        throw error;
    }
}

// POST 请求
async function createUser(userData) {
    try {
        const data = await api.post('/users', userData);
        return data;
    } catch (error) {
        console.error('创建用户失败:', error);
        throw error;
    }
}

// PUT 请求
async function updateUser(id, userData) {
    return api.put(`/users/${id}`, userData);
}

// DELETE 请求
async function deleteUser(id) {
    return api.delete(`/users/${id}`);
}

// 并发请求
async function fetchAll() {
    const [users, posts] = await Promise.all([
        api.get('/users'),
        api.get('/posts')
    ]);
    return { users, posts };
}

// 下载文件
async function downloadFile(url, filepath) {
    const response = await axios({
        method: 'GET',
        url: url,
        responseType: 'stream'
    });
    
    const writer = require('fs').createWriteStream(filepath);
    response.data.pipe(writer);
    
    return new Promise((resolve, reject) => {
        writer.on('finish', resolve);
        writer.on('error', reject);
    });
}
```

## 验证库

### joi

```javascript
// 安装: npm install joi

const Joi = require('joi');

// 定义验证规则
const userSchema = Joi.object({
    name: Joi.string()
        .min(2)
        .max(50)
        .required()
        .messages({
            'string.base': '姓名必须是字符串',
            'string.min': '姓名至少 {#limit} 个字符',
            'string.max': '姓名最多 {#limit} 个字符',
            'any.required': '姓名是必填项'
        }),
    
    email: Joi.string()
        .email()
        .required(),
    
    password: Joi.string()
        .pattern(/^[a-zA-Z0-9]{8,30}$/)
        .required(),
    
    age: Joi.number()
        .integer()
        .min(0)
        .max(150),
    
    phone: Joi.string()
        .pattern(/^1[3-9]\d{9}$/),
    
    role: Joi.string()
        .valid('admin', 'user', 'guest')
        .default('user'),
    
    createdAt: Joi.date()
        .default(Date.now)
});

// 验证数据
function validateUser(data) {
    const { error, value } = userSchema.validate(data, {
        abortEarly: false,  // 返回所有错误
        stripUnknown: true  // 移除未知字段
    });
    
    if (error) {
        const errors = error.details.map(detail => ({
            field: detail.path.join('.'),
            message: detail.message
        }));
        return { valid: false, errors };
    }
    
    return { valid: true, value };
}

// 使用示例
const result = validateUser({
    name: '张三',
    email: 'zhangsan@example.com',
    password: 'password123',
    age: 25
});

console.log(result);

// 常用验证规则
const commonSchemas = {
    // 字符串
    string: Joi.string(),
    email: Joi.string().email(),
    url: Joi.string().uri(),
    uuid: Joi.string().uuid(),
    
    // 数字
    number: Joi.number(),
    integer: Joi.number().integer(),
    positive: Joi.number().positive(),
    
    // 日期
    date: Joi.date(),
    dateString: Joi.date().iso(),
    
    // 数组
    array: Joi.array(),
    stringArray: Joi.array().items(Joi.string()),
    
    // 对象
    object: Joi.object(),
    
    // 布尔
    boolean: Joi.boolean(),
    
    // 允许任意值
    any: Joi.any()
};
```

## 配置管理

### dotenv

```javascript
// 安装: npm install dotenv

// 在项目入口文件加载
require('dotenv').config();

// 或指定文件路径
require('dotenv').config({ path: '.env.local' });

// 使用环境变量
console.log(process.env.NODE_ENV);
console.log(process.env.DB_HOST);

// config.js - 配置管理
const dotenv = require('dotenv');

// 根据环境加载不同配置
const env = process.env.NODE_ENV || 'development';
dotenv.config({ path: `.env.${env}` });

const config = {
    env: env,
    port: parseInt(process.env.PORT) || 3000,
    
    database: {
        host: process.env.DB_HOST || 'localhost',
        port: parseInt(process.env.DB_PORT) || 3306,
        name: process.env.DB_NAME || 'app',
        user: process.env.DB_USER || 'root',
        password: process.env.DB_PASSWORD || ''
    },
    
    jwt: {
        secret: process.env.JWT_SECRET || 'default-secret',
        expiresIn: process.env.JWT_EXPIRES_IN || '7d'
    },
    
    redis: {
        host: process.env.REDIS_HOST || 'localhost',
        port: parseInt(process.env.REDIS_PORT) || 6379
    }
};

module.exports = config;
```

## 日志库

### winston

```javascript
// 安装: npm install winston

const winston = require('winston');

// 创建 logger
const logger = winston.createLogger({
    level: process.env.LOG_LEVEL || 'info',
    format: winston.format.combine(
        winston.format.timestamp({ format: 'YYYY-MM-DD HH:mm:ss' }),
        winston.format.errors({ stack: true }),
        winston.format.json()
    ),
    defaultMeta: { service: 'my-app' },
    transports: [
        // 错误日志单独文件
        new winston.transports.File({ 
            filename: 'logs/error.log', 
            level: 'error' 
        }),
        // 所有日志
        new winston.transports.File({ 
            filename: 'logs/combined.log' 
        })
    ]
});

// 开发环境输出到控制台
if (process.env.NODE_ENV !== 'production') {
    logger.add(new winston.transports.Console({
        format: winston.format.combine(
            winston.format.colorize(),
            winston.format.printf(({ level, message, timestamp, ...meta }) => {
                return `${timestamp} [${level}]: ${message} ${
                    Object.keys(meta).length ? JSON.stringify(meta) : ''
                }`;
            })
        )
    }));
}

// 使用
logger.error('错误信息', { error: new Error('出错了') });
logger.warn('警告信息');
logger.info('普通信息');
logger.http('HTTP 请求', { method: 'GET', url: '/api/users' });
logger.debug('调试信息', { data: { foo: 'bar' } });

// 创建子 logger
const userLogger = logger.child({ module: 'user-service' });
userLogger.info('用户登录', { userId: 123 });
```

## 实用工具

### commander

```javascript
// 安装: npm install commander

const { Command } = require('commander');
const program = new Command();

program
    .name('mycli')
    .description('CLI 工具示例')
    .version('1.0.0');

// 定义命令
program
    .command('greet <name>')
    .description('问候用户')
    .option('-t, --title <title>', '称呼', '先生/女士')
    .action((name, options) => {
        console.log(`你好，${options.title} ${name}！`);
    });

program
    .command('list')
    .description('列出文件')
    .option('-a, --all', '显示隐藏文件')
    .action((options) => {
        const fs = require('fs');
        const files = fs.readdirSync('.');
        files.forEach(file => {
            if (options.all || !file.startsWith('.')) {
                console.log(file);
            }
        });
    });

program
    .command('create <project>')
    .description('创建新项目')
    .option('-t, --template <template>', '模板名称', 'default')
    .action((project, options) => {
        console.log(`创建项目: ${project}`);
        console.log(`使用模板: ${options.template}`);
    });

// 解析命令行参数
program.parse();
```

### chalk

```javascript
// 安装: npm install chalk@4

const chalk = require('chalk');

// 基本颜色
console.log(chalk.blue('蓝色文字'));
console.log(chalk.red('红色文字'));
console.log(chalk.green('绿色文字'));
console.log(chalk.yellow('黄色文字'));

// 背景色
console.log(chalk.bgBlue('蓝色背景'));
console.log(chalk.bgRed.white('白字红底'));

// 样式
console.log(chalk.bold('粗体'));
console.log(chalk.underline('下划线'));
console.log(chalk.italic('斜体'));
console.log(chalk.strikethrough('删除线'));

// 组合使用
console.log(chalk.bold.red('粗体红色'));
console.log(chalk.bgYellow.blue.bold('蓝字黄底粗体'));

// 模板字符串
console.log(chalk`{red 错误}: {gray 文件不存在}`);
console.log(chalk`{green 成功}: {blue.bold 操作完成}`);

// 条件颜色
const success = true;
console.log(success ? chalk.green('成功') : chalk.red('失败'));

// 日志级别颜色
const log = {
    info: (msg) => console.log(chalk.blue('[INFO]'), msg),
    warn: (msg) => console.log(chalk.yellow('[WARN]'), msg),
    error: (msg) => console.log(chalk.red('[ERROR]'), msg),
    success: (msg) => console.log(chalk.green('[SUCCESS]'), msg)
};

log.info('普通信息');
log.warn('警告信息');
log.error('错误信息');
log.success('成功信息');
```
