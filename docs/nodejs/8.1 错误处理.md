# 8.1 错误处理

良好的错误处理是应用稳定性的保障。本节将介绍 Node.js 中的错误处理最佳实践。

## 错误类型

### 同步错误

```javascript
// 同步错误使用 try/catch
function divide(a, b) {
    if (b === 0) {
        throw new Error('除数不能为零');
    }
    return a / b;
}

try {
    const result = divide(10, 0);
    console.log(result);
} catch (error) {
    console.error('错误:', error.message);
}
```

### 异步错误

```javascript
// Promise 错误
async function fetchData() {
    const response = await fetch('/api/data');
    if (!response.ok) {
        throw new Error(`HTTP ${response.status}`);
    }
    return response.json();
}

// 使用 try/catch
try {
    const data = await fetchData();
} catch (error) {
    console.error('获取数据失败:', error);
}

// 使用 .catch()
fetchData()
    .then(data => console.log(data))
    .catch(error => console.error(error));
```

## 自定义错误类

```javascript
// 基础错误类
class AppError extends Error {
    constructor(message, statusCode = 500) {
        super(message);
        this.statusCode = statusCode;
        this.status = `${statusCode}`.startsWith('4') ? 'fail' : 'error';
        this.isOperational = true;
        Error.captureStackTrace(this, this.constructor);
    }
}

// 特定错误类
class NotFoundError extends AppError {
    constructor(resource = '资源') {
        super(`${resource}不存在`, 404);
    }
}

class ValidationError extends AppError {
    constructor(errors) {
        super('验证失败', 400);
        this.errors = errors;
    }
}

class UnauthorizedError extends AppError {
    constructor(message = '未授权') {
        super(message, 401);
    }
}

class ForbiddenError extends AppError {
    constructor(message = '禁止访问') {
        super(message, 403);
    }
}

class ConflictError extends AppError {
    constructor(message = '资源冲突') {
        super(message, 409);
    }
}

module.exports = {
    AppError,
    NotFoundError,
    ValidationError,
    UnauthorizedError,
    ForbiddenError,
    ConflictError
};
```

## Express 错误处理

### 错误处理中间件

```javascript
const { AppError } = require('./errors');

// 异步错误包装器
const asyncHandler = (fn) => (req, res, next) => {
    Promise.resolve(fn(req, res, next)).catch(next);
};

// 全局错误处理中间件
function errorHandler(err, req, res, next) {
    err.statusCode = err.statusCode || 500;
    err.status = err.status || 'error';
    
    // 开发环境返回详细信息
    if (process.env.NODE_ENV === 'development') {
        return res.status(err.statusCode).json({
            status: err.status,
            message: err.message,
            stack: err.stack,
            error: err
        });
    }
    
    // 生产环境
    if (err.isOperational) {
        return res.status(err.statusCode).json({
            status: err.status,
            message: err.message
        });
    }
    
    // 未知错误
    console.error('未知错误:', err);
    return res.status(500).json({
        status: 'error',
        message: '服务器内部错误'
    });
}

// 404 处理
function notFoundHandler(req, res, next) {
    next(new NotFoundError(`路径 ${req.originalUrl}`));
}

module.exports = { asyncHandler, errorHandler, notFoundHandler };
```

### 使用错误处理

```javascript
const express = require('express');
const { asyncHandler, errorHandler, notFoundHandler } = require('./middleware/error');
const { NotFoundError, ValidationError } = require('./errors');

const app = express();

// 路由
app.get('/users/:id', asyncHandler(async (req, res) => {
    const user = await User.findById(req.params.id);
    
    if (!user) {
        throw new NotFoundError('用户');
    }
    
    res.json(user);
}));

app.post('/users', asyncHandler(async (req, res) => {
    const { name, email } = req.body;
    
    // 验证
    const errors = [];
    if (!name) errors.push({ field: 'name', message: '姓名必填' });
    if (!email) errors.push({ field: 'email', message: '邮箱必填' });
    
    if (errors.length > 0) {
        throw new ValidationError(errors);
    }
    
    const user = await User.create(req.body);
    res.status(201).json(user);
}));

// 404 处理
app.use(notFoundHandler);

// 错误处理
app.use(errorHandler);

app.listen(3000);
```

## 未捕获错误处理

```javascript
// 未捕获的 Promise 拒绝
process.on('unhandledRejection', (reason, promise) => {
    console.error('未处理的 Promise 拒绝:', reason);
    // 生产环境可以发送到日志服务
});

// 未捕获的异常
process.on('uncaughtException', (error) => {
    console.error('未捕获的异常:', error);
    // 严重错误，建议退出进程
    process.exit(1);
});

// 优雅关闭
process.on('SIGTERM', async () => {
    console.log('收到 SIGTERM 信号');
    // 关闭数据库连接
    await mongoose.connection.close();
    // 关闭服务器
    server.close(() => {
        console.log('服务器已关闭');
        process.exit(0);
    });
});
```

## 小结

本节学习了：

1. **错误类型**：同步错误、异步错误
2. **自定义错误类**：AppError、NotFoundError 等
3. **Express 错误处理**：中间件、asyncHandler
4. **全局错误处理**：unhandledRejection、uncaughtException

> **提示**：良好的错误处理应该区分可预期的操作错误和不可预期的程序错误。操作错误应该友好提示用户，程序错误应该记录日志并修复。
