# 2.4 解构与展开运算符

解构赋值和展开运算符是 ES6 引入的重要特性，可以更简洁地处理对象和数组。本节将详细介绍它们的使用方法。

## 数组解构

### 基本语法

```javascript
// 基本解构
const [a, b, c] = [1, 2, 3];
console.log(a, b, c);  // 1 2 3

// 从数组中提取值
const colors = ['红', '绿', '蓝'];
const [red, green, blue] = colors;
console.log(red);    // 红
console.log(green);  // 绿
console.log(blue);   // 蓝

// 跳过元素
const [first, , third] = [1, 2, 3];
console.log(first);   // 1
console.log(third);   // 3

// 默认值
const [x, y, z = 0] = [1, 2];
console.log(x, y, z);  // 1 2 0

const [m, n = 10] = [1, undefined];
console.log(m, n);  // 1 10

const [p, q = 10] = [1, null];
console.log(p, q);  // 1 null（null 不触发默认值）
```

### 剩余元素

```javascript
// 剩余元素
const [head, ...tail] = [1, 2, 3, 4, 5];
console.log(head);  // 1
console.log(tail);  // [2, 3, 4, 5]

// 必须放在最后
// const [...rest, last] = [1, 2, 3];  // 错误

// 克隆数组
const original = [1, 2, 3];
const cloned = [...original];
console.log(cloned);        // [1, 2, 3]
console.log(cloned === original);  // false
```

### 嵌套解构

```javascript
// 嵌套数组解构
const nested = [1, [2, 3], [4, [5, 6]]];

const [a, [b, c], [d, [e, f]]] = nested;
console.log(a, b, c, d, e, f);  // 1 2 3 4 5 6

// 部分解构
const [x, [y]] = nested;
console.log(x, y);  // 1 2
```

### 交换变量

```javascript
// 交换变量值
let a = 1;
let b = 2;

[a, b] = [b, a];

console.log(a);  // 2
console.log(b);  // 1

// 交换对象属性
const obj = { x: 1, y: 2 };
[obj.x, obj.y] = [obj.y, obj.x];
console.log(obj);  // { x: 2, y: 1 }
```

### 函数返回值解构

```javascript
// 解构函数返回值
function getCoordinates() {
    return [10, 20, 30];
}

const [x, y, z] = getCoordinates();
console.log(x, y, z);  // 10 20 30

// 忽略某些返回值
function getUser() {
    return ['张三', 25, 'zhangsan@example.com'];
}

const [name, , email] = getUser();
console.log(name, email);  // 张三 zhangsan@example.com
```

## 对象解构

### 基本语法

```javascript
// 基本解构
const person = {
    name: '张三',
    age: 25,
    email: 'zhangsan@example.com'
};

const { name, age, email } = person;
console.log(name);   // 张三
console.log(age);    // 25
console.log(email);  // zhangsan@example.com

// 顺序无关
const { email: mail, name: userName } = person;
console.log(userName);  // 张三
console.log(mail);      // zhangsan@example.com

// 重命名变量
const { name: personName, age: personAge } = person;
console.log(personName);  // 张三
console.log(personAge);   // 25
```

### 默认值

```javascript
// 默认值
const user = {
    name: '张三',
    age: 25
};

const { name, age, email = '未设置' } = user;
console.log(email);  // 未设置

// 重命名 + 默认值
const { name: userName, phone = '无' } = user;
console.log(userName);  // 张三
console.log(phone);     // 无
```

### 剩余属性

```javascript
// 剩余属性
const { a, b, ...rest } = { a: 1, b: 2, c: 3, d: 4 };
console.log(a);     // 1
console.log(b);     // 2
console.log(rest);  // { c: 3, d: 4 }

// 克隆对象
const original = { x: 1, y: 2, z: 3 };
const cloned = { ...original };
console.log(cloned);  // { x: 1, y: 2, z: 3 }
console.log(cloned === original);  // false
```

### 嵌套解构

```javascript
// 嵌套对象解构
const company = {
    name: '科技公司',
    address: {
        city: '北京',
        district: '海淀区'
    },
    employees: [
        { name: '张三', position: '工程师' },
        { name: '李四', position: '设计师' }
    ]
};

const {
    name: companyName,
    address: { city, district },
    employees: [{ name: firstEmployee }]
} = company;

console.log(companyName);     // 科技公司
console.log(city);            // 北京
console.log(district);        // 海淀区
console.log(firstEmployee);   // 张三
```

### 函数参数解构

```javascript
// 函数参数解构
function greet({ name, age, email = '未设置' }) {
    console.log(`姓名: ${name}`);
    console.log(`年龄: ${age}`);
    console.log(`邮箱: ${email}`);
}

greet({ name: '张三', age: 25 });

// 带默认值的参数解构
function createUser({
    name = '匿名',
    age = 0,
    profile: { avatar = 'default.png', bio = '' } = {}
} = {}) {
    return { name, age, avatar, bio };
}

console.log(createUser());
// { name: '匿名', age: 0, avatar: 'default.png', bio: '' }

console.log(createUser({ name: '张三', profile: { bio: '你好' } }));
// { name: '张三', age: 0, avatar: 'default.png', bio: '你好' }
```

## 展开运算符

### 数组展开

```javascript
// 合并数组
const arr1 = [1, 2, 3];
const arr2 = [4, 5, 6];

const merged = [...arr1, ...arr2];
console.log(merged);  // [1, 2, 3, 4, 5, 6]

// 复制数组
const original = [1, 2, 3];
const copy = [...original];
console.log(copy);  // [1, 2, 3]

// 在任意位置插入
const arr = [1, 5];
const newArr = [...arr.slice(0, 1), 2, 3, 4, ...arr.slice(1)];
console.log(newArr);  // [1, 2, 3, 4, 5]

// 将字符串转为数组
const chars = [...'hello'];
console.log(chars);  // ['h', 'e', 'l', 'l', 'o']

// 将 NodeList 转为数组
// const divs = [...document.querySelectorAll('div')];

// Math 函数应用
const numbers = [1, 5, 3, 9, 2];
console.log(Math.max(...numbers));  // 9
console.log(Math.min(...numbers));  // 1
```

### 对象展开

```javascript
// 合并对象
const obj1 = { a: 1, b: 2 };
const obj2 = { c: 3, d: 4 };

const merged = { ...obj1, ...obj2 };
console.log(merged);  // { a: 1, b: 2, c: 3, d: 4 }

// 覆盖属性
const defaults = {
    theme: 'light',
    language: 'zh-CN',
    notifications: true
};

const userConfig = {
    theme: 'dark'
};

const config = { ...defaults, ...userConfig };
console.log(config);
// { theme: 'dark', language: 'zh-CN', notifications: true }

// 添加属性
const person = { name: '张三', age: 25 };
const personWithEmail = { ...person, email: 'zhangsan@example.com' };
console.log(personWithEmail);
// { name: '张三', age: 25, email: 'zhangsan@example.com' }

// 修改属性
const updatedPerson = { ...person, age: 26 };
console.log(updatedPerson);  // { name: '张三', age: 26 }

// 浅拷贝
const original = {
    name: '张三',
    address: { city: '北京' }
};

const copy = { ...original };
copy.name = '李四';
copy.address.city = '上海';

console.log(original.name);          // 张三
console.log(original.address.city);  // 上海（嵌套对象是引用）
```

### 函数调用展开

```javascript
// 展开数组作为参数
function sum(a, b, c) {
    return a + b + c;
}

const numbers = [1, 2, 3];
console.log(sum(...numbers));  // 6

// 与 apply 对比
console.log(sum.apply(null, numbers));  // 6

// 构造函数
const dateFields = [2024, 0, 1];  // 2024年1月1日
const date = new Date(...dateFields);
console.log(date.toDateString());  // Mon Jan 01 2024

// push 多个元素
const arr = [1, 2, 3];
const toAdd = [4, 5, 6];
arr.push(...toAdd);
console.log(arr);  // [1, 2, 3, 4, 5, 6]
```

## 实际应用场景

### 处理 API 响应

```javascript
// 解构 API 响应
async function fetchUser(id) {
    const response = await fetch(`/api/users/${id}`);
    const { data: { name, email, profile: { avatar } } } = await response.json();
    
    return { name, email, avatar };
}

// 解构数组响应
async function fetchUsers() {
    const response = await fetch('/api/users');
    const [first, second, ...others] = await response.json();
    
    return { first, second, others };
}
```

### React/Redux 中的使用

```javascript
// React 组件 props 解构
function UserCard({ name, age, email, address: { city } }) {
    return `
        <div class="user-card">
            <h2>${name}</h2>
            <p>年龄: ${age}</p>
            <p>邮箱: ${email}</p>
            <p>城市: ${city}</p>
        </div>
    `;
}

// Redux reducer 中的不可变更新
const initialState = {
    users: [],
    loading: false,
    error: null
};

function usersReducer(state = initialState, action) {
    switch (action.type) {
        case 'FETCH_USERS_START':
            return { ...state, loading: true, error: null };
        
        case 'FETCH_USERS_SUCCESS':
            return { ...state, loading: false, users: action.payload };
        
        case 'FETCH_USERS_ERROR':
            return { ...state, loading: false, error: action.payload };
        
        case 'ADD_USER':
            return { ...state, users: [...state.users, action.payload] };
        
        case 'UPDATE_USER':
            return {
                ...state,
                users: state.users.map(user =>
                    user.id === action.payload.id
                        ? { ...user, ...action.payload }
                        : user
                )
            };
        
        case 'DELETE_USER':
            return {
                ...state,
                users: state.users.filter(user => user.id !== action.payload)
            };
        
        default:
            return state;
    }
}
```

### 数组操作

```javascript
// 数组去重
const arr = [1, 2, 2, 3, 3, 3, 4];
const unique = [...new Set(arr)];
console.log(unique);  // [1, 2, 3, 4]

// 数组交集
const arr1 = [1, 2, 3, 4];
const arr2 = [3, 4, 5, 6];
const intersection = arr1.filter(x => arr2.includes(x));
console.log(intersection);  // [3, 4]

// 数组并集
const union = [...new Set([...arr1, ...arr2])];
console.log(union);  // [1, 2, 3, 4, 5, 6]

// 数组差集
const difference = arr1.filter(x => !arr2.includes(x));
console.log(difference);  // [1, 2]

// 二维数组转一维
const nested = [[1, 2], [3, 4], [5, 6]];
const flattened = [].concat(...nested);
console.log(flattened);  // [1, 2, 3, 4, 5, 6]

// 或使用 flat
console.log(nested.flat());  // [1, 2, 3, 4, 5, 6]
```

### 函数参数处理

```javascript
// 必需参数检查
function required(paramName) {
    throw new Error(`参数 ${paramName} 是必需的`);
}

function createUser({
    name = required('name'),
    age = required('age'),
    email = '未设置'
}) {
    return { name, age, email };
}

// createUser({});  // 错误：参数 name 是必需的

// 可选链 + 解构
function processResponse(response) {
    const { data: { user: { name = '匿名' } = {} } = {} } = response || {};
    return name;
}

console.log(processResponse(null));  // 匿名
console.log(processResponse({}));  // 匿名
console.log(processResponse({ data: {} }));  // 匿名
console.log(processResponse({ data: { user: {} } }));  // 匿名
console.log(processResponse({ data: { user: { name: '张三' } } }));  // 张三
```

本节学习了：

1. **数组解构**：基本语法、默认值、剩余元素、嵌套解构
2. **对象解构**：基本语法、重命名、默认值、嵌套解构

> **提示**：解构赋值和展开运算符是现代 JavaScript 开发的基础技能，熟练掌握它们可以让代码更简洁、更易读。注意展开运算符只进行浅拷贝，嵌套对象需要递归处理或使用其他深拷贝方法。
