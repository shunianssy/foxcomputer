# 4.4 事件循环

事件循环是 Node.js 异步非阻塞 I/O 的核心机制。本节将详细介绍事件循环的工作原理。

## 什么是事件循环

Node.js 是单线程的，但通过事件循环可以处理大量并发请求。事件循环负责执行代码、收集和处理事件、执行队列中的子任务。

```javascript
// 单线程示例
console.log('1. 开始');

setTimeout(() => {
    console.log('2. 定时器回调');
}, 0);

Promise.resolve().then(() => {
    console.log('3. Promise 回调');
});

console.log('4. 结束');

// 输出顺序: 1 -> 4 -> 3 -> 2
```

## 事件循环阶段

事件循环分为六个阶段：

```text
   ┌───────────────────────┐
┌─>│        timers         │ 定时器阶段
│  └───────────┬───────────┘
│  ┌───────────┴───────────┐
│  │     pending callbacks │ 待定回调阶段
│  └───────────┬───────────┘
│  ┌───────────┴───────────┐
│  │       idle, prepare   │ 空闲阶段
│  └───────────┬───────────┘
│  ┌───────────┴───────────┐
│  │         poll          │ 轮询阶段
│  └───────────┬───────────┘
│  ┌───────────┴───────────┐
│  │        check          │ 检查阶段
│  └───────────┬───────────┘
│  ┌───────────┴───────────┐
└──┤    close callbacks    │ 关闭回调阶段
   └───────────────────────┘
```

### 各阶段详解

```javascript
// 1. timers 阶段 - 执行 setTimeout/setInterval 回调
setTimeout(() => {
    console.log('timers 阶段');
}, 0);

// 2. pending callbacks 阶段 - 执行系统操作的回调
// 如 TCP 错误、信号处理等

// 3. idle, prepare 阶段 - 内部使用

// 4. poll 阶段 - 执行 I/O 回调
const fs = require('fs');
fs.readFile('file.txt', () => {
    console.log('poll 阶段 - I/O 回调');
});

// 5. check 阶段 - 执行 setImmediate 回调
setImmediate(() => {
    console.log('check 阶段');
});

// 6. close callbacks 阶段 - 执行关闭回调
socket.on('close', () => {
    console.log('close callbacks 阶段');
});
```

## 宏任务和微任务

### 宏任务（Macrotask）

```javascript
// 宏任务包括：
// - setTimeout
// - setInterval
// - setImmediate（Node.js）
// - I/O 操作
// - UI 渲染（浏览器）

setTimeout(() => console.log('宏任务 1'), 0);
setTimeout(() => console.log('宏任务 2'), 0);
setImmediate(() => console.log('宏任务 3'));
```

### 微任务（Microtask）

```javascript
// 微任务包括：
// - Promise.then/catch/finally
// - process.nextTick（Node.js）
// - queueMicrotask
// - MutationObserver（浏览器）

Promise.resolve().then(() => console.log('微任务 1'));
Promise.resolve().then(() => console.log('微任务 2'));
process.nextTick(() => console.log('nextTick'));
```

### 执行顺序

```javascript
// 执行顺序：同步代码 -> 微任务 -> 宏任务

console.log('1. 同步代码');

setTimeout(() => {
    console.log('2. setTimeout');
}, 0);

Promise.resolve().then(() => {
    console.log('3. Promise');
});

process.nextTick(() => {
    console.log('4. nextTick');
});

console.log('5. 同步代码');

// 输出顺序：
// 1. 同步代码
// 5. 同步代码
// 4. nextTick（优先级最高的微任务）
// 3. Promise
// 2. setTimeout
```

## process.nextTick

```javascript
// process.nextTick 优先级高于 Promise
process.nextTick(() => {
    console.log('nextTick 1');
});

Promise.resolve().then(() => {
    console.log('Promise 1');
});

process.nextTick(() => {
    console.log('nextTick 2');
});

// 输出：nextTick 1 -> nextTick 2 -> Promise 1

// nextTick 递归调用会导致 I/O 饥饿
function recursiveNextTick() {
    process.nextTick(recursiveNextTick);
}
// recursiveNextTick();  // 会阻塞事件循环

// 使用 setImmediate 代替
function recursiveImmediate() {
    setImmediate(recursiveImmediate);
}
recursiveImmediate();  // 允许 I/O 执行
```

## setImmediate vs setTimeout

```javascript
// 在 I/O 回调中
const fs = require('fs');

fs.readFile('file.txt', () => {
    setTimeout(() => {
        console.log('setTimeout');
    }, 0);
    
    setImmediate(() => {
        console.log('setImmediate');
    });
});

// 输出顺序：setImmediate -> setTimeout
// 因为 I/O 回调在 poll 阶段执行，下一个阶段是 check（setImmediate）

// 在主模块中
setTimeout(() => {
    console.log('setTimeout');
}, 0);

setImmediate(() => {
    console.log('setImmediate');
});

// 输出顺序不确定，取决于进程性能
```

## 阻塞事件循环

### 阻塞示例

```javascript
// CPU 密集型操作会阻塞事件循环
function heavyComputation(n) {
    let result = 0;
    for (let i = 0; i < n; i++) {
        result += Math.random();
    }
    return result;
}

// 这会阻塞事件循环
console.log('开始');
const result = heavyComputation(1000000000);  // 阻塞
console.log('结束');

// 其他请求会被阻塞
setTimeout(() => {
    console.log('定时器');
}, 100);
```

### 解决方案

```javascript
// 方案1：使用 setImmediate 分片执行
function chunkedComputation(data, chunkSize = 1000) {
    return new Promise((resolve) => {
        let index = 0;
        let result = 0;
        
        function processChunk() {
            const end = Math.min(index + chunkSize, data.length);
            
            while (index < end) {
                result += data[index];
                index++;
            }
            
            if (index < data.length) {
                setImmediate(processChunk);
            } else {
                resolve(result);
            }
        }
        
        processChunk();
    });
}

// 方案2：使用 Worker Threads
const { Worker, isMainThread, parentPort, workerData } = require('worker_threads');

if (isMainThread) {
    const worker = new Worker(__filename, {
        workerData: { n: 1000000000 }
    });
    
    worker.on('message', (result) => {
        console.log('结果:', result);
    });
    
    worker.on('error', (error) => {
        console.error('错误:', error);
    });
    
    worker.on('exit', (code) => {
        if (code !== 0) {
            console.error('Worker 退出码:', code);
        }
    });
} else {
    // Worker 线程中执行
    let result = 0;
    for (let i = 0; i < workerData.n; i++) {
        result += Math.random();
    }
    parentPort.postMessage(result);
}

// 方案3：使用 child_process
const { fork } = require('child_process');

function runInChildProcess(scriptPath, data) {
    return new Promise((resolve, reject) => {
        const child = fork(scriptPath);
        
        child.send(data);
        
        child.on('message', (result) => {
            resolve(result);
            child.kill();
        });
        
        child.on('error', reject);
    });
}
```

## 监控事件循环

```javascript
// 监控事件循环延迟
const { monitorEventLoopDelay } = require('perf_hooks');

const h = monitorEventLoopDelay({
    resolution: 10  // 毫秒
});

h.enable();

// 定期检查
setInterval(() => {
    console.log('事件循环延迟统计:');
    console.log('  最小:', h.min, 'ms');
    console.log('  最大:', h.max, 'ms');
    console.log('  平均:', h.mean, 'ms');
    console.log('  99%', h.percentile(99), 'ms');
    
    h.reset();
}, 10000);

// 使用 perf_hooks 检查
const { performance, PerformanceObserver } = require('perf_hooks');

const obs = new PerformanceObserver((list) => {
    const entries = list.getEntries();
    entries.forEach((entry) => {
        console.log(`${entry.name}: ${entry.duration}ms`);
    });
});

obs.observe({ entryTypes: ['measure', 'function'] });

// 测量函数执行时间
function measureAsync(name, fn) {
    return async function(...args) {
        const start = performance.now();
        const result = await fn.apply(this, args);
        const end = performance.now();
        console.log(`${name}: ${(end - start).toFixed(2)}ms`);
        return result;
    };
}
```

## 最佳实践

```javascript
// 1. 避免阻塞操作
// 不好的做法
app.get('/compute', (req, res) => {
    const result = heavyComputation();  // 阻塞
    res.json({ result });
});

// 好的做法
app.get('/compute', async (req, res) => {
    const result = await runInWorker(heavyComputation);
    res.json({ result });
});

// 2. 合理使用 nextTick 和 setImmediate
// 需要立即执行用 nextTick
process.nextTick(() => {
    // 在 I/O 之前执行
});

// 需要让出控制权用 setImmediate
setImmediate(() => {
    // 让其他 I/O 有机会执行
});

// 3. 分批处理大量数据
async function processLargeArray(array, processFn, batchSize = 100) {
    for (let i = 0; i < array.length; i += batchSize) {
        const batch = array.slice(i, i + batchSize);
        
        for (const item of batch) {
            await processFn(item);
        }
        
        // 让出控制权
        await new Promise(resolve => setImmediate(resolve));
    }
}

// 4. 使用 Promise 而不是回调
// 不好的做法
function badExample(callback) {
    fs.readFile('file.txt', (err, data) => {
        if (err) return callback(err);
        callback(null, data);
    });
}

// 好的做法
async function goodExample() {
    const data = await fs.promises.readFile('file.txt');
    return data;
}
```

## 小结

本节学习了：

1. **事件循环阶段**：timers、pending callbacks、poll、check、close callbacks
2. **宏任务和微任务**：执行顺序和优先级
3. **process.nextTick**：优先级最高的微任务
4. **setImmediate vs setTimeout**：执行时机差异
5. **阻塞问题**：识别和解决方案
6. **监控方法**：性能监控和优化

> **提示**：理解事件循环对于编写高性能 Node.js 应用至关重要。避免在主线程执行 CPU 密集型操作，合理使用 Worker Threads 和 setImmediate 来保持事件循环的响应性。
