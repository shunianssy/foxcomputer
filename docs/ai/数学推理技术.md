# 数学推理 (Mathematical Reasoning)

## 概述

数学推理是大语言模型（LLM）的核心能力之一，指的是模型理解和解决数学问题的能力。这包括算术运算、代数求解、几何推理、概率统计等多个领域。随着 Chain-of-Thought（思维链）等技术的发展，大语言模型的数学推理能力得到了显著提升。

## Chain-of-Thought (思维链)

### CoT 的核心思想

Chain-of-Thought 是提升数学推理能力的关键技术，通过让模型"逐步思考"来提高复杂问题的解决准确率。

```
传统提示 vs CoT 提示：

传统提示：
Q: 小明有 5 个苹果，给了小红 2 个，又买了 3 个，现在有几个？
A: 6 个

CoT 提示：
Q: 小明有 5 个苹果，给了小红 2 个，又买了 3 个，现在有几个？
A: 让我们一步步思考：
   1. 小明最初有 5 个苹果
   2. 给了小红 2 个后，剩下：5 - 2 = 3 个
   3. 又买了 3 个，现在有：3 + 3 = 6 个
   答案是 6 个。
```

### CoT 实现

```python
"""
Chain-of-Thought 数学推理实现
"""
from typing import List, Dict, Optional
from dataclasses import dataclass

@dataclass
class MathStep:
    """数学推理步骤"""
    step_number: int
    description: str
    expression: str
    result: Optional[float] = None


class ChainOfThoughtReasoner:
    """
    思维链数学推理器
    """
    
    def __init__(self, llm_client):
        """
        初始化推理器
        
        Args:
            llm_client: LLM 客户端
        """
        self.llm = llm_client
        
        # CoT 提示模板
        self.cot_prompt = """请一步步解决以下数学问题，展示完整的推理过程：

问题：{question}

请按以下格式回答：
步骤1: [描述] [计算表达式] = [结果]
步骤2: ...
...
最终答案: [答案]

现在开始解答："""

        self.zero_shot_cot = """{question}

请一步步思考并解答。"""
        
    def solve(self, question: str, 
              use_zero_shot: bool = True) -> str:
        """
        解决数学问题
        
        Args:
            question: 数学问题
            use_zero_shot: 是否使用零样本 CoT
            
        Returns:
            解答过程和答案
        """
        if use_zero_shot:
            prompt = self.zero_shot_cot.format(question=question)
        else:
            prompt = self.cot_prompt.format(question=question)
            
        response = self.llm.chat([
            {"role": "user", "content": prompt}
        ])
        
        return response
    
    def solve_with_verification(self, question: str) -> Dict:
        """
        带验证的问题求解
        
        Args:
            question: 数学问题
            
        Returns:
            包含解答和验证结果的字典
        """
        # 第一次求解
        solution1 = self.solve(question)
        
        # 第二次求解（验证）
        verify_prompt = f"""问题：{question}

以下是之前的解答：
{solution1}

请验证上述解答是否正确。如果正确，说明原因；如果错误，请给出正确解答。"""

        verification = self.llm.chat([
            {"role": "user", "content": verify_prompt}
        ])
        
        return {
            "question": question,
            "solution": solution1,
            "verification": verification
        }

### Few-shot CoT

```python
"""
Few-shot Chain-of-Thought 示例
"""

# 数学推理的 Few-shot 示例
MATH_COT_EXAMPLES = [
    {
        "question": "Roger 有 5 个网球。他又买了 2 罐网球，每罐有 3 个。他现在有多少个网球？",
        "solution": """让我们一步步思考：
步骤1: Roger 最初有 5 个网球
步骤2: 他买了 2 罐，每罐 3 个，所以新买了 2 × 3 = 6 个
步骤3: 总共有 5 + 6 = 11 个网球
答案是 11。"""
    },
    {
        "question": "一个篮子里有 24 个苹果，小明拿走了其中的 1/3，小红又拿走了剩下的 1/4。篮子里还剩多少个苹果？",
        "solution": """让我们一步步思考：
步骤1: 篮子里最初有 24 个苹果
步骤2: 小明拿走了 24 × (1/3) = 8 个，剩下 24 - 8 = 16 个
步骤3: 小红拿走了 16 × (1/4) = 4 个，剩下 16 - 4 = 12 个
答案是 12。"""
    },
    {
        "question": "一辆汽车以 60 公里/小时的速度行驶了 2.5 小时，然后以 80 公里/小时的速度行驶了 1.5 小时。这辆汽车总共行驶了多少公里？",
        "solution": """让我们一步步思考：
步骤1: 第一段路程：60 × 2.5 = 150 公里
步骤2: 第二段路程：80 × 1.5 = 120 公里
步骤3: 总路程：150 + 120 = 270 公里
答案是 270 公里。"""
    }
]


class FewShotCoTReasoner:
    """
    Few-shot CoT 推理器
    使用示例引导模型进行推理
    """
    
    def __init__(self, llm_client, examples: List[Dict] = None):
        """
        初始化推理器
        
        Args:
            llm_client: LLM 客户端
            examples: Few-shot 示例
        """
        self.llm = llm_client
        self.examples = examples or MATH_COT_EXAMPLES
        
    def build_prompt(self, question: str, 
                     num_examples: int = 2) -> str:
        """
        构建 Few-shot 提示
        
        Args:
            question: 目标问题
            num_examples: 使用的示例数量
            
        Returns:
            完整提示
        """
        prompt_parts = []
        
        # 添加示例
        for i, example in enumerate(self.examples[:num_examples]):
            prompt_parts.append(f"问题：{example['question']}")
            prompt_parts.append(f"解答：{example['solution']}")
            prompt_parts.append("")  # 空行分隔
            
        # 添加目标问题
        prompt_parts.append(f"问题：{question}")
        prompt_parts.append("解答：")
        
        return "\n".join(prompt_parts)
    
    def solve(self, question: str, 
              num_examples: int = 2) -> str:
        """
        使用 Few-shot CoT 解决问题
        """
        prompt = self.build_prompt(question, num_examples)
        return self.llm.chat([{"role": "user", "content": prompt}])
```

## 高级推理技术

### 1. Tree of Thoughts (思维树)

```python
"""
Tree of Thoughts (ToT) 实现
通过探索多条推理路径来解决问题
"""
from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass
import json

@dataclass
class ThoughtNode:
    """思维节点"""
    thought: str           # 当前思考
    evaluation: float      # 评估分数 (0-1)
    children: List['ThoughtNode'] = None
    
    def __post_init__(self):
        if self.children is None:
            self.children = []


class TreeOfThoughts:
    """
    思维树推理器
    探索多条推理路径，选择最优解
    """
    
    def __init__(self, llm_client, 
                 max_depth: int = 3,
                 beam_width: int = 3):
        """
        初始化思维树
        
        Args:
            llm_client: LLM 客户端
            max_depth: 最大搜索深度
            beam_width: 每层保留的节点数
        """
        self.llm = llm_client
        self.max_depth = max_depth
        self.beam_width = beam_width
        
    def generate_thoughts(self, problem: str, 
                          current_state: str = "",
                          num_thoughts: int = 3) -> List[str]:
        """
        生成多个可能的下一步思考
        
        Args:
            problem: 原始问题
            current_state: 当前状态/已进行的思考
            num_thoughts: 生成的思考数量
            
        Returns:
            思考列表
        """
        prompt = f"""问题：{problem}

当前思考过程：
{current_state if current_state else "（刚开始）"}

请生成 {num_thoughts} 个可能的下一步思考方向，每个方向应该是解决问题的一个合理步骤。
以 JSON 列表格式返回：["思考1", "思考2", "思考3"]"""

        response = self.llm.chat([
            {"role": "user", "content": prompt}
        ])
        
        try:
            # 尝试解析 JSON
            thoughts = json.loads(response)
            return thoughts[:num_thoughts]
        except:
            # 如果解析失败，按行分割
            return response.strip().split('\n')[:num_thoughts]
    
    def evaluate_thought(self, problem: str, 
                         thought: str,
                         current_state: str) -> float:
        """
        评估思考的质量
        
        Args:
            problem: 原始问题
            thought: 待评估的思考
            current_state: 当前状态
            
        Returns:
            评估分数 (0-1)
        """
        prompt = f"""问题：{problem}

当前思考过程：
{current_state}

下一步思考：{thought}

请评估这个下一步思考的质量，考虑：
1. 是否朝着正确方向前进
2. 是否有助于解决问题
3. 逻辑是否合理

只返回一个 0 到 1 之间的分数，不要其他内容。"""

        response = self.llm.chat([
            {"role": "user", "content": prompt}
        ])
        
        try:
            score = float(response.strip())
            return max(0, min(1, score))  # 限制在 [0, 1]
        except:
            return 0.5
    
    def solve(self, problem: str) -> Tuple[str, List[str]]:
        """
        使用思维树解决问题
        
        Args:
            problem: 数学问题
            
        Returns:
            (最终答案, 推理路径)
        """
        # 初始化
        current_nodes = [ThoughtNode(thought="", evaluation=1.0)]
        best_path = []
        
        for depth in range(self.max_depth):
            next_nodes = []
            
            for node in current_nodes:
                # 构建当前状态
                current_state = self._build_state(node)
                
                # 生成下一步思考
                thoughts = self.generate_thoughts(
                    problem, current_state, self.beam_width
                )
                
                for thought in thoughts:
                    # 评估思考
                    score = self.evaluate_thought(
                        problem, thought, current_state
                    )
                    
                    # 创建新节点
                    new_node = ThoughtNode(
                        thought=thought,
                        evaluation=score * node.evaluation
                    )
                    node.children.append(new_node)
                    next_nodes.append(new_node)
            
            # 选择最优的几个节点继续
            next_nodes.sort(key=lambda x: x.evaluation, reverse=True)
            current_nodes = next_nodes[:self.beam_width]
            
            # 记录最佳路径
            if current_nodes:
                best_path.append(current_nodes[0].thought)
        
        # 生成最终答案
        final_prompt = f"""问题：{problem}

推理过程：
{chr(10).join(f'{i+1}. {t}' for i, t in enumerate(best_path) if t)}

请根据以上推理给出最终答案："""

        final_answer = self.llm.chat([
            {"role": "user", "content": final_prompt}
        ])
        
        return final_answer, best_path
    
    def _build_state(self, node: ThoughtNode) -> str:
        """构建从根到当前节点的状态"""
        path = []
        current = node
        while current:
            if current.thought:
                path.append(current.thought)
            # 这里简化处理，实际需要追踪父节点
            break
        return "\n".join(reversed(path))


# 使用示例
if __name__ == "__main__":
    # 示例：需要实际的 LLM 客户端
    # tot = TreeOfThoughts(llm_client)
    # answer, path = tot.solve("一个数加上 5 后乘以 3 等于 30，求这个数。")
    # print(f"答案: {answer}")
    # print(f"推理路径: {path}")

### 2. Self-Consistency (自一致性)

```python
"""
Self-Consistency 自一致性推理
通过多次采样并选择最一致的答案
"""
from typing import List, Dict, Tuple
from collections import Counter
import re

class SelfConsistencyReasoner:
    """
    自一致性推理器
    多次采样，选择最一致的答案
    """
    
    def __init__(self, llm_client, 
                 num_samples: int = 10,
                 temperature: float = 0.7):
        """
        初始化自一致性推理器
        
        Args:
            llm_client: LLM 客户端
            num_samples: 采样次数
            temperature: 采样温度
        """
        self.llm = llm_client
        self.num_samples = num_samples
        self.temperature = temperature
        
    def extract_answer(self, response: str) -> str:
        """
        从响应中提取答案
        
        Args:
            response: 模型响应
            
        Returns:
            提取的答案
        """
        # 尝试匹配常见答案格式
        patterns = [
            r'答案是[：:]\s*(.+)',
            r'答案[是为][：:]\s*(.+)',
            r'最终答案[：:]\s*(.+)',
            r'因此[，,]?\s*(.+)',
            r'所以[，,]?\s*(.+)',
            r'=\s*(\d+(?:\.\d+)?)\s*$',  # 数字答案
        ]
        
        for pattern in patterns:
            match = re.search(pattern, response)
            if match:
                return match.group(1).strip()
                
        # 如果没有匹配，返回最后一行
        lines = response.strip().split('\n')
        return lines[-1].strip()
    
    def solve(self, question: str) -> Dict:
        """
        使用自一致性解决问题
        
        Args:
            question: 数学问题
            
        Returns:
            包含答案和详细信息的字典
        """
        prompt = f"""{question}

请一步步思考并解答，最后明确给出答案。"""
        
        # 多次采样
        responses = []
        answers = []
        
        for _ in range(self.num_samples):
            response = self.llm.chat(
                [{"role": "user", "content": prompt}],
                temperature=self.temperature
            )
            responses.append(response)
            answers.append(self.extract_answer(response))
        
        # 统计答案频率
        answer_counts = Counter(answers)
        most_common = answer_counts.most_common(1)[0]
        
        return {
            "question": question,
            "best_answer": most_common[0],
            "confidence": most_common[1] / self.num_samples,
            "answer_distribution": dict(answer_counts),
            "all_responses": responses
        }
    
    def solve_with_analysis(self, question: str) -> Dict:
        """
        带详细分析的问题求解
        """
        result = self.solve(question)
        
        # 分析答案分布
        analysis = f"""问题：{question}

自一致性分析：
- 采样次数：{self.num_samples}
- 最常见答案：{result['best_answer']}
- 置信度：{result['confidence']:.1%}

答案分布：
"""
        for answer, count in sorted(
            result['answer_distribution'].items(),
            key=lambda x: x[1],
            reverse=True
        ):
            analysis += f"  - {answer}: {count} 次 ({count/self.num_samples:.1%})\n"
            
        result['analysis'] = analysis
        return result


# 使用示例

### 3. Program-of-Thoughts (程序思维)

```python
"""
Program-of-Thoughts (PoT) 实现
通过生成代码来解决数学问题
"""
from typing import List, Dict, Optional
import subprocess
import tempfile
import os

class ProgramOfThoughts:
    """
    程序思维推理器
    生成并执行代码来解决数学问题
    """
    
    def __init__(self, llm_client):
        """
        初始化 PoT 推理器
        
        Args:
            llm_client: LLM 客户端
        """
        self.llm = llm_client
        
        # 代码生成提示
        self.code_prompt = """请编写 Python 代码来解决以下数学问题。
代码应该：
1. 定义必要的变量
2. 进行计算
3. 打印最终答案

问题：{question}

请只输出 Python 代码，不要其他内容。代码如下："""

        # 带解释的代码生成
        self.explained_code_prompt = """请编写 Python 代码来解决以下数学问题，并添加注释解释每一步。

问题：{question}

请按以下格式输出：
```python
# 代码实现
...
```

解释：
..."""
        
    def generate_code(self, question: str) -> str:
        """
        生成解决问题的代码
        
        Args:
            question: 数学问题
            
        Returns:
            Python 代码
        """
        response = self.llm.chat([
            {"role": "user", "content": self.code_prompt.format(question=question)}
        ])
        
        # 提取代码块
        if '```python' in response:
            code = response.split('```python')[1].split('```')[0]
        elif '```' in response:
            code = response.split('```')[1].split('```')[0]
        else:
            code = response
            
        return code.strip()
    
    def execute_code(self, code: str, 
                     timeout: int = 10) -> Dict:
        """
        执行代码并获取结果
        
        Args:
            code: Python 代码
            timeout: 超时时间（秒）
            
        Returns:
            执行结果
        """
        try:
            # 创建临时文件
            with tempfile.NamedTemporaryFile(
                mode='w', 
                suffix='.py', 
                delete=False
            ) as f:
                f.write(code)
                temp_path = f.name
                
            try:
                # 执行代码
                result = subprocess.run(
                    ['python', temp_path],
                    capture_output=True,
                    text=True,
                    timeout=timeout
                )
                
                return {
                    "success": result.returncode == 0,
                    "output": result.stdout,
                    "error": result.stderr,
                    "code": code
                }
            finally:
                # 清理临时文件
                os.unlink(temp_path)
                
        except subprocess.TimeoutExpired:
            return {
                "success": False,
                "output": "",
                "error": "执行超时",
                "code": code
            }
        except Exception as e:
            return {
                "success": False,
                "output": "",
                "error": str(e),
                "code": code
            }
    
    def solve(self, question: str) -> Dict:
        """
        使用 PoT 解决问题
        
        Args:
            question: 数学问题
            
        Returns:
            解决结果
        """
        # 生成代码
        code = self.generate_code(question)
        
        # 执行代码
        execution = self.execute_code(code)
        
        # 如果执行失败，尝试修复
        if not execution['success']:
            fix_prompt = f"""以下代码执行时出错：

代码：
{code}

错误：
{execution['error']}

请修复代码并重新输出完整的 Python 代码："""

            fixed_code = self.llm.chat([
                {"role": "user", "content": fix_prompt}
            ])
            
            if '```python' in fixed_code:
                fixed_code = fixed_code.split('```python')[1].split('```')[0]
            elif '```' in fixed_code:
                fixed_code = fixed_code.split('```')[1].split('```')[0]
                
            execution = self.execute_code(fixed_code.strip())
            code = fixed_code.strip()
        
        return {
            "question": question,
            "code": code,
            "answer": execution['output'].strip() if execution['success'] else None,
            "success": execution['success'],
            "error": execution.get('error')
        }

class MathEvaluator:
    """
    数学推理评估器
    """
    
    def __init__(self, llm_client):
        """
        初始化评估器
        
        Args:
            llm_client: LLM 客户端
        """
        self.llm = llm_client
        
    def evaluate_single(self, question: str, 
                        expected_answer: str,
                        model_answer: str) -> Dict:
        """
        评估单个问题的答案
        
        Args:
            question: 问题
            expected_answer: 期望答案
            model_answer: 模型答案
            
        Returns:
            评估结果
        """
        # 提取数值答案
        def extract_number(text):
            import re
            numbers = re.findall(r'-?\d+\.?\d*', text)
            return float(numbers[-1]) if numbers else None
        
        expected_num = extract_number(expected_answer)
        model_num = extract_number(model_answer)
        
        is_correct = False
        if expected_num is not None and model_num is not None:
            # 允许小误差
            is_correct = abs(expected_num - model_num) < 0.01
        
        return {
            "question": question,
            "expected": expected_answer,
            "model": model_answer,
            "expected_num": expected_num,
            "model_num": model_num,
            "is_correct": is_correct
        }
    
    def evaluate_dataset(self, 
                         dataset: List[Dict],
                         solver) -> Dict:
        """
        评估整个数据集
        
        Args:
            dataset: 数据集 [{"question": ..., "answer": ...}, ...]
            solver: 问题求解函数
            
        Returns:
            评估统计
        """
        results = []
        correct = 0
        
        for item in dataset:
            model_answer = solver(item['question'])
            eval_result = self.evaluate_single(
                item['question'],
                item['answer'],
                model_answer
            )
            results.append(eval_result)
            
            if eval_result['is_correct']:
                correct += 1
        
        return {
            "total": len(dataset),
            "correct": correct,
            "accuracy": correct / len(dataset) if dataset else 0,
            "details": results
        }
```

"""

def decompose_problem(problem: str, llm_client) -> List[str]:
    """
    将复杂问题分解为子问题
    
    Args:
        problem: 原始问题
        llm_client: LLM 客户端
        
    Returns:
        子问题列表
    """
    prompt = f"""请将以下数学问题分解为更简单的子问题：

问题：{problem}

请列出需要解决的子问题（每行一个）："""

    response = llm_client.chat([
        {"role": "user", "content": prompt}
    ])
    
    # 解析子问题
    subproblems = [
        line.strip() 
        for line in response.strip().split('\n')
        if line.strip() and not line.startswith('#')
    ]
    
    return subproblems
```

class FormulaRetriever:
    """
    数学公式检索器
    """
    
    def __init__(self, llm_client, formulas: Dict = None):
        self.llm = llm_client
        self.formulas = formulas or MATH_FORMULAS
        
    def find_relevant_formulas(self, problem: str) -> List[Dict]:
        """
        找出与问题相关的公式
        
        Args:
            problem: 数学问题
            
        Returns:
            相关公式列表
        """
        prompt = f"""问题：{problem}

可用的数学公式：
{chr(10).join(f'- {name}: {info["formula"]}' for name, info in self.formulas.items())}

请列出解决这个问题可能需要的公式名称（每行一个）："""

        response = self.llm.chat([
            {"role": "user", "content": prompt}
        ])
        
        # 匹配公式
        relevant = []
        for name in self.formulas:
            if name in response:
                relevant.append({
                    "name": name,
                    **self.formulas[name]
                })
                
        return relevant
```
