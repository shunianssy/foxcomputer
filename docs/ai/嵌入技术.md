# 嵌入技术 (Embedding)

## 概述

嵌入模型是一种将高维、离散的输入数据（如文本、图像、音频等）转换成低维、连续的数值向量的机器学习技术。每个对象（如一个词、一句话、一张图）被映射为一个浮点数数组，这个数组就是该对象的"嵌入向量"或"嵌入表示"。

## 核心概念

### 什么是嵌入？

嵌入是将离散对象映射到连续向量空间的过程：

```
文本: "人工智能改变世界"
        ↓ 嵌入模型
向量: [0.123, -0.456, 0.789, ..., 0.234]  # 例如 768 维
```

### 嵌入的关键特性

1. **语义相似性**：语义相近的内容在向量空间中距离更近
2. **维度压缩**：将高维稀疏表示压缩为低维稠密向量
3. **可计算性**：向量之间可以进行数学运算

## 嵌入向量维度

不同模型的嵌入维度：

| 模型 | 维度 | 适用场景 |
|------|------|----------|
| Word2Vec | 300 | 词级别表示 |
| GloVe | 300 | 词级别表示 |
| BERT-Base | 768 | 句子/文档表示 |
| BERT-Large | 1024 | 句子/文档表示 |
| Sentence-BERT | 384-768 | 语义相似度 |
| OpenAI text-embedding-3-small | 1536 | 通用文本嵌入 |
| OpenAI text-embedding-3-large | 3072 | 高质量文本嵌入 |

## 嵌入技术的发展历程

```
词嵌入时代 (2013-2018)
├── Word2Vec (2013): Skip-gram, CBOW
├── GloVe (2014): 全局词向量
└── FastText (2016): 子词嵌入

上下文嵌入时代 (2018-至今)
├── ELMo (2018): 双向 LSTM
├── BERT (2018): 双向 Transformer
├── GPT 系列: 单向 Transformer
└── 现代嵌入模型: Sentence-BERT, E5, BGE 等
```

### 1. Word2Vec

```python
"""
Word2Vec 实现
包含 Skip-gram 和 CBOW 两种架构
"""
import numpy as np
from collections import Counter
import torch
import torch.nn as nn
import torch.optim as optim

class Word2VecSkipGram(nn.Module):
    """
    Skip-gram 模型
    根据中心词预测上下文词
    """
    def __init__(self, vocab_size, embedding_dim):
        super().__init__()
        # 中心词嵌入矩阵
        self.center_embeddings = nn.Embedding(vocab_size, embedding_dim)
        # 上下文词嵌入矩阵
        self.context_embeddings = nn.Embedding(vocab_size, embedding_dim)
        
        # 初始化
        init_range = 0.5 / embedding_dim
        self.center_embeddings.weight.data.uniform_(-init_range, init_range)
        self.context_embeddings.weight.data.uniform_(-init_range, init_range)
        
    def forward(self, center_word, context_word):
        """
        前向传播
        
        Args:
            center_word: 中心词索引 [batch_size]
            context_word: 上下文词索引 [batch_size]
            
        Returns:
            score: 点积得分 [batch_size]
        """
        center_emb = self.center_embeddings(center_word)  # [batch, dim]
        context_emb = self.context_embeddings(context_word)  # [batch, dim]
        
        # 计算点积
        score = torch.sum(center_emb * context_emb, dim=1)
        return score


class Word2VecTrainer:
    """
    Word2Vec 训练器
    使用负采样优化
    """
    def __init__(self, vocab_size, embedding_dim=100, 
                 window_size=5, neg_samples=5):
        self.vocab_size = vocab_size
        self.embedding_dim = embedding_dim
        self.window_size = window_size
        self.neg_samples = neg_samples
        
        self.model = Word2VecSkipGram(vocab_size, embedding_dim)
        self.optimizer = optim.SGD(self.model.parameters(), lr=0.025)
        
    def train_pair(self, center_word, context_word, negative_words):
        """
        训练一个词对
        
        Args:
            center_word: 中心词索引
            context_word: 正样本上下文词索引
            negative_words: 负样本词索引列表
        """
        self.optimizer.zero_grad()
        
        # 正样本得分
        pos_score = self.model(
            torch.tensor([center_word]),
            torch.tensor([context_word])
        )
        pos_loss = -torch.log(torch.sigmoid(pos_score) + 1e-10)
        
        # 负样本得分
        neg_loss = 0
        for neg_word in negative_words:
            neg_score = self.model(
                torch.tensor([center_word]),
                torch.tensor([neg_word])
            )
            neg_loss += -torch.log(torch.sigmoid(-neg_score) + 1e-10)
        
        # 总损失
        total_loss = pos_loss + neg_loss
        total_loss.backward()
        self.optimizer.step()
        
        return total_loss.item()
    
    def get_embedding(self, word_idx):
        """获取词嵌入"""
        with torch.no_grad():
            return self.model.center_embeddings.weight[word_idx].numpy()
```

### 2. Sentence-BERT

```python
"""
Sentence-BERT 实现
用于生成句子级别的嵌入向量
"""
from sentence_transformers import SentenceTransformer
import numpy as np
from typing import List, Union

class SentenceEmbedding:
    """
    句子嵌入类
    封装 Sentence-Transformers 库
    """
    def __init__(self, model_name: str = 'all-MiniLM-L6-v2'):
        """
        初始化嵌入模型
        
        Args:
            model_name: 模型名称，常用选项:
                - 'all-MiniLM-L6-v2': 快速，384维，适合大多数场景
                - 'all-mpnet-base-v2': 高质量，768维
                - 'paraphrase-multilingual-MiniLM-L12-v2': 多语言支持
                - 'text2vec-chinese': 中文优化
        """
        self.model = SentenceTransformer(model_name)
        self.model_name = model_name
        
    def encode(self, sentences: Union[str, List[str]], 
               normalize: bool = True) -> np.ndarray:
        """
        将句子编码为嵌入向量
        
        Args:
            sentences: 单个句子或句子列表
            normalize: 是否对向量进行 L2 归一化
            
        Returns:
            嵌入向量数组
        """
        if isinstance(sentences, str):
            sentences = [sentences]
            
        embeddings = self.model.encode(sentences, normalize_embeddings=normalize)
        return embeddings
    
    def similarity(self, sentence1: str, sentence2: str) -> float:
        """
        计算两个句子的语义相似度
        
        Args:
            sentence1: 第一个句子
            sentence2: 第二个句子
            
        Returns:
            相似度分数 [0, 1]
        """
        embeddings = self.encode([sentence1, sentence2])
        
        # 计算余弦相似度
        similarity = np.dot(embeddings[0], embeddings[1])
        return float(similarity)
    
    def most_similar(self, query: str, candidates: List[str], 
                     top_k: int = 5) -> List[tuple]:
        """
        找出与查询最相似的候选句子
        
        Args:
            query: 查询句子
            candidates: 候选句子列表
            top_k: 返回前 k 个结果
            
        Returns:
            [(句子, 相似度), ...] 列表
        """
        query_emb = self.encode(query)
        candidate_embs = self.encode(candidates)
        
        # 计算相似度
        similarities = np.dot(candidate_embs, query_emb)
        
        # 获取 top-k
        top_indices = np.argsort(similarities)[::-1][:top_k]
        
        results = [
            (candidates[i], float(similarities[i]))
            for i in top_indices
        ]
        return results


# 使用示例
if __name__ == "__main__":
    # 创建嵌入模型
    embedder = SentenceEmbedding('all-MiniLM-L6-v2')
    
    # 示例句子
    sentences = [
        "人工智能正在改变世界",
        "AI技术正在重塑各行各业",
        "今天天气很好",
        "机器学习是人工智能的核心技术"
    ]
    
    # 生成嵌入
    embeddings = embedder.encode(sentences)
    print(f"嵌入形状: {embeddings.shape}")  # (4, 384)
    
    # 计算相似度
    sim = embedder.similarity(sentences[0], sentences[1])
    print(f"'{sentences[0]}' 与 '{sentences[1]}' 相似度: {sim:.4f}")
    
    # 查找最相似句子
    query = "深度学习技术"
    results = embedder.most_similar(query, sentences, top_k=2)
    print(f"\n与 '{query}' 最相似的句子:")
    for sentence, score in results:
        print(f"  [{score:.4f}] {sentence}")
```

```python
"""
OpenAI Embeddings API 使用示例
"""
import os
from openai import OpenAI
import numpy as np
from typing import List, Union

class OpenAIEmbedding:
    """
    OpenAI 嵌入模型封装
    """
    def __init__(self, model: str = "text-embedding-3-small"):
        """
        初始化 OpenAI 客户端
        
        Args:
            model: 模型名称
                - text-embedding-3-small: 1536维，性价比高
                - text-embedding-3-large: 3072维，最高质量
                - text-embedding-ada-002: 旧版模型
        """
        self.client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))
        self.model = model
        
    def encode(self, texts: Union[str, List[str]]) -> np.ndarray:
        """
        生成文本嵌入
        
        Args:
            texts: 单个文本或文本列表
            
        Returns:
            嵌入向量数组
        """
        if isinstance(texts, str):
            texts = [texts]
            
        response = self.client.embeddings.create(
            input=texts,
            model=self.model
        )
        
        embeddings = np.array([item.embedding for item in response.data])
        return embeddings
    
    def encode_with_dimensions(self, texts: Union[str, List[str]], 
                               dimensions: int = 512) -> np.ndarray:
        """
        生成指定维度的嵌入（仅支持 text-embedding-3 系列）
        
        Args:
            texts: 文本
            dimensions: 目标维度（最小 256）
            
        Returns:
            嵌入向量数组
        """
        if isinstance(texts, str):
            texts = [texts]
            
        response = self.client.embeddings.create(
            input=texts,
            model=self.model,
            dimensions=dimensions
        )
        
        embeddings = np.array([item.embedding for item in response.data])
        return embeddings
```

## 相似度计算方法

```python
"""
向量相似度计算方法
"""
import numpy as np
from typing import List

def cosine_similarity(vec1: np.ndarray, vec2: np.ndarray) -> float:
    """
    余弦相似度
    衡量两个向量方向的相似程度
    
    Args:
        vec1: 第一个向量
        vec2: 第二个向量
        
    Returns:
        相似度 [-1, 1]，1 表示完全相同方向
    """
    dot_product = np.dot(vec1, vec2)
    norm1 = np.linalg.norm(vec1)
    norm2 = np.linalg.norm(vec2)
    
    if norm1 == 0 or norm2 == 0:
        return 0.0
    
    return float(dot_product / (norm1 * norm2))


def euclidean_distance(vec1: np.ndarray, vec2: np.ndarray) -> float:
    """
    欧氏距离
    衡量两个向量之间的直线距离
    
    Returns:
        距离值，越小越相似
    """
    return float(np.linalg.norm(vec1 - vec2))


def manhattan_distance(vec1: np.ndarray, vec2: np.ndarray) -> float:
    """
    曼哈顿距离
    各维度差值绝对值之和
    
    Returns:
        距离值，越小越相似
    """
    return float(np.sum(np.abs(vec1 - vec2)))


def dot_product_similarity(vec1: np.ndarray, vec2: np.ndarray) -> float:
    """
    点积相似度
    适用于已归一化的向量
    
    Returns:
        点积值
    """
    return float(np.dot(vec1, vec2))


def batch_cosine_similarity(query: np.ndarray, 
                            candidates: np.ndarray) -> np.ndarray:
    """
    批量计算余弦相似度
    
    Args:
        query: 查询向量 [dim]
        candidates: 候选向量矩阵 [n, dim]
        
    Returns:
        相似度数组 [n]
    """
    # 归一化
    query_norm = query / (np.linalg.norm(query) + 1e-10)
    candidates_norm = candidates / (np.linalg.norm(candidates, axis=1, keepdims=True) + 1e-10)
    
    # 批量点积
    similarities = np.dot(candidates_norm, query_norm)
    return similarities
```

## 向量数据库集成

```python
"""
向量数据库使用示例
使用 FAISS 进行高效向量检索
"""
import numpy as np
import faiss
from typing import List, Tuple

class VectorStore:
    """
    基于 FAISS 的向量存储
    """
    def __init__(self, dimension: int, index_type: str = 'flat'):
        """
        初始化向量存储
        
        Args:
            dimension: 向量维度
            index_type: 索引类型
                - 'flat': 暴力搜索，精确但慢
                - 'ivf': 倒排索引，快速近似
                - 'hnsw': 层次导航小世界图，高效近似
        """
        self.dimension = dimension
        self.index_type = index_type
        self.id_map = {}  # ID 到索引的映射
        self.index_counter = 0
        
        # 创建索引
        if index_type == 'flat':
            self.index = faiss.IndexFlatIP(dimension)  # 内积索引
        elif index_type == 'ivf':
            quantizer = faiss.IndexFlatIP(dimension)
            self.index = faiss.IndexIVFFlat(quantizer, dimension, 100)
        elif index_type == 'hnsw':
            self.index = faiss.IndexHNSWFlat(dimension, 32)
        else:
            raise ValueError(f"不支持的索引类型: {index_type}")
            
    def add(self, vectors: np.ndarray, ids: List[str] = None) -> None:
        """
        添加向量到索引
        
        Args:
            vectors: 向量数组 [n, dim]
            ids: 可选的 ID 列表
        """
        # 确保是 float32 类型
        vectors = vectors.astype('float32')
        
        # 归一化向量（用于余弦相似度）
        faiss.normalize_L2(vectors)
        
        # 添加到索引
        start_idx = self.index_counter
        self.index.add(vectors)
        
        # 更新 ID 映射
        n = len(vectors)
        if ids:
            for i, id_ in enumerate(ids):
                self.id_map[start_idx + i] = id_
        self.index_counter += n
        
    def search(self, query: np.ndarray, k: int = 5) -> List[Tuple[str, float]]:
        """
        搜索最相似的向量
        
        Args:
            query: 查询向量 [dim] 或 [n, dim]
            k: 返回数量
            
        Returns:
            [(id, 相似度), ...] 列表
        """
        if query.ndim == 1:
            query = query.reshape(1, -1)
            
        query = query.astype('float32')
        faiss.normalize_L2(query)
        
        # 搜索
        similarities, indices = self.index.search(query, k)
        
        # 整理结果
        results = []
        for i in range(min(k, len(indices[0]))):
            idx = int(indices[0][i])
            sim = float(similarities[0][i])
            id_ = self.id_map.get(idx, str(idx))
            results.append((id_, sim))
            
        return results
    
    def save(self, path: str) -> None:
        """保存索引到文件"""
        faiss.write_index(self.index, path)
        
    def load(self, path: str) -> None:
        """从文件加载索引"""
        self.index = faiss.read_index(path)


# 使用示例
if __name__ == "__main__":
    # 创建向量存储
    store = VectorStore(dimension=384, index_type='flat')
    
    # 模拟数据
    np.random.seed(42)
    vectors = np.random.randn(100, 384).astype('float32')
    ids = [f"doc_{i}" for i in range(100)]
    
    # 添加向量
    store.add(vectors, ids)
    print(f"已添加 {store.index_counter} 个向量")
    
    # 搜索
    query = np.random.randn(384)
    results = store.search(query, k=5)
    
    print("\n搜索结果:")
    for id_, sim in results:
        print(f"  {id_}: {sim:.4f}")
```

```python
"""
语义搜索示例
"""
class SemanticSearch:
    """语义搜索引擎"""
    
    def __init__(self, embedder, vector_store):
        self.embedder = embedder
        self.vector_store = vector_store
        
    def index_documents(self, documents: List[str]) -> None:
        """索引文档"""
        embeddings = self.embedder.encode(documents)
        ids = [f"doc_{i}" for i in range(len(documents))]
        self.vector_store.add(embeddings, ids)
        
    def search(self, query: str, k: int = 5) -> List[Tuple[str, float]]:
        """搜索相关文档"""
        query_emb = self.embedder.encode(query)
        return self.vector_store.search(query_emb, k)
```

### 2. RAG (检索增强生成)

```python
"""
RAG 系统中的嵌入应用
"""
class RAGRetriever:
    """RAG 检索器"""
    
    def __init__(self, embedder, vector_store, documents):
        self.embedder = embedder
        self.vector_store = vector_store
        self.documents = documents
        
        # 初始化时索引所有文档
        self._index_documents()
        
    def _index_documents(self):
        """索引文档"""
        embeddings = self.embedder.encode(self.documents)
        ids = list(range(len(self.documents)))
        self.vector_store.add(embeddings, ids)
        
    def retrieve(self, query: str, k: int = 3) -> List[str]:
        """检索相关文档片段"""
        query_emb = self.embedder.encode(query)
        results = self.vector_store.search(query_emb, k)
        
        # 返回文档内容
        return [self.documents[int(idx)] for idx, _ in results]
```

### 3. 聚类分析

```python
"""
文本聚类示例
"""
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score

def cluster_texts(embedder, texts: List[str], n_clusters: int = 5):
    """
    对文本进行聚类
    
    Args:
        embedder: 嵌入模型
        texts: 文本列表
        n_clusters: 聚类数量
        
    Returns:
        聚类标签和轮廓系数
    """
    # 生成嵌入
    embeddings = embedder.encode(texts)
    
    # K-Means 聚类
    kmeans = KMeans(n_clusters=n_clusters, random_state=42)
    labels = kmeans.fit_predict(embeddings)
    
    # 评估聚类质量
    score = silhouette_score(embeddings, labels)
    
    return labels, score
```

### 3. 批量处理

```python
def batch_encode(embedder, texts: List[str], batch_size: int = 32):
    """批量编码，避免内存溢出"""
    all_embeddings = []
    
    for i in range(0, len(texts), batch_size):
        batch = texts[i:i + batch_size]
        embeddings = embedder.encode(batch)
        all_embeddings.append(embeddings)
        
    return np.vstack(all_embeddings)
```
