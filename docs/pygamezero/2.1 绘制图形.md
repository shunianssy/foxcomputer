# 2.1 绘制图形

Pygame Zero 提供了丰富的绘图函数，可以轻松绘制各种基本图形。

## 基本图形绘制

### 绘制矩形

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

def draw():
    screen.fill((0, 0, 0))
    
    # 绘制矩形
    screen.draw.rect(Rect(100, 100, 200, 150), (255, 0, 0))
    
    # 绘制填充矩形
    screen.draw.filled_rect(Rect(400, 100, 200, 150), (0, 255, 0))

pgzrun.go()
```

## 图形属性

### 颜色

Pygame Zero 使用 RGB 颜色模式，每个颜色由三个值组成（红、绿、蓝），范围是 0-255。

```python
# 常用颜色
RED = (255, 0, 0)
GREEN = (0, 255, 0)
BLUE = (0, 0, 255)
WHITE = (255, 255, 255)
BLACK = (0, 0, 0)
YELLOW = (255, 255, 0)
CYAN = (0, 255, 255)
MAGENTA = (255, 0, 255)

# 自定义颜色
ORANGE = (255, 165, 0)
PURPLE = (128, 0, 128)
PINK = (255, 192, 203)

def draw():
    screen.fill(BLACK)
    screen.draw.rect(Rect(100, 100, 100, 100), RED)
    screen.draw.rect(Rect(250, 100, 100, 100), GREEN)
    screen.draw.rect(Rect(400, 100, 100, 100), BLUE)
```

## 复杂图形示例

### 绘制房子

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

def draw():
    screen.fill((135, 206, 235))  # 天空蓝背景
    
    # 地面
    screen.draw.filled_rect(Rect(0, 450, 800, 150), (34, 139, 34))
    
    # 房子主体
    screen.draw.filled_rect(Rect(300, 250, 200, 200), (139, 69, 19))
    
    # 屋顶
    roof_points = [(280, 250), (400, 150), (520, 250)]
    screen.draw.filled_polygon(roof_points, (178, 34, 34))
    
    # 门
    screen.draw.filled_rect(Rect(380, 350, 40, 100), (101, 67, 33))
    
    # 窗户
    screen.draw.filled_rect(Rect(320, 280, 50, 50), (135, 206, 250))
    screen.draw.filled_rect(Rect(430, 280, 50, 50), (135, 206, 250))
    
    # 窗框
    screen.draw.line((345, 280), (345, 330), (139, 69, 19), 3)
    screen.draw.line((320, 305), (370, 305), (139, 69, 19), 3)
    screen.draw.line((455, 280), (455, 330), (139, 69, 19), 3)
    screen.draw.line((430, 305), (480, 305), (139, 69, 19), 3)

pgzrun.go()
```

## 动态图形

### 移动的矩形

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

rect_x = 100
rect_y = 100
rect_speed = 5

def draw():
    screen.fill((0, 0, 0))
    screen.draw.rect(Rect(rect_x, rect_y, 100, 100), (255, 0, 0))

def update():
    global rect_x, rect_y
    
    if keyboard.left:
        rect_x -= rect_speed
    if keyboard.right:
        rect_x += rect_speed
    if keyboard.up:
        rect_y -= rect_speed
    if keyboard.down:
        rect_y += rect_speed
    
    # 边界检测
    rect_x = max(0, min(WIDTH - 100, rect_x))
    rect_y = max(0, min(HEIGHT - 100, rect_y))

pgzrun.go()
```

## 图形组合

### 创建角色

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

def draw_character(x, y):
    # 身体
    screen.draw.filled_rect(Rect(x - 25, y, 50, 80), (0, 0, 255))
    
    # 头
    screen.draw.filled_circle((x, y - 20), 30, (255, 200, 150))
    
    # 眼睛
    screen.draw.filled_circle((x - 10, y - 25), 5, (0, 0, 0))
    screen.draw.filled_circle((x + 10, y - 25), 5, (0, 0, 0))
    
    # 嘴巴
    screen.draw.line((x - 10, y - 10), (x + 10, y - 10), (0, 0, 0), 2)

def draw():
    screen.fill((255, 255, 255))
    draw_character(400, 300)

pgzrun.go()
```
