# 2.3 文字渲染

在游戏中，文字是重要的 UI 元素，用于显示分数、提示信息等。

## 基本文字显示

### 简单文字

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

def draw():
    screen.fill((0, 0, 0))
    
    # 显示文字
    screen.draw.text("Hello, Pygame Zero!", (200, 280))

pgzrun.go()
```

### 文字参数

```python
import pgzrun
```

## 动态文字

### 显示变量

```python
import pgzrun
import time

WIDTH = 800
HEIGHT = 600

start_time = time.time()

def draw():
    # 计算运行时间
    elapsed = time.time() - start_time
    
    screen.fill((0, 0, 0))
    
    # 显示变量
    screen.draw.text(f"运行时间: {elapsed:.1f} 秒", (200, 280), 
                    fontsize=40, color=(255, 255, 0))

pgzrun.go()
```

### 计数器

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

counter = 0

def draw():
    screen.fill((0, 0, 0))
    screen.draw.text(f"计数器: {counter}", (300, 280), 
                    fontsize=40, color=(255, 255, 0))

    counter += 1

pgzrun.go()
```

### 打字机效果

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

text = "欢迎使用 Pygame Zero！"
displayed_text = ""
index = 0
timer = 0

def draw():
    screen.fill((0, 0, 0))
    screen.draw.text(displayed_text, (200, 280), fontsize=40, 
                    color=(255, 255, 0))

def update():
    global displayed_text, index, timer
    
    timer += 1
    if timer >= 5 and index < len(text):  # 每 5 帧显示一个字符
        displayed_text += text[index]
        index += 1
        timer = 0

pgzrun.go()
```

### 描边文字

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

def draw():
    screen.fill((0, 0, 0))
    
    text = "描边文字"
    x, y = 200, 200
    fontsize = 60
    
    # 绘制描边（8个方向）
    for dx, dy in [(-2, -2), (-2, 0), (-2, 2), (0, -2), 
                   (0, 2), (2, -2), (2, 0), (2, 2)]:
        screen.draw.text(text, (x + dx, y + dy), fontsize=fontsize, 
                        color=(0, 0, 255))
    
    # 绘制主文字
    screen.draw.text(text, (x, y), fontsize=fontsize, color=(255, 255, 255))

pgzrun.go()
```

### 渐变文字

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

def draw():
    screen.fill((0, 0, 0))
    
    text = "渐变文字"
    x, y = 200, 200
    fontsize = 60
    
    # 创建渐变效果
    for i in range(len(text)):
        r = int(255 * i / len(text))
        g = int(255 * (1 - i / len(text)))
        b = 128
        color = (r, g, b)
        screen.draw.text(text[i], (x + i * 50, y), fontsize=fontsize, 
                        color=color)

pgzrun.go()
```

```python
import pgzrun

HEIGHT = 600
```

```python
    screen.fill((0, 0, 0))
    
    # 当前分数
    screen.draw.text(f"分数: {score}", (20, 20), fontsize=30, 
                    color=(255, 255, 0))

                    color=(150, 150, 150))

def update():
    global score
    if keyboard.space:
        score += 10

pgzrun.go()
```
