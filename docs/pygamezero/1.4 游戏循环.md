# 1.4 游戏循环

游戏循环是游戏开发的核心概念。Pygame Zero 自动处理游戏循环，让我们专注于游戏逻辑。

## 什么是游戏循环？

游戏循环是一个不断重复的过程，通常每秒执行 60 次（60 FPS）。每次循环包含两个主要步骤：

1. **更新（Update）**: 处理游戏逻辑
2. **绘制（Draw）**: 渲染游戏画面

```
┌─────────────┐
│ 开始游戏 │
└──────┬──────┘
 │
 ▼
┌─────────────┐
│ update() │ ← 处理输入、移动、碰撞等
└──────┬──────┘
 │
 ▼
┌─────────────┐
│ draw() │ ← 绘制所有游戏元素
└──────┬──────┘
 │
 ▼
┌─────────────┐
│ 等待下一帧 │
└──────┬──────┘
 │
 ▼
 （循环）
```

## update() 函数

`update()` 函数负责处理游戏逻辑，每秒调用 60 次。

### 基本用法

```python
def update():
 # 游戏逻辑代码
 pass
```

### 常见用途

```python
def update():
 # 1. 处理输入
 if keyboard.left:
 player.x -= 5
 
 # 2. 移动对象
 ball.x += ball_speed_x
 ball.y += ball_speed_y
 
 # 3. 碰撞检测
 if player.colliderect(enemy):
 game_over()
 
 # 4. 更新游戏状态
 timer -= 1
 if timer <= 0:
 next_level()
```

### 示例：移动的方块

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

# 方块属性
box_x = 100
box_y = 100
box_size = 50
speed = 5

def draw():
 screen.fill((0, 0, 0))
 screen.draw.rect(Rect(box_x, box_y, box_size, box_size), (255, 0, 0))

def update():
 global box_x, box_y
 
 # 键盘控制
 if keyboard.left:
 box_x -= speed
 if keyboard.right:
 box_x += speed
 if keyboard.up:
 box_y -= speed
 if keyboard.down:
 box_y += speed
 
 # 边界限制
 box_x = max(0, min(WIDTH - box_size, box_x))
 box_y = max(0, min(HEIGHT - box_size, box_y))

pgzrun.go()
```

## draw() 函数

`draw()` 函数负责渲染游戏画面，每秒调用 60 次。

### 基本用法

```python
def draw():
 # 清空屏幕
 screen.fill((0, 0, 0))
 
 # 绘制游戏元素
 player.draw()
 enemy.draw()
 
 # 绘制 UI
 screen.draw.text(f"分数: {score}", (10, 10))
```

### 重要规则

1. **必须清空屏幕**: 每次绘制前都要清空屏幕
2. **绘制顺序**: 后绘制的元素会覆盖先绘制的
3. **不处理逻辑**: `draw()` 只负责绘制，不处理游戏逻辑

### 示例：绘制多个元素

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

def draw():
 # 清空屏幕（必须）
 screen.fill((0, 0, 0))
 
 # 绘制背景
 screen.draw.circle((400, 300), 200, (50, 50, 50))
 
 # 绘制玩家
 screen.draw.rect(Rect(350, 500, 100, 20), (0, 255, 0))
 
 # 绘制敌人
 screen.draw.circle((400, 100), 30, (255, 0, 0))
 
 # 绘制 UI
 screen.draw.text("游戏示例", (350, 50), fontsize=30)
 screen.draw.text("按空格键开始", (350, 550), fontsize=20)

def update():
 # 游戏逻辑
 pass

pgzrun.go()
```

## 时间控制

### 使用 clock 对象

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

# 创建计时器
timer = 0

def update():
 global timer
 
 # 每秒增加 60（因为 update 每秒调用 60 次）
 timer += 1
 
 # 每 60 帧（约 1 秒）执行一次
 if timer % 60 == 0:
 print("1 秒过去了")
 
 # 每 180 帧（约 3 秒）执行一次
 if timer % 180 == 0:
 print("3 秒过去了")

def draw():
 screen.fill((0, 0, 0))
 screen.draw.text(f"计时器: {timer}", (10, 10))

pgzrun.go()
```

### 使用 time 模块

```python
import pgzrun
import time

WIDTH = 800
HEIGHT = 600

start_time = time.time()

def update():
 # 计算经过的时间（秒）
 elapsed = time.time() - start_time
 
 # 每 1 秒执行一次
 if int(elapsed) > int(elapsed - 1/60):
 print(f"经过时间: {elapsed:.1f} 秒")

def draw():
 elapsed = time.time() - start_time
 screen.fill((0, 0, 0))
 screen.draw.text(f"时间: {elapsed:.1f} 秒", (10, 10))

pgzrun.go()
```

## 完整示例：弹球游戏

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

# 游戏对象
paddle = Rect(350, 550, 100, 20)
ball = Rect(390, 290, 20, 20)

# 速度
ball_speed_x = 5
ball_speed_y = 5

# 分数
score = 0

def draw():
 # 清空屏幕
 screen.fill((0, 0, 0))
 
 # 绘制挡板
 screen.draw.rect(paddle, (0, 255, 0))
 
 # 绘制球
 screen.draw.circle((ball.centerx, ball.centery), 10, (255, 0, 0))
 
 # 绘制分数
 screen.draw.text(f"分数: {score}", (10, 10), fontsize=30)

def update():
 global ball_speed_x, ball_speed_y, score
 
 # 移动球
 ball.x += ball_speed_x
 ball.y += ball_speed_y
 
 # 墙壁碰撞
 if ball.left <= 0 or ball.right >= WIDTH:
 ball_speed_x *= -1
 if ball.top <= 0:
 ball_speed_y *= -1
 
 # 挡板碰撞
 if paddle.colliderect(ball) and ball_speed_y > 0:
 ball_speed_y *= -1
 score += 1
 
 # 游戏结束
 if ball.bottom >= HEIGHT:
 ball.center = (400, 300)
 ball_speed_x = 5
 ball_speed_y = 5
 score = 0
 
 # 挡板移动
 if keyboard.left and paddle.left > 0:
 paddle.x -= 8
 if keyboard.right and paddle.right < WIDTH:
 paddle.x += 8

pgzrun.go()
```

## 游戏循环最佳实践

### 1. 分离逻辑和渲染

```python
# 好的做法
def update():
 move_player()
 check_collisions()
 update_score()

def draw():
 screen.fill((0, 0, 0))
 player.draw()
 enemy.draw()
 draw_ui()

# 不好的做法
def draw():
 screen.fill((0, 0, 0))
 player.x += 5 # 不要在 draw 中处理逻辑
 player.draw()
```

### 2. 使用全局变量

```python
# 声明全局变量
score = 0
player_x = 100

def update():
 global score, player_x # 必须声明
 player_x += 5
 score += 1
```

### 3. 保持代码简洁

```python
# 将复杂逻辑拆分成函数
def update():
 handle_input()
 move_objects()
 check_collisions()
 update_game_state()

def handle_input():
 if keyboard.left:
 player.x -= 5
```
