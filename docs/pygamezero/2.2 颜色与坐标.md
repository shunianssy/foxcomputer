# 2.2 颜色与坐标

理解颜色和坐标系统是游戏开发的基础。

## 坐标系统

### 屏幕坐标

Pygame Zero 使用屏幕坐标系：

- 原点 (0, 0) 在屏幕左上角
- X 轴向右增加
- Y 轴向下增加

```
(0, 0) ───────────────► X 轴
  │
  │
  │
  │
  ▼
  Y 轴
```

### 坐标示例

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

def draw():
    screen.fill((0, 0, 0))
    
    # 左上角
    screen.draw.circle((50, 50), 20, (255, 0, 0))
    screen.draw.text("(50, 50)", (80, 40))
    
    # 右上角
    screen.draw.circle((750, 50), 20, (0, 255, 0))
    screen.draw.text("(750, 50)", (680, 40))
    
    # 左下角
    screen.draw.circle((50, 550), 20, (0, 0, 255))
    screen.draw.text("(50, 550)", (80, 540))
    
    # 右下角
    screen.draw.circle((750, 550), 20, (255, 255, 0))
    screen.draw.text("(750, 550)", (680, 540))
    
    # 中心
    screen.draw.circle((400, 300), 30, (255, 0, 255))
    screen.draw.text("(400, 300)", (440, 290))

pgzrun.go()
```

### 相对坐标

```python
import pgzrun
```

## 颜色系统

### RGB 颜色模式

Pygame Zero 使用 RGB 颜色模式，每个颜色由三个值组成：

- R (Red): 红色分量 (0-255)
- G (Green): 绿色分量 (0-255)
- B (Blue): 蓝色分量 (0-255)

```python
# RGB 颜色格式
color = (R, G, B)

# 示例
RED = (255, 0, 0)      # 纯红
GREEN = (0, 255, 0)    # 纯绿
BLUE = (0, 0, 255)     # 纯蓝
WHITE = (255, 255, 255) # 白色
BLACK = (0, 0, 0)       # 黑色
```

### 颜色渐变

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

def draw():
    screen.fill((0, 0, 0))
    
    # 创建渐变效果
    for i in range(100):
        # 从红色到绿色的渐变
        r = 255 - i * 2
        g = i * 2
        b = 0
        color = (r, g, b)
        screen.draw.rect(Rect(50 + i * 7, 100, 7, 400), color)

pgzrun.go()
```

### 动态颜色

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

color_shift = 0

def draw():
    screen.fill((0, 0, 0))
    
    # 动态变化的颜色
    r = int(127 + 127 * __import__('math').sin(color_shift))
    g = int(127 + 127 * __import__('math').sin(color_shift + 2))
    b = int(127 + 127 * __import__('math').sin(color_shift + 4))
    
    color = (r, g, b)
    screen.draw.filled_circle((400, 300), 150, color)

def update():
    global color_shift
    color_shift += 0.05

pgzrun.go()
```

## Rect 对象

`Rect` 对象表示一个矩形区域，包含位置和大小信息。

### 创建 Rect

```python
# Rect(x, y, width, height)
rect = Rect(100, 100, 200, 150)
```

### Rect 属性

```python
rect = Rect(100, 100, 200, 150)

# 位置
print(rect.x)      # 100 - 左上角 X 坐标
print(rect.y)      # 100 - 左上角 Y 坐标

# 大小
print(rect.width)  # 200 - 宽度
print(rect.height) # 150 - 高度

# 边界
print(rect.left)   # 100 - 左边界
print(rect.right)  # 300 - 右边界
print(rect.top)    # 100 - 上边界
print(rect.bottom) # 250 - 下边界

# 中心
print(rect.centerx) # 200 - 中心 X 坐标
print(rect.centery) # 175 - 中心 Y 坐标
print(rect.center)  # (200, 175) - 中心坐标

# 角落
print(rect.topleft)     # (100, 100) - 左上角
print(rect.topright)    # (300, 100) - 右上角
print(rect.bottomleft)  # (100, 250) - 左下角
print(rect.bottomright) # (300, 250) - 右下角
```

### Rect 方法

```python
rect = Rect(100, 100, 200, 150)

# 移动
rect.move_ip(50, 50)  # 向右下移动 50 像素

# 调整大小
rect.inflate_ip(20, 20)  # 宽度和高度各增加 20 像素

# 居中
rect.center = (400, 300)  # 将矩形中心移到 (400, 300)

# 碰撞检测
other_rect = Rect(200, 200, 100, 100)
if rect.colliderect(other_rect):
    print("矩形碰撞！")
```

### Rect 示例

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

# 创建多个矩形
rect1 = Rect(100, 100, 150, 100)
rect2 = Rect(300, 200, 200, 150)
rect3 = Rect(500, 350, 120, 120)

def draw():
    screen.fill((0, 0, 0))
    
    # 绘制矩形
    screen.draw.rect(rect1, (255, 0, 0))
    screen.draw.rect(rect2, (0, 255, 0))
    screen.draw.rect(rect3, (0, 0, 255))
    
    # 显示矩形信息
    screen.draw.text(f"Rect1: {rect1}", (10, 10))
    screen.draw.text(f"Rect2: {rect2}", (10, 30))
    screen.draw.text(f"Rect3: {rect3}", (10, 50))
    
    # 检测碰撞
    if rect1.colliderect(rect2):
        screen.draw.text("Rect1 和 Rect2 碰撞！", (10, 500), color=(255, 255, 0))
    if rect2.colliderect(rect3):
        screen.draw.text("Rect2 和 Rect3 碰撞！", (10, 530), color=(255, 255, 0))

pgzrun.go()
```

## 实用函数

### 颜色转换

```python
def rgb_to_hex(r, g, b):
    """将 RGB 转换为十六进制颜色"""
    return f"#{r:02x}{g:02x}{b:02x}"

def hex_to_rgb(hex_color):
    """将十六进制颜色转换为 RGB"""
    hex_color = hex_color.lstrip('#')
    return tuple(int(hex_color[i:i+2], 16) for i in (0, 2, 4))

# 使用示例
print(rgb_to_hex(255, 0, 0))    # 输出: #ff0000
print(hex_to_rgb("#00ff00"))    # 输出: (0, 255, 0)
```

### 颜色混合

```python
def blend_colors(color1, color2, ratio=0.5):
    """混合两种颜色"""
    r = int(color1[0] * (1 - ratio) + color2[0] * ratio)
    g = int(color1[1] * (1 - ratio) + color2[1] * ratio)
    b = int(color1[2] * (1 - ratio) + color2[2] * ratio)
    return (r, g, b)

# 使用示例
red = (255, 0, 0)
blue = (0, 0, 255)
purple = blend_colors(red, blue, 0.5)  # 输出: (127, 0, 127)
```
