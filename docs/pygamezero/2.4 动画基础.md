# 2.4 动画基础

动画让游戏画面更加生动有趣。本章将学习如何创建各种动画效果。

## 基本动画原理

动画是通过快速连续显示一系列略有不同的图像来产生运动错觉。在 Pygame Zero 中，我们通过在 `update()` 函数中改变对象的位置、大小、颜色等属性来创建动画。

### 移动动画

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

# 球的位置
ball_x = 100
ball_y = 300
ball_speed_x = 5

def draw():
    screen.fill((0, 0, 0))
    screen.draw.circle((ball_x, ball_y), 50, (255, 0, 0))

def update():
    global ball_x
    ball_x += ball_speed_x
    
    # 边界反弹
    if ball_x > WIDTH - 50 or ball_x < 50:
        ball_speed_x *= -1

pgzrun.go()
```

### 弹跳动画

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

# 球的位置和速度
ball_x = 400
ball_y = 100
ball_speed_x = 5
ball_speed_y = 0
gravity = 0.5

def draw():
    screen.fill((0, 0, 0))
    screen.draw.circle((ball_x, ball_y), 30, (255, 0, 0))

def update():
    global ball_y, ball_speed_y
    
    # 应用重力
    ball_speed_y += gravity
    ball_y += ball_speed_y
    
    # 地面反弹
    if ball_y > HEIGHT - 30:
        ball_y = HEIGHT - 30
        ball_speed_y *= -0.8  # 能量损失

pgzrun.go()
```

## 颜色动画

### 颜色循环

```python
import pgzrun
import math

WIDTH = 800
HEIGHT = 600

color_shift = 0

def draw():
    screen.fill((0, 0, 0))
    
    # 动态颜色
    r = int(127 + 127 * math.sin(color_shift))
    g = int(127 + 127 * math.sin(color_shift + 2))
    b = int(127 + 127 * math.sin(color_shift + 4))
    
    color = (r, g, b)
    screen.draw.filled_circle((400, 300), 150, color)

def update():
    global color_shift
    color_shift += 0.05

pgzrun.go()
```

### 脉冲动画

```python
import pgzrun
import math

WIDTH = 800
HEIGHT = 600

timer = 0

def draw():
    screen.fill((0, 0, 0))
    
    # 多个圆圈产生脉冲效果
    for i in range(5):
        radius = int(50 + 30 * math.sin(timer * 0.1 + i))
        alpha = int(255 * (1 - i / 5))
        color = (alpha, 0, 0)
        screen.draw.circle((400, 300), radius, color, 3)

def update():
    global timer
    timer += 1

pgzrun.go()
```

## 旋转动画

### 旋转矩形

```python
import pgzrun
import math

WIDTH = 800
HEIGHT = 600

angle = 0

def draw():
    screen.fill((0, 0, 0))
    
    # 计算旋转后的顶点
    cx, cy = 400, 300  # 中心点
    w, h = 200, 100    # 宽度和高度
    
    # 四个顶点
    points = [
        (-w/2, -h/2),
        (w/2, -h/2),
        (w/2, h/2),
        (-w/2, h/2)
    ]
    
    # 旋转并转换到屏幕坐标
    rotated_points = []
    for x, y in points:
        # 旋转
        rx = x * math.cos(angle) - y * math.sin(angle)
        ry = x * math.sin(angle) + y * math.cos(angle)
        # 平移
        rotated_points.append((cx + rx, cy + ry))
    
    screen.draw.polygon(rotated_points, (255, 0, 0))

def update():
    global angle
    angle += 0.02

pgzrun.go()
```

## 路径动画

### 圆形路径

```python
import pgzrun
import math

WIDTH = 800
HEIGHT = 600

angle = 0

def draw():
    screen.fill((0, 0, 0))
    
    # 绘制路径
    screen.draw.circle((400, 300), 200, (100, 100, 100), 2)
    
    # 计算圆周上的位置
    x = 400 + 200 * math.cos(angle)
    y = 300 + 200 * math.sin(angle)
    
    # 绘制移动的球
    screen.draw.filled_circle((x, y), 20, (255, 0, 0))

def update():
    global angle
    angle += 0.05

pgzrun.go()
```

### 波浪路径

```python
import pgzrun
import math

WIDTH = 800
HEIGHT = 600

x = 0
time_counter = 0

def draw():
    screen.fill((0, 0, 0))
    
    # 计算波浪路径上的位置
    y = 300 + 100 * math.sin(x * 0.02 + time_counter * 0.1)
    
    # 绘制轨迹
    screen.draw.line((0, y), (800, y), (100, 100, 100), 2)
    
    # 绘制移动的球
    screen.draw.filled_circle((x, y), 20, (255, 0, 0))

def update():
    global x, time_counter
    x += 5
    time_counter += 1
    
    if x > WIDTH:
        x = 0

pgzrun.go()
```

## 复合动画

### 综合示例

```python
import pgzrun
import math

WIDTH = 800
HEIGHT = 600

timer = 0

def draw():
    screen.fill((0, 0, 0))
    
    # 1. 颜色变化
    r = int(127 + 127 * math.sin(timer * 0.05))
    g = int(127 + 127 * math.sin(timer * 0.05 + 2))
    b = int(127 + 127 * math.sin(timer * 0.05 + 4))
    color = (r, g, b)
    
    # 2. 大小变化
    radius = int(50 + 30 * math.sin(timer * 0.05))
    
    # 3. 位置变化（圆形路径）
    x = 400 + 150 * math.cos(timer * 0.03)
    y = 300 + 150 * math.sin(timer * 0.03)
    
    # 绘制
    screen.draw.filled_circle((x, y), radius, color)
    
    # 显示信息
    screen.draw.text(f"计时器: {timer}", (10, 10), fontsize=20, 
                    color=(255, 255, 255))

def update():
    global timer
    timer += 1

pgzrun.go()
```

## 动画控制

### 暂停和恢复

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

ball_x = 100
ball_speed = 5
is_paused = False

def draw():
    screen.fill((0, 0, 0))
    
    screen.draw.circle((ball_x, 300), 50, (255, 0, 0))
    
    # 显示状态
    status = "暂停" if is_paused else "运行中"
    screen.draw.text(f"状态: {status}", (10, 10), fontsize=24, 
                    color=(255, 255, 255))
    screen.draw.text("按 P 暂停/恢复", (10, 40), fontsize=20, 
                    color=(150, 150, 150))

def update():
    global ball_x, is_paused
    
    if not is_paused:
        ball_x += ball_speed
        
        if ball_x > WIDTH - 50 or ball_x < 50:
            ball_speed *= -1
    
    if keyboard.p:
        is_paused = not is_paused

pgzrun.go()
```

### 速度控制

```python
import pgzrun

WIDTH = 800
HEIGHT = 600

ball_x = 100
ball_speed = 5

def draw():
    screen.fill((0, 0, 0))
    
    screen.draw.circle((ball_x, 300), 50, (255, 0, 0))
    
    # 显示速度
    screen.draw.text(f"速度: {ball_speed}", (10, 10), fontsize=24, 
                    color=(255, 255, 255))
    screen.draw.text("按 ↑ 加速, 按 ↓ 减速", (10, 40), fontsize=20, 
                    color=(150, 150, 150))

def update():
    global ball_x, ball_speed
    
    ball_x += ball_speed
    
    if ball_x > WIDTH - 50 or ball_x < 50:
        ball_speed *= -1
    
    # 速度控制
    if keyboard.up:
        ball_speed += 0.5
    if keyboard.down:
        ball_speed -= 0.5
    
    # 限制速度范围
    ball_speed = max(1, min(20, ball_speed))

pgzrun.go()
```
