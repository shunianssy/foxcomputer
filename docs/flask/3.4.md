# 3.4 错误处理：优雅地认错

如果用户访问了不存在的页面，默认的 404 页面很丑。我们可以自定义它

## 自定义错误页面

### 404 错误

```python
from flask import render_template

@app.errorhandler(404)
def page_not_found(e):
    # e 是错误对象
    return render_template('404.html'), 404
```

### 500 错误

```python
@app.errorhandler(500)
def internal_server_error(e):
    return "服务器冒烟了，请稍后再试", 500
```

### 其他常见错误

```python
# 403 禁止访问
@app.errorhandler(403)
def forbidden(e):
    return render_template('403.html'), 403

# 400 错误请求
@app.errorhandler(400)
def bad_request(e):
    return render_template('400.html'), 400

# 405 方法不允许
@app.errorhandler(405)
def method_not_allowed(e):
    return render_template('405.html'), 405

# 429 请求过多
@app.errorhandler(429)
def too_many_requests(e):
    return '请求太频繁，请稍后再试', 429
```

## 统一错误处理

### 处理所有 HTTP 错误

```python
@app.errorhandler(Exception)
def handle_exception(e):
    # 记录错误日志
    app.logger.error(f'发生错误: {str(e)}')
    
    # 如果是 HTTP 异常，返回对应状态码
    if isinstance(e, HTTPException):
        return render_template('error.html', error=e), e.code
    
    # 其他异常返回 500
    return render_template('error.html', error=e), 500
```

### 处理特定异常类型

```python
from sqlalchemy.exc import IntegrityError

@app.errorhandler(IntegrityError)
def handle_integrity_error(e):
    return jsonify({'error': '数据已存在或违反约束'}), 409

@app.errorhandler(ValueError)
def handle_value_error(e):
    return jsonify({'error': str(e)}), 400
```

## API 错误响应

### 统一错误格式

```python
from flask import jsonify
from werkzeug.exceptions import HTTPException

class APIError(Exception):
    """自定义 API 错误"""
    def __init__(self, message, status_code=400, payload=None):
        super().__init__()
        self.message = message
        self.status_code = status_code
        self.payload = payload
    
    def to_dict(self):
        rv = dict(self.payload or ())
        rv['error'] = self.message
        rv['status'] = 'error'
        return rv

@app.errorhandler(APIError)
def handle_api_error(error):
    response = jsonify(error.to_dict())
    response.status_code = error.status_code
    return response

# 使用示例
@app.route('/api/user/<int:id>')
def get_user(id):
    user = User.query.get(id)
    if not user:
        raise APIError('用户不存在', status_code=404)
    return jsonify(user.to_dict())
```

### 统一响应格式

```python
def api_response(data=None, message='success', status=200):
    """统一 API 响应格式"""
    return jsonify({
        'status': 'success' if status < 400 else 'error',
        'message': message,
        'data': data
    }), status

@app.route('/api/users')
def get_users():
    users = User.query.all()
    return api_response([u.to_dict() for u in users])

@app.route('/api/users/<int:id>')
def get_user(id):
    user = User.query.get(id)
    if not user:
        return api_response(message='用户不存在', status=404)
    return api_response(user.to_dict())
```

## 使用 abort 中止请求

### 基本用法

```python
from flask import abort

@app.route('/user/<int:id>')
def get_user(id):
    if id > 100:
        abort(404)  # 返回 404 错误
    return f'用户 {id}'
```

### 自定义错误消息

```python
from werkzeug.exceptions import NotFound

@app.route('/user/<int:id>')
def get_user(id):
    user = User.query.get(id)
    if not user:
        abort(NotFound(description='找不到该用户'))
    return render_template('user.html', user=user)
```

### API 中使用 abort

```python
from werkzeug.exceptions import HTTPException

@app.route('/api/user/<int:id>')
def get_user(id):
    user = User.query.get(id)
    if not user:
        abort(Response(
            json.dumps({'error': '用户不存在'}),
            status=404,
            mimetype='application/json'
        ))
    return jsonify(user.to_dict())
```

## 错误日志记录

### 记录错误到文件

```python
import logging
from logging.handlers import RotatingFileHandler

# 配置日志
handler = RotatingFileHandler(
    'logs/error.log', 
    maxBytes=10000000,  # 10MB
    backupCount=5
)
handler.setLevel(logging.ERROR)
handler.setFormatter(logging.Formatter(
    '%(asctime)s %(levelname)s: %(message)s [in %(pathname)s:%(lineno)d]'
))
app.logger.addHandler(handler)

@app.errorhandler(Exception)
def handle_exception(e):
    # 记录错误
    app.logger.error(f'Unhandled exception: {str(e)}', exc_info=True)
    return render_template('500.html'), 500
```

### 发送错误邮件

```python
import smtplib
from email.mime.text import MIMEText

def send_error_email(error, request):
    """发送错误邮件"""
    msg = MIMEText(f"""
    错误信息: {str(error)}
    请求路径: {request.path}
    请求方法: {request.method}
    客户端IP: {request.remote_addr}
    """)
    msg['Subject'] = 'Flask 应用错误'
    msg['From'] = 'app@example.com'
    msg['To'] = 'admin@example.com'
    
    # 发送邮件
    with smtplib.SMTP('smtp.example.com') as server:
        server.send_message(msg)

@app.errorhandler(500)
def handle_500(e):
    send_error_email(e, request)
    return render_template('500.html'), 500
```

if app.debug:

## HTTP 状态码速查

| 状态码 | 说明 | 使用场景 |
|--------|------|----------|
| 400 | Bad Request | 请求参数错误 |
| 401 | Unauthorized | 未认证 |
| 403 | Forbidden | 无权限 |
| 404 | Not Found | 资源不存在 |
| 405 | Method Not Allowed | 方法不允许 |
| 409 | Conflict | 资源冲突 |
| 422 | Unprocessable Entity | 验证失败 |
| 429 | Too Many Requests | 请求过多 |
| 500 | Internal Server Error | 服务器错误 |
| 502 | Bad Gateway | 网关错误 |
| 503 | Service Unavailable | 服务不可用 |

## 小结

| 功能 | 代码 |
|------|------|
| 自定义错误页面 | `@app.errorhandler(404)` |
| 中止请求 | `abort(404)` |
| 自定义错误类 | `class APIError(Exception)` |
| 记录错误日志 | `app.logger.error()` |
| 统一错误格式 | `jsonify({'error': message})` |

> 💡 **提示**：生产环境中，不要向用户展示详细的错误信息（如堆栈跟踪），这可能会暴露敏感信息。记录详细日志，向用户显示友好的错误页面。
