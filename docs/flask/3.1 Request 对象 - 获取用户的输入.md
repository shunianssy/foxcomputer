# 3.1 Request 对象：获取用户的输入

当用户在浏览器里点击按钮或输入文字时，数据会通过 request 对象传给 Flask。你只需要导入并使用它

## 导入 Request

```python
from flask import request
```

## 查询参数 (GET)

获取 URL 问号后面的数据，如 `?page=1&search=flask`

```python
@app.route('/search')
def search():
    # 获取单个参数
    keyword = request.args.get('q')
    
    # 获取参数，带默认值
    page = request.args.get('page', '1', type=int)
    
    # 获取参数列表（多个同名参数）
    tags = request.args.getlist('tag')
    
    # 获取所有参数
    all_args = request.args.to_dict()
    
    return f'搜索关键词: {keyword}, 页码: {page}'
```

访问：`http://localhost:5000/search?q=flask&page=2&tag=python&tag=web`

## JSON 数据

处理 API 请求中的 JSON 数据

```python
from flask import jsonify

@app.route('/api/users', methods=['POST'])
def create_user():
    # 方式一
    data = request.get_json()
    
    # 方式二（简写）
    data = request.json
    
    # 检查是否为 JSON 请求
    if request.is_json:
        username = data.get('username')
        return jsonify({'status': 'success', 'username': username})
    
    return jsonify({'error': '需要 JSON 数据'}), 400
```

## 文件上传

### 基本文件上传

```python
import os
from werkzeug.utils import secure_filename

UPLOAD_FOLDER = 'uploads'
ALLOWED_EXTENSIONS = {'txt', 'pdf', 'png', 'jpg', 'jpeg', 'gif'}

app.config['UPLOAD_FOLDER'] = UPLOAD_FOLDER

def allowed_file(filename):
    """检查文件扩展名是否允许"""
    return '.' in filename and \
           filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS

@app.route('/upload', methods=['GET', 'POST'])
def upload_file():
    if request.method == 'POST':
        # 检查是否有文件
        if 'file' not in request.files:
            return '没有选择文件'
        
        file = request.files['file']
        
        # 检查文件名
        if file.filename == '':
            return '没有选择文件'
        
        if file and allowed_file(file.filename):
            # 安全地获取文件名
            filename = secure_filename(file.filename)
            # 确保上传目录存在
            os.makedirs(app.config['UPLOAD_FOLDER'], exist_ok=True)
            # 保存文件
            filepath = os.path.join(app.config['UPLOAD_FOLDER'], filename)
            file.save(filepath)
            return f'文件上传成功: {filename}'
        
        return '不支持的文件类型'
    
    # GET 请求显示上传表单
    return '''
        <form method="post" enctype="multipart/form-data">
            <input type="file" name="file">
            <button type="submit">上传</button>
        </form>
    '''
```

### 多文件上传

```python
@app.route('/upload-multiple', methods=['POST'])
def upload_multiple():
    files = request.files.getlist('files')
    
    for file in files:
        if file and allowed_file(file.filename):
            filename = secure_filename(file.filename)
            file.save(os.path.join(app.config['UPLOAD_FOLDER'], filename))
    
    return f'成功上传 {len(files)} 个文件'
```

## Cookie 操作

### 设置 Cookie

```python
from flask import make_response

@app.route('/set_cookie')
def set_cookie():
    resp = make_response('Cookie 已设置')
    # 设置 Cookie
    resp.set_cookie('username', 'john', max_age=3600)  # 1小时过期
    resp.set_cookie('theme', 'dark', expires=None, path='/', httponly=True)
    return resp
```

### 读取 Cookie

```python
@app.route('/get_cookie')
def get_cookie():
    username = request.cookies.get('username')
    theme = request.cookies.get('theme', 'light')  # 默认值
    return f'用户名: {username}, 主题: {theme}'
```

### 删除 Cookie

```python
@app.route('/delete_cookie')
def delete_cookie():
    resp = make_response('Cookie 已删除')
    resp.delete_cookie('username')
    return resp
```

## Request 对象属性一览

```python
@app.route('/info')
def request_info():
    info = {
        # 请求方法
        'method': request.method,
        
        # URL 相关
        'url': request.url,           # 完整 URL
        'base_url': request.base_url, # 不含查询参数
        'path': request.path,         # 路径部分
        'query_string': request.query_string.decode(),  # 查询字符串
        
        # 请求头
        'user_agent': request.user_agent.string,
        'content_type': request.content_type,
        'content_length': request.content_length,
        
        # 客户端信息
        'remote_addr': request.remote_addr,  # 客户端 IP
        
        # 数据相关
        'is_json': request.is_json,
        'cookies': dict(request.cookies),
    }
    
    return jsonify(info)
```

## 获取客户端 IP

```python
@app.route('/ip')
def get_ip():
    # 直接获取（可能被代理修改）
    ip = request.remote_addr
    
    # 考虑代理的情况
    if request.headers.get('X-Forwarded-For'):
        ip = request.headers.get('X-Forwarded-For').split(',')[0]
    elif request.headers.get('X-Real-IP'):
        ip = request.headers.get('X-Real-IP')
    
    return f'您的 IP: {ip}'
```

## 小结

| 数据来源 | 获取方式 |
|----------|----------|
| GET 参数 | `request.args.get('key')` |
| POST 表单 | `request.form.get('key')` |
| JSON 数据 | `request.get_json()` |
| 文件 | `request.files['key']` |
| Cookie | `request.cookies.get('key')` |
| 请求头 | `request.headers.get('key')` |
| 客户端 IP | `request.remote_addr` |

> 💡 **提示**：使用 `get()` 方法获取数据时，可以提供默认值，避免 KeyError 异常。
