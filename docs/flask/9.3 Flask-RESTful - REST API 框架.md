# 9.3 Flask-RESTful：REST API 框架

Flask-RESTful 是一个扩展，帮助你快速构建 RESTful API。

## 安装

```bash
pip install flask-restful
```

## 基本使用

```python
from flask import Flask, request
from flask_restful import Resource, Api

app = Flask(__name__)
api = Api(app)

# 定义资源
class HelloWorld(Resource):
    def get(self):
        return {'message': 'Hello, World!'}

# 添加路由
api.add_resource(HelloWorld, '/api/hello')

if __name__ == '__main__':
    app.run(debug=True)
```

## CRUD 操作

```python
from flask_restful import fields, marshal_with

# 模拟数据库
todos = {}

# 输出字段定义
todo_fields = {
    'id': fields.Integer,
    'task': fields.String,
    'done': fields.Boolean,
    'uri': fields.Url('todo')
}

class TodoList(Resource):
    """获取所有任务 / 创建新任务"""
    
    @marshal_with(todo_fields)
    def get(self):
        return list(todos.values())
    
    def post(self):
        todo_id = len(todos) + 1
        todos[todo_id] = {
            'id': todo_id,
            'task': request.json['task'],
            'done': False
        }
        return {'id': todo_id}, 201

class Todo(Resource):
    """单个任务操作"""
    
    @marshal_with(todo_fields)
    def get(self, todo_id):
        return todos.get(todo_id, (None, 404))
    
    def put(self, todo_id):
        todos[todo_id]['task'] = request.json.get('task')
        todos[todo_id]['done'] = request.json.get('done', False)
        return {'message': '更新成功'}
    
    def delete(self, todo_id):
        if todo_id in todos:
            del todos[todo_id]
        return '', 204

# 注册路由
api.add_resource(TodoList, '/api/todos')
api.add_resource(Todo, '/api/todos/<int:todo_id>')
```

## 请求解析

```python
from flask_restful import reqparse

# 创建请求解析器
parser = reqparse.RequestParser()
parser.add_argument('task', type=str, required=True, help='任务内容不能为空')
parser.add_argument('done', type=bool, default=False)
parser.add_argument('priority', type=int, choices=(1, 2, 3), help='优先级必须是 1、2 或 3')

class TodoCreate(Resource):
    def post(self):
        # 解析请求参数
        args = parser.parse_args()
        
        todo = {
            'task': args['task'],
            'done': args['done'],
            'priority': args['priority']
        }
        
        return todo, 201
```

## 错误处理

```python
from flask_restful import abort

class TodoResource(Resource):
    def get(self, todo_id):
        if todo_id not in todos:
            abort(404, message=f"任务 {todo_id} 不存在")
        return todos[todo_id]

# 自定义错误处理
@app.errorhandler(404)
def not_found(error):
    return {'error': '资源未找到'}, 404

@app.errorhandler(400)
def bad_request(error):
    return {'error': str(error)}, 400
```

## 认证保护

```python
from functools import wraps

def authenticate(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        token = request.headers.get('Authorization')
        if not token or not verify_token(token):
            return {'error': '未授权'}, 401
        return func(*args, **kwargs)
    return wrapper

class ProtectedResource(Resource):
    method_decorators = [authenticate]
    
    def get(self):
        return {'message': '这是受保护的资源'}
```

## API 响应格式

```python
def api_response(data=None, message='成功', code=200):
    """统一 API 响应格式"""
    return {
        'code': code,
        'message': message,
        'data': data
    }, code

class UserResource(Resource):
    def get(self, user_id):
        user = User.query.get(user_id)
        if not user:
            return api_response(None, '用户不存在', 404)
        return api_response(user.to_dict())
```
