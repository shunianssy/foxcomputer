# 7.3 SQL 注入防护：安全的数据库操作

SQL 注入是最常见的 Web 安全漏洞之一，攻击者可以通过恶意 SQL 代码获取或破坏数据。

## 危险示例（不要这样做！）

```python
# 危险！直接拼接 SQL
@app.route('/user/<username>')
def get_user(username):
    query = f"SELECT * FROM users WHERE username = '{username}'"
    # 攻击者输入: ' OR '1'='1
    # 实际执行: SELECT * FROM users WHERE username = '' OR '1'='1'
    return db.execute(query)
```

## 安全做法：参数化查询

### 使用 SQLAlchemy ORM（推荐）

```python
from flask_sqlalchemy import SQLAlchemy
from sqlalchemy import text

db = SQLAlchemy(app)

# 方式一：ORM 查询（最安全）
user = User.query.filter_by(username=username).first()

# 方式二：参数化原生 SQL
result = db.session.execute(
    text("SELECT * FROM users WHERE username = :username"),
    {'username': username}
)
```

### 使用原始参数化查询

```python
import sqlite3

@app.route('/user/<username>')
def get_user(username):
    conn = sqlite3.connect('database.db')
    cursor = conn.cursor()
    
    # 使用参数化查询
    cursor.execute("SELECT * FROM users WHERE username = ?", (username,))
    user = cursor.fetchone()
    
    conn.close()
    return user
```

## 输入验证

```python
from wtforms import Form, StringField, validators

class UserForm(Form):
    username = StringField('用户名', [
        validators.Length(min=3, max=20),
        validators.Regexp(r'^[a-zA-Z0-9_]+$', message='只能包含字母、数字和下划线')
    ])

@app.route('/user', methods=['POST'])
def create_user():
    form = UserForm(request.form)
    if form.validate():
        # 验证通过后再进行数据库操作
        user = User(username=form.username.data)
        db.session.add(user)
        db.session.commit()
```

```python
# 为应用创建专用数据库用户，只授予必要权限
# CREATE USER 'flask_app'@'localhost' IDENTIFIED BY 'password';
# GRANT SELECT, INSERT, UPDATE, DELETE ON myapp.* TO 'flask_app'@'localhost';

# 配置中使用受限用户
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql://flask_app:password@localhost/myapp'
```

## 安全检查函数

```python
def is_safe_input(value):
    """检查输入是否包含可疑 SQL 关键字"""
    dangerous_keywords = ['DROP', 'DELETE', 'TRUNCATE', '--', ';', '/*']
    value_upper = value.upper()
    return not any(keyword in value_upper for keyword in dangerous_keywords)
```
