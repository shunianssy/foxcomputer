# 4.2 数据库操作：增删改查 (CRUD)

有了模型，我们就可以在视图函数里操控数据了

## 创建数据 (Create)

### 添加单条记录

```python
from app import db
from models import User

# 创建新用户
new_user = User(username='shunx', email='shunx@shunx.top')
db.session.add(new_user)
db.session.commit()  # 必须 commit 才会真正写入数据库
```

### 批量添加

```python
users = [
    User(username='user1', email='user1@example.com'),
    User(username='user2', email='user2@example.com'),
    User(username='user3', email='user3@example.com'),
]
db.session.add_all(users)
db.session.commit()
```

### 添加关联数据

```python
# 一对多
user = User(username='john')
article = Article(title='第一篇文章', author=user)
db.session.add(article)
db.session.commit()

# 多对多
article = Article(title='Python 教程')
tag1 = Tag(name='Python')
tag2 = Tag(name='教程')
article.tags.extend([tag1, tag2])
db.session.add(article)
db.session.commit()
```

## 查询数据 (Read)

### 基本查询

```python
# 获取所有记录
users = User.query.all()

# 获取单条记录（按主键）
user = User.query.get(1)
user = User.query.get_or_404(1)  # 找不到返回 404

# 获取第一条记录
user = User.query.first()

# 按条件查询
user = User.query.filter_by(username='shunx').first()
user = User.query.filter(User.username == 'shunx').first()
```

### 条件过滤

```python
from sqlalchemy import or_, and_, not_

# 等于
User.query.filter(User.username == 'john').all()

# 不等于
User.query.filter(User.username != 'john').all()

# 模糊查询
User.query.filter(User.username.like('%john%')).all()
User.query.filter(User.username.ilike('%JOHN%')).all()  # 不区分大小写

# IN 查询
User.query.filter(User.username.in_(['john', 'jane'])).all()

# NOT IN
User.query.filter(~User.username.in_(['john', 'jane'])).all()

# 范围查询
User.query.filter(User.id.between(1, 10)).all()

# 空值判断
User.query.filter(User.email.is_(None)).all()
User.query.filter(User.email.isnot(None)).all()

# AND 条件
User.query.filter(User.username == 'john', User.email.like('%@gmail.com')).all()
User.query.filter(and_(User.username == 'john', User.email.like('%@gmail.com'))).all()

# OR 条件
User.query.filter(or_(User.username == 'john', User.username == 'jane')).all()

# NOT 条件
User.query.filter(not_(User.username == 'john')).all()
```

### 排序

```python
# 升序
User.query.order_by(User.created_at).all()

# 降序
User.query.order_by(User.created_at.desc()).all()

# 多字段排序
User.query.order_by(User.is_admin.desc(), User.created_at).all()
```

### 分页

```python
# 方式一：limit 和 offset
users = User.query.limit(10).offset(0).all()  # 第1页，每页10条
users = User.query.limit(10).offset(10).all()  # 第2页

# 方式二：paginate（推荐）
page = User.query.paginate(page=1, per_page=10)
print(page.items)      # 当前页数据
print(page.total)      # 总记录数
print(page.pages)      # 总页数
print(page.has_next)   # 是否有下一页
print(page.has_prev)   # 是否有上一页
print(page.next_num)   # 下一页页码

# 遍历所有页
for user in page.items:
    print(user.username)
```

### 聚合查询

```python
from sqlalchemy import func

# 计数
count = User.query.count()
count = db.session.query(func.count(User.id)).scalar()

# 求和
total = db.session.query(func.sum(User.score)).scalar()

# 平均值
avg = db.session.query(func.avg(User.score)).scalar()

# 最大值
max_score = db.session.query(func.max(User.score)).scalar()

# 最小值
min_score = db.session.query(func.min(User.score)).scalar()

# 分组统计
result = db.session.query(
    User.role, 
    func.count(User.id)
).group_by(User.role).all()
```

### 关联查询

```python
# Join 查询
results = db.session.query(User, Article).join(Article).all()

# 外连接
results = db.session.query(User, Article).outerjoin(Article).all()

# 预加载（避免 N+1 问题）
users = User.query.options(db.joinedload(User.articles)).all()

# 子查询
subquery = db.session.query(Article.author_id, func.count('*').label('count')).group_by(Article.author_id).subquery()
results = db.session.query(User, subquery.c.count).outerjoin(subquery, User.id == subquery.c.author_id).all()
```

## 更新数据 (Update)

### 方式一：修改属性后提交

```python
user = User.query.get(1)
user.email = 'new_email@example.com'
db.session.commit()
```

### 方式二：批量更新

```python
# 更新所有匹配的记录
User.query.filter(User.active == False).update({'active': True})
db.session.commit()

# 使用模型属性
User.query.filter(User.id > 10).update({User.views: User.views + 1})
db.session.commit()
```

### 更新关联数据

```python
article = Article.query.get(1)
article.tags.clear()  # 清除所有标签
article.tags.append(Tag.query.get(1))
article.tags.append(Tag.query.get(2))
db.session.commit()
```

## 删除数据 (Delete)

### 删除单条记录

```python
user = User.query.get(1)
db.session.delete(user)
db.session.commit()
```

### 批量删除

```python
User.query.filter(User.active == False).delete()
db.session.commit()
```

### 级联删除

在模型中配置级联删除：

```python
class User(db.Model):
    __tablename__ = 'users'
    
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80))
    # 级联删除：删除用户时同时删除其所有文章
    articles = db.relationship('Article', backref='author', cascade='all, delete-orphan')
```

## 事务管理

### 手动事务

```python
try:
    # 开始事务
    user = User(username='test')
    db.session.add(user)
    
    article = Article(title='Test')
    db.session.add(article)
    
    # 提交事务
    db.session.commit()
except Exception as e:
    # 回滚事务
    db.session.rollback()
    raise e
```

### 上下文管理器

```python
from contextlib import contextmanager

@contextmanager
def db_session():
    try:
        yield db.session
        db.session.commit()
    except:
        db.session.rollback()
        raise
    finally:
        db.session.remove()

# 使用
with db_session():
    user = User(username='test')
    db.session.add(user)
```

## 常用查询模式

### 获取或创建

```python
def get_or_create(model, **kwargs):
    instance = model.query.filter_by(**kwargs).first()
    if instance:
        return instance, False
    instance = model(**kwargs)
    db.session.add(instance)
    db.session.commit()
    return instance, True

# 使用
user, created = get_or_create(User, username='john')
```

### 更新或创建

```python
def update_or_create(model, defaults=None, **kwargs):
    instance = model.query.filter_by(**kwargs).first()
    if instance:
        for key, value in (defaults or {}).items():
            setattr(instance, key, value)
        db.session.commit()
        return instance, False
    kwargs.update(defaults or {})
    instance = model(**kwargs)
    db.session.add(instance)
    db.session.commit()
    return instance, True
```

## 小结

| 操作 | 代码 |
|------|------|
| 添加 | `db.session.add(obj)` |
| 批量添加 | `db.session.add_all([objs])` |
| 查询所有 | `Model.query.all()` |
| 条件查询 | `Model.query.filter_by(name='x')` |
| 排序 | `order_by(Model.field)` |
| 分页 | `paginate(page=1, per_page=10)` |
| 更新 | `obj.attr = value; db.session.commit()` |
| 删除 | `db.session.delete(obj)` |
| 提交 | `db.session.commit()` |
| 回滚 | `db.session.rollback()` |

> **提示**：不要忘记 `db.session.commit()`
