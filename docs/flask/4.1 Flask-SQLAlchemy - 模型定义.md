## 数据库配置

```python

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///test.db'
app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False

db = SQLAlchemy(app)
```

## 定义模型

### 基本模型

```python
class User(db.Model):
    __tablename__ = 'users'
    
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80), unique=True, nullable=False)
    email = db.Column(db.String(120), unique=True, nullable=False)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    
    def __repr__(self):
        return f'<User {self.username}>'
    
    def to_dict(self):
        return {
            'id': self.id,
            'username': self.username,
            'email': self.email
        }
```

### 常用字段类型

| 类型 | Python 类型 | 说明 |
|------|-------------|------|
| `Integer` | int | 整数 |
| `String(size)` | str | 定长字符串 |
| `Text` | str | 不定长文本 |
| `Float` | float | 浮点数 |
| `Boolean` | bool | 布尔值 |
| `DateTime` | datetime | 日期时间 |
| `Date` | date | 日期 |
| `Time` | time | 时间 |
| `LargeBinary` | bytes | 二进制数据 |
| `JSON` | dict/list | JSON 数据 |

### 字段选项

```python
class Article(db.Model):
    __tablename__ = 'articles'
    
    id = db.Column(db.Integer, primary_key=True)
    title = db.Column(db.String(200), nullable=False)
    content = db.Column(db.Text)
    views = db.Column(db.Integer, default=0)
    is_published = db.Column(db.Boolean, default=False)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    updated_at = db.Column(db.DateTime, onupdate=datetime.utcnow)
    author_id = db.Column(db.Integer, db.ForeignKey('users.id'))
    
    # 唯一约束
    slug = db.Column(db.String(100), unique=True)
    
    # 索引
    __table_args__ = (
        db.Index('idx_created_at', 'created_at'),
        db.Index('idx_author_title', 'author_id', 'title'),
    )
```

### 字段参数说明

| 参数 | 说明 |
|------|------|
| `primary_key` | 主键 |
| `unique` | 唯一约束 |
| `nullable` | 是否允许为空 |
| `default` | 默认值 |
| `index` | 创建索引 |
| `unique` | 唯一约束 |
| `server_default` | 数据库层面的默认值 |

## 关系定义

### 一对多关系

```python
class User(db.Model):
    __tablename__ = 'users'
    
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80))
    # 一对多：一个用户有多篇文章
    articles = db.relationship('Article', backref='author', lazy='dynamic')

class Article(db.Model):
    __tablename__ = 'articles'
    
    id = db.Column(db.Integer, primary_key=True)
    title = db.Column(db.String(200))
    # 外键
    author_id = db.Column(db.Integer, db.ForeignKey('users.id'))
```

使用：

```python
# 获取用户的所有文章
user = User.query.get(1)
articles = user.articles.all()

# 获取文章的作者
article = Article.query.get(1)
author = article.author
```

### 一对一关系

```python
class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80))
    # 一对一：uselist=False
    profile = db.relationship('Profile', backref='user', uselist=False)

class Profile(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    bio = db.Column(db.Text)
    user_id = db.Column(db.Integer, db.ForeignKey('users.id'))
```

### 多对多关系

```python
# 关联表
article_tags = db.Table('article_tags',
    db.Column('article_id', db.Integer, db.ForeignKey('articles.id'), primary_key=True),
    db.Column('tag_id', db.Integer, db.ForeignKey('tags.id'), primary_key=True)
)

class Article(db.Model):
    __tablename__ = 'articles'
    
    id = db.Column(db.Integer, primary_key=True)
    title = db.Column(db.String(200))
    tags = db.relationship('Tag', secondary=article_tags, backref='articles')

class Tag(db.Model):
    __tablename__ = 'tags'
    
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(50), unique=True)
```

使用：

```python
# 添加标签
article = Article.query.get(1)
tag = Tag(name='Python')
article.tags.append(tag)
db.session.commit()

# 获取文章的所有标签
for tag in article.tags:
    print(tag.name)

# 获取标签下的所有文章
for article in tag.articles:
    print(article.title)
```

### 自引用关系

```python
class Category(db.Model):
    __tablename__ = 'categories'
    
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(50))
    parent_id = db.Column(db.Integer, db.ForeignKey('categories.id'))
    
    # 自引用：父分类
    parent = db.relationship('Category', remote_side=[id], backref='children')
```

## 模型方法

### 实例方法

```python
class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80))
    password_hash = db.Column(db.String(128))
    
    def set_password(self, password):
        """设置密码"""
        from werkzeug.security import generate_password_hash
        self.password_hash = generate_password_hash(password)
    
    def check_password(self, password):
        """验证密码"""
        from werkzeug.security import check_password_hash
        return check_password_hash(self.password_hash, password)
    
    def to_dict(self):
        """转换为字典"""
        return {
            'id': self.id,
            'username': self.username
        }
```

### 类方法

```python
class User(db.Model):
    # ...
    
    @classmethod
    def find_by_username(cls, username):
        """根据用户名查找"""
        return cls.query.filter_by(username=username).first()
    
    @classmethod
    def create(cls, username, email):
        """创建用户"""
        user = cls(username=username, email=email)
        db.session.add(user)
        db.session.commit()
        return user
```

### 静态方法

```python
class User(db.Model):
    # ...
    
    @staticmethod
    def get_all_usernames():
        """获取所有用户名"""
        return [u.username for u in User.query.all()]
```

## 模型继承

### 单表继承

```python
class Animal(db.Model):
    __tablename__ = 'animals'
    
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(50))
    type = db.Column(db.String(20))  # 区分类型
    
    __mapper_args__ = {
        'polymorphic_on': type,
        'polymorphic_identity': 'animal'
    }

class Dog(Animal):
    __mapper_args__ = {
        'polymorphic_identity': 'dog'
    }

class Cat(Animal):
    __mapper_args__ = {
        'polymorphic_identity': 'cat'
    }
```

### 多表继承

```python
class Person(db.Model):
    __tablename__ = 'persons'
    
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(50))
    type = db.Column(db.String(20))
    
    __mapper_args__ = {
        'polymorphic_on': type,
        'polymorphic_identity': 'person'
    }

class Employee(Person):
    __tablename__ = 'employees'
    
    id = db.Column(db.Integer, db.ForeignKey('persons.id'), primary_key=True)
    salary = db.Column(db.Float)
    
    __mapper_args__ = {
        'polymorphic_identity': 'employee'
    }
```

## 小结

| 功能 | 代码 |
|------|------|
| 定义模型 | `class User(db.Model)` |
| 主键 | `primary_key=True` |
| 外键 | `db.ForeignKey('table.id')` |
| 一对多 | `db.relationship('Model', backref='name')` |
| 多对多 | `secondary=association_table` |
| 创建表 | `db.create_all()` |

> **提示**：使用 `db.relationship()` 定义关系时
