# 8.2 测试客户端：模拟请求测试

Flask 提供了测试客户端，可以在不运行服务器的情况下模拟 HTTP 请求。

## 基本请求测试

```python
import pytest
from app import app

@pytest.fixture
def client():
    app.config['TESTING'] = True
    with app.test_client() as client:
        yield client

def test_get_request(client):
    """测试 GET 请求"""
    response = client.get('/api/users')
    assert response.status_code == 200
    assert response.content_type == 'application/json'

def test_post_request(client):
    """测试 POST 请求"""
    response = client.post('/api/users', json={
        'name': '张三',
        'email': 'zhangsan@example.com'
    })
    assert response.status_code == 201
    data = response.get_json()
    assert data['name'] == '张三'

def test_put_request(client):
    """测试 PUT 请求"""
    response = client.put('/api/users/1', json={
        'name': '李四'
    })
    assert response.status_code == 200

def test_delete_request(client):
    """测试 DELETE 请求"""
    response = client.delete('/api/users/1')
    assert response.status_code == 204
```

```python
def test_cookies(client):
    """测试 Cookie"""
    response = client.get('/set-cookie')
    
    # 检查是否设置了 Cookie
    assert 'my_cookie' in response.cookies
    assert response.cookies['my_cookie'] == 'cookie_value'
```

## 认证测试

```python
import base64

def test_basic_auth(client):
    """测试 Basic 认证"""
    credentials = base64.b64encode(b'admin:password').decode('utf-8')
    response = client.get('/protected', headers={
        'Authorization': f'Basic {credentials}'
    })
    assert response.status_code == 200

def test_token_auth(client):
    """测试 Token 认证"""
    response = client.get('/api/profile', headers={
        'Authorization': 'Bearer your-jwt-token'
    })
    assert response.status_code == 200
```

## 测试文件上传

```python
import io

def test_file_upload(client):
    """测试文件上传"""
    data = {
        'file': (io.BytesIO(b'file content'), 'test.txt')
    }
    response = client.post('/upload', 
        data=data,
        content_type='multipart/form-data'
    )
    assert response.status_code == 200
```
