# 4.1 信号系统

## 什么是信号？

信号（Signal）是 Godot 的观察者模式实现，用于节点间的通信。它允许一个节点向其他节点发送通知，而不需要直接引用它们。

```
传统方式（紧耦合）：
节点A ──直接调用──> 节点B的方法
问题：A 需要知道 B 的存在

信号方式（松耦合）：
节点A ──发射信号──> 信号 ──通知──> 节点B、C、D...
优点：A 不需要知道谁在监听
```

## 内置信号

### 常用内置信号

```gdscript
extends Node

func _ready():
 # 节点信号
 ready.connect(_on_ready)
 tree_entered.connect(_on_tree_entered)
 tree_exited.connect(_on_tree_exited)
 
 # 子节点信号
 child_entered_tree.connect(_on_child_entered)
 child_exiting_tree.connect(_on_child_exiting)

func _on_ready():
 print("节点准备完成")

func _on_tree_entered():
 print("进入场景树")

func _on_tree_exited():
 print("离开场景树")

func _on_child_entered(node: Node):
 print("子节点进入: ", node.name)
```

## 连接信号

### 在编辑器中连接

1. 选择节点
2. 在节点面板中选择"信号"标签
3. 双击要连接的信号
4. 选择接收节点和方法名

### 在代码中连接

```gdscript
extends Node2D

@onready var button = $Button
@onready var player = $Player

func _ready():
 # 基本连接
 button.pressed.connect(_on_button_pressed)
 
 # 带参数绑定
 button.pressed.connect(_on_button_pressed_with_param.bind("custom_data"))
 
 # 连接到其他节点的方法
 button.pressed.connect(player.jump)

func _on_button_pressed():
 print("按钮被按下")

func _on_button_pressed_with_param(data: String):
 print("按钮被按下，数据: ", data)
```

### 连接选项

```gdscript
func _ready():
 # 默认连接（一次性？否，延迟？否）
 button.pressed.connect(_on_pressed)
 
 # 一次性连接（触发后自动断开）
 button.pressed.connect(_on_pressed, CONNECT_ONE_SHOT)
 
 # 延迟连接（在空闲时调用）
 button.pressed.connect(_on_pressed, CONNECT_DEFERRED)
 
 # 引用计数（多次连接只触发一次）
 button.pressed.connect(_on_pressed, CONNECT_REFERENCE_COUNTED)
 
 # 组合选项
 button.pressed.connect(_on_pressed, CONNECT_ONE_SHOT | CONNECT_DEFERRED)
```

### 断开连接

```gdscript
func _ready():
 button.pressed.connect(_on_pressed)

func disable_button():
 # 检查是否已连接
 if button.pressed.is_connected(_on_pressed):
 # 断开连接
 button.pressed.disconnect(_on_pressed)
 
 # 断开所有连接
 button.pressed.disconnect_all()
```

## 信号参数

### 带参数的信号

```gdscript
extends Node

# 自定义带参数的信号
signal health_changed(new_health: int, max_health: int)
signal item_collected(item_name: String, count: int)
signal position_updated(position: Vector2)

func take_damage(amount: int):
 current_health -= amount
 # 发射带参数的信号
 health_changed.emit(current_health, max_health)

func collect_item(name: String, amount: int):
 item_collected.emit(name, amount)
```

### 接收信号参数

```gdscript
extends Node

@onready var player = $Player

func _ready():
 # 连接带参数的信号
 player.health_changed.connect(_on_health_changed)
 player.item_collected.connect(_on_item_collected)

# 方法参数与信号参数对应
func _on_health_changed(new_health: int, max_health: int):
 print("血量: ", new_health, "/", max_health)
 update_health_bar(new_health, max_health)

func _on_item_collected(item_name: String, count: int):
 print("获得物品: ", item_name, " x", count)
```

### 绑定额外参数

```gdscript
extends Node

func _ready():
 var enemy1 = $Enemy1
 var enemy2 = $Enemy2
 
 # 绑定额外参数来区分不同的敌人
 enemy1.died.connect(_on_enemy_died.bind("Enemy1"))
 enemy2.died.connect(_on_enemy_died.bind("Enemy2"))

func _on_enemy_died(enemy_name: String):
 print("敌人死亡: ", enemy_name)
```

## 信号的应用场景

### UI 更新

```gdscript
# player.gd
extends CharacterBody2D

signal health_changed(new_health: int)
signal score_changed(new_score: int)

var health: int = 100
var score: int = 0

func take_damage(amount: int):
 health -= amount
 health_changed.emit(health)

func add_score(points: int):
 score += points
 score_changed.emit(score)

# hud.gd
extends CanvasLayer

@onready var health_bar = $HealthBar
@onready var score_label = $ScoreLabel
@onready var player = $"../Player"

func _ready():
 player.health_changed.connect(_on_health_changed)
 player.score_changed.connect(_on_score_changed)

func _on_health_changed(new_health: int):
 health_bar.value = new_health

func _on_score_changed(new_score: int):
 score_label.text = "分数: " + str(new_score)
```

### 敌人系统

```gdscript
# enemy.gd
extends CharacterBody2D

signal died
signal damaged(amount: int, current_health: int)
signal target_detected(target: Node2D)

@export var max_health: int = 100
var current_health: int

func _ready():
 current_health = max_health

func take_damage(amount: int):
 current_health -= amount
 damaged.emit(amount, current_health)
 
 if current_health <= 0:
 die()

func die():
 died.emit()
 queue_free()

func _on_detection_area_body_entered(body: Node2D):
 if body.is_in_group("player"):
 target_detected.emit(body)

# enemy_spawner.gd
extends Node2D

const EnemyScene = preload("res://scenes/enemy.tscn")

func spawn_enemy():
 var enemy = EnemyScene.instantiate()
 
 # 连接敌人信号
 enemy.died.connect(_on_enemy_died)
 enemy.damaged.connect(_on_enemy_damaged)
 
 add_child(enemy)

func _on_enemy_died():
 print("敌人死亡")
 update_score()

func _on_enemy_damaged(amount: int, current_health: int):
 print("敌人受伤: -", amount, " 剩余: ", current_health)
```

## 信号最佳实践

### 命名规范

```gdscript
# 信号命名使用过去式或事件描述
signal health_changed(new_health: int) # 好
signal died # 好
signal item_collected(item: Item) # 好

# 避免使用动词原形
signal change_health # 不推荐
signal die # 不推荐
```

### 避免循环信号

```gdscript
# 错误示例：可能导致无限循环
signal a_triggered
signal b_triggered

func _ready():
 a_triggered.connect(_on_a_triggered)
 b_triggered.connect(_on_b_triggered)

func _on_a_triggered():
 b_triggered.emit() # A -> B

func _on_b_triggered():
 a_triggered.emit() # B -> A (循环!)

# 正确做法：使用标志位或重新设计
var is_processing: bool = false

func _on_a_triggered():
 if is_processing:
 return
 is_processing = true
 # 处理逻辑
 is_processing = false
```

### 清理信号连接

```gdscript
extends Node

var connected_node: Node

func setup_connection(node: Node):
 connected_node = node
 node.some_signal.connect(_on_some_signal)

func _exit_tree():
 # 离开场景树时断开连接
 if connected_node and connected_node.some_signal.is_connected(_on_some_signal):
 connected_node.some_signal.disconnect(_on_some_signal)
```
