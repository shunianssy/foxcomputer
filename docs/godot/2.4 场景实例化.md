# 2.4 场景实例化

## 什么是场景实例化？

场景实例化是将一个场景文件（.tscn）作为子节点添加到另一个场景中的过程。这使得场景可以像预制体一样被复用。

```
主场景 (Main.tscn)
├── Player (Player.tscn 的实例)
├── Enemies
│ ├── Enemy1 (Enemy.tscn 的实例)
│ ├── Enemy2 (Enemy.tscn 的实例)
│ └── Enemy3 (Enemy.tscn 的实例)
└── Items
 ├── Coin1 (Coin.tscn 的实例)
 └── Coin2 (Coin.tscn 的实例)
```

## 创建预制体

### 步骤 1：创建可复用场景

1. 创建新场景（`场景` → `新建场景`）
2. 选择合适的根节点类型
3. 添加子节点和脚本
4. 保存为 `.tscn` 文件

### 示例：创建敌人预制体

```
Enemy.tscn 结构：
Enemy (CharacterBody2D)
├── Sprite2D
├── CollisionShape2D
├── DetectionArea (Area2D)
│ └── CollisionShape2D
└── HealthBar (ProgressBar)
```

```gdscript
# enemy.gd
extends CharacterBody2D

signal died
signal health_changed(new_health: int)

@export var max_health: int = 100
@export var speed: float = 100.0
@export var damage: int = 10

var current_health: int

func _ready():
 current_health = max_health
 update_health_bar()

func take_damage(amount: int):
 current_health -= amount
 health_changed.emit(current_health)
 update_health_bar()
 
 if current_health <= 0:
 die()

func die():
 died.emit()
 queue_free()

func update_health_bar():
 var health_bar = $HealthBar
 health_bar.value = (float(current_health) / max_health) * 100
```

## 在编辑器中实例化

### 方法 1：拖放实例化

1. 在文件系统面板中找到 `.tscn` 文件
2. 将其拖放到场景树中
3. 场景会作为子节点添加

### 方法 2：实例化子场景

1. 右键点击父节点
2. 选择 `实例化子场景`
3. 选择要实例化的 `.tscn` 文件

### 方法 3：继承场景

1. `场景` → `新建继承场景`
2. 选择父场景
3. 可以覆盖父场景的属性

## 在代码中实例化

### 基本实例化

```gdscript
extends Node2D

# 预加载场景（推荐）
const EnemyScene = preload("res://scenes/enemy.tscn")

func spawn_enemy():
 # 实例化场景
 var enemy = EnemyScene.instantiate()
 
 # 设置属性
 enemy.global_position = Vector2(100, 100)
 enemy.max_health = 150
 
 # 添加到场景树
 add_child(enemy)
```

### 动态加载场景

```gdscript
extends Node2D

# 对于大型资源，使用动态加载
var enemy_scene: PackedScene

func _ready():
 # 异步加载
 load_scene_async("res://scenes/enemy.tscn")

func load_scene_async(path: String):
 ResourceLoader.load_threaded_request(path)
 set_process(true)

func _process(delta):
 var status = ResourceLoader.load_threaded_get_status("res://scenes/enemy.tscn")
 if status == ResourceLoader.THREAD_LOAD_LOADED:
 enemy_scene = ResourceLoader.load_threaded_get("res://scenes/enemy.tscn")
 set_process(false)
 print("场景加载完成")
```

### 对象池模式

```gdscript
extends Node2D

const BulletScene = preload("res://scenes/bullet.tscn")
var bullet_pool: Array[Node] = []
@export var pool_size: int = 20

func _ready():
 # 初始化对象池
 for i in range(pool_size):
 var bullet = BulletScene.instantiate()
 bullet.visible = false
 add_child(bullet)
 bullet_pool.append(bullet)

func get_bullet() -> Node:
 # 从池中获取可用对象
 for bullet in bullet_pool:
 if not bullet.visible:
 return bullet
 
 # 池中没有可用对象，创建新的
 var new_bullet = BulletScene.instantiate()
 add_child(new_bullet)
 bullet_pool.append(new_bullet)
 return new_bullet

func fire_bullet(pos: Vector2, direction: Vector2):
 var bullet = get_bullet()
 bullet.global_position = pos
 bullet.direction = direction
 bullet.visible = true
 bullet.activate()
```

## 实例化信号连接

### 连接实例的信号

```gdscript
extends Node2D

const EnemyScene = preload("res://scenes/enemy.tscn")

func spawn_enemy():
 var enemy = EnemyScene.instantiate()
 
 # 连接信号
 enemy.died.connect(_on_enemy_died)
 enemy.health_changed.connect(_on_enemy_health_changed)
 
 add_child(enemy)

func _on_enemy_died():
 print("敌人死亡")
 update_score(100)

func _on_enemy_health_changed(new_health: int):
 print("敌人血量: ", new_health)
```

### 批量实例化

```gdscript
extends Node2D

const EnemyScene = preload("res://scenes/enemy.tscn")

@export var spawn_points: Array[Marker2D] = []

func spawn_wave(enemy_count: int):
 for i in range(enemy_count):
 var enemy = EnemyScene.instantiate()
 
 # 随机选择出生点
 var spawn_point = spawn_points[randi() % spawn_points.size()]
 enemy.global_position = spawn_point.global_position
 
 # 连接信号
 enemy.died.connect(_on_enemy_died.bind(i))
 
 add_child(enemy)
 
 # 延迟生成
 await get_tree().create_timer(0.5).timeout

func _on_enemy_died(enemy_index: int):
 print("敌人 ", enemy_index, " 死亡")
```

## 实例属性覆盖

### 覆盖导出属性

```gdscript

```

### 使用 setter 函数

```gdscript
# enemy.gd
extends CharacterBody2D

@export var enemy_type: String = "normal":
 set(value):
 enemy_type = value
 apply_enemy_type()

func apply_enemy_type():
 match enemy_type:
 "fast":
 speed = 200.0
 max_health = 50
 $Sprite2D.modulate = Color.BLUE
 "tank":
 speed = 50.0
 max_health = 300
 $Sprite2D.modulate = Color.RED
 _:
 speed = 100.0
 max_health = 100
 $Sprite2D.modulate = Color.WHITE
```

## 实例管理

### 管理多个实例

```gdscript
extends Node2D

const EnemyScene = preload("res://scenes/enemy.tscn")

var enemies: Array[Node] = []

func spawn_enemies(count: int):
 for i in range(count):
 var enemy = EnemyScene.instantiate()
 enemy.name = "Enemy_" + str(i)
 add_child(enemy)
 enemies.append(enemy)

func get_enemy_by_name(name: String) -> Node:
 for enemy in enemies:
 if enemy.name == name:
 return enemy
 return null

func clear_all_enemies():
 for enemy in enemies:
 enemy.queue_free()
 enemies.clear()

func get_active_enemies() -> Array:
 return enemies.filter(func(e): return is_instance_valid(e) and e.visible)
```

### 使用组管理实例

```gdscript
# 在敌人场景中
func _ready():
 add_to_group("enemies")

# 在管理器中
func damage_all_enemies(amount: int):
 for enemy in get_tree().get_nodes_in_group("enemies"):
 enemy.take_damage(amount)

func count_enemies() -> int:
 return get_tree().get_nodes_in_group("enemies").size()
```

## 实战示例：敌人生成器

```gdscript
# enemy_spawner.gd
extends Node2D

const EnemyScene = preload("res://scenes/enemy.tscn")

@export var spawn_interval: float = 2.0
@export var max_enemies: int = 10
@export var spawn_area: Rect2 = Rect2(0, 0, 800, 600)

var spawn_timer: float = 0.0
var enemy_count: int = 0

signal enemy_spawned(enemy: Node)
signal max_enemies_reached

func _process(delta):
 if enemy_count >= max_enemies:
 return
 
 spawn_timer += delta
 if spawn_timer >= spawn_interval:
 spawn_timer = 0.0
 spawn_enemy()

func spawn_enemy():
 var enemy = EnemyScene.instantiate()
 
 # 随机位置
 var spawn_pos = Vector2(
 randf_range(spawn_area.position.x, spawn_area.end.x),
 randf_range(spawn_area.position.y, spawn_area.end.y)
 )
 enemy.global_position = spawn_pos
 
 # 连接信号
 enemy.died.connect(_on_enemy_died)
 
 # 添加到场景
 get_parent().add_child(enemy)
 enemy_count += 1
 
 enemy_spawned.emit(enemy)
 
 if enemy_count >= max_enemies:
 max_enemies_reached.emit()

func _on_enemy_died():
 enemy_count -= 1

func reset():
 enemy_count = 0
 spawn_timer = 0.0
```

## 最佳实践

### 场景设计原则

```text
1. 单一职责：每个场景只做一件事
2. 可配置：使用 @export 暴露可配置属性
3. 独立性：场景应该能够独立运行和测试
4. 信号通信：使用信号而不是直接调用
5. 合理层级：避免过深的节点嵌套
```

### 性能优化

```gdscript
# 1. 使用预加载
const Scene = preload("res://path/to/scene.tscn")

# 2. 使用对象池
var pool: Array[Node] = []

# 3. 批量操作
func batch_spawn(count: int):
 var enemies = []
 for i in range(count):
 enemies.append(EnemyScene.instantiate())
 
 for enemy in enemies:
 add_child(enemy)

# 4. 延迟实例化
func spawn_delayed():
 await get_tree().create_timer(0.1).timeout
 var enemy = EnemyScene.instantiate()
 add_child(enemy)
```
