# 3.4 输入处理

## 🎮 输入系统概述

Godot 提供了灵活的输入系统，支持：

- 键盘输入
- 鼠标输入
- 触摸输入
- 手柄输入
- 自定义输入映射

## 🗺️ 输入映射

### 创建输入动作

1. 打开 `项目` → `项目设置`
2. 选择 `输入映射` 标签
3. 输入动作名称并点击 `添加`

### 绑定按键

```text
动作：move_right
├── 键盘：D
├── 键盘：右箭头
├── 手柄：D-Pad Right
└── 手柄：Left Stick X+
```

### 代码中定义输入映射

```gdscript
func _ready():
    # 添加输入动作
    InputMap.add_action("custom_action")
    
    # 添加按键绑定
    var key = InputEventKey.new()
    key.keycode = KEY_E
    InputMap.action_add_event("custom_action", key)
```

## ⌨️ 键盘输入

### 检测按键

```gdscript
extends Node2D

func _process(delta):
    # 检测按键是否被按下（持续检测）
    if Input.is_action_pressed("move_right"):
        position.x += speed * delta
    
    # 检测按键刚按下（只触发一次）
    if Input.is_action_just_pressed("jump"):
        jump()
    
    # 检测按键刚释放
    if Input.is_action_just_released("jump"):
        print("跳跃键释放")
```

### 获取轴输入

```gdscript
func _physics_process(delta):
    # 获取水平轴输入 (-1, 0, 1)
    var horizontal = Input.get_axis("move_left", "move_right")
    
    # 获取垂直轴输入
    var vertical = Input.get_axis("move_up", "move_down")
    
    # 组合成方向向量
    var direction = Vector2(horizontal, vertical).normalized()
    
    # 应用移动
    velocity = direction * speed
```

### 获取向量输入

```gdscript
func _physics_process(delta):
    # 直接获取归一化的移动向量
    var movement = Input.get_vector(
        "move_left", "move_right",
        "move_up", "move_down"
    )
    
    # movement 已经是归一化的向量
    velocity = movement * speed
```

### 原始键盘事件

```gdscript
func _input(event):
    if event is InputEventKey:
        # 检测特定按键
        if event.keycode == KEY_SPACE:
            if event.pressed:
                print("空格键按下")
            else:
                print("空格键释放")
        
        # 检测组合键
        if event.keycode == KEY_S and event.ctrl_pressed:
            print("Ctrl+S 保存")

func _unhandled_key_input(event):
    # 优化的键盘输入处理
    if event.pressed:
        match event.keycode:
            KEY_W, KEY_UP:
                move_up()
            KEY_S, KEY_DOWN:
                move_down()
            KEY_A, KEY_LEFT:
                move_left()
            KEY_D, KEY_RIGHT:
                move_right()
```

func _input(event):
    if event is InputEventMouseButton:

### 鼠标光标

```gdscript
func _ready():
    # 隐藏鼠标
    Input.mouse_mode = Input.MOUSE_MODE_HIDDEN
    
    # 捕获鼠标（FPS 风格）
    Input.mouse_mode = Input.MOUSE_MODE_CAPTURED
    
    # 恢复正常
    Input.mouse_mode = Input.MOUSE_MODE_VISIBLE
    
    # 限制在窗口内
    Input.mouse_mode = Input.MOUSE_MODE_CONFINED

func _input(event):
    if event is InputEventKey and event.keycode == KEY_ESCAPE:
        # 按 ESC 释放鼠标
        if Input.mouse_mode == Input.MOUSE_MODE_CAPTURED:
            Input.mouse_mode = Input.MOUSE_MODE_VISIBLE
```

## 👆 触摸输入

### 触摸事件

```gdscript
func _input(event):
    if event is InputEventScreenTouch:
        if event.pressed:
            print("触摸开始: ", event.index, " 位置: ", event.position)
        else:
            print("触摸结束: ", event.index)
    
    elif event is InputEventScreenDrag:
        print("触摸拖拽: ", event.index)
        print("位置: ", event.position)
        print("相对移动: ", event.relative)
        print("速度: ", event.velocity)
```

### 多点触控

```gdscript
var touches: Dictionary = {}

func _input(event):
    if event is InputEventScreenTouch:
        if event.pressed:
            touches[event.index] = event.position
        else:
            touches.erase(event.index)
        
        # 检测捏合手势
        if touches.size() == 2:
            check_pinch()
    
    elif event is InputEventScreenDrag:
        if touches.has(event.index):
            touches[event.index] = event.position

func check_pinch():
    var touch_array = touches.values()
    var distance = touch_array[0].distance_to(touch_array[1])
    print("两指距离: ", distance)
```

### 触摸虚拟摇杆

```gdscript
extends Control

@export var max_distance: float = 100.0
var touch_index: int = -1
var origin: Vector2
var direction: Vector2 = Vector2.ZERO

signal joystick_moved(direction: Vector2)

func _input(event):
    if event is InputEventScreenTouch:
        if event.pressed and touch_index == -1:
            # 检测触摸是否在摇杆区域内
            var local_pos = to_local(event.position)
            if local_pos.length() < max_distance * 2:
                touch_index = event.index
                origin = event.position
    
    elif event is InputEventScreenDrag:
        if event.index == touch_index:
            # 计算方向
            var offset = event.position - origin
            direction = offset.limit_length(max_distance) / max_distance
            joystick_moved.emit(direction)
            
            # 更新摇杆视觉位置
            $Knob.position = direction * max_distance

func _on_touch_ended():
    touch_index = -1
    direction = Vector2.ZERO
    joystick_moved.emit(direction)
    $Knob.position = Vector2.ZERO
```

### 手柄摇杆

```gdscript
func _process(delta):
    # 获取左摇杆
    var left_stick = Vector2(
        Input.get_joy_axis(0, JOY_AXIS_LEFT_X),
        Input.get_joy_axis(0, JOY_AXIS_LEFT_Y)
    )
    
    # 获取右摇杆
    var right_stick = Vector2(
        Input.get_joy_axis(0, JOY_AXIS_RIGHT_X),
        Input.get_joy_axis(0, JOY_AXIS_RIGHT_Y)
    )
    
    # 应用死区（避免漂移）
    left_stick = apply_deadzone(left_stick, 0.2)
    
    # 使用摇杆移动
    velocity = left_stick * speed

func apply_deadzone(input: Vector2, deadzone: float) -> Vector2:
    if input.length() < deadzone:
        return Vector2.ZERO
    return input.normalized() * ((input.length() - deadzone) / (1.0 - deadzone))
```

## 🎯 实战示例

### 完整玩家控制器

```gdscript
extends CharacterBody2D

# 移动参数
@export var speed: float = 300.0
@export var jump_force: float = 400.0
@export var acceleration: float = 1500.0
@export var friction: float = 1200.0

# 状态
var gravity: float

func _ready():
    gravity = ProjectSettings.get_setting("physics/2d/default_gravity")

func _physics_process(delta):
    # 应用重力
    if not is_on_floor():
        velocity.y += gravity * delta
    
    # 处理跳跃
    handle_jump()
    
    # 处理移动
    handle_movement(delta)
    
    # 应用移动
    move_and_slide()

func handle_jump():
    # 普通跳跃
    if Input.is_action_just_pressed("jump") and is_on_floor():
        velocity.y = -jump_force
    
    # 可变跳跃高度（松开跳跃键降低跳跃）
    if Input.is_action_just_released("jump") and velocity.y < 0:
        velocity.y *= 0.5

func handle_movement(delta):
    # 获取输入方向
    var direction = Input.get_axis("move_left", "move_right")
    
    if direction:
        # 加速
        velocity.x = move_toward(velocity.x, direction * speed, acceleration * delta)
    else:
        # 减速
        velocity.x = move_toward(velocity.x, 0, friction * delta)

func _input(event):
    # 处理攻击输入（使用 _input 确保响应及时）
    if event.is_action_pressed("attack"):
        attack()

func attack():
    print("攻击!")
```

### 相机控制

```gdscript
extends Camera2D

@export var sensitivity: float = 0.005

func _ready():
    # 捕获鼠标
    Input.mouse_mode = Input.MOUSE_MODE_CAPTURED

func _input(event):
    if event is InputEventMouseMotion:
        # 旋转相机
        rotation -= event.relative.x * sensitivity
        
        # 限制俯仰角
        var pitch = -event.relative.y * sensitivity
        pitch = clamp(pitch, -PI/3, PI/3)
        offset.y = tan(pitch) * 100
    
    if event is InputEventKey:
        if event.keycode == KEY_ESCAPE:
            # 释放鼠标
            Input.mouse_mode = Input.MOUSE_MODE_VISIBLE

func _unhandled_input(event):
    if event is InputEventMouseButton:
        if event.pressed and Input.mouse_mode == Input.MOUSE_MODE_VISIBLE:
            # 重新捕获鼠标
            Input.mouse_mode = Input.MOUSE_MODE_CAPTURED
```

### 输入映射命名规范

```text
移动类：
├── move_left, move_right
├── move_up, move_down
└── jump

动作类：
├── attack, attack_secondary
├── interact
├── dodge
└── skill_1, skill_2, skill_3

系统类：
├── pause
├── inventory
├── map
└── menu_confirm, menu_cancel
```

### 输入处理建议

```gdscript
# 使用输入映射而不是硬编码按键
# 好
if Input.is_action_pressed("jump"):
    jump()

# 避免
if Input.is_key_pressed(KEY_SPACE):
    jump()

# 使用 _unhandled_input 处理游戏操作
# 使用 _input 处理 UI 操作

# 使用轴输入处理移动
var direction = Input.get_axis("move_left", "move_right")

# 使用向量输入处理 2D 移动
var movement = Input.get_vector("move_left", "move_right", "move_up", "move_down")
```
