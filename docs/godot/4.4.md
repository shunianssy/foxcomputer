# 4.4 全局单例

## 🌐 什么是全局单例？

全局单例（Global Singleton）是在整个游戏生命周期中唯一存在的节点，可以被任何脚本直接访问，无需通过节点路径查找。

```
场景树结构：
Root
├── GameManager (自动加载)
├── AudioManager (自动加载)
├── SaveManager (自动加载)
└── Main (当前场景)
    ├── Player
    └── Enemies
```

## 🔧 创建自动加载

### 步骤 1：创建脚本

```gdscript
# game_manager.gd
extends Node

# 全局变量
var score: int = 0
var current_level: int = 1
var is_paused: bool = false

# 信号
signal score_changed(new_score: int)
signal game_paused
signal game_resumed

func _ready():
    print("GameManager 初始化")

func add_score(points: int):
    score += points
    score_changed.emit(score)

func pause_game():
    is_paused = true
    get_tree().paused = true
    game_paused.emit()

func resume_game():
    is_paused = false
    get_tree().paused = false
    game_resumed.emit()
```

### 步骤 2：配置自动加载

1. 打开 `项目` → `项目设置`
2. 选择 `自动加载` 标签
3. 点击文件夹图标选择脚本文件
4. 设置节点名称（如 `GameManager`）
5. 点击 `添加`

### 步骤 3：使用自动加载

```gdscript
# 在任何脚本中直接访问
extends Node

func _ready():
    # 直接使用全局单例名称
    GameManager.add_score(100)
    
    # 连接信号
    GameManager.score_changed.connect(_on_score_changed)

func _on_score_changed(new_score: int):
    print("分数: ", new_score)
```

## 📊 常用全局单例示例

### 游戏管理器

```gdscript
# game_manager.gd
extends Node

signal game_started
signal game_over(final_score: int)
signal level_changed(level: int)

var score: int = 0
var high_score: int = 0
var current_level: int = 1
var lives: int = 3

func _ready():
    # 加载最高分
    load_high_score()

func start_game():
    score = 0
    lives = 3
    current_level = 1
    game_started.emit()

func add_score(points: int):
    score += points
    if score > high_score:
        high_score = score
        save_high_score()

func lose_life():
    lives -= 1
    if lives <= 0:
        end_game()

func end_game():
    game_over.emit(score)

func next_level():
    current_level += 1
    level_changed.emit(current_level)

func save_high_score():
    var file = FileAccess.open("user://highscore.save", FileAccess.WRITE)
    file.store_32(high_score)

func load_high_score():
    if FileAccess.file_exists("user://highscore.save"):
        var file = FileAccess.open("user://highscore.save", FileAccess.READ)
        high_score = file.get_32()
```

### 音频管理器

```gdscript
# audio_manager.gd
extends Node

var master_volume: float = 1.0
var music_volume: float = 1.0
var sfx_volume: float = 1.0

var music_player: AudioStreamPlayer
var sfx_players: Array[AudioStreamPlayer] = []
const MAX_SFX_PLAYERS = 8

func _ready():
    # 创建音乐播放器
    music_player = AudioStreamPlayer.new()
    music_player.bus = "Music"
    add_child(music_player)
    
    # 创建音效播放器池
    for i in MAX_SFX_PLAYERS:
        var player = AudioStreamPlayer.new()
        player.bus = "SFX"
        add_child(player)
        sfx_players.append(player)
    
    # 加载音量设置
    load_settings()

func play_music(stream: AudioStream, loop: bool = true):
    if music_player.playing:
        music_player.stop()
    
    music_player.stream = stream
    music_player.stream.loop = loop
    music_player.volume_db = linear_to_db(music_volume * master_volume)
    music_player.play()

func stop_music():
    music_player.stop()

func play_sfx(stream: AudioStream, volume_scale: float = 1.0):
    # 找到空闲的播放器
    for player in sfx_players:
        if not player.playing:
            player.stream = stream
            player.volume_db = linear_to_db(sfx_volume * master_volume * volume_scale)
            player.play()
            return

func set_master_volume(value: float):
    master_volume = value
    AudioServer.set_bus_volume_db(AudioServer.get_bus_index("Master"), linear_to_db(value))
    save_settings()

func set_music_volume(value: float):
    music_volume = value
    music_player.volume_db = linear_to_db(value * master_volume)
    save_settings()

func set_sfx_volume(value: float):
    sfx_volume = value
    save_settings()

func save_settings():
    var config = ConfigFile.new()
    config.set_value("audio", "master", master_volume)
    config.set_value("audio", "music", music_volume)
    config.set_value("audio", "sfx", sfx_volume)
    config.save("user://settings.cfg")

func load_settings():
    var config = ConfigFile.new()
    if config.load("user://settings.cfg") == OK:
        master_volume = config.get_value("audio", "master", 1.0)
        music_volume = config.get_value("audio", "music", 1.0)
        sfx_volume = config.get_value("audio", "sfx", 1.0)
```

### 存档管理器

```gdscript
# save_manager.gd
extends Node

const SAVE_PATH = "user://savegame.save"

signal game_saved
signal game_loaded

var current_save: Dictionary = {}

func save_game():
    current_save = {
        "timestamp": Time.get_datetime_string_from_system(),
        "player": save_player_data(),
        "inventory": save_inventory_data(),
        "world": save_world_data()
    }
    
    var file = FileAccess.open(SAVE_PATH, FileAccess.WRITE)
    file.store_string(JSON.stringify(current_save))
    game_saved.emit()
    print("游戏已保存")

func load_game() -> bool:
    if not FileAccess.file_exists(SAVE_PATH):
        return false
    
    var file = FileAccess.open(SAVE_PATH, FileAccess.READ)
    var json_string = file.get_as_text()
    
    var json = JSON.new()
    var error = json.parse(json_string)
    
    if error != OK:
        print("加载存档失败: ", json.get_error_message())
        return false
    
    current_save = json.data
    load_player_data(current_save.get("player", {}))
    load_inventory_data(current_save.get("inventory", {}))
    load_world_data(current_save.get("world", {}))
    
    game_loaded.emit()
    print("游戏已加载: ", current_save.get("timestamp"))
    return true

func has_save() -> bool:
    return FileAccess.file_exists(SAVE_PATH)

func delete_save():
    if FileAccess.file_exists(SAVE_PATH):
        DirAccess.remove_absolute(SAVE_PATH)
    current_save.clear()

func save_player_data() -> Dictionary:
    var player = get_tree().get_first_node_in_group("player")
    if player:
        return {
            "position": {
                "x": player.global_position.x,
                "y": player.global_position.y
            },
            "health": player.health,
            "score": player.score
        }
    return {}

func load_player_data(data: Dictionary):
    var player = get_tree().get_first_node_in_group("player")
    if player and data.size() > 0:
        player.global_position = Vector2(data.position.x, data.position.y)
        player.health = data.get("health", 100)
        player.score = data.get("score", 0)

func save_inventory_data() -> Dictionary:

    return {}

func load_inventory_data(data: Dictionary):

    pass

func save_world_data() -> Dictionary:
    # 保存世界状态
    return {}

func load_world_data(data: Dictionary):
    # 加载世界状态

```

### 场景管理器

```gdscript
# scene_manager.gd
extends Node

signal scene_changed(scene_name: String)
signal transition_started
signal transition_completed

var current_scene: Node
var transition_data: Dictionary = {}

func _ready():
    var root = get_tree().root
    current_scene = root.get_child(root.get_child_count() - 1)

func change_scene(scene_path: String, data: Dictionary = {}):
    transition_data = data
    transition_started.emit()
    
    # 播放过渡动画
    await play_transition_out()
    
    # 卸载当前场景
    current_scene.queue_free()
    
    # 加载新场景
    var new_scene = load(scene_path)
    current_scene = new_scene.instantiate()
    get_tree().root.add_child(current_scene)
    get_tree().current_scene = current_scene
    
    # 播放进入动画
    await play_transition_in()
    
    var scene_name = current_scene.name
    scene_changed.emit(scene_name)
    transition_completed.emit()

func reload_current_scene():
    get_tree().reload_current_scene()

func get_transition_data() -> Dictionary:
    var data = transition_data.duplicate()
    transition_data.clear()
    return data

func play_transition_out():
    # 实现淡出效果
    var tween = create_tween()
    # tween.tween_property(...)
    await tween.finished

func play_transition_in():
    # 实现淡入效果
    var tween = create_tween()
    # tween.tween_property(...)
    await tween.finished
```

## 🎯 使用示例

### 在游戏中使用多个单例

```gdscript
# player.gd
extends CharacterBody2D

@export var max_health: int = 100
var health: int

func _ready():
    health = max_health
    EventBus.player_damaged.connect(_on_player_damaged)

func take_damage(amount: int):
    health -= amount
    EventBus.player_damaged.emit(amount)
    
    AudioManager.play_sfx(preload("res://sounds/hurt.wav"))
    
    if health <= 0:
        die()

func die():
    EventBus.player_died.emit()
    GameManager.lose_life()
    AudioManager.play_sfx(preload("res://sounds/death.wav"))

func collect_item(item_name: String):
    EventBus.item_collected.emit(item_name, 1)
    GameManager.add_score(10)
    AudioManager.play_sfx(preload("res://sounds/pickup.wav"))
```

## 📊 自动加载顺序

自动加载的节点按配置顺序添加到场景树：

```
配置顺序：
1. EventBus
2. GameManager
3. AudioManager
4. SaveManager
5. SceneManager

场景树顺序：
Root
├── EventBus
├── GameManager
├── AudioManager
├── SaveManager
├── SceneManager
└── CurrentScene
```

## 💡 最佳实践

### 单例职责划分

```gdscript
# 好：职责单一
GameManager      # 游戏状态管理
AudioManager     # 音频播放管理
SaveManager      # 存档管理
EventBus         # 全局事件通信

# 避免：职责混乱
GameController   # 既管理游戏状态，又处理音频，还负责存档...
```

### 避免过度使用

```gdscript
# 差：所有通信都通过全局单例
EventBus.player_moved.emit(position)
EventBus.player_stopped.emit()
EventBus.player_jumped.emit()

# 好：只在必要时使用全局单例
# 局部通信使用信号
signal moved(position)
signal jumped

# 全局事件使用 EventBus
EventBus.game_over.emit(score)
```

### 使用类名

```gdscript
# game_manager.gd
extends Node

class_name GameManager

# 这样可以在代码中获得类型提示
var manager: GameManager = GameManager
```
