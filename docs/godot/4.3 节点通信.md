# 4.3 节点通信

## 📡 节点通信概述

在 Godot 中，节点之间的通信是游戏开发的核心。不同的通信方式适用于不同的场景：

```
通信方式对比：

┌─────────────┬─────────────────┬─────────────────┐
│    方式     │      优点       │      缺点       │
├─────────────┼─────────────────┼─────────────────┤
│ 直接引用    │ 简单直接        │ 耦合度高        │
│ 信号        │ 解耦、灵活      │ 需要连接设置    │
│ 组          │ 批量操作        │ 性能开销        │
│ 全局单例    │ 全局访问        │ 可能过度使用    │
│ 节点路径    │ 编辑器配置      │ 路径依赖        │
└─────────────┴─────────────────┴─────────────────┘
```

## 👨‍👧 父子节点通信

### 子节点访问父节点

```gdscript
# 子节点脚本
extends Sprite2D

func _ready():
    # 获取直接父节点
    var parent = get_parent()
    
    # 调用父节点方法
    if parent.has_method("on_child_ready"):
        parent.on_child_ready(self)
    
    # 访问父节点属性
    if "player_speed" in parent:
        var speed = parent.player_speed

# 向上查找特定类型的父节点
func get_parent_of_type(type: String) -> Node:
    var current = get_parent()
    while current:
        if current.is_class(type):
            return current
        current = current.get_parent()
    return null

# 使用示例
func _ready():
    var player = get_parent_of_type("CharacterBody2D")
    if player:
        player.take_damage(10)
```

### 父节点访问子节点

```gdscript
# 父节点脚本
extends Node2D

@onready var sprite = $Sprite2D
@onready var collision = $CollisionShape2D

func _ready():
    # 直接访问子节点
    sprite.modulate = Color.RED
    
    # 遍历所有子节点
    for child in get_children():
        print("子节点: ", child.name)
    
    # 查找特定名称的子节点
    var found = find_child("Sprite2D", true, false)
    
    # 通过路径访问嵌套子节点
    var nested = $Enemies/Enemy1/Sprite2D
```

### 使用信号通信（推荐）

```gdscript
# 子节点：weapon.gd
extends Node2D

signal attack_performed(damage: int)
signal weapon_broke

@export var damage: int = 10
@export var durability: int = 100

func attack():
    durability -= 1
    attack_performed.emit(damage)
    
    if durability <= 0:
        weapon_broke.emit()

# 父节点：player.gd
extends CharacterBody2D

@onready var weapon = $Weapon

func _ready():
    weapon.attack_performed.connect(_on_attack_performed)
    weapon.weapon_broke.connect(_on_weapon_broke)

func _on_attack_performed(damage: int):
    print("造成伤害: ", damage)

func _on_weapon_broke():
    print("武器损坏")
    equip_new_weapon()
```

## 👥 同级节点通信

### 通过父节点中转

```gdscript
# 场景结构：
# Main
# ├── Player
# └── Enemy

# player.gd
extends CharacterBody2D

signal attacked(damage: int)

func attack():
    attacked.emit(10)

# enemy.gd
extends CharacterBody2D

@onready var player = $"../Player"

func _ready():
    player.attacked.connect(_on_player_attacked)

func _on_player_attacked(damage: int):
    take_damage(damage)
```

### 使用组通信

```gdscript
# player.gd
extends CharacterBody2D

func _ready():
    add_to_group("player")

func heal_all_enemies():
    # 通知所有敌人
    get_tree().call_group("enemies", "heal", 50)

# enemy.gd
extends CharacterBody2D

func _ready():
    add_to_group("enemies")

func heal(amount: int):
    health += amount
    print("%s 恢复了 %d 生命值" % [name, amount])
```

### 直接引用同级节点

```gdscript
# enemy_spawner.gd
extends Node2D

@export var player_path: NodePath

@onready var player = get_node(player_path)

func _ready():
    if player:
        player.died.connect(_on_player_died)

func _on_player_died():
```

## 🌐 跨场景通信

### 使用全局单例

```gdscript
# game_events.gd (自动加载)
extends Node

signal player_died

signal score_changed(new_score: int)

var score: int = 0
var current_level: int = 1

    level_completed.emit(current_level)

    GameEvents.player_died.emit()

    GameEvents.score_changed.connect(_on_score_changed)

func _on_player_died():
    show_game_over()

func _on_score_changed(new_score: int):
    update_score_display(new_score)
```

## 📊 通信模式对比

### 模式 1：直接调用

```gdscript
# 紧耦合方式
extends Node2D

@onready var player = $Player
@onready var enemy = $Enemy

func _ready():
    # 直接调用方法
    player.take_damage(10)
    enemy.set_target(player)
```

**适用场景**：紧密相关的节点，如玩家和其组件

### 模式 2：信号通信

```gdscript
# 松耦合方式
extends Node2D

@onready var player = $Player
@onready var enemy = $Enemy

func _ready():
    # 通过信号通信
    player.attacked.connect(enemy._on_player_attacked)
    enemy.died.connect(player._on_enemy_killed)
```

**适用场景**：需要解耦的节点，如玩家和敌人

### 模式 3：事件总线

```gdscript
# 全局事件总线
# event_bus.gd (自动加载)
extends Node

signal enemy_spawned(enemy: Node)
signal enemy_died(enemy: Node)
signal player_damaged(amount: int)

# enemy.gd
func die():
    EventBus.enemy_died.emit(self)
    queue_free()

# player.gd
func _ready():
    EventBus.enemy_died.connect(_on_enemy_died)

func _on_enemy_died(enemy: Node):
    add_score(enemy.score_value)
```

**适用场景**：全局事件，如成就系统、统计系统

## 🎯 实战示例

### 玩家与敌人交互

```gdscript
# player.gd
extends CharacterBody2D

signal health_changed(new_health: int)
signal died

@export var max_health: int = 100
var current_health: int

@onready var hitbox = $Hitbox

func _ready():
    current_health = max_health
    hitbox.body_entered.connect(_on_hitbox_body_entered)

func _on_hitbox_body_entered(body: Node2D):
    if body.is_in_group("enemies"):
        var enemy = body
        if enemy.has_method("get_damage"):
            take_damage(enemy.get_damage())

func take_damage(amount: int):
    current_health -= amount
    health_changed.emit(current_health)
    
    if current_health <= 0:
        die()

func die():
    died.emit()
    queue_free()

# enemy.gd
extends CharacterBody2D

signal died

@export var damage: int = 10
@export var score_value: int = 100

func get_damage() -> int:
    return damage

func kill():
    died.emit()
    queue_free()

# game_manager.gd
extends Node

@onready var player = $Player

func _ready():
    player.died.connect(_on_player_died)
    
    # 连接所有敌人的信号
    for enemy in get_tree().get_nodes_in_group("enemies"):
        enemy.died.connect(_on_enemy_died.bind(enemy))

func _on_player_died():
    show_game_over()

func _on_enemy_died(enemy: Node):
    add_score(enemy.score_value)
```

### UI 与游戏逻辑分离

```gdscript
# player_stats.gd
extends Node

signal health_changed(current: int, maximum: int)
signal score_changed(new_score: int)
signal level_changed(new_level: int)

var health: int = 100
var max_health: int = 100
var score: int = 0
var level: int = 1

func take_damage(amount: int):
    health = max(0, health - amount)
    health_changed.emit(health, max_health)

func heal(amount: int):
    health = min(max_health, health + amount)
    health_changed.emit(health, max_health)

func add_score(points: int):
    score += points
    score_changed.emit(score)
    
    if score >= level * 1000:
        level_up()

func level_up():
    level += 1
    level_changed.emit(level)

# player.gd
extends CharacterBody2D

@onready var stats = $PlayerStats

func _ready():
    stats.health_changed.connect(_on_health_changed)

func _on_health_changed(current: int, maximum: int):
    if current <= 0:
        die()

# hud.gd
extends CanvasLayer

@onready var player = $"../Player"
@onready var stats = player.stats

func _ready():
    stats.health_changed.connect(_update_health_bar)
    stats.score_changed.connect(_update_score)
    stats.level_changed.connect(_update_level)

func _update_health_bar(current: int, maximum: int):
    $HealthBar.value = float(current) / maximum * 100
    $HealthLabel.text = "%d/%d" % [current, maximum]

func _update_score(new_score: int):
    $ScoreLabel.text = "分数: %d" % new_score

func _update_level(new_level: int):
    $LevelLabel.text = "等级: %d" % new_level
```

### 场景间数据传递

```gdscript
# scene_manager.gd (自动加载)
extends Node

var transition_data: Dictionary = {}

func change_scene(scene_path: String, data: Dictionary = {}):
    transition_data = data
    get_tree().change_scene_to_file(scene_path)

func get_transition_data() -> Dictionary:
    var data = transition_data.duplicate()
    transition_data.clear()
    return data

# level_select.gd
extends Control

func _on_level_selected(level_id: int):
    SceneManager.change_scene("res://scenes/game.tscn", {
        "level_id": level_id,
        "difficulty": "normal"
    })

# game.gd
extends Node2D

var level_id: int
var difficulty: String

func _ready():
    var data = SceneManager.get_transition_data()
    level_id = data.get("level_id", 1)
    difficulty = data.get("difficulty", "normal")
    
    load_level(level_id)
    set_difficulty(difficulty)
```

## 💡 最佳实践

### 选择合适的通信方式

```gdscript
# 父子节点：直接引用或信号
@onready var child = $Child
child.do_something()

# 同级节点：信号或父节点中转
sibling.some_signal.connect(_on_signal)

# 跨场景：全局单例
GlobalEvents.some_event.emit()

# 批量操作：组
get_tree().call_group("enemies", "method")
```

### 避免循环引用

```gdscript
# 差：循环引用
# A 引用 B，B 引用 A
@onready var other = $Other  # A.gd
@onready var self_ref = $".."  # B.gd

# 好：使用信号打破循环
# A 发射信号，B 监听信号
signal request_action
request_action.emit()
```

### 使用接口模式

```gdscript
# 定义接口约定
# 任何可以受伤的节点都应该实现这些方法
func take_damage(amount: int):
    pass

func get_health() -> int:
    return 0

# 使用时检查
func attack_target(target: Node):
    if target.has_method("take_damage"):
        target.take_damage(damage)
```
