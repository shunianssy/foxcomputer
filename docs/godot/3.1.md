# 3.1 GDScript 基础

## 📝 GDScript 简介

GDScript 是 Godot 的官方脚本语言，专门为游戏开发设计：

- **语法简洁**：类似 Python，易于学习
- **类型可选**：支持动态类型和静态类型
- **与引擎集成**：直接访问 Godot API
- **高性能**：针对 Godot 优化

## 🔤 基本语法

### 注释

```gdscript
# 这是单行注释

"""
这是
多行注释
"""
```

### 变量声明

```gdscript
# 动态类型（类型自动推断）
var name = "Player"
var health = 100
var speed = 200.0
var is_alive = true

# 静态类型（推荐）
var name: String = "Player"
var health: int = 100
var speed: float = 200.0
var is_alive: bool = true

# 先声明后赋值
var position: Vector2
position = Vector2(100, 100)

# 常量
const MAX_HEALTH: int = 100
const GRAVITY: float = 980.0
const PLAYER_NAME: String = "Hero"
```

### 导出变量

```gdscript
# 在检查器中显示并可编辑
@export var speed: float = 200.0
@export var health: int = 100
@export var player_name: String = "Player"

# 导出枚举
enum EnemyType { NORMAL, FAST, TANK }
@export var enemy_type: EnemyType = EnemyType.NORMAL

# 导出范围
@export_range(0, 100) var progress: int = 0
@export_range(0.0, 1.0) var volume: float = 0.5

# 导出文件路径
@export_file("*.tscn") var scene_path: String
@export_file("*.png") var texture_path: String

# 导出颜色
@export var main_color: Color = Color.WHITE

# 导出节点引用
@export var target: NodePath
```

### 集合类型

```gdscript
# 数组
var numbers: Array = [1, 2, 3, 4, 5]
var mixed: Array = [1, "two", 3.0, true]
```

## 🔄 控制流

### 条件语句

```gdscript
# if-elif-else
var health = 50

if health > 80:
    print("健康状态良好")
elif health > 50:
    print("健康状态一般")
elif health > 20:
    print("健康状态较差")
else:
    print("濒临死亡")

# 三元运算符
var status = "alive" if health > 0 else "dead"

# match 语句（类似 switch）
var enemy_type = "fast"

match enemy_type:
    "normal":
        print("普通敌人")
    "fast":
        print("快速敌人")
    "tank":
        print("坦克敌人")
    _:
        print("未知类型")

# match 范围
var score = 85

match score:
    90..100:
        print("优秀")
    80..89:
        print("良好")
    60..79:
        print("及格")
    _:
        print("不及格")

# match 数组
var result = [1, 2]

match result:
    [1, 2]:
        print("匹配 [1, 2]")
    [1, ..]:
        print("匹配以1开头的数组")
    _:
        print("其他情况")
```

### 循环语句

```gdscript
# for 循环
for i in range(5):
    print(i)  # 0, 1, 2, 3, 4

for i in range(1, 5):
    print(i)  # 1, 2, 3, 4

for i in range(0, 10, 2):
    print(i)  # 0, 2, 4, 6, 8

# 遍历数组
var fruits = ["apple", "banana", "orange"]
for fruit in fruits:
    print(fruit)

# 遍历数组索引
for i in fruits.size():
    print("Index %d: %s" % [i, fruits[i]])

# 遍历字典
var player = {"name": "Hero", "level": 10}
for key in player:
    print(key, ": ", player[key])

for key, value in player:
    print(key, ": ", value)

# while 循环
var count = 0
while count < 5:
    print(count)
    count += 1

# 循环控制
for i in range(10):
    if i == 3:
        continue  # 跳过本次循环
    if i == 7:
        break     # 退出循环
    print(i)
```

## 📦 函数

### 函数定义

```gdscript
# 基本函数
func say_hello():
    print("Hello!")

# 带参数的函数
func greet(name: String):
    print("Hello, " + name + "!")

# 带返回值的函数
func add(a: int, b: int) -> int:
    return a + b

# 默认参数
func create_enemy(name: String, health: int = 100, speed: float = 100.0):
    print("Created: ", name, " HP: ", health, " Speed: ", speed)

# 调用
create_enemy("Goblin")                    # 使用默认值
create_enemy("Orc", 200)                  # 覆盖部分默认值
create_enemy("Dragon", 500, 150.0)        # 覆盖所有默认值
```

### 静态函数

```gdscript
# 静态函数属于类，不属于实例
static func calculate_damage(base: int, multiplier: float) -> int:
    return int(base * multiplier)

# 调用静态函数
var damage = MyClass.calculate_damage(10, 1.5)
```

### Lambda 表达式

```gdscript
# 匿名函数
var add = func(a: int, b: int): return a + b
print(add.call(1, 2))  # 输出: 3

# 作为参数传递
var numbers = [1, 2, 3, 4, 5]
var doubled = numbers.map(func(n): return n * 2)
var even = numbers.filter(func(n): return n % 2 == 0)
var sum = numbers.reduce(func(acc, n): return acc + n, 0)
```

## 🏛️ 类和继承

### 类定义

```gdscript
# player.gd
extends CharacterBody2D

class_name Player

# 属性
@export var speed: float = 200.0
var health: int = 100

# 信号
signal health_changed(new_health: int)
signal died

# 初始化
func _init():
    print("Player 初始化")

func _ready():
    print("Player 准备完成")

# 方法
func take_damage(amount: int):
    health -= amount
    health_changed.emit(health)
    
    if health <= 0:
        die()

func die():
    died.emit()
    queue_free()
```

### 继承

```gdscript
# enemy.gd - 父类
extends CharacterBody2D

class_name Enemy

@export var health: int = 100
@export var damage: int = 10

func take_damage(amount: int):
    health -= amount
    if health <= 0:
        die()

func die():
    queue_free()

func attack():
    print("Enemy attacks!")

# boss_enemy.gd - 子类
extends "res://scripts/enemy.gd"

class_name BossEnemy

@export var phase: int = 1

# 重写父类方法
func die():
    # 调用父类方法
    super.die()
    print("Boss defeated!")

# 新增方法
func special_attack():
    print("Boss uses special attack!")
```

### 内部类

```gdscript
# 在一个文件中定义多个类
extends Node

class_name GameManager

# 内部类
class InventoryItem:
    var name: String
    var count: int
    
    func _init(item_name: String, item_count: int):
        name = item_name
        count = item_count
    
    func add(amount: int):
        count += amount

# 使用内部类
var item = GameManager.InventoryItem.new("Potion", 5)
```

## 🔧 类型转换

```gdscript
# 基本类型转换

var text: String = "123"
var num: int = int(text)  # 123

# 对象类型检查
if node is Sprite2D:
    node.texture = new_texture

var sprite = node as Sprite2D
if sprite:
    sprite.modulate = Color.RED

# 安全类型转换
var collision = body as CharacterBody2D
if collision and collision.has_method("take_damage"):
    collision.take_damage(10)
```

## 💡 最佳实践

### 命名规范

```gdscript
# 常量：全大写下划线
const MAX_SPEED: float = 500.0
const DEFAULT_HEALTH: int = 100

# 变量：小写下划线
var current_health: int = 100
var move_speed: float = 200.0

# 函数：小写下划线
func calculate_damage():
    pass

# 类名：大驼峰
class_name PlayerController
class_name EnemySpawner

# 私有成员：下划线前缀
var _internal_value: int = 0
func _helper_function():
    pass
```

### 类型提示

```gdscript
# 推荐：始终使用类型提示
var health: int = 100
var position: Vector2 = Vector2.ZERO

func get_player() -> Player:
    return $Player as Player

func deal_damage(target: Node, amount: int) -> bool:
    if target.has_method("take_damage"):
        target.take_damage(amount)
        return true
    return false
```
