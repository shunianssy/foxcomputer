# 4.2 自定义信号

## 📝 定义自定义信号

### 基本语法

```gdscript
extends Node

# 无参数信号
signal game_started
signal paused
signal resumed

# 带参数的信号
signal health_changed(new_health: int)
signal item_collected(item_name: String, count: int)
signal position_changed(new_position: Vector2)

# 带多个参数的信号
signal enemy_spawned(enemy: Node, position: Vector2, level: int)
signal damage_dealt(target: Node, amount: float, type: String)
```

### 类型提示

```gdscript
extends Node

# 推荐使用类型提示
signal health_changed(new_health: int, max_health: int)
signal player_moved(direction: Vector2, speed: float)

# 无类型提示（不推荐，但合法）
signal something_happened(value)
```

## 📡 发射信号

### emit() 方法

```gdscript
extends CharacterBody2D

signal health_changed(new_health: int)
signal died

var health: int = 100

func take_damage(amount: int):
    health -= amount
    health_changed.emit(health)
    
    if health <= 0:
        died.emit()

func heal(amount: int):
    health += amount
    health_changed.emit(health)
```

### 条件发射

```gdscript
extends Node

signal level_up(new_level: int)

var experience: int = 0
var level: int = 1
var experience_to_next: int = 100

func add_experience(amount: int):
    experience += amount
    
    while experience >= experience_to_next:
        experience -= experience_to_next
        level += 1
        experience_to_next = int(experience_to_next * 1.5)
        level_up.emit(level)
```

## 🔗 连接自定义信号

### 基本连接

```gdscript
# player.gd
extends CharacterBody2D

signal health_changed(new_health: int)
signal died

# hud.gd
extends CanvasLayer

@onready var player = $"../Player"

func _ready():
    player.health_changed.connect(_on_health_changed)
    player.died.connect(_on_player_died)

func _on_health_changed(new_health: int):
    update_health_display(new_health)

func _on_player_died():
    show_game_over_screen()
```

### 使用 Callable

```gdscript
extends Node

@onready var player = $Player

func _ready():
    # 使用 Callable
    var callback = Callable(self, "_on_health_changed")
    player.health_changed.connect(callback)
    
    # 使用 lambda
    player.health_changed.connect(func(health): print("Health: ", health))

func _on_health_changed(new_health: int):
    print("Health changed to: ", new_health)
```

## 🎯 实战示例

### 血量系统

```gdscript
# health_component.gd
extends Node

class_name HealthComponent

signal health_changed(current: int, max_health: int)
signal health_depleted
signal damage_taken(amount: int)

@export var max_health: int = 100
var current_health: int

func _ready():
    current_health = max_health

func take_damage(amount: int):
    var actual_damage = mini(amount, current_health)
    current_health -= actual_damage
    
    damage_taken.emit(actual_damage)
    health_changed.emit(current_health, max_health)
    
    if current_health <= 0:
        health_depleted.emit()

func heal(amount: int):
    var actual_heal = mini(amount, max_health - current_health)
    current_health += actual_heal
    health_changed.emit(current_health, max_health)

# player.gd
extends CharacterBody2D

@onready var health_component = $HealthComponent

func _ready():
    health_component.health_changed.connect(_on_health_changed)
    health_component.health_depleted.connect(_on_health_depleted)

func _on_health_changed(current: int, max_hp: int):
    print("血量: %d/%d" % [current, max_hp])

func _on_health_depleted():
    print("玩家死亡")
    die()
```

### 物品系统

```gdscript
# inventory.gd
extends Node

signal item_added(item: Dictionary)
signal item_removed(item_name: String, count: int)
signal inventory_full

var items: Dictionary = {}
@export var max_slots: int = 20

func add_item(item_name: String, count: int = 1) -> bool:
    if get_total_items() >= max_slots and not items.has(item_name):
        inventory_full.emit()
        return false
    
    if items.has(item_name):
        items[item_name] += count
    else:
        items[item_name] = count
    
    item_added.emit({"name": item_name, "count": count})
    return true

func remove_item(item_name: String, count: int = 1) -> bool:
    if not items.has(item_name) or items[item_name] < count:
        return false
    
    items[item_name] -= count
    item_removed.emit(item_name, count)
    
    if items[item_name] <= 0:
        items.erase(item_name)
    
    return true

func get_total_items() -> int:
    var total = 0
    for count in items.values():
        total += count
    return total
```

## 📊 信号参数绑定

```gdscript
extends Node

signal button_clicked(button_id: int)

func _ready():
    for i in range(5):
        var button = create_button(i)
        # 绑定按钮 ID
        button.pressed.connect(_on_button_clicked.bind(i))

func _on_button_clicked(button_id: int):
    print("按钮 %d 被点击" % button_id)
```

## 💡 最佳实践

### 命名规范

```gdscript
# 信号命名使用过去式或事件描述
signal health_changed(new_health: int)    # 好
signal died                               # 好
signal item_collected(item: Item)         # 好

# 避免使用动词原形
signal change_health                      # 不推荐
signal die                                # 不推荐
```

### 避免循环信号

```gdscript
# 使用标志位避免循环
var is_processing: bool = false

func _on_a_triggered():
    if is_processing:
        return
    is_processing = true
    # 处理逻辑
    is_processing = false
```
