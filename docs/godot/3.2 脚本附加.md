# 3.2 脚本附加

## 什么是脚本附加？

脚本附加是将 GDScript 文件与节点关联的过程。附加脚本后，节点将获得脚本中定义的属性和方法。

```
节点 + 脚本 = 功能完整的游戏对象

Sprite2D 节点
 ↓ 附加脚本
Sprite2D + player.gd
 ↓ 结果
具有移动、动画、碰撞检测功能的角色
```

## 创建和附加脚本

### 方法 1：通过编辑器

1. 在场景树中选择节点
2. 点击右上角的"附加脚本"按钮（或右键 → 附加脚本）
3. 配置脚本选项：
 - **脚本路径**：选择保存位置
 - **模板**：选择模板类型
 - **语言**：GDScript

### 方法 2：拖放脚本

1. 在文件系统面板中找到 `.gd` 文件
2. 将其拖放到场景树中的节点上

### 方法 3：通过代码

```gdscript
# 动态加载并设置脚本
var script = load("res://scripts/player.gd")
node.set_script(script)
```

## 🏗 脚本模板

### 默认模板

```gdscript
extends Node2D

# Called when the node enters the scene tree for the first time.
func _ready():
 pass # Replace with function body.

# Called every frame. 'delta' is the elapsed time since the previous frame.
func _process(delta):
 pass
```

### CharacterBody2D 模板

```gdscript
extends CharacterBody2D

const SPEED = 300.0
const JUMP_VELOCITY = -400.0

# Get the gravity from the project settings to be synced with RigidBody nodes.
var gravity = ProjectSettings.get_setting("physics/2d/default_gravity")

func _physics_process(delta):
 # Add the gravity.
 if not is_on_floor():
 velocity.y += gravity * delta

 # Handle jump.
 if Input.is_action_just_pressed("ui_accept") and is_on_floor():
 velocity.y = JUMP_VELOCITY

 # Get the input direction and handle the movement/deceleration.
 var direction = Input.get_axis("ui_left", "ui_right")
 if direction:
 velocity.x = direction * SPEED
 else:
 velocity.x = move_toward(velocity.x, 0, SPEED)

 move_and_slide()
```

## extends 关键字

### 继承节点类型

```gdscript
# 继承 Node2D
extends Node2D

# 继承 CharacterBody2D
extends CharacterBody2D

# 继承 Control
extends Control

# 继承自定义脚本
extends "res://scripts/base_enemy.gd"
```

### 继承的作用

```gdscript
extends Sprite2D

func _ready():

 texture = load("res://icon.svg")
 position = Vector2(100, 100)
```

## 脚本属性访问

### 访问节点属性

```gdscript
extends Sprite2D

func _ready():
 # 直接访问属性
 position = Vector2(100, 100)
 rotation = 0.5
 scale = Vector2(2, 2)
 modulate = Color(1, 0.5, 0)
 
 # 使用 self 明确表示当前节点
 self.visible = true
```

### 访问子节点

```gdscript
extends Node2D

func _ready():
 # 方法 1：使用 $ 语法
 var sprite = $Sprite2D
 var collision = $CollisionShape2D
 
 # 方法 2：使用 get_node
 var sprite2 = get_node("Sprite2D")
 
 # 方法 3：使用路径
 var child = get_node("Enemies/Enemy1")
 
 # 方法 4：使用 @onready 缓存（推荐）
 @onready var sprite = $Sprite2D
 @onready var collision = $CollisionShape2D
```

### @onready 关键字

```gdscript
extends CharacterBody2D

# 在 _ready 之前执行，用于缓存节点引用
@onready var sprite = $Sprite2D
@onready var collision = $CollisionShape2D
@onready var animation_player = $AnimationPlayer

# 可以使用表达式
@onready var gun_position = $GunPosition.global_position

func _ready():
 # 此时 @onready 变量已经初始化
 sprite.modulate = Color.RED
 animation_player.play("idle")
```

## 脚本通信

### 从脚本访问其他节点

```gdscript
extends Node2D

@onready var player = $Player

func _ready():
 # 访问子节点
 player.take_damage(10)
 
 # 访问父节点
 var parent = get_parent()
 
 # 访问兄弟节点
 var sibling = get_parent().get_node("Sibling")
 
 # 访问根节点
 var root = get_tree().root
 
 # 访问当前场景
 var current_scene = get_tree().current_scene
```

### 使用节点路径导出

```gdscript
extends Node2D

# 在检查器中设置目标节点
@export var target_path: NodePath

@onready var target = get_node(target_path)

func _ready():
 if target:
 target.do_something()

# 或者直接导出节点
@export var target: Node

func _ready():
 if target:
 target.do_something()
```

## 多脚本协作

### 场景结构

```
Player (CharacterBody2D) - player.gd
├── Sprite2D
├── CollisionShape2D
├── AnimationPlayer
└── StateMachine (Node) - state_machine.gd
 ├── IdleState (Node) - idle_state.gd
 ├── WalkState (Node) - walk_state.gd
 └── JumpState (Node) - jump_state.gd
```

### 脚本间通信

```gdscript
# player.gd
extends CharacterBody2D

signal health_changed(new_health: int)

@export var max_health: int = 100
var current_health: int

@onready var state_machine = $StateMachine

func _ready():
 current_health = max_health
 state_machine.initialize(self)

func take_damage(amount: int):
 current_health -= amount
 health_changed.emit(current_health)

# state_machine.gd
extends Node

var current_state: Node
var player: CharacterBody2D

func initialize(owner: CharacterBody2D):
 player = owner
 change_state("Idle")

func change_state(state_name: String):
 if current_state:
 current_state.exit()
 
 current_state = get_node(state_name)
 current_state.enter(player)
```

## 脚本继承

### 创建基类脚本

```gdscript
# base_enemy.gd
extends CharacterBody2D

class_name BaseEnemy

signal died

@export var health: int = 100
@export var damage: int = 10
@export var speed: float = 100.0

var target: Node2D

func _ready():
 find_target()

func find_target():
 target = get_tree().get_first_node_in_group("player")

func take_damage(amount: int):
 health -= amount
 if health <= 0:
 die()

func die():
 died.emit()
 queue_free()

func attack():
 if target and target.has_method("take_damage"):
 target.take_damage(damage)
```

### 创建子类脚本

```gdscript
# flying_enemy.gd
extends "res://scripts/base_enemy.gd"

class_name FlyingEnemy

@export var fly_height: float = 50.0
@export var hover_speed: float = 2.0

var time: float = 0.0

func _physics_process(delta):
 # 调用父类方法
 super._physics_process(delta)
 
 # 添加飞行行为
 time += delta
 position.y += sin(time * hover_speed) * fly_height * delta

# 重写父类方法
func die():
 # 添加特效
 create_explosion()
 # 调用父类方法
 super.die()

func create_explosion():
 var explosion = preload("res://scenes/explosion.tscn").instantiate()
 get_tree().current_scene.add_child(explosion)
 explosion.global_position = global_position
```

## 动态脚本操作

### 运行时更换脚本

```gdscript
extends Node

func change_enemy_behavior(enemy: Node, new_script_path: String):
 # 保存当前状态
 var position = enemy.global_position
 var health = enemy.health if "health" in enemy else 100
 
 # 更换脚本
 var new_script = load(new_script_path)
 enemy.set_script(new_script)
 
 # 恢复状态
 enemy.global_position = position
 if "health" in enemy:
 enemy.health = health
```

### 检查脚本和方法

```gdscript
func interact_with_node(node: Node):
 # 检查是否有脚本
 if node.get_script():
 print("节点有脚本")
 
 # 检查是否有特定方法
 if node.has_method("interact"):
 node.interact()
 
 # 检查是否有特定属性
 if "health" in node:
 print("血量: ", node.health)
 
 # 检查是否是特定类型
 if node is CharacterBody2D:
 print("这是一个物理角色")
 
 # 检查脚本类名
 if node.get_script() and node.get_script().get_global_name() == "Player":
 print("这是玩家")
```

## 脚本组织最佳实践

### 文件结构

```
├── player/
│ ├── player.gd # 玩家主脚本
│ ├── player_input.gd # 输入处理
│ └── player_animation.gd # 动画控制
├── enemies/
│ ├── base_enemy.gd # 敌人基类
│ ├── flying_enemy.gd # 飞行敌人
│ └── tank_enemy.gd # 坦克敌人
├── systems/
│ ├── game_manager.gd # 游戏管理器
│ └── audio_manager.gd # 音频管理器
└── utils/
 ├── helpers.gd # 辅助函数
 └── constants.gd # 常量定义
```

### 单一职责原则

```gdscript
# 不推荐：一个脚本做太多事
extends CharacterBody2D

func _physics_process(delta):
 handle_movement(delta)
 handle_animation()
 handle_input()
 handle_camera()
 handle_audio()
 handle_ui()

# 推荐：分离职责
# player_movement.gd
extends Node
class_name PlayerMovement

var player: CharacterBody2D

func move(delta: float):
 # 只处理移动
 pass

# player.gd
extends CharacterBody2D

@onready var movement = $Movement

func _physics_process(delta):
 movement.move(delta)
```
