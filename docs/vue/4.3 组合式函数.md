# 4.3 组合式函数

## 组合式函数概述

组合式函数（Composables）是 Vue 3 中复用有状态逻辑的主要方式。它是一个函数，可以用来封装和复用响应式状态。

## 基础示例

### 简单的组合式函数

```javascript
// composables/useCounter.js
import { ref, computed } from 'vue'

export function useCounter(initialValue = 0) {
 const count = ref(initialValue)
 
 const doubled = computed(() => count.value * 2)
 
 function increment() {
 count.value++
 }
 
 function decrement() {
 count.value--
 }
 
 function reset() {
 count.value = initialValue
 }
 
 return {
 count,
 doubled,
 increment,
 decrement,
 reset
 }
}
```

```vue
<!-- 使用 -->
<script setup>
import { useCounter } from '@/composables/useCounter'

const { count, doubled, increment, decrement, reset } = useCounter(10)
</script>

<template>
 <div>
 <p>Count: {{ count }}</p>
 <p>Doubled: {{ doubled }}</p>
 <button @click="increment">+1</button>
 <button @click="decrement">-1</button>
 <button @click="reset">Reset</button>
 </div>
</template>
```

## 常用组合式函数

### useMouse - 鼠标位置追踪

```javascript
// composables/useMouse.js
import { ref, onMounted, onUnmounted } from 'vue'

export function useMouse() {
 const x = ref(0)
 const y = ref(0)
 
 function update(event) {
 x.value = event.clientX
 y.value = event.clientY
 }
 
 onMounted(() => {
 window.addEventListener('mousemove', update)
 })
 
 onUnmounted(() => {
 window.removeEventListener('mousemove', update)
 })
 
 return { x, y }
}
```

```vue
<script setup>
import { useMouse } from '@/composables/useMouse'

const { x, y } = useMouse()
</script>

<template>
 <p>鼠标位置: {{ x }}, {{ y }}</p>
</template>
```

### useFetch - 数据请求

```javascript
// composables/useFetch.js
import { ref, watchEffect, toValue } from 'vue'

export function useFetch(url) {
 const data = ref(null)
 const error = ref(null)
 const loading = ref(false)
 
 async function fetchData() {
 loading.value = true
 error.value = null
 
 try {
 const response = await fetch(toValue(url))
 
 if (!response.ok) {
 throw new Error(`HTTP error! status: ${response.status}`)
 }
 
 data.value = await response.json()
 } catch (e) {
 error.value = e
 } finally {
 loading.value = false
 }
 }
 
 // 当 URL 变化时自动重新请求
 watchEffect(() => {
 fetchData()
 })
 
 return {
 data,
 error,
 loading,
 refetch: fetchData
 }
}
```

```vue
<script setup>
import { ref } from 'vue'
import { useFetch } from '@/composables/useFetch'

const userId = ref(1)
const { data, error, loading } = useFetch(
 () => `https://jsonplaceholder.typicode.com/users/${userId.value}`
)
</script>

<template>
 <div>
 <button @click="userId++">下一个用户</button>
 
 <div v-if="loading">加载中...</div>
 <div v-else-if="error">错误: {{ error.message }}</div>
 <div v-else>
 <h3>{{ data.name }}</h3>
 <p>{{ data.email }}</p>
 </div>
 </div>
</template>
```

### useLocalStorage - 本地存储

```javascript
// composables/useLocalStorage.js
import { ref, watch } from 'vue'

export function useLocalStorage(key, defaultValue) {
 // 从 localStorage 读取初始值
 const storedValue = localStorage.getItem(key)
 const data = ref(storedValue ? JSON.parse(storedValue) : defaultValue)
 
 // 监听变化并同步到 localStorage
 watch(data, (newValue) => {
 localStorage.setItem(key, JSON.stringify(newValue))
 }, { deep: true })
 
 // 移除存储
 function remove() {
 localStorage.removeItem(key)
 data.value = defaultValue
 }
 
 return {
 data,
 remove
 }
}
```

```vue
<script setup>
import { useLocalStorage } from '@/composables/useLocalStorage'

const { data: theme } = useLocalStorage('theme', 'light')
const { data: user } = useLocalStorage('user', { name: '', email: '' })

function toggleTheme() {
 theme.value = theme.value === 'light' ? 'dark' : 'light'
}
</script>

<template>
 <div>
 <p>主题: {{ theme }}</p>
 <button @click="toggleTheme">切换主题</button>
 
 <input v-model="user.name" placeholder="姓名" />
 <input v-model="user.email" placeholder="邮箱" />
 </div>
</template>
```

### useDebounce - 防抖

```javascript
// composables/useDebounce.js
import { ref, watch } from 'vue'

export function useDebounce(value, delay = 300) {
 const debouncedValue = ref(value.value)
 let timeout = null
 
 watch(value, (newValue) => {
 clearTimeout(timeout)
 timeout = setTimeout(() => {
 debouncedValue.value = newValue
 }, delay)
 })
 
 return debouncedValue
}

// 或者返回一个防抖函数
export function useDebounceFn(fn, delay = 300) {
 let timeout = null
 
 return function (...args) {
 clearTimeout(timeout)
 timeout = setTimeout(() => {
 fn.apply(this, args)
 }, delay)
 }
}
```

```vue
<script setup>
import { ref } from 'vue'
import { useDebounce, useDebounceFn } from '@/composables/useDebounce'

const searchQuery = ref('')
const debouncedQuery = useDebounce(searchQuery, 500)

// 使用防抖函数
const debouncedSearch = useDebounceFn((query) => {
 console.log('搜索:', query)
}, 500)

function handleInput() {
 debouncedSearch(searchQuery.value)
}
</script>

<template>
 <div>
 <input v-model="searchQuery" @input="handleInput" placeholder="搜索..." />
 <p>防抖后的值: {{ debouncedQuery }}</p>
 </div>
</template>
```

### useToggle - 切换状态

```javascript
// composables/useToggle.js
import { ref } from 'vue'

export function useToggle(initialValue = false) {
 const value = ref(initialValue)
 
 function toggle() {
 value.value = !value.value
 }
 
 function setTrue() {
 value.value = true
 }
 
 function setFalse() {
 value.value = false
 }
 
 return {
 value,
 toggle,
 setTrue,
 setFalse
 }
}
```

```vue
<script setup>
import { useToggle } from '@/composables/useToggle'

const { value: isOpen, toggle, setFalse: close } = useToggle()
</script>

<template>
 <div>
 <button @click="toggle">切换</button>
 <button @click="close">关闭</button>
 
 <div v-if="isOpen" class="modal">
 弹窗内容
 </div>
 </div>
</template>
```

### useEventListener - 事件监听

```javascript
// composables/useEventListener.js
import { onMounted, onUnmounted } from 'vue'

export function useEventListener(target, event, callback, options = {}) {
 onMounted(() => {
 target.addEventListener(event, callback, options)
 })
 
 onUnmounted(() => {
 target.removeEventListener(event, callback, options)
 })
}
```

```vue
<script setup>
import { ref } from 'vue'
import { useEventListener } from '@/composables/useEventListener'

const windowWidth = ref(window.innerWidth)

useEventListener(window, 'resize', () => {
 windowWidth.value = window.innerWidth
})
</script>

<template>
 <p>窗口宽度: {{ windowWidth }}px</p>
</template>
```

## 实际应用示例

### 表单验证

```javascript
// composables/useForm.js
import { ref, reactive, computed } from 'vue'

export function useForm(initialValues, validationRules) {
 const values = reactive({ ...initialValues })
 const errors = reactive({})
 const touched = reactive({})
 const isSubmitting = ref(false)
 
 // 验证单个字段
 function validateField(field) {
 const rules = validationRules[field]
 if (!rules) return true
 
 for (const rule of rules) {
 const error = rule(values[field], values)
 if (error) {
 errors[field] = error
 return false
 }
 }
 
 errors[field] = ''
 return true
 }
 
 // 验证所有字段
 function validateAll() {
 let isValid = true
 
 for (const field in validationRules) {
 if (!validateField(field)) {
 isValid = false
 }
 }
 
 return isValid
 }
 
 // 设置字段为已触碰
 function touch(field) {
 touched[field] = true
 validateField(field)
 }
 
 // 提交表单
 async function handleSubmit(callback) {
 isSubmitting.value = true
 
 // 触碰所有字段
 for (const field in validationRules) {
 touched[field] = true
 }
 
 if (validateAll()) {
 await callback(values)
 }
 
 isSubmitting.value = false
 }
 
 // 重置表单
 function reset() {
 Object.assign(values, initialValues)
 Object.keys(errors).forEach(key => errors[key] = '')
 Object.keys(touched).forEach(key => touched[key] = false)
 }
 
 const isValid = computed(() => {
 return Object.values(errors).every(error => !error)
 })
 
 return {
 values,
 errors,
 touched,
 isSubmitting,
 isValid,
 validateField,
 validateAll,
 touch,
 handleSubmit,
 reset
 }
}
```

```vue
<script setup>
import { useForm } from '@/composables/useForm'

const { values, errors, touched, isSubmitting, touch, handleSubmit, reset } = useForm(
 // 初始值
 {
 email: '',
 password: '',
 confirmPassword: ''
 },
 // 验证规则
 {
 email: [
 (value) => !value && '请输入邮箱',
 (value) => !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value) && '请输入有效的邮箱'
 ],
 password: [
 (value) => !value && '请输入密码',
 (value) => value.length < 6 && '密码至少 6 个字符'
 ],
 confirmPassword: [
 (value) => !value && '请确认密码',
 (value, values) => value !== values.password && '两次密码不一致'
 ]
 }
)

async function submitForm(values) {
 console.log('提交:', values)
 // 发送请求...
}
</script>

<template>
 <form @submit.prevent="handleSubmit(submitForm)">
 <div>
 <input
 v-model="values.email"
 @blur="touch('email')"
 placeholder="邮箱"
 />
 <span v-if="touched.email && errors.email" class="error">
 {{ errors.email }}
 </span>
 </div>
 
 <div>
 <input
 v-model="values.password"
 type="password"
 @blur="touch('password')"
 placeholder="密码"
 />
 <span v-if="touched.password && errors.password" class="error">
 {{ errors.password }}
 </span>
 </div>
 
 <div>
 <input
 v-model="values.confirmPassword"
 type="password"
 @blur="touch('confirmPassword')"
 placeholder="确认密码"
 />
 <span v-if="touched.confirmPassword && errors.confirmPassword" class="error">
 {{ errors.confirmPassword }}
 </span>
 </div>
 
 <button type="submit" :disabled="isSubmitting">
 {{ isSubmitting ? '提交中...' : '提交' }}
 </button>
 <button type="button" @click="reset">重置</button>
 </form>
</template>
```

### 分页

```javascript
// composables/usePagination.js
import { ref, computed } from 'vue'

export function usePagination(items, pageSize = 10) {
 const currentPage = ref(1)
 const itemsPerPage = ref(pageSize)
 
 const totalPages = computed(() => {
 return Math.ceil(items.value.length / itemsPerPage.value)
 })
 
 const paginatedItems = computed(() => {
 const start = (currentPage.value - 1) * itemsPerPage.value
 const end = start + itemsPerPage.value
 return items.value.slice(start, end)
 })
 
 const pageInfo = computed(() => ({
 currentPage: currentPage.value,
 totalPages: totalPages.value,
 totalItems: items.value.length,
 start: (currentPage.value - 1) * itemsPerPage.value + 1,
 end: Math.min(currentPage.value * itemsPerPage.value, items.value.length)
 }))
 
 function goToPage(page) {
 if (page >= 1 && page <= totalPages.value) {
 currentPage.value = page
 }
 }
 
 function nextPage() {
 goToPage(currentPage.value + 1)
 }
 
 function prevPage() {
 goToPage(currentPage.value - 1)
 }
 
 function firstPage() {
 goToPage(1)
 }
 
 function lastPage() {
 goToPage(totalPages.value)
 }
 
 return {
 currentPage,
 itemsPerPage,
 totalPages,
 paginatedItems,
 pageInfo,
 goToPage,
 nextPage,
 prevPage,
 firstPage,
 lastPage
 }
}
```

```vue
<script setup>
import { ref } from 'vue'
import { usePagination } from '@/composables/usePagination'

const items = ref(Array.from({ length: 100 }, (_, i) => ({
 id: i + 1,
 name: `项目 ${i + 1}`
})))

const {
 paginatedItems,
 pageInfo,
 currentPage,
 totalPages,
 goToPage,
 nextPage,
 prevPage
} = usePagination(items, 10)
</script>

<template>
 <div>
 <ul>
 <li v-for="item in paginatedItems" :key="item.id">
 {{ item.name }}
 </li>
 </ul>
 
 <div class="pagination">
 <button @click="prevPage" :disabled="currentPage === 1">上一页</button>
 
 <button
 v-for="page in totalPages"
 :key="page"
 @click="goToPage(page)"
 :class="{ active: page === currentPage }"
 >
 {{ page }}
 </button>
 
 <button @click="nextPage" :disabled="currentPage === totalPages">下一页</button>
 </div>
 
 <p>
 显示 {{ pageInfo.start }}-{{ pageInfo.end }} 条，
 共 {{ pageInfo.totalItems }} 条
 </p>
 </div>
</template>
```

## 总结

| 特点 | 说明 |
|------|------|
| 命名约定 | 以 `use` 开头 |
| 返回值 | 返回响应式引用和方法 |
| 生命周期 | 可以使用生命周期钩子 |
| 组合使用 | 可以在组合式函数中调用其他组合式函数 |

**最佳实践**：
- 命名以 `use` 开头
- 返回响应式引用和方法
- 封装可复用的逻辑
- 保持函数职责单一
