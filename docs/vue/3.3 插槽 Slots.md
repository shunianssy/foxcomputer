# 3.3 插槽 Slots

## 插槽概述

插槽（Slots）是 Vue 组件的内容分发机制，允许父组件向子组件传递模板内容。

## 默认插槽

### 基础用法

```vue
<!-- Card.vue -->
<script setup>
defineProps({
 title: String
})
</script>

<template>
 <div class="card">
 <div class="card-header">
 <h3>{{ title }}</h3>
 </div>
 <div class="card-body">
 <!-- 插槽出口 -->
 <slot />
 </div>
 </div>
</template>

<style scoped>
.card {
 border: 1px solid #ddd;
 border-radius: 8px;
 overflow: hidden;
}

.card-header {
 padding: 16px;
 background: #f5f5f5;
}

.card-body {
 padding: 16px;
}
</style>
```

```vue
<!-- 使用 -->
<script setup>
import Card from './Card.vue'
</script>

<template>
 <Card title="用户信息">
 <!-- 插槽内容 -->
 <p>姓名：张三</p>
 <p>年龄：25</p>
 </Card>
</template>
```

### 默认内容

```vue
<!-- Button.vue -->
<script setup>
defineProps({
 type: {
 type: String,
 default: 'button'
 }
})
</script>

<template>
 <button :type="type" class="btn">
 <!-- 如果没有提供内容，显示默认内容 -->
 <slot>
 点击按钮
 </slot>
 </button>
</template>
```

```vue
<!-- 使用 -->
<script setup>
import Button from './Button.vue'
</script>

<template>
 <!-- 使用默认内容 -->
 <Button />
 
 <!-- 自定义内容 -->
 <Button>提交</Button>
 <Button>取消</Button>
</template>
```

## 具名插槽

当需要多个插槽时，使用具名插槽：

```vue
<!-- Layout.vue -->
<script setup>
</script>

<template>
 <div class="layout">
 <header class="header">
 <slot name="header" />
 </header>
 
 <main class="main">
 <slot />
 </main>
 
 <footer class="footer">
 <slot name="footer" />
 </footer>
 </div>
</template>

<style scoped>
.layout {
 min-height: 100vh;
 display: flex;
 flex-direction: column;
}

.header {
 padding: 16px;
 background: #42b883;
 color: white;
}

.main {
 flex: 1;
 padding: 20px;
}

.footer {
 padding: 16px;
 background: #f5f5f5;
 text-align: center;
}
</style>
```

```vue
<!-- 使用 -->
<script setup>
import Layout from './Layout.vue'
</script>

<template>
 <Layout>
 <!-- 具名插槽使用 v-slot:name 或 #name -->
 <template #header>
 <h1>网站标题</h1>
 <nav>
 <a href="/">首页</a>
 <a href="/about">关于</a>
 </nav>
 </template>

 <!-- 默认插槽 -->
 <p>主要内容</p>

 <template #footer>
 <p>&copy; 2024 My Website</p>
 </template>
 </Layout>
</template>
```

## 作用域插槽

作用域插槽允许子组件向插槽传递数据：

### 基础用法

```vue
<!-- List.vue -->
<script setup>
const items = [
 { id: 1, name: '苹果', price: 5 },
 { id: 2, name: '香蕉', price: 3 },
 { id: 3, name: '橙子', price: 4 }
]
</script>

<template>
 <ul class="list">
 <li v-for="item in items" :key="item.id">
 <!-- 通过插槽属性传递数据 -->
 <slot :item="item" :index="item.id">
 {{ item.name }}
 </slot>
 </li>
 </ul>
</template>
```

```vue
<!-- 使用 -->
<script setup>
import List from './List.vue'
</script>

<template>
 <List v-slot="slotProps">
 <!-- 接收插槽传递的数据 -->
 {{ slotProps.item.name }} - ¥{{ slotProps.item.price }}
 </List>
</template>
```

### 解构插槽 Props

```vue
<template>
 <!-- 解构插槽属性 -->
 <List v-slot="{ item, index }">
 {{ index }}. {{ item.name }} - ¥{{ item.price }}
 </List>
</template>
```

### 具名作用域插槽

```vue
<!-- Table.vue -->
<script setup>
defineProps({
 columns: Array,
 data: Array
})
</script>

<template>
 <table>
 <thead>
 <tr>
 <th v-for="col in columns" :key="col.key">
 <slot name="header" :column="col">
 {{ col.title }}
 </slot>
 </th>
 </tr>
 </thead>
 <tbody>
 <tr v-for="(row, rowIndex) in data" :key="row.id">
 <td v-for="col in columns" :key="col.key">
 <slot :name="col.key" :row="row" :index="rowIndex">
 {{ row[col.key] }}
 </slot>
 </td>
 </tr>
 </tbody>
 </table>
</template>
```

```vue
<!-- 使用 -->
<script setup>
import Table from './Table.vue'

const columns = [
 { key: 'name', title: '姓名' },
 { key: 'age', title: '年龄' },
 { key: 'actions', title: '操作' }
]

const data = [
 { id: 1, name: '张三', age: 25 },
 { id: 2, name: '李四', age: 30 }
]
</script>

<template>
 <Table :columns="columns" :data="data">
 <!-- 自定义表头 -->
 <template #header="{ column }">
 <strong>{{ column.title }}</strong>
 </template>

 <!-- 自定义姓名列 -->
 <template #name="{ row }">
 <span style="color: blue;">{{ row.name }}</span>
 </template>

 <!-- 自定义操作列 -->
 <template #actions="{ row, index }">
 <button @click="edit(row)">编辑</button>
 <button @click="remove(index)">删除</button>
 </template>
 </Table>
</template>
```

## 实际应用示例

### 可复用弹窗组件

```vue
<!-- Modal.vue -->
<script setup>
import { watch } from 'vue'

const props = defineProps({
 modelValue: {
 type: Boolean,
 default: false
 },
 title: {
 type: String,
 default: ''
 },
 width: {
 type: String,
 default: '500px'
 }
})

const emit = defineEmits(['update:modelValue', 'close', 'confirm'])

function close() {
 emit('update:modelValue', false)
 emit('close')
}

function confirm() {
 emit('confirm')
 close()
}

// 禁止背景滚动
watch(() => props.modelValue, (val) => {
 document.body.style.overflow = val ? 'hidden' : ''
})
</script>

<template>
 <Teleport to="body">
 <Transition name="modal">
 <div v-if="modelValue" class="modal-overlay" @click.self="close">
 <div class="modal-container" :style="{ width }">
 <!-- 头部插槽 -->
 <div class="modal-header">
 <slot name="header">
 <h3>{{ title }}</h3>
 </slot>
 <button class="close-btn" @click="close">&times;</button>
 </div>

 <!-- 内容插槽 -->
 <div class="modal-body">
 <slot />
 </div>

 <!-- 底部插槽 -->
 <div class="modal-footer">
 <slot name="footer">
 <button class="btn btn-secondary" @click="close">取消</button>
 <button class="btn btn-primary" @click="confirm">确定</button>
 </slot>
 </div>
 </div>
 </div>
 </Transition>

</template>
```

### 使用弹窗组件

```vue
<script setup>
import { ref } from 'vue'
import Modal from './Modal.vue'

const showModal = ref(false)
const showConfirm = ref(false)

function handleConfirm() {
 alert('确认操作')
}

function handleDelete() {
 alert('删除成功')
 showConfirm.value = false
}
</script>

<template>
 <button @click="showModal = true">打开弹窗</button>

 <!-- 基础弹窗 -->
 <Modal v-model="showModal" title="提示">
 <p>这是一个弹窗内容</p>
 </Modal>

 <!-- 自定义弹窗 -->
 <Modal v-model="showConfirm" title="确认删除">
 <p>确定要删除这条数据吗？</p>
 <template #footer>
 <button @click="showConfirm = false">取消</button>
 <button @click="handleDelete" style="background: red; color: white;">
 确认删除
 </button>
 </template>
 </Modal>
</template>
```

### 标签页组件

```vue
<!-- Tabs.vue -->
<script setup>
import { ref, provide, computed } from 'vue'

const props = defineProps({
 modelValue: String
})

const emit = defineEmits(['update:modelValue'])

const tabs = ref([])
const activeTab = computed({
 get: () => props.modelValue,
 set: (val) => emit('update:modelValue', val)
})

function registerTab(tab) {
 tabs.value.push(tab)
 if (!activeTab.value) {
 activeTab.value = tab.name
 }
}

provide('tabs', {
 registerTab,
 activeTab
})
</script>

<template>
 <div class="tabs">
 <div class="tabs-header">
 <slot name="header">
 <button
 v-for="tab in tabs"
 :key="tab.name"
 :class="['tab-btn', { active: activeTab === tab.name }]"
 @click="activeTab = tab.name"
 >
 {{ tab.label }}
 </button>
 </slot>
 </div>
 <div class="tabs-content">
 <slot />
 </div>
 </div>
</template>

<style scoped>
.tabs-header {
 display: flex;
 border-bottom: 1px solid #ddd;
}

.tab-btn {
 padding: 12px 20px;
 background: none;
 border: none;
 cursor: pointer;
 border-bottom: 2px solid transparent;
 transition: all 0.3s;
}

.tab-btn:hover {
 background: #f5f5f5;
}

.tab-btn.active {
 color: #42b883;
 border-bottom-color: #42b883;
}

.tabs-content {
 padding: 20px;
}
</style>
```

```vue
<!-- Tab.vue -->
<script setup>
import { inject, onMounted, computed } from 'vue'

const props = defineProps({
 name: {
 type: String,
 required: true
 },
 label: {
 type: String,
 required: true
 }
})

const { registerTab, activeTab } = inject('tabs')

onMounted(() => {
 registerTab({
 name: props.name,
 label: props.label
 })
})

const isActive = computed(() => activeTab.value === props.name)
</script>

<template>
 <div v-show="isActive" class="tab-panel">
 <slot />
 </div>
</template>
```

## 总结

| 类型 | 语法 | 说明 |
|------|------|------|
| 默认插槽 | `<slot />` | 单个内容分发点 |
| 具名插槽 | `<slot name="xxx" />` | 多个内容分发点 |
| 作用域插槽 | `<slot :data="xxx" />` | 子组件向父组件传递数据 |

**最佳实践**：
- 使用具名插槽组织复杂布局
- 作用域插槽用于列表、表格等需要自定义渲染的组件
- 提供合理的默认内容提升组件易用性
