# 1.3 Hello World

让我们创建第一个 Vue.js 应用，体验 Vue 的响应式特性。

## 最简单的 Vue 应用

创建一个 `index.html` 文件：

```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Vue Hello World</title>
</head>
<body>
  <div id="app">
    <h1>{{ message }}</h1>
    <button @click="count++">点击次数: {{ count }}</button>
  </div>

  <script type="module">
    import { createApp, ref } from 'https://unpkg.com/vue@3/dist/vue.esm-browser.js'
    
    createApp({
      setup() {
        const message = ref('Hello Vue!')
        const count = ref(0)
        return { message, count }
      }
    }).mount('#app')
  </script>
</body>
</html>
```

直接在浏览器中打开这个 HTML 文件，你就能看到第一个 Vue 应用了！

## 使用 Vite 项目

如果你已经创建了 Vue 项目，让我们看看项目中的文件：

### main.js - 应用入口

```javascript
import { createApp } from 'vue'
import App from './App.vue'

// 创建 Vue 应用实例并挂载
createApp(App).mount('#app')
```

### App.vue - 根组件

```vue
<script setup>
import { ref } from 'vue'

// 响应式数据
const message = ref('Hello Vue!')
const count = ref(0)

// 方法
function increment() {
  count.value++
}
</script>

<template>
  <div class="app">
    <h1>{{ message }}</h1>
    <p>计数器: {{ count }}</p>
    <button @click="increment">点击 +1</button>
  </div>
</template>

<style scoped>
.app {
  text-align: center;
  padding: 20px;
}

h1 {
  color: #42b883;
}

button {
  padding: 10px 20px;
  font-size: 16px;
  cursor: pointer;
}
</style>
```

## Vue 应用的核心概念

### 1. 响应式数据

```vue
<script setup>
import { ref, reactive } from 'vue'

// ref 用于基本类型
const count = ref(0)

// reactive 用于对象
const user = reactive({
  name: '张三',
  age: 25
})
</script>

<template>
  <p>计数: {{ count }}</p>
  <p>用户: {{ user.name }}, {{ user.age }}岁</p>
</template>
```

### 2. 模板语法

```vue
<template>
  <!-- 文本插值 -->
  <p>{{ message }}</p>
  
  <!-- 属性绑定 -->
  <img :src="imageUrl" :alt="imageAlt">
  
  <!-- 事件绑定 -->
  <button @click="handleClick">点击</button>
  
  <!-- 双向绑定 -->
  <input v-model="inputText">
  
  <!-- 条件渲染 -->
  <p v-if="showMessage">显示这段文字</p>
  
  <!-- 列表渲染 -->
  <ul>
    <li v-for="item in items" :key="item.id">
      {{ item.name }}
    </li>
  </ul>
</template>
```

### 3. 计算属性

```vue
<script setup>
import { ref, computed } from 'vue'

const firstName = ref('张')
const lastName = ref('三')

// 计算属性
const fullName = computed(() => {
  return firstName.value + lastName.value
})
</script>

<template>
  <p>全名: {{ fullName }}</p>
</template>
```

### 4. 方法

```vue
<script setup>
import { ref } from 'vue'

const count = ref(0)

function increment() {
  count.value++
}

function decrement() {
  count.value--
}
</script>

<template>
  <p>计数: {{ count }}</p>
  <button @click="increment">+1</button>
  <button @click="decrement">-1</button>
</template>
```

## 完整示例：待办事项

```vue
<script setup>
import { ref } from 'vue'

const newTodo = ref('')
const todos = ref([])

function addTodo() {
  if (newTodo.value.trim()) {
    todos.value.push({
      id: Date.now(),
      text: newTodo.value,
      done: false
    })
    newTodo.value = ''
  }
}

function removeTodo(id) {
  todos.value = todos.value.filter(todo => todo.id !== id)
}

function toggleTodo(todo) {
  todo.done = !todo.done
}
</script>

<template>
  <div class="todo-app">
    <h1>待办事项</h1>
    
    <div class="input-group">
      <input 
        v-model="newTodo" 
        @keyup.enter="addTodo"
        placeholder="输入待办事项"
      >
      <button @click="addTodo">添加</button>
    </div>
    
    <ul class="todo-list">
      <li 
        v-for="todo in todos" 
        :key="todo.id"
        :class="{ done: todo.done }"
      >
        <input 
          type="checkbox" 
          :checked="todo.done"
          @change="toggleTodo(todo)"
        >
        <span>{{ todo.text }}</span>
        <button @click="removeTodo(todo.id)">删除</button>
      </li>
    </ul>
    
    <p v-if="todos.length === 0">暂无待办事项</p>
  </div>
</template>

<style scoped>
.todo-app {
  max-width: 400px;
  margin: 0 auto;
  padding: 20px;
}

.input-group {
  display: flex;
  gap: 10px;
  margin-bottom: 20px;
}

input {
  flex: 1;
  padding: 8px;
}

button {
  padding: 8px 16px;
  cursor: pointer;
}

.todo-list {
  list-style: none;
  padding: 0;
}

.todo-list li {
  display: flex;
  align-items: center;
  gap: 10px;
  padding: 10px;
  border-bottom: 1px solid #eee;
}

.todo-list li.done span {
  text-decoration: line-through;
  color: #999;
}
</style>
```
