# 3.2 Props 与 Emits

## 📤 Props - 父传子

Props 用于父组件向子组件传递数据。

### 基础用法

```vue
<!-- ChildComponent.vue -->
<script setup>
// 定义 props
defineProps({
  title: String,
  count: Number
})
</script>

<template>
  <h2>{{ title }}</h2>
  <p>计数: {{ count }}</p>
</template>
```

```vue
<!-- ParentComponent.vue -->
<script setup>
import ChildComponent from './ChildComponent.vue'
</script>

<template>
  <ChildComponent title="Hello Vue" :count="10" />
</template>
```

### Props 类型定义

```vue
<script setup>
defineProps({
  // 基础类型检查
  name: String,
  age: Number,
  isActive: Boolean,
  
  // 多个可能的类型
  id: [String, Number],
  
  // 必填字段
  title: {
    type: String,
    required: true
  },
  
  // 带默认值
  count: {
    type: Number,
    default: 0
  },
  
  // 对象/数组默认值需要使用工厂函数
  user: {
    type: Object,
    default: () => ({
      name: '默认用户',
      age: 0
    })
  },
  
  // 自定义验证
  status: {
    type: String,
    validator: (value) => ['active', 'inactive', 'pending'].includes(value)
  }
})
</script>
```

### Props 类型列表

| 类型 | 说明 |
|------|------|
| `String` | 字符串 |
| `Number` | 数字 |
| `Boolean` | 布尔值 |
| `Array` | 数组 |
| `Object` | 对象 |
| `Date` | 日期 |
| `Function` | 函数 |
| `Symbol` | Symbol |

### 使用 TypeScript 定义 Props

```vue
<script setup lang="ts">
interface Props {
  title: string
  count?: number
  user: {
    name: string
    age: number
  }
}

const props = defineProps<Props>()
</script>
```

### Props 的单向数据流

```vue
<script setup>
const props = defineProps({
  count: Number
})

// ❌ 不要直接修改 props
// props.count++

// ✅ 使用本地数据
import { ref, computed } from 'vue'

const localCount = ref(props.count)

// ✅ 使用计算属性
const doubledCount = computed(() => props.count * 2)
</script>
```

## 📥 Emits - 子传父

Emits 用于子组件向父组件发送事件。

### 基础用法

```vue
<!-- ChildComponent.vue -->
<script setup>
// 定义 emits
const emit = defineEmits(['update', 'delete'])

function handleUpdate() {
  emit('update', { id: 1, name: '更新数据' })
}

function handleDelete(id) {
  emit('delete', id)
}
</script>

<template>
  <button @click="handleUpdate">更新</button>
  <button @click="handleDelete(1)">删除</button>
</template>
```

```vue
<!-- ParentComponent.vue -->
<script setup>
import ChildComponent from './ChildComponent.vue'

function onUpdate(data) {
  console.log('更新:', data)
}

function onDelete(id) {
  console.log('删除:', id)
}
</script>

<template>
  <ChildComponent 
    @update="onUpdate" 
    @delete="onDelete" 
  />
</template>
```

### Emits 验证

```vue
<script setup>
const emit = defineEmits({
  // 无验证
  click: null,
  
  // 带验证
  submit: (payload) => {
    if (payload.email && payload.password) {
      return true
    }
    console.warn('无效的提交数据')
    return false
  },
  
  // 验证参数类型
  update: (value) => typeof value === 'number'
})

function handleSubmit() {
  emit('submit', {
    email: 'test@example.com',
    password: '123456'
  })
}
</script>
```

### 使用 TypeScript 定义 Emits

```vue
<script setup lang="ts">
interface Emits {
  (e: 'update', value: number): void
  (e: 'delete', id: number): void
  (e: 'submit', payload: { email: string; password: string }): void
}

const emit = defineEmits<Emits>()
</script>
```

## 🔄 v-model 实现

### 单个 v-model

```vue
<!-- CustomInput.vue -->
<script setup>
const props = defineProps({
  modelValue: String
})

const emit = defineEmits(['update:modelValue'])

function onInput(event) {
  emit('update:modelValue', event.target.value)
}
</script>

<template>
  <input
    :value="modelValue"
    @input="onInput"
  />
</template>
```

```vue
<!-- 使用 -->
<script setup>
import { ref } from 'vue'
import CustomInput from './CustomInput.vue'

const text = ref('')
</script>

<template>
  <CustomInput v-model="text" />
  <p>输入内容: {{ text }}</p>
</template>
```

### 使用 computed 简化

```vue
<!-- CustomInput.vue -->
<script setup>
import { computed } from 'vue'

const props = defineProps({
  modelValue: String
})

const emit = defineEmits(['update:modelValue'])

const value = computed({
  get: () => props.modelValue,
  set: (val) => emit('update:modelValue', val)
})
</script>

<template>
  <input v-model="value" />
</template>
```

### 多个 v-model

```vue
<!-- UserForm.vue -->
<script setup>
const props = defineProps({
  firstName: String,
  lastName: String
})

const emit = defineEmits(['update:firstName', 'update:lastName'])
</script>

<template>
  <input
    :value="firstName"
    @input="$emit('update:firstName', $event.target.value)"
  />
  <input
    :value="lastName"
    @input="$emit('update:lastName', $event.target.value)"
  />
</template>
```

```vue
<!-- 使用 -->
<script setup>
import { ref } from 'vue'
import UserForm from './UserForm.vue'

const user = ref({
  firstName: '张',
  lastName: '三'
})
</script>

<template>
  <UserForm 
    v-model:first-name="user.firstName"
    v-model:last-name="user.lastName"
  />
</template>
```

## 🎨 实际应用示例

### 可复用表格组件

```vue
<!-- DataTable.vue -->
<script setup>
defineProps({
  columns: {
    type: Array,
    required: true
  },
  data: {
    type: Array,
    required: true
  },
  loading: {
    type: Boolean,
    default: false
  }
})

const emit = defineEmits(['row-click', 'sort'])

function handleRowClick(row, index) {
  emit('row-click', { row, index })
}

function handleSort(column) {
  if (column.sortable) {
    emit('sort', column.key)
  }
}
</script>

<template>
  <div class="data-table">
    <table>
      <thead>
        <tr>
          <th
            v-for="col in columns"
            :key="col.key"
            :class="{ sortable: col.sortable }"
            @click="handleSort(col)"
          >
            {{ col.title }}
            <span v-if="col.sortable">↕</span>
          </th>
        </tr>
      </thead>
      <tbody>
        <tr v-if="loading">
          <td :colspan="columns.length" class="loading">
            加载中...
          </td>
        </tr>
        <tr v-else v-for="(row, index) in data" :key="row.id" @click="handleRowClick(row, index)">
          <td v-for="col in columns" :key="col.key">
            <slot :name="col.key" :row="row" :value="row[col.key]">
              {{ row[col.key] }}
            </slot>
          </td>
        </tr>
      </tbody>
    </table>
  </div>
</template>

<style scoped>
.data-table table {
  width: 100%;
  border-collapse: collapse;
}

th, td {
  padding: 12px;
  border: 1px solid #eee;
  text-align: left;
}

th.sortable {
  cursor: pointer;
}

th.sortable:hover {
  background: #f5f5f5;
}

tbody tr:hover {
  background: #fafafa;
  cursor: pointer;
}

.loading {
  text-align: center;
  color: #999;
}
</style>
```

### 使用表格组件

```vue
<script setup>
import { ref } from 'vue'
import DataTable from './DataTable.vue'

const columns = [
  { key: 'id', title: 'ID', sortable: true },

  { id: 2, name: '李四', email: 'li@example.com', status: 'inactive' }

const loading = ref(false)

  console.log('点击行:', row, index)

  console.log('排序:', key)

      <span :class="['status', value]">

.status.active {
```

### 计数器组件

```vue
<!-- Counter.vue -->
<script setup>
const props = defineProps({
  modelValue: {
    type: Number,
    default: 0
  },
  min: {
    type: Number,
    default: 0
  },
  max: {
    type: Number,
    default: 100
  },
  step: {
    type: Number,
    default: 1
  }
})

const emit = defineEmits(['update:modelValue', 'change'])

function increment() {
  const newValue = Math.min(props.modelValue + props.step, props.max)
  emit('update:modelValue', newValue)
  emit('change', newValue)
}

function decrement() {
  const newValue = Math.max(props.modelValue - props.step, props.min)
  emit('update:modelValue', newValue)
  emit('change', newValue)
}
</script>

<template>
  <div class="counter">
    <button 
      @click="decrement" 
      :disabled="modelValue <= min"
    >
      -
    </button>
    <span class="value">{{ modelValue }}</span>
    <button 
      @click="increment" 
      :disabled="modelValue >= max"
    >
      +
    </button>
  </div>
</template>

<style scoped>
.counter {
  display: inline-flex;
  align-items: center;
  gap: 10px;
}

button {
  width: 32px;
  height: 32px;
  border: 1px solid #ddd;
  background: white;
  cursor: pointer;
  border-radius: 4px;
}

button:hover:not(:disabled) {
  background: #f5f5f5;
}

button:disabled {
  opacity: 0.5;
  cursor: not-allowed;
}

.value {
  min-width: 40px;
  text-align: center;
  font-size: 18px;
}
</style>
```

## 📝 总结

| 概念 | 说明 |
|------|------|
| `defineProps` | 定义组件接收的属性 |
| `defineEmits` | 定义组件发出的事件 |
| `v-model` | 双向数据绑定语法糖 |
| 单向数据流 | Props 只能父传子，不能子修改父 |

**最佳实践**：
- Props 使用对象语法定义类型和默认值
- 复杂对象使用 TypeScript 接口定义
- Emits 使用验证函数确保数据正确
- 不要直接修改 Props，使用本地数据或计算属性
