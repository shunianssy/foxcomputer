# 3.4 DOM 操作

DOM（Document Object Model）是网页的编程接口，JavaScript 通过 DOM 操作网页内容。

## 什么是 DOM？

DOM 将 HTML 文档表示为一个树形结构：

```
Document
└── html
 ├── head
 │ ├── title
 │ └── meta
 └── body
 ├── header
 │ └── nav
 ├── main
 │ ├── h1
 │ └── p
 └── footer
```

## 获取元素

### 通过 ID 获取

```javascript
// 获取单个元素
const element = document.getElementById('header');
console.log(element);
```

### 通过选择器获取

```javascript
// 获取第一个匹配的元素
const firstItem = document.querySelector('.item');
const header = document.querySelector('#header');
const links = document.querySelector('nav a');

// 获取所有匹配的元素
const allItems = document.querySelectorAll('.item');
const allLinks = document.querySelectorAll('a');

// 遍历 NodeList
allItems.forEach(item => {
 console.log(item);
});
```

### 通过标签名/类名获取

```javascript
// 通过标签名
const paragraphs = document.getElementsByTagName('p');

// 通过类名
const items = document.getElementsByClassName('item');

// 注意：返回的是 HTMLCollection，不是数组
// 需要转换为数组才能使用 forEach
Array.from(items).forEach(item => {
 console.log(item);
});
```

### 获取元素关系

```javascript
const element = document.querySelector('.item');

// 父元素
const parent = element.parentElement;

// 子元素
const children = element.children;
const firstChild = element.firstElementChild;
const lastChild = element.lastElementChild;

// 兄弟元素
const nextSibling = element.nextElementSibling;
const prevSibling = element.previousElementSibling;
```

## 操作元素内容

### 获取/设置文本

```javascript
const element = document.getElementById('title');

// 获取文本（不包含 HTML 标签）
const text = element.textContent;
console.log(text);

// 设置文本
element.textContent = '新标题';

// 获取文本（会触发重排）
const text2 = element.innerText;

// 设置文本（会触发重排）
element.innerText = '新标题';
```

### 获取/设置 HTML

```javascript
const element = document.getElementById('content');

// 获取 HTML
const html = element.innerHTML;
console.log(html);

// 设置 HTML
element.innerHTML = '<strong>加粗文字</strong>';

// 安全设置（防止 XSS）
element.textContent = userInput; // 推荐
```

### 获取/设置值

```javascript
const input = document.querySelector('input');
const textarea = document.querySelector('textarea');
const select = document.querySelector('select');

// 获取值
const value = input.value;

// 设置值
input.value = '默认值';

// 获取选中状态
const checkbox = document.querySelector('input[type="checkbox"]');
const isChecked = checkbox.checked;
```

## 操作元素属性

### 获取/设置属性

```javascript
const link = document.querySelector('a');

// 获取属性
const href = link.getAttribute('href');
const target = link.getAttribute('target');

// 设置属性
link.setAttribute('href', 'https://example.com');
link.setAttribute('target', '_blank');

// 移除属性
link.removeAttribute('target');

// 检查属性是否存在
const hasTarget = link.hasAttribute('target');
```

### data 属性

```html
<div id="user" data-id="123" data-name="张三" data-user-age="25">
 用户信息
</div>
```

```javascript
const user = document.getElementById('user');

// 获取 data 属性
console.log(user.dataset.id); // '123'
console.log(user.dataset.name); // '张三'
console.log(user.dataset.userAge); // '25'（驼峰命名）

// 设置 data 属性
user.dataset.role = 'admin';
// HTML: data-role="admin"
```

## 操作元素样式

### style 属性

```javascript
const element = document.getElementById('box');

// 设置单个样式
element.style.backgroundColor = 'red';
element.style.fontSize = '16px';
element.style.marginTop = '10px';

// 注意：使用驼峰命名法
// CSS: background-color → JS: backgroundColor
// CSS: font-size → JS: fontSize

// 获取计算样式
const computedStyle = getComputedStyle(element);
console.log(computedStyle.backgroundColor);
```

### classList 操作

```javascript
const element = document.getElementById('box');

// 添加类
element.classList.add('active');
element.classList.add('highlight', 'featured');

// 移除类
element.classList.remove('active');
element.classList.remove('highlight', 'featured');

// 切换类
element.classList.toggle('active');
element.classList.toggle('active', true); // 强制添加
element.classList.toggle('active', false); // 强制移除

// 检查是否包含类
const hasClass = element.classList.contains('active');

// 替换类
element.classList.replace('old-class', 'new-class');
```

### className 操作

```javascript
const element = document.getElementById('box');

// 获取所有类名
console.log(element.className); // 'box active highlight'

// 设置类名（覆盖所有）
element.className = 'new-class';

// 添加类名
element.className += ' another-class';
```

## 创建和删除元素

### 创建元素

```javascript
// 创建元素
const div = document.createElement('div');
div.id = 'new-box';
div.className = 'box';
div.textContent = '新创建的元素';

// 创建文本节点
const text = document.createTextNode('文本内容');

// 创建文档片段（性能优化）
const fragment = document.createDocumentFragment();
for (let i = 0; i < 5; i++) {
 const li = document.createElement('li');
 li.textContent = `项目 ${i + 1}`;
 fragment.appendChild(li);
}
```

### 插入元素

```javascript
const parent = document.getElementById('list');
const newElement = document.createElement('li');
newElement.textContent = '新项目';

// 追加到末尾
parent.appendChild(newElement);

// 插入到指定位置
const referenceElement = parent.children[0];
parent.insertBefore(newElement, referenceElement);

// 插入到末尾（现代方法）
parent.append(newElement);

// 插入到开头
parent.prepend(newElement);

// 在元素后面插入
referenceElement.after(newElement);

// 在元素前面插入
referenceElement.before(newElement);
```

### 删除元素

```javascript
const element = document.getElementById('old-box');

// 删除元素（现代方法）
element.remove();

// 通过父元素删除
const parent = element.parentElement;
parent.removeChild(element);

// 清空元素内容
element.innerHTML = '';
element.textContent = '';
```

### 克隆元素

```javascript
const original = document.getElementById('box');

// 浅克隆（只克隆元素本身）
const clone = original.cloneNode();

// 深克隆（克隆元素及其内容）
const deepClone = original.cloneNode(true);
```

## 表单操作

### 获取表单数据

```html
<form id="myForm">
 <input type="text" name="username" value="张三">
 <input type="email" name="email" value="test@example.com">
 <input type="checkbox" name="agree" checked>
 <select name="city">
 <option value="beijing">北京</option>
 <option value="shanghai" selected>上海</option>
 </select>
</form>
```

```javascript
const form = document.getElementById('myForm');

// 获取表单元素
const username = form.elements.username;
const email = form.elements['email'];

// 获取值
console.log(username.value); // '张三'
console.log(email.value); // 'test@example.com'

// 获取选中状态
const agree = form.elements.agree;
console.log(agree.checked); // true

// 获取下拉框选中值
const city = form.elements.city;
console.log(city.value); // 'shanghai'

// 获取表单数据
const formData = new FormData(form);
console.log(formData.get('username')); // '张三'
```

### 表单验证

```javascript
const form = document.getElementById('myForm');

form.addEventListener('submit', function(e) {
 e.preventDefault(); // 阻止表单提交
 
 const username = form.elements.username.value.trim();
 const email = form.elements.email.value.trim();
 
 // 验证用户名
 if (username.length < 3) {
 alert('用户名至少3个字符');
 return;
 }
 
 // 验证邮箱
 const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
 if (!emailRegex.test(email)) {
 alert('请输入有效的邮箱地址');
 return;
 }
 
 // 验证通过，提交表单
 console.log('表单验证通过');
 form.submit();
});
```

## 实战示例

### 动态列表

```html
<div id="app">
 <input type="text" id="input" placeholder="输入内容">
 <button id="addBtn">添加</button>
 <ul id="list"></ul>
</div>
```

```javascript
const input = document.getElementById('input');
const addBtn = document.getElementById('addBtn');
const list = document.getElementById('list');

// 添加项目
function addItem(text) {
 const li = document.createElement('li');
 li.innerHTML = `
 <span>${text}</span>
 <button class="delete-btn">删除</button>
 `;
 
 // 删除按钮事件
 const deleteBtn = li.querySelector('.delete-btn');
 deleteBtn.addEventListener('click', function() {
 li.remove();
 });
 
 list.appendChild(li);
}

// 添加按钮事件
addBtn.addEventListener('click', function() {
 const text = input.value.trim();
 if (text) {
 addItem(text);
 input.value = '';
 input.focus();
 }
});

// 回车添加
input.addEventListener('keypress', function(e) {
 if (e.key === 'Enter') {
 addBtn.click();
 }
});
```

### 标签页切换

```html
<div class="tabs">
 <div class="tab-buttons">
 <button class="tab-btn active" data-tab="tab1">标签1</button>
 <button class="tab-btn" data-tab="tab2">标签2</button>
 <button class="tab-btn" data-tab="tab3">标签3</button>
 </div>
 <div class="tab-content">
 <div id="tab1" class="tab-panel active">内容1</div>
 <div id="tab2" class="tab-panel">内容2</div>
 <div id="tab3" class="tab-panel">内容3</div>
 </div>
</div>
```

```javascript
const tabBtns = document.querySelectorAll('.tab-btn');
const tabPanels = document.querySelectorAll('.tab-panel');

tabBtns.forEach(btn => {
 btn.addEventListener('click', function() {
 // 移除所有活动状态
 tabBtns.forEach(b => b.classList.remove('active'));
 tabPanels.forEach(p => p.classList.remove('active'));
 
 // 添加当前活动状态
 this.classList.add('active');
 const tabId = this.dataset.tab;
 document.getElementById(tabId).classList.add('active');
 });
});
```
