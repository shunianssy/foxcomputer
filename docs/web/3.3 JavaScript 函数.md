# 3.3 JavaScript 函数

函数是 JavaScript 中最重要的概念之一，它是一段可重复使用的代码块。

## 函数定义

### 函数声明

```javascript
// 函数声明
function greet() {
 console.log('你好！');
}

// 调用函数
greet(); // 你好！
```

### 函数表达式

```javascript
// 函数表达式
const greet = function() {
 console.log('你好！');
};

// 调用函数
greet();
```

### 箭头函数（ES6）

```javascript
// 箭头函数
const greet = () => {
 console.log('你好！');
};

// 简写（单行可省略大括号）
const greet = () => console.log('你好！');

// 单参数可省略括号
const double = x => x * 2;

// 多参数需要括号
const add = (a, b) => a + b;

// 多行需要大括号和 return
const calculate = (a, b) => {
 const sum = a + b;
 return sum * 2;
};
```

### 函数对比

| 类型 | 提升 | this | arguments | 适用场景 |
|------|------|------|-----------|----------|
| 函数声明 | 是 | 动态 | 有 | 通用 |
| 函数表达式 | 否 | 动态 | 有 | 回调、闭包 |
| 箭头函数 | 否 | 词法 | 无 | 简洁回调 |

## 参数

### 基本参数

```javascript
function greet(name) {
 console.log('你好，' + name);
}

greet('张三'); // 你好，张三
greet(); // 你好，undefined
```

### 默认参数

```javascript
// ES6 默认参数
function greet(name = '游客') {
 console.log('你好，' + name);
}

greet('张三'); // 你好，张三
greet(); // 你好，游客
```

### 剩余参数

```javascript
// 剩余参数（...args）
function sum(...numbers) {
 return numbers.reduce((total, num) => total + num, 0);
}

console.log(sum(1, 2, 3)); // 6
console.log(sum(1, 2, 3, 4, 5)); // 15
```

### 参数解构

```javascript
// 对象解构
function showUser({ name, age, city = '北京' }) {
 console.log(`${name}, ${age}岁, 来自${city}`);
}

showUser({ name: '张三', age: 25 });
// 张三, 25岁, 来自北京

// 数组解构
function showCoords([x, y]) {
 console.log(`坐标：(${x}, ${y})`);
}

showCoords([10, 20]); // 坐标：(10, 20)
```

## 返回值

### return 语句

```javascript
function add(a, b) {
 return a + b;
}

const result = add(3, 5);
console.log(result); // 8

// 没有返回值，返回 undefined
function noReturn() {
 console.log('没有返回值');
}

console.log(noReturn()); // undefined
```

### 提前返回

```javascript
function divide(a, b) {
 if (b === 0) {
 return '除数不能为0';
 }
 return a / b;
}

console.log(divide(10, 2)); // 5
console.log(divide(10, 0)); // '除数不能为0'
```

## 作用域

### 全局作用域

```javascript
// 全局变量
const globalVar = '全局变量';

function showGlobal() {
 console.log(globalVar); // 可以访问
}

showGlobal(); // 全局变量
console.log(globalVar); // 全局变量
```

### 函数作用域

```javascript
function myFunction() {
 // 函数作用域变量
 const localVar = '局部变量';
 console.log(localVar); // 可以访问
}

myFunction(); // 局部变量
// console.log(localVar); // 错误！无法访问
```

### 块级作用域

```javascript
if (true) {
 const blockVar = '块级变量';
 let blockLet = '块级 let';
 var blockVar2 = '块级 var';
 console.log(blockVar); // 可以访问
}

// console.log(blockVar); // 错误！无法访问
// console.log(blockLet); // 错误！无法访问
console.log(blockVar2); // 块级 var（var 没有块级作用域）
```

### 作用域链

```javascript
const global = '全局';

function outer() {
 const outerVar = '外部';
 
 function inner() {
 const innerVar = '内部';
 console.log(innerVar); // 内部
 console.log(outerVar); // 外部（向上一级查找）
 console.log(global); // 全局（向上一级查找）
 }
 
 inner();
}

outer();
```

## 闭包

闭包是指函数可以访问其创建时所在作用域的变量。

```javascript
function createCounter() {
 let count = 0;
 
 return function() {
 count++;
 return count;
 };
}

const counter = createCounter();
console.log(counter()); // 1
console.log(counter()); // 2
console.log(counter()); // 3
```

### 闭包应用

**1. 私有变量：**

```javascript
function createWallet(initialBalance) {
 let balance = initialBalance;
 
 return {
 getBalance: function() {
 return balance;
 },
 deposit: function(amount) {
 balance += amount;
 return balance;
 },
 withdraw: function(amount) {
 if (amount > balance) {
 return '余额不足';
 }
 balance -= amount;
 return balance;
 }
 };
}

const wallet = createWallet(100);
console.log(wallet.getBalance()); // 100
wallet.deposit(50);
console.log(wallet.getBalance()); // 150
// console.log(balance); // 错误！无法直接访问
```

**2. 函数工厂：**

```javascript
function createMultiplier(multiplier) {
 return function(number) {
 return number * multiplier;
 };
}

const double = createMultiplier(2);
const triple = createMultiplier(3);

console.log(double(5)); // 10
console.log(triple(5)); // 15
```

## 高阶函数

高阶函数是指接收函数作为参数或返回函数的函数。

### 函数作为参数

```javascript
// 数组方法
const numbers = [1, 2, 3, 4, 5];

// forEach
numbers.forEach(function(num) {
 console.log(num * 2);
});

// map
const doubled = numbers.map(function(num) {
 return num * 2;
});

// filter
const evens = numbers.filter(function(num) {
 return num % 2 === 0;
});

// 简写
const doubled = numbers.map(num => num * 2);
const evens = numbers.filter(num => num % 2 === 0);
```

### 自定义高阶函数

```javascript
// 创建一个高阶函数
function repeat(n, action) {
 for (let i = 0; i < n; i++) {
 action(i);
 }
}

repeat(3, function(i) {
 console.log(`执行第 ${i + 1} 次`);
});
// 执行第 1 次
// 执行第 2 次
// 执行第 3 次
```

## this 关键字

### this 的指向

```javascript
// 1. 全局作用域
console.log(this); // window（浏览器）

// 2. 普通函数
function showThis() {
 console.log(this);
}
showThis(); // window（严格模式下是 undefined）

// 3. 对象方法
const person = {
 name: '张三',
 greet: function() {
 console.log(this.name); // 张三
 }
};
person.greet();

// 4. 箭头函数（没有自己的 this）
const person2 = {
 name: '李四',
 greet: function() {
 const arrow = () => console.log(this.name);
 arrow(); // 李四（继承外层 this）
 }
};
person2.greet();
```

### 改变 this 指向

```javascript
const person = { name: '张三' };

function greet() {
 console.log(this.name);
}

// call - 立即调用
greet.call(person); // 张三

// apply - 立即调用（参数为数组）
function introduce(age, city) {
 console.log(`${this.name}, ${age}岁, 来自${city}`);
}
introduce.apply(person, [25, '北京']); // 张三, 25岁, 来自北京

// bind - 返回新函数
const boundGreet = greet.bind(person);
boundGreet(); // 张三
```

## 立即执行函数（IIFE）

```javascript
// 立即执行函数表达式
(function() {
 const privateVar = '私有变量';
 console.log('立即执行！');
})();

// 箭头函数形式
(() => {
 console.log('箭头函数 IIFE');
})();

// 带参数
((name) => {
 console.log('你好，' + name);
})('张三');
```
