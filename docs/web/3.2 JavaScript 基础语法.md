# 3.2 JavaScript 基础语法

本章介绍 JavaScript 的基础语法，包括变量、数据类型、运算符和流程控制。

## 变量

变量是存储数据的容器。

### 声明变量

```javascript
// let - 可变变量（推荐）
let name = '张三';
name = '李四';  // 可以重新赋值

// const - 常量（推荐）
const PI = 3.14159;
// PI = 3.14;  // 错误！不能重新赋值

// var - 旧语法（不推荐）
var age = 25;
age = 26;  // 可以重新赋值
```

### let vs const vs var

| 特性 | let | const | var |
|------|-----|-------|-----|
| 作用域 | 块级 | 块级 | 函数级 |
| 重复声明 | ❌ 不允许 | ❌ 不允许 | ✅ 允许 |
| 重新赋值 | ✅ 允许 | ❌ 不允许 | ✅ 允许 |
| 变量提升 | ❌ 不提升 | ❌ 不提升 | ✅ 提升 |
| 暂时性死区 | ✅ 有 | ✅ 有 | ❌ 没有 |

### 命名规范

```javascript
// 推荐：驼峰命名法
let userName = '张三';
let userAge = 25;
let isLoggedIn = true;

// 常量：全大写，下划线分隔
const MAX_SIZE = 100;
const API_URL = 'https://api.example.com';

// 私有变量：下划线开头
let _privateVar = '私有';

// 布尔值：is/has/can 开头
let isVisible = true;
let hasPermission = false;
let canEdit = true;
```

## 数据类型

JavaScript 有 7 种基本数据类型和 1 种复杂数据类型。

### 基本数据类型

```javascript
// 1. Number - 数字
let integer = 42;           // 整数
let float = 3.14;           // 浮点数
let negative = -10;         // 负数
let infinity = Infinity;    // 无穷大
let notANumber = NaN;       // 非数字

// 2. String - 字符串
let single = '单引号';
let double = "双引号";
let template = `模板字符串，可以包含变量：${integer}`;

// 3. Boolean - 布尔值
let isTrue = true;
let isFalse = false;

// 4. undefined - 未定义
let notDefined;             // undefined
let explicitUndefined = undefined;

// 5. null - 空值
let empty = null;

// 6. Symbol - 符号（ES6）
let sym = Symbol('description');
let sym2 = Symbol('description');
console.log(sym === sym2);  // false（唯一）

// 7. BigInt - 大整数（ES2020）
let bigNumber = 9007199254740991n;
let bigNumber2 = BigInt(9007199254740991);
```

### 复杂数据类型

```javascript
// Object - 对象
const person = {
    name: '张三',
    age: 25,
    isStudent: true,
    sayHello: function() {
        return '你好！';
    }
};

// Array - 数组（特殊的对象）
const numbers = [1, 2, 3, 4, 5];
const fruits = ['苹果', '香蕉', '橙子'];
const mixed = [1, 'hello', true, { name: '张三' }];

// Function - 函数（特殊的对象）
function greet(name) {
    return '你好，' + name;
}
```

### 类型检测

```javascript
// typeof 运算符
console.log(typeof 42);           // 'number'
console.log(typeof 'hello');      // 'string'
console.log(typeof true);         // 'boolean'
console.log(typeof undefined);    // 'undefined'
console.log(typeof null);         // 'object'（历史遗留问题）
console.log(typeof {});           // 'object'
console.log(typeof []);           // 'object'
console.log(typeof function(){}); // 'function'

// 检测数组
console.log(Array.isArray([1, 2, 3]));  // true

// 检测 null
console.log(value === null);  // 推荐方式

// instanceof 检测原型
console.log([] instanceof Array);  // true
console.log({} instanceof Object); // true
```

### 类型转换

```javascript
// 转字符串
String(123);        // '123'
(123).toString();   // '123'
123 + '';           // '123'

// 转数字
Number('123');      // 123
parseInt('123.45'); // 123
parseFloat('123.45'); // 123.45
+'123';             // 123

// 转布尔值
Boolean(1);         // true
Boolean(0);         // false
Boolean('');        // false
Boolean('hello');   // true
!!1;                // true（双重否定）
```

## 运算符

### 算术运算符

```javascript
let a = 10;
let b = 3;

console.log(a + b);  // 13 加法
console.log(a - b);  // 7  减法
console.log(a * b);  // 30 乘法
console.log(a / b);  // 3.333... 除法
console.log(a % b);  // 1  取余
console.log(a ** b); // 1000 幂运算

// 自增/自减
let x = 5;
console.log(x++);    // 5（先返回，后加）
console.log(x);      // 6
console.log(++x);    // 7（先加，后返回）

let y = 5;
console.log(y--);    // 5
console.log(--y);    // 3
```

### 赋值运算符

```javascript
let x = 10;

x += 5;   // x = x + 5  → 15
x -= 3;   // x = x - 3  → 12
x *= 2;   // x = x * 2  → 24
x /= 4;   // x = x / 4  → 6
x %= 4;   // x = x % 4  → 2
x **= 3;  // x = x ** 3 → 8
```

### 比较运算符

```javascript
// 相等比较
console.log(5 == '5');   // true（类型转换后比较）
console.log(5 === '5');  // false（严格比较，推荐）
console.log(5 != '5');   // false
console.log(5 !== '5');  // true（严格不等于，推荐）

// 大小比较
console.log(10 > 5);    // true
console.log(10 >= 10);  // true
console.log(5 < 10);    // true
console.log(5 <= 5);    // true

// 字符串比较（按 Unicode 编码）
console.log('a' < 'b');  // true
console.log('abc' < 'abd'); // true
```

### 逻辑运算符

```javascript
// 与（AND）
console.log(true && true);   // true
console.log(true && false);  // false

// 或（OR）
console.log(true || false);  // true
console.log(false || false); // false

// 非（NOT）
console.log(!true);   // false
console.log(!false);  // true

// 短路求值
let name = '张三';
console.log(name && '用户名存在');  // '用户名存在'
console.log(name || '匿名用户');    // '张三'

// 空值合并运算符（ES2020）
let value = null;
console.log(value ?? '默认值');  // '默认值'

// 可选链运算符（ES2020）
const user = { name: '张三' };
console.log(user?.address?.city);  // undefined（不会报错）
```

### 三元运算符

```javascript
// 条件 ? 真值 : 假值
let age = 18;
let status = age >= 18 ? '成年' : '未成年';
console.log(status);  // '成年'

// 嵌套使用（不推荐过度嵌套）
let score = 85;
let grade = score >= 90 ? 'A' :
            score >= 80 ? 'B' :
            score >= 70 ? 'C' : 'D';
console.log(grade);  // 'B'
```

## 流程控制

### 条件语句

**if 语句：**

```javascript
let age = 18;

if (age >= 18) {
    console.log('成年人');
}

if (age >= 18) {
    console.log('成年人');
} else {
    console.log('未成年人');
}

if (age < 12) {
    console.log('儿童');
} else if (age < 18) {
    console.log('青少年');
} else {
    console.log('成年人');
}
```

**switch 语句：**

```javascript
let day = 3;
let dayName;

switch (day) {
    case 1:
        dayName = '星期一';
        break;
    case 2:
        dayName = '星期二';
        break;
    case 3:
        dayName = '星期三';
        break;
    case 4:
        dayName = '星期四';
        break;
    case 5:
        dayName = '星期五';
        break;
    case 6:
    case 7:
        dayName = '周末';
        break;
    default:
        dayName = '无效日期';
}

console.log(dayName);  // '星期三'
```

### 循环语句

**for 循环：**

```javascript
// 基本形式
for (let i = 0; i < 5; i++) {
    console.log(i);  // 0, 1, 2, 3, 4
}

// 遍历数组
const fruits = ['苹果', '香蕉', '橙子'];
for (let i = 0; i < fruits.length; i++) {
    console.log(fruits[i]);
}

// for...of（遍历值）
for (const fruit of fruits) {
    console.log(fruit);
}

// for...in（遍历索引/键）
for (const index in fruits) {
    console.log(index, fruits[index]);
}
```

**while 循环：**

```javascript
// while
let i = 0;
while (i < 5) {
    console.log(i);
    i++;
}

// do...while（至少执行一次）
let j = 0;
do {
    console.log(j);
    j++;
} while (j < 5);
```

**循环控制：**

```javascript
// break - 跳出循环
for (let i = 0; i < 10; i++) {
    if (i === 5) break;
    console.log(i);  // 0, 1, 2, 3, 4
}

// continue - 跳过本次迭代
for (let i = 0; i < 5; i++) {
    if (i === 2) continue;
    console.log(i);  // 0, 1, 3, 4
}
```

### 数组方法遍历

```javascript
const numbers = [1, 2, 3, 4, 5];

// forEach - 遍历
numbers.forEach(function(num, index) {
    console.log(index, num);
});

// map - 映射（返回新数组）
const doubled = numbers.map(num => num * 2);
console.log(doubled);  // [2, 4, 6, 8, 10]

// filter - 过滤
const evens = numbers.filter(num => num % 2 === 0);
console.log(evens);  // [2, 4]

// find - 查找
const found = numbers.find(num => num > 3);
console.log(found);  // 4

// reduce - 归约
const sum = numbers.reduce((acc, num) => acc + num, 0);
console.log(sum);  // 15
```
