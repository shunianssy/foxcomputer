# 4.2 JavaScript 异步编程

JavaScript 是单线程语言，异步编程是处理耗时操作的关键技术。

## 同步与异步

### 同步代码

```javascript
console.log('开始');
console.log('执行中');
console.log('结束');
// 输出顺序：开始 -> 执行中 -> 结束
```

### 异步代码

```javascript
console.log('开始');
setTimeout(() => {
    console.log('异步操作');
}, 1000);
console.log('结束');
// 输出顺序：开始 -> 结束 -> 异步操作（1秒后）
```

## 回调函数

回调函数是最基本的异步处理方式：

```javascript
function fetchData(callback) {
    setTimeout(() => {
        const data = { name: '张三', age: 25 };
        callback(data);
    }, 1000);
}

fetchData(function(data) {
    console.log('获取到数据：', data);
});
```

### 回调地狱

多层嵌套的回调函数难以维护：

```javascript
getUserInfo(function(user) {
    getOrders(user.id, function(orders) {
        getOrderDetails(orders[0].id, function(details) {
            getProducts(details.products, function(products) {
                console.log(products);
            });
        });
    });
});
```

## Promise

Promise 是处理异步操作的现代方式。

### 创建 Promise

```javascript
const promise = new Promise((resolve, reject) => {
    setTimeout(() => {
        const success = true;
        if (success) {
            resolve('操作成功');
        } else {
            reject('操作失败');
        }
    }, 1000);
});
```

### 使用 Promise

```javascript
promise
    .then(result => {
        console.log('成功：', result);
    })
    .catch(error => {
        console.log('失败：', error);
    })
    .finally(() => {
        console.log('操作完成');
    });
```

### 链式调用

```javascript
fetch('/api/user')
    .then(response => response.json())
    .then(user => fetch(`/api/orders/${user.id}`))
    .then(response => response.json())
    .then(orders => console.log(orders))
    .catch(error => console.error('错误：', error));
```

### Promise 静态方法

```javascript
// Promise.resolve - 立即成功的 Promise
Promise.resolve('立即成功').then(console.log);

// Promise.reject - 立即失败的 Promise
Promise.reject('立即失败').catch(console.log);

// Promise.all - 所有 Promise 都成功
Promise.all([
    fetch('/api/user'),
    fetch('/api/products'),
    fetch('/api/orders')
])
.then(responses => {
    console.log('所有请求完成');
})
.catch(error => {
    console.log('有一个请求失败');
});

// Promise.allSettled - 等待所有 Promise 完成
Promise.allSettled([
    Promise.resolve('成功1'),
    Promise.reject('失败1'),
    Promise.resolve('成功2')
])
.then(results => {
    results.forEach(result => {
        if (result.status === 'fulfilled') {
            console.log('成功：', result.value);
        } else {
            console.log('失败：', result.reason);
        }
    });
});

// Promise.race - 返回最先完成的 Promise
Promise.race([
    fetch('/api/server1'),
    fetch('/api/server2')
])
.then(response => {
    console.log('最快的响应');
});

// Promise.any - 返回最先成功的 Promise
Promise.any([
    Promise.reject('失败1'),
    Promise.resolve('成功1'),
    Promise.resolve('成功2')
])
.then(result => {
    console.log('第一个成功：', result);
});
```

## async/await

async/await 是基于 Promise 的语法糖，让异步代码看起来像同步代码。

### async 函数

```javascript
async function getData() {
    return '数据';
}

// async 函数返回 Promise
getData().then(console.log);
```

### await 表达式

```javascript
async function fetchUser() {
    try {
        const response = await fetch('/api/user');
        const user = await response.json();
        console.log(user);
    } catch (error) {
        console.error('错误：', error);
    }
}
```

### 并行执行

```javascript
async function fetchAll() {
    // 并行执行
    const [users, products, orders] = await Promise.all([
        fetch('/api/users').then(r => r.json()),
        fetch('/api/products').then(r => r.json()),
        fetch('/api/orders').then(r => r.json())
    ]);
    
    console.log({ users, products, orders });
}
```

### 错误处理

```javascript
async function handleRequest() {
    try {
        const user = await fetchUser();
        const orders = await fetchOrders(user.id);
        return orders;
    } catch (error) {
        console.error('请求失败：', error);
        throw error;
    } finally {
        console.log('请求完成');
    }
}
```

## 定时器

### setTimeout

```javascript
// 延迟执行
const timerId = setTimeout(() => {
    console.log('1秒后执行');
}, 1000);

// 取消定时器
clearTimeout(timerId);
```

### setInterval

```javascript
// 定时重复执行
const intervalId = setInterval(() => {
    console.log('每秒执行一次');
}, 1000);

// 取消定时器
clearInterval(intervalId);
```

### 实际应用

```javascript
// 倒计时
function countdown(seconds) {
    return new Promise(resolve => {
        const timer = setInterval(() => {
            console.log(seconds);
            seconds--;
            if (seconds < 0) {
                clearInterval(timer);
                resolve('倒计时结束');
            }
        }, 1000);
    });
}

countdown(5).then(console.log);
```

## 事件循环

JavaScript 使用事件循环处理异步操作：

```javascript
console.log('1. 同步代码开始');

setTimeout(() => {
    console.log('2. 宏任务：setTimeout');
}, 0);

Promise.resolve().then(() => {
    console.log('3. 微任务：Promise');
});

console.log('4. 同步代码结束');

// 输出顺序：1 -> 4 -> 3 -> 2
```

### 宏任务与微任务

| 宏任务 | 微任务 |
|--------|--------|
| setTimeout | Promise.then |
| setInterval | Promise.catch |
| setImmediate (Node.js) | Promise.finally |
| I/O 操作 | process.nextTick (Node.js) |
| UI 渲染 | MutationObserver |

执行顺序：
1. 执行同步代码
2. 执行所有微任务
3. 执行一个宏任务
4. 重复步骤 2-3

## 实际应用示例

### 封装 AJAX 请求

```javascript
function request(url, options = {}) {
    return new Promise((resolve, reject) => {
        const xhr = new XMLHttpRequest();
        
        xhr.open(options.method || 'GET', url);
        
        xhr.onload = function() {
            if (xhr.status >= 200 && xhr.status < 300) {
                resolve(JSON.parse(xhr.responseText));
            } else {
                reject(new Error(xhr.statusText));
            }
        };
        
        xhr.onerror = function() {
            reject(new Error('网络错误'));
        };
        
        xhr.send(options.body ? JSON.stringify(options.body) : null);
    });
}

// 使用
request('/api/user')
    .then(user => console.log(user))
    .catch(error => console.error(error));
```

### 带超时的请求

```javascript
function fetchWithTimeout(url, timeout = 5000) {
    return Promise.race([
        fetch(url),
        new Promise((_, reject) => {
            setTimeout(() => {
                reject(new Error('请求超时'));
            }, timeout);
        })
    ]);
}

fetchWithTimeout('/api/user', 3000)
    .then(response => response.json())
    .then(data => console.log(data))
    .catch(error => console.error(error));
```

### 重试机制

```javascript
async function fetchWithRetry(url, retries = 3) {
    for (let i = 0; i < retries; i++) {
        try {
            const response = await fetch(url);
            if (!response.ok) {
                throw new Error(`HTTP ${response.status}`);
            }
            return await response.json();
        } catch (error) {
            console.log(`第 ${i + 1} 次尝试失败：`, error.message);
            if (i === retries - 1) {
                throw error;
            }
            await new Promise(resolve => setTimeout(resolve, 1000));
        }
    }
}
```

### 并发控制

```javascript
async function limitConcurrency(tasks, limit) {
    const results = [];
    const executing = [];
    
    for (const task of tasks) {
        const promise = Promise.resolve().then(() => task());
        results.push(promise);
        
        if (limit <= tasks.length) {
            const exec = promise.then(() => {
                executing.splice(executing.indexOf(exec), 1);
            });
            executing.push(exec);
            
            if (executing.length >= limit) {
                await Promise.race(executing);
            }
        }
    }
    
    return Promise.all(results);
}

// 同时最多执行 3 个任务
const tasks = urls.map(url => () => fetch(url));
const results = await limitConcurrency(tasks, 3);
```
