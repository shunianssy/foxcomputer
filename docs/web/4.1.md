# 4.1 JavaScript 事件

JavaScript 事件是网页交互的核心，允许我们对用户的操作做出响应。

## 事件概述

事件是用户或浏览器执行的某种动作，例如：
- 点击按钮
- 提交表单
- 页面加载完成
- 鼠标移动
- 键盘按键

## 事件监听

### 1. 内联事件处理（不推荐）

```html
<button onclick="alert('你点击了按钮！')">点击我</button>
```

### 2. DOM 属性方式

```html
<button id="myBtn">点击我</button>

<script>
const btn = document.getElementById('myBtn');
btn.onclick = function() {
    alert('你点击了按钮！');
};
</script>
```

### 3. addEventListener 方法（推荐）

```html
<button id="myBtn">点击我</button>

<script>
const btn = document.getElementById('myBtn');

// 添加事件监听器
btn.addEventListener('click', function() {
    alert('你点击了按钮！');
});

// 可以添加多个监听器
btn.addEventListener('click', function() {
    console.log('第二个监听器');
});
</script>
```

### 移除事件监听器

```javascript
function handleClick() {
    console.log('按钮被点击');
}

// 添加监听器
btn.addEventListener('click', handleClick);

// 移除监听器
btn.removeEventListener('click', handleClick);
```

## 常用事件类型

### 鼠标事件

| 事件 | 描述 |
|------|------|
| click | 单击 |
| dblclick | 双击 |
| mousedown | 鼠标按下 |
| mouseup | 鼠标释放 |
| mousemove | 鼠标移动 |
| mouseover | 鼠标移入 |
| mouseout | 鼠标移出 |
| contextmenu | 右键菜单 |

```html
<div id="box" style="width: 200px; height: 200px; background: #ccc;">
    鼠标操作区域
</div>

<script>
const box = document.getElementById('box');

box.addEventListener('mouseover', function() {
    this.style.background = '#f00';
});

box.addEventListener('mouseout', function() {
    this.style.background = '#ccc';
});
</script>
```

### 键盘事件

| 事件 | 描述 |
|------|------|
| keydown | 按键按下 |
| keyup | 按键释放 |
| keypress | 按键按下（已废弃） |

```html
<input id="input" placeholder="按下键盘试试">

<script>
const input = document.getElementById('input');

input.addEventListener('keydown', function(e) {
    console.log('按下键：', e.key);
    console.log('键码：', e.code);
});
</script>
```

### 表单事件

| 事件 | 描述 |
|------|------|
| submit | 表单提交 |
| change | 值改变（失焦时触发） |
| input | 值改变（实时触发） |
| focus | 获得焦点 |
| blur | 失去焦点 |

```html
<form id="myForm">
    <input type="text" name="username" placeholder="用户名">
    <input type="email" name="email" placeholder="邮箱">
    <button type="submit">提交</button>
</form>

<script>
const form = document.getElementById('myForm');

// 表单提交事件
form.addEventListener('submit', function(e) {
    e.preventDefault(); // 阻止默认提交行为
    
    const formData = new FormData(this);
    console.log('用户名：', formData.get('username'));
    console.log('邮箱：', formData.get('email'));
});

// 输入事件
form.querySelector('input[name="username"]')
    .addEventListener('input', function(e) {
        console.log('当前值：', e.target.value);
    });
</script>
```

### 文档/窗口事件

| 事件 | 描述 |
|------|------|
| load | 页面加载完成 |
| DOMContentLoaded | DOM 加载完成 |
| resize | 窗口大小改变 |
| scroll | 页面滚动 |
| unload | 页面卸载 |

```javascript
// DOM 加载完成
document.addEventListener('DOMContentLoaded', function() {
    console.log('DOM 已加载');
});

// 页面完全加载（包括图片等资源）
window.addEventListener('load', function() {
    console.log('页面已完全加载');
});

// 窗口大小改变
window.addEventListener('resize', function() {
    console.log('窗口大小：', window.innerWidth, window.innerHeight);
});

// 页面滚动
window.addEventListener('scroll', function() {
    console.log('滚动位置：', window.scrollY);
});
```

## 事件对象

事件处理函数会接收一个事件对象，包含事件的详细信息：

```javascript
btn.addEventListener('click', function(e) {
    console.log('事件类型：', e.type);
    console.log('目标元素：', e.target);
    console.log('当前元素：', e.currentTarget);
    console.log('鼠标位置：', e.clientX, e.clientY);
    console.log('时间戳：', e.timeStamp);
});
```

### 常用方法

```javascript
// 阻止默认行为
link.addEventListener('click', function(e) {
    e.preventDefault();
    console.log('链接跳转被阻止');
});

// 阻止事件冒泡
child.addEventListener('click', function(e) {
    e.stopPropagation();
    console.log('事件不会冒泡到父元素');
});
```

## 事件冒泡与捕获

### 事件传播三个阶段

1. **捕获阶段**：从 window 向下传播到目标元素
2. **目标阶段**：到达目标元素
3. **冒泡阶段**：从目标元素向上冒泡到 window

```html
<div id="outer" style="padding: 50px; background: red;">
    外层
    <div id="inner" style="padding: 50px; background: blue;">
        内层
    </div>
</div>

<script>
const outer = document.getElementById('outer');
const inner = document.getElementById('inner');

// 冒泡阶段（默认，第三个参数为 false 或省略）
outer.addEventListener('click', function() {
    console.log('外层 - 冒泡');
});

inner.addEventListener('click', function() {
    console.log('内层 - 冒泡');
});

// 捕获阶段（第三个参数为 true）
outer.addEventListener('click', function() {
    console.log('外层 - 捕获');
}, true);

inner.addEventListener('click', function() {
    console.log('内层 - 捕获');
}, true);

// 点击内层时的输出顺序：
// 外层 - 捕获
// 内层 - 捕获
// 内层 - 冒泡
// 外层 - 冒泡
</script>
```

## 事件委托

事件委托利用事件冒泡机制，在父元素上统一处理子元素的事件：

```html
<ul id="list">
    <li>项目 1</li>
    <li>项目 2</li>
    <li>项目 3</li>
</ul>
<button id="addBtn">添加项目</button>

<script>
const list = document.getElementById('list');
const addBtn = document.getElementById('addBtn');

// 事件委托：在父元素上监听
list.addEventListener('click', function(e) {
    // 检查点击的是否为 li 元素
    if (e.target.tagName === 'LI') {
        console.log('点击了：', e.target.textContent);
        e.target.style.color = 'red';
    }
});

// 动态添加的项目也能响应事件
addBtn.addEventListener('click', function() {
    const li = document.createElement('li');
    li.textContent = '新项目 ' + (list.children.length + 1);
    list.appendChild(li);
});
</script>
```

### 事件委托的优势

1. 减少事件监听器数量，提高性能
2. 动态添加的元素也能响应事件
3. 代码更简洁

## 自定义事件

创建和触发自定义事件：

```javascript
// 创建自定义事件
const myEvent = new CustomEvent('myEvent', {
    detail: {
        message: '这是自定义数据'
    },
    bubbles: true
});

// 监听自定义事件
document.addEventListener('myEvent', function(e) {
    console.log('收到自定义事件：', e.detail.message);
});

// 触发自定义事件
document.dispatchEvent(myEvent);
```

## 实际应用示例

### 拖拽功能

```html
<div id="draggable" style="width: 100px; height: 100px; background: #42b883; cursor: move;">
    拖拽我
</div>

<script>
const draggable = document.getElementById('draggable');
let isDragging = false;
let startX, startY, initialX, initialY;

draggable.addEventListener('mousedown', function(e) {
    isDragging = true;
    startX = e.clientX;
    startY = e.clientY;
    const rect = this.getBoundingClientRect();
    initialX = rect.left;
    initialY = rect.top;
});

document.addEventListener('mousemove', function(e) {
    if (!isDragging) return;
    
    const dx = e.clientX - startX;
    const dy = e.clientY - startY;
    
    draggable.style.position = 'absolute';
    draggable.style.left = (initialX + dx) + 'px';
    draggable.style.top = (initialY + dy) + 'px';
});

document.addEventListener('mouseup', function() {
    isDragging = false;
});
</script>
```

### 键盘快捷键

```javascript
document.addEventListener('keydown', function(e) {
    // Ctrl + S 保存
    if (e.ctrlKey && e.key === 's') {
        e.preventDefault();
        console.log('保存内容');
    }
    
    // Ctrl + Shift + N 新建
    if (e.ctrlKey && e.shiftKey && e.key === 'N') {
        e.preventDefault();
        console.log('新建文件');
    }
    
    // Escape 关闭
    if (e.key === 'Escape') {
        console.log('关闭弹窗');
    }
});
```
