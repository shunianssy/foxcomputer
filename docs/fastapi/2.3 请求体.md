# 2.3 请求体

请求体（Request Body）是客户端发送给服务器的数据，通常用于 POST、PUT、PATCH 等请求。FastAPI 使用 Pydantic 模型来处理请求体。

## 基本请求体

### 定义模型

```python
from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()

class Item(BaseModel):
 """商品模型"""
 name: str
 price: float
 description: str | None = None
 tax: float | None = None

@app.post("/items/")
async def create_item(item: Item):
 """
 创建商品
 
 请求体示例：
 {
 "name": "iPhone 15",
 "price": 5999.0,
 "description": "最新款 iPhone",
 "tax": 0.1
 }
 """
 return item
```

### 请求体验证

FastAPI 会自动验证请求体：

- 字段类型验证
- 必填字段检查
- 额外字段处理（默认忽略）

## 字段验证

使用 `Field` 进行字段验证：

```python
from fastapi import FastAPI
from pydantic import BaseModel, Field
from typing import Annotated

app = FastAPI()

class Item(BaseModel):
 name: Annotated[str, Field(
 min_length=1,
 max_length=100,
 description="商品名称",
 examples=["iPhone 15"]
 )]
 price: Annotated[float, Field(
 gt=0,
 le=100000,
 description="商品价格",
 examples=[5999.0]
 )]
 description: str | None = Field(
 default=None,
 max_length=500,
 description="商品描述"
 )
 tax: float | None = Field(
 default=None,
 ge=0,
 le=1,
 description="税率（0-1之间）"
 )

@app.post("/items/")
async def create_item(item: Item):
 return item
```

## 嵌套模型

### 对象嵌套

```python
from pydantic import BaseModel

class Image(BaseModel):
 """图片模型"""
 url: str
 name: str

class Item(BaseModel):
 """商品模型"""
 name: str
 price: float
 image: Image | None = None # 嵌套对象

@app.post("/items/")
async def create_item(item: Item):
 """
 请求体示例：
 {
 "name": "iPhone",
 "price": 5999,
 "image": {
 "url": "https://example.com/iphone.jpg",
 "name": "iPhone 图片"
 }
 }
 """
 return item
```

### 列表嵌套

```python
class Item(BaseModel):
 name: str
 price: float

class Order(BaseModel):
 """订单模型"""
 customer_name: str
 items: list[Item] # 嵌套列表
 total: float

@app.post("/orders/")
async def create_order(order: Order):
 """
 请求体示例：
 {
 "customer_name": "张三",
 "items": [
 {"name": "iPhone", "price": 5999},
 {"name": "AirPods", "price": 1299}
 ],
 "total": 7298
 }
 """
 return order
```

## 多个请求体参数

```python
from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()

class Item(BaseModel):
 name: str
 price: float

class User(BaseModel):
 username: str
 email: str

@app.post("/items/")
async def create_item(item: Item, user: User):
 """
 多个请求体参数
 
 请求体示例：
 {
 "item": {
 "name": "iPhone",
 "price": 5999
 },
 "user": {
 "username": "zhangsan",
 "email": "zhangsan@example.com"
 }
 }
 """
 return {"item": item, "user": user}
```

## 单个值请求体

使用 `Body` 获取单个值：

```python
from fastapi import FastAPI, Body
from pydantic import BaseModel

app = FastAPI()

class Item(BaseModel):
 name: str
 price: float

@app.post("/items/")
async def create_item(
 item: Item,
 importance: int = Body(default=1, gt=0, lt=10)
):
 """
 混合使用模型和单个值
 
 请求体示例：
 {
 "item": {
 "name": "iPhone",
 "price": 5999
 },
 "importance": 5
 }
 """
 return {"item": item, "importance": importance}
```

## 嵌入单个请求体

```python
from fastapi import FastAPI, Body
from pydantic import BaseModel

app = FastAPI()

class Item(BaseModel):
 name: str
 price: float

@app.post("/items/")
async def create_item(item: Item = Body(embed=True)):
 """
 嵌入请求体
 
 请求体示例：
 {
 "item": {
 "name": "iPhone",
 "price": 5999
 }
 }
 
 如果 embed=False（默认），请求体是：
 {
 "name": "iPhone",
 "price": 5999
 }
 """
 return item
```

## 表单数据

处理表单提交：

```python
from fastapi import FastAPI, Form

app = FastAPI()

@app.post("/login/")
async def login(
 username: str = Form(...),
 password: str = Form(...),
 remember: bool = Form(default=False)
):
 """
 处理表单数据
 
 Content-Type: application/x-www-form-urlencoded
 """
 return {
 "username": username,
 "remember": remember
 }
```

## 文件上传

### 单文件上传

```python
from fastapi import FastAPI, File, UploadFile

app = FastAPI()

@app.post("/upload/")
async def upload_file(file: UploadFile = File(...)):
 """
 上传单个文件
 
 - file.filename: 文件名
 - file.content_type: 文件类型
 - file.file: 文件对象
 """
 contents = await file.read()
 return {
 "filename": file.filename,
 "content_type": file.content_type,
 "size": len(contents)
 }
```

### 多文件上传

```python
from fastapi import FastAPI, File, UploadFile
from typing import List

app = FastAPI()

@app.post("/upload/")
async def upload_files(files: List[UploadFile] = File(...)):
 """
 上传多个文件
 """
 results = []
 for file in files:
 contents = await file.read()
 results.append({
 "filename": file.filename,
 "size": len(contents)
 })
 return {"files": results}
```

### 文件与表单混合

```python
from fastapi import FastAPI, File, UploadFile, Form

app = FastAPI()

@app.post("/upload/")
async def upload_with_metadata(
 file: UploadFile = File(...),
 description: str = Form(...),
 tags: str = Form(default="")
):
 """
 文件与表单数据混合
 """
 return {
 "filename": file.filename,
 "description": description,
 "tags": tags.split(",") if tags else []
 }
```

## 请求体示例

添加请求体示例：

```python
from pydantic import BaseModel, Field

class Item(BaseModel):
 name: str
 price: float
 description: str | None = None

 model_config = {
 "json_schema_extra": {
 "examples": [
 {
 "name": "iPhone 15",
 "price": 5999.0,
 "description": "最新款 iPhone"
 }
 ]
 }
 }

# 或者使用 Field
class Item(BaseModel):
 name: str = Field(examples=["iPhone 15"])
 price: float = Field(examples=[5999.0])
 description: str | None = Field(default=None, examples=["最新款 iPhone"])
```

## 访问原始请求体

```python
from fastapi import FastAPI, Request

app = FastAPI()

@app.post("/items/")
async def create_item(request: Request):
 """
 访问原始请求体
 """
 # 获取原始字节
 body = await request.body()
 
 # 获取 JSON
 json_data = await request.json()
 
 # 获取表单数据
 form_data = await request.form()
 
 return {"body": body.decode(), "json": json_data}
```

## 完整示例

```python
from fastapi import FastAPI, HTTPException, File, UploadFile, Form
from pydantic import BaseModel, Field, EmailStr
from typing import Annotated, Optional
from datetime import datetime

app = FastAPI()

class Address(BaseModel):
 """地址模型"""
 province: str
 city: str
 detail: str

class UserCreate(BaseModel):
 """用户创建模型"""
 username: Annotated[str, Field(
 min_length=3,
 max_length=20,
 pattern="^[a-zA-Z0-9_]+$",
 description="用户名，只能包含字母、数字和下划线"
 )]
 email: EmailStr
 password: Annotated[str, Field(
 min_length=8,
 max_length=50,
 description="密码，至少8个字符"
 )]
 age: Annotated[int, Field(
 ge=0,
 le=150,
 description="年龄"
 )]
 address: Address | None = None
 tags: list[str] = Field(default_factory=list)

 model_config = {
 "json_schema_extra": {
 "examples": [
 {
 "username": "zhangsan",
 "email": "zhangsan@example.com",
 "password": "password123",
 "age": 25,
 "address": {
 "province": "北京",
 "city": "北京",
 "detail": "朝阳区xxx街道"
 },
 "tags": ["developer", "python"]
 }
 ]
 }
 }

@app.post("/users/", status_code=201)
async def create_user(user: UserCreate):
 """
 创建用户
 
 包含完整的字段验证和嵌套模型
 """
 # 模拟检查用户名是否存在
 if user.username == "admin":
 raise HTTPException(
 status_code=400,
 detail="用户名已存在"
 )
 
 return {
 "id": 1,
 "username": user.username,
 "email": user.email,
 "age": user.age,
 "address": user.address,
 "tags": user.tags,
 "created_at": datetime.now()
 }
```

## 小结

请求体处理要点：

1. **Pydantic 模型**：使用 `BaseModel` 定义请求体结构
2. **字段验证**：使用 `Field` 进行详细验证
3. **嵌套模型**：支持对象和列表嵌套
4. **表单数据**：使用 `Form` 处理表单提交
5. **文件上传**：使用 `File` 和 `UploadFile` 处理文件

> **提示**：使用 Pydantic 模型可以自动生成文档、验证数据，大大简化了请求处理的代码量。
