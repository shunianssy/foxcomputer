# 2.4 请求头与 Cookie

HTTP 请求头和 Cookie 是 Web 开发中常用的功能，用于传递元信息和用户状态。FastAPI 提供了便捷的方式来处理它们。

## 请求头

### 基本用法

```python
from fastapi import FastAPI, Header

app = FastAPI()

@app.get("/items/")
async def read_items(user_agent: str | None = Header(default=None)):
    """
    获取 User-Agent 请求头
    
    注意：请求头参数名中的连字符(-)会自动转换为下划线(_)
    User-Agent -> user_agent
    """
    return {"User-Agent": user_agent}
```

### 带默认值的请求头

```python
@app.get("/items/")
async def read_items(
    host: str = Header(default="localhost"),
    content_type: str = Header(default="application/json")
):
    """带默认值的请求头"""
    return {
        "Host": host,
        "Content-Type": content_type
    }
```

### 请求头类型转换

```python
@app.get("/items/")
async def read_items(
    content_length: int | None = Header(default=None),
    x_request_id: str | None = Header(default=None)
):
    """
    FastAPI 会自动进行类型转换
    
    Content-Length: "123" -> content_length: 123
    """
    return {
        "content_length": content_length,
        "x_request_id": x_request_id
    }
```

### 自动转换规则

FastAPI 会自动处理请求头名称：

```python
@app.get("/items/")
async def read_items(
    # 以下写法都可以获取 X-Token 请求头
    x_token: str | None = Header(default=None),      # 推荐
    x_token: str | None = Header(default=None, alias="X-Token"),
):
    """
    请求头名称转换：
    - 连字符转下划线：X-Api-Key -> x_api_key
    - 大写转小写：Content-Type -> content_type
    
    客户端发送：X-Token: abc123
    """
    return {"x_token": x_token}
```

### 重复的请求头

```python
@app.get("/items/")
async def read_items(
    x_token: list[str] | None = Header(default=None)
):
    """
    获取重复的请求头
    
    请求：
    X-Token: foo
    X-Token: bar
    
    结果：x_token = ["foo", "bar"]
    """
    return {"X-Token": x_token}
```

### 请求头验证

```python
from fastapi import FastAPI, Header, HTTPException

app = FastAPI()

@app.get("/items/")
async def read_items(
    x_api_key: str = Header(
        ...,
        min_length=10,
        max_length=50,
        description="API 密钥"
    )
):
    """
    验证请求头
    
    如果 X-Api-Key 不存在或长度不符合要求，返回 422 错误
    """
    if x_api_key != "valid-api-key":
        raise HTTPException(status_code=401, detail="Invalid API Key")
    return {"x_api_key": x_api_key}
```

## Cookie

### 基本用法

```python
from fastapi import FastAPI, Cookie

app = FastAPI()

@app.get("/items/")
async def read_items(
    session_id: str | None = Cookie(default=None)
):
    """
    获取 Cookie
    
    Cookie: session_id=abc123
    
    结果：session_id = "abc123"
    """
    return {"session_id": session_id}
```

### 带验证的 Cookie

```python
from fastapi import FastAPI, Cookie, HTTPException

app = FastAPI()

@app.get("/protected/")
async def protected_route(
    auth_token: str = Cookie(
        ...,
        min_length=20,
        description="认证令牌"
    )
):
    """
    必需的 Cookie
    
    如果没有 auth_token Cookie，返回 422 错误
    """
    # 验证 token
    if not is_valid_token(auth_token):
        raise HTTPException(status_code=401, detail="Invalid token")
    return {"message": "Access granted"}

def is_valid_token(token: str) -> bool:
    """模拟 token 验证"""
    return token.startswith("valid_")
```

### 设置 Cookie

使用 `Response` 对象设置 Cookie：

```python
from fastapi import FastAPI, Response

app = FastAPI()

@app.post("/login/")
async def login(response: Response):
    """
    设置 Cookie
    
    set_cookie 参数：
    - key: Cookie 名称
    - value: Cookie 值
    - max_age: 过期时间（秒）
    - expires: 过期时间（datetime 或秒数）
    - path: Cookie 路径
    - domain: Cookie 域名
    - secure: 仅 HTTPS 传输
    - httponly: 禁止 JavaScript 访问
    - samesite: 同站策略（"strict", "lax", "none"）
    """
    response.set_cookie(
        key="session_id",
        value="abc123xyz",
        max_age=3600,
        httponly=True,
        secure=True,
        samesite="strict"
    )
    return {"message": "Login successful"}

@app.post("/logout/")
async def logout(response: Response):
    """删除 Cookie"""
    response.delete_cookie("session_id")
    return {"message": "Logout successful"}
```

## 实用示例

### API 认证中间件

```python
from fastapi import FastAPI, Header, HTTPException, Depends
from typing import Annotated

app = FastAPI()

async def verify_api_key(
    x_api_key: Annotated[str, Header(description="API 密钥")]
):
    """
    验证 API Key 的依赖函数
    """
    valid_keys = {"key1", "key2", "key3"}
    if x_api_key not in valid_keys:
        raise HTTPException(
            status_code=401,
            detail="Invalid API Key"
        )
    return x_api_key

@app.get("/protected/", dependencies=[Depends(verify_api_key)])
async def protected_route():
    """需要 API Key 的路由"""
    return {"message": "Access granted"}

@app.get("/user-info/")
async def user_info(api_key: str = Depends(verify_api_key)):
    """获取 API Key 信息"""
    return {"api_key": api_key, "user": "张三"}
```

### 用户会话管理

```python
from fastapi import FastAPI, Cookie, HTTPException, Response
from pydantic import BaseModel
from typing import Optional
import uuid
import time

app = FastAPI()

# 模拟会话存储
sessions: dict[str, dict] = {}

class LoginRequest(BaseModel):
    username: str
    password: str

@app.post("/login/")
async def login(request: LoginRequest, response: Response):
    """
    用户登录，创建会话
    """
    # 模拟验证
    if request.username != "admin" or request.password != "password":
        raise HTTPException(status_code=401, detail="用户名或密码错误")
    
    # 创建会话
    session_id = str(uuid.uuid4())
    sessions[session_id] = {
        "username": request.username,
        "created_at": time.time()
    }
    
    # 设置 Cookie
    response.set_cookie(
        key="session_id",
        value=session_id,
        max_age=3600,  # 1小时
        httponly=True,
        secure=True,
        samesite="lax"
    )
    
    return {"message": "登录成功", "session_id": session_id}

@app.get("/me/")
async def get_current_user(
    session_id: Optional[str] = Cookie(default=None)
):
    """
    获取当前用户信息
    """
    if not session_id or session_id not in sessions:
        raise HTTPException(status_code=401, detail="未登录")
    
    session = sessions[session_id]
    
    # 检查会话是否过期
    if time.time() - session["created_at"] > 3600:
        del sessions[session_id]
        raise HTTPException(status_code=401, detail="会话已过期")
    
    return {
        "username": session["username"],
        "logged_in_at": session["created_at"]
    }

@app.post("/logout/")
async def logout(
    session_id: Optional[str] = Cookie(default=None),
    response: Response = None
):
    """
    用户登出
    """
    if session_id and session_id in sessions:
        del sessions[session_id]
    
    response.delete_cookie("session_id")
    return {"message": "已登出"}
```

### 自定义请求头验证

```python
from fastapi import FastAPI, Header, HTTPException
from typing import Annotated
import hashlib
import time

app = FastAPI()

@app.post("/api/webhook/")
async def webhook(
    x_signature: Annotated[str, Header(description="请求签名")],
    x_timestamp: Annotated[int, Header(description="时间戳")]
):
    """
    验证 Webhook 签名
    
    客户端需要发送：
    - X-Signature: HMAC 签名
    - X-Timestamp: 请求时间戳
    """
    # 验证时间戳（防止重放攻击）
    current_time = int(time.time())
    if abs(current_time - x_timestamp) > 300:  # 5分钟有效期
        raise HTTPException(status_code=400, detail="请求已过期")
    
    # 验证签名
    expected_signature = calculate_signature(x_timestamp)
    if x_signature != expected_signature:
        raise HTTPException(status_code=401, detail="签名验证失败")
    
    return {"message": "Webhook received"}

def calculate_signature(timestamp: int) -> str:
    """计算签名"""
    secret = "your-secret-key"
    data = f"{timestamp}{secret}"
    return hashlib.sha256(data.encode()).hexdigest()
```

## 小结

请求头与 Cookie 要点：

1. **请求头**：使用 `Header()` 获取，自动转换命名格式
2. **Cookie**：使用 `Cookie()` 获取，需要设置 `default=None` 使其可选
3. **设置 Cookie**：使用 `Response.set_cookie()`
4. **删除 Cookie**：使用 `Response.delete_cookie()`
5. **验证**：可以像其他参数一样进行类型验证

> 💡 **提示**：对于敏感信息（如认证令牌），建议使用 HttpOnly 和 Secure 标志的 Cookie，或使用 Authorization 请求头。
