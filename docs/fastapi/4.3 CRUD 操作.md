## 数据库配置

```python
# database.py
from sqlalchemy import create_engine
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker

SQLALCHEMY_DATABASE_URL = "sqlite:///./app.db"

engine = create_engine(
 SQLALCHEMY_DATABASE_URL,
 connect_args={"check_same_thread": False}
)

SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

Base = declarative_base()

def get_db():
 """获取数据库会话"""
 db = SessionLocal()
 try:
 yield db
 finally:
 db.close()
```

## Pydantic 模型

```python
# schemas.py
from pydantic import BaseModel, Field, ConfigDict
from typing import Optional
from datetime import datetime

class ItemBase(BaseModel):
 """商品基础模型"""
 name: str = Field(..., min_length=1, max_length=100, description="商品名称")
 description: Optional[str] = Field(None, max_length=500, description="商品描述")
 price: float = Field(..., gt=0, description="商品价格")
 quantity: int = Field(default=0, ge=0, description="库存数量")

class ItemCreate(ItemBase):
 """创建商品模型"""
 pass

class ItemUpdate(BaseModel):
 """更新商品模型"""
 name: Optional[str] = Field(None, min_length=1, max_length=100)
 description: Optional[str] = Field(None, max_length=500)
 price: Optional[float] = Field(None, gt=0)
 quantity: Optional[int] = Field(None, ge=0)
 is_active: Optional[bool] = None

class Item(ItemBase):
 """商品响应模型"""
 id: int
 is_active: bool
 created_at: datetime
 updated_at: Optional[datetime] = None
 
 model_config = ConfigDict(from_attributes=True)

class ItemList(BaseModel):
 """商品列表响应"""
 total: int
 items: list[Item]
```

## CRUD 封装

```python
# crud.py
from sqlalchemy.orm import Session
from sqlalchemy import or_
from typing import List, Optional
from models import Item
from schemas import ItemCreate, ItemUpdate

class ItemCRUD:
 """商品 CRUD 操作封装"""
 
 @staticmethod
 def create(db: Session, item: ItemCreate) -> Item:
 """
 创建商品
 
 Args:
 db: 数据库会话
 item: 商品创建数据
 
 Returns:
 创建的商品对象
 """
 db_item = Item(**item.model_dump())
 db.add(db_item)
 db.commit()
 db.refresh(db_item)
 return db_item
 
 @staticmethod
 def get_by_id(db: Session, item_id: int) -> Optional[Item]:
 """
 根据 ID 获取商品
 
 Args:
 db: 数据库会话
 item_id: 商品 ID
 
 Returns:
 商品对象或 None
 """
 return db.query(Item).filter(Item.id == item_id).first()
 
 @staticmethod
 def get_list(
 db: Session,
 skip: int = 0,
 limit: int = 10,
 name: str = None,
 min_price: float = None,
 max_price: float = None,
 is_active: bool = None
 ) -> List[Item]:
 """
 获取商品列表
 
 Args:
 db: 数据库会话
 skip: 跳过数量
 limit: 返回数量
 name: 名称筛选（模糊匹配）
 min_price: 最低价格
 max_price: 最高价格
 is_active: 是否激活
 
 Returns:
 商品列表
 """
 query = db.query(Item)
 
 # 名称筛选
 if name:
 query = query.filter(Item.name.ilike(f"%{name}%"))
 
 # 价格筛选
 if min_price is not None:
 query = query.filter(Item.price >= min_price)
 if max_price is not None:
 query = query.filter(Item.price <= max_price)
 
 # 状态筛选
 if is_active is not None:
 query = query.filter(Item.is_active == is_active)
 
 return query.offset(skip).limit(limit).all()
 
 @staticmethod
 def count(
 db: Session,
 name: str = None,
 min_price: float = None,
 max_price: float = None,
 is_active: bool = None
 ) -> int:
 """
 统计商品数量
 
 Args:
 db: 数据库会话
 name: 名称筛选
 min_price: 最低价格
 max_price: 最高价格
 is_active: 是否激活
 
 Returns:
 商品数量
 """
 from sqlalchemy import func
 
 query = db.query(func.count(Item.id))
 
 if name:
 query = query.filter(Item.name.ilike(f"%{name}%"))
 if min_price is not None:
 query = query.filter(Item.price >= min_price)
 if max_price is not None:
 query = query.filter(Item.price <= max_price)
 if is_active is not None:
 query = query.filter(Item.is_active == is_active)
 
 return query.scalar()
 
 @staticmethod
 def update(
 db: Session,
 item_id: int,
 item: ItemUpdate
 ) -> Optional[Item]:
 """
 更新商品
 
 Args:
 db: 数据库会话
 item_id: 商品 ID
 item: 更新数据
 
 Returns:
 更新后的商品或 None
 """
 db_item = db.query(Item).filter(Item.id == item_id).first()
 if not db_item:
 return None
 
 # 只更新提供的字段
 update_data = item.model_dump(exclude_unset=True)
 for key, value in update_data.items():
 setattr(db_item, key, value)
 
 db.commit()
 db.refresh(db_item)
 return db_item
 
 @staticmethod
 def delete(db: Session, item_id: int) -> bool:
 """
 删除商品
 
 Args:
 db: 数据库会话
 item_id: 商品 ID
 
 Returns:
 是否删除成功
 """
 db_item = db.query(Item).filter(Item.id == item_id).first()
 if not db_item:
 return False
 
 db.delete(db_item)
 db.commit()
 return True
 
 @staticmethod
 def bulk_create(db: Session, items: List[ItemCreate]) -> List[Item]:
 """
 批量创建商品
 
 Args:
 db: 数据库会话
 items: 商品列表
 
 Returns:
 创建的商品列表
 """
 db_items = [Item(**item.model_dump()) for item in items]
 db.add_all(db_items)
 db.commit()
 for db_item in db_items:
 db.refresh(db_item)
 return db_items
 
 @staticmethod
 def bulk_update_quantity(
 db: Session,
 updates: dict[int, int]
 ) -> List[Item]:
 """
 批量更新库存
 
 Args:
 db: 数据库会话
 updates: {商品ID: 新库存数量}
 
 Returns:
 更新后的商品列表
 """
 updated_items = []
 for item_id, quantity in updates.items():
 db_item = db.query(Item).filter(Item.id == item_id).first()
 if db_item:
 db_item.quantity = quantity
 updated_items.append(db_item)
 
 db.commit()
 return updated_items
 
 @staticmethod
 def exists(db: Session, item_id: int) -> bool:
 """
 检查商品是否存在
 
 Args:
 db: 数据库会话
 item_id: 商品 ID
 
 Returns:
 是否存在
 """
 return db.query(Item).filter(Item.id == item_id).first() is not None
```

## 路由实现

```python
# routers/items.py
from fastapi import APIRouter, Depends, HTTPException, status, Query
from sqlalchemy.orm import Session
from typing import Optional

from database import get_db
from models import Item
from schemas import Item, ItemCreate, ItemUpdate, ItemList
from crud import ItemCRUD

router = APIRouter(prefix="/items", tags=["商品管理"])

@router.post(
 "/",
 response_model=Item,
 status_code=status.HTTP_201_CREATED,
 summary="创建商品"
)
async def create_item(
 item: ItemCreate,
 db: Session = Depends(get_db)
):
 """
 创建新商品
 
 - **name**: 商品名称（必填）
 - **price**: 商品价格（必填，大于0）
 - **description**: 商品描述（可选）
 - **quantity**: 库存数量（默认0）
 """
 return ItemCRUD.create(db, item)

@router.get("/", response_model=ItemList, summary="获取商品列表")
async def list_items(
 skip: int = Query(default=0, ge=0, description="跳过数量"),
 limit: int = Query(default=10, ge=1, le=100, description="返回数量"),
 name: Optional[str] = Query(default=None, description="商品名称（模糊搜索）"),
 min_price: Optional[float] = Query(default=None, ge=0, description="最低价格"),
 max_price: Optional[float] = Query(default=None, ge=0, description="最高价格"),
 is_active: Optional[bool] = Query(default=None, description="是否激活"),
 db: Session = Depends(get_db)
):
 """
 获取商品列表，支持分页和筛选
 """
 items = ItemCRUD.get_list(
 db,
 skip=skip,
 limit=limit,
 name=name,
 min_price=min_price,
 max_price=max_price,
 is_active=is_active
 )
 total = ItemCRUD.count(
 db,
 name=name,
 min_price=min_price,
 max_price=max_price,
 is_active=is_active
 )
 return ItemList(total=total, items=items)

@router.get("/{item_id}", response_model=Item, summary="获取单个商品")
async def read_item(
 item_id: int,
 db: Session = Depends(get_db)
):
 """
 根据 ID 获取商品详情
 """
 item = ItemCRUD.get_by_id(db, item_id)
 if not item:
 raise HTTPException(
 status_code=status.HTTP_404_NOT_FOUND,
 detail=f"商品 ID {item_id} 不存在"
 )
 return item

@router.put("/{item_id}", response_model=Item, summary="更新商品")
async def update_item(
 item_id: int,
 item: ItemUpdate,
 db: Session = Depends(get_db)
):
 """
 更新商品信息（部分更新）
 """
 db_item = ItemCRUD.update(db, item_id, item)
 if not db_item:
 raise HTTPException(
 status_code=status.HTTP_404_NOT_FOUND,
 detail=f"商品 ID {item_id} 不存在"
 )
 return db_item

@router.delete(
 "/{item_id}",
 status_code=status.HTTP_204_NO_CONTENT,
 summary="删除商品"
)
async def delete_item(
 item_id: int,
 db: Session = Depends(get_db)
):
 """
 删除商品
 """
 if not ItemCRUD.delete(db, item_id):
 raise HTTPException(
 status_code=status.HTTP_404_NOT_FOUND,
 detail=f"商品 ID {item_id} 不存在"
 )

@router.post("/bulk", response_model=list[Item], summary="批量创建商品")
async def bulk_create_items(
 items: list[ItemCreate],
 db: Session = Depends(get_db)
):
 """
 批量创建商品
 """
 return ItemCRUD.bulk_create(db, items)
```

## 事务处理

```python
from sqlalchemy.orm import Session
from fastapi import HTTPException

def transfer_stock(
 db: Session,
 from_item_id: int,
 to_item_id: int,
 quantity: int
):
 """
 库存调拨（事务示例）
 """
 try:
 # 获取商品
 from_item = db.query(Item).filter(Item.id == from_item_id).first()
 to_item = db.query(Item).filter(Item.id == to_item_id).first()
 
 if not from_item or not to_item:
 raise HTTPException(status_code=404, detail="商品不存在")
 
 if from_item.quantity < quantity:
 raise HTTPException(status_code=400, detail="库存不足")
 
 # 更新库存
 from_item.quantity -= quantity
 to_item.quantity += quantity
 
 db.commit()
 
 return {"message": "调拨成功"}
 
 except Exception as e:
 db.rollback()
 raise e
```

## 小结

CRUD 操作要点：

1. **模型分离**：ORM 模型和 Pydantic 模型分离
2. **CRUD 封装**：封装数据库操作，提高代码复用
3. **部分更新**：使用 `exclude_unset=True` 实现部分更新
4. **分页筛选**：支持多种筛选条件
5. **事务处理**：使用 `db.commit()` 和 `db.rollback()` 管理事务

> **提示**：将 CRUD 操作封装成独立模块，可以使路由代码更加简洁，也便于单元测试。
