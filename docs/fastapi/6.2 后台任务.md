# 6.2 后台任务

后台任务允许你在响应返回后执行额外的操作，如发送邮件、处理数据等。FastAPI 提供了简单易用的后台任务支持。

## 基本用法

### 使用 BackgroundTasks

```python
from fastapi import FastAPI, BackgroundTasks

app = FastAPI()

def send_email(email: str, message: str):
 """发送邮件（模拟）"""
 print(f"发送邮件到 {email}: {message}")
 # 实际发送邮件的逻辑

@app.post("/send-notification")
async def send_notification(email: str, background_tasks: BackgroundTasks):
 """
 发送通知
 
 邮件发送会在响应返回后执行
 """
 background_tasks.add_task(send_email, email, "您有新的通知")
 return {"message": "通知已发送"}
```

### 多个后台任务

```python
@app.post("/process-data")
async def process_data(background_tasks: BackgroundTasks):
 """多个后台任务"""
 
 def task1():
 print("执行任务 1")
 
 def task2(name: str):
 print(f"执行任务 2: {name}")
 
 def task3(data: dict):
 print(f"执行任务 3: {data}")
 
 # 添加多个任务
 background_tasks.add_task(task1)
 background_tasks.add_task(task2, name="测试")
 background_tasks.add_task(task3, data={"key": "value"})
 
 return {"message": "任务已添加"}
```

## 实际应用场景

### 发送邮件

```python
from fastapi import FastAPI, BackgroundTasks
from pydantic import BaseModel, EmailStr

app = FastAPI()

class EmailRequest(BaseModel):
 to: EmailStr
 subject: str
 body: str

def send_email_task(to: str, subject: str, body: str):
 """发送邮件任务"""
 # 实际的邮件发送逻辑
 import smtplib
 from email.mime.text import MIMEText
 
 # 模拟发送
 print(f"发送邮件到 {to}: {subject}")

@app.post("/send-email")
async def send_email(
 request: EmailRequest,
 background_tasks: BackgroundTasks
):
 """发送邮件接口"""
 background_tasks.add_task(
```

### 数据处理

```python
from fastapi import FastAPI, BackgroundTasks, UploadFile, File
import csv
import io

app = FastAPI()

def process_csv_file(content: bytes):
 """处理 CSV 文件"""
 decoded = content.decode('utf-8')
 reader = csv.DictReader(io.StringIO(decoded))
 
 for row in reader:
 # 处理每一行数据
 print(f"处理: {row}")

@app.post("/upload-csv")
async def upload_csv(
 file: UploadFile = File(...),
 background_tasks: BackgroundTasks = None
):
 """上传并处理 CSV 文件"""
 content = await file.read()
 
 # 在后台处理文件
 background_tasks.add_task(process_csv_file, content)
 
 return {
 "filename": file.filename,
 "message": "文件已上传，正在后台处理"
 }
```

### 日志记录

```python
from fastapi import FastAPI, BackgroundTasks, Request
import logging
from datetime import datetime

app = FastAPI()
logger = logging.getLogger(__name__)

def log_request(request_info: dict):
 """记录请求日志"""
 logger.info(f"Request: {request_info}")

@app.middleware("http")
async def log_requests(request: Request, call_next):
 response = await call_next(request)
 
 # 后台记录日志
 background_tasks = BackgroundTasks()
 background_tasks.add_task(
 log_request,
 {
 "method": request.method,
 "url": str(request.url),
 "status": response.status_code,
 "time": datetime.now().isoformat()
 }
 )
 
 return response
```

## 后台任务与依赖注入

```python
from fastapi import FastAPI, BackgroundTasks, Depends

app = FastAPI()

def get_background_tasks():
 """获取后台任务"""
 return BackgroundTasks()

def write_log(message: str):
 """写日志"""
 with open("app.log", "a") as f:
 f.write(f"{message}\n")

@app.post("/log")
async def log_message(
 message: str,
 background_tasks: BackgroundTasks = Depends(get_background_tasks)
):
 """记录日志"""
 background_tasks.add_task(write_log, message)
 return {"message": "日志已记录"}
```

## Celery 集成

对于复杂的后台任务，推荐使用 Celery：

### 安装

```bash
pip install celery redis
```

### 配置 Celery

```python
# celery_app.py
from celery import Celery

celery = Celery(
 "worker",
 broker="redis://localhost:6379/0",
 backend="redis://localhost:6379/1"
)

celery.conf.update(
 task_serializer="json",
 accept_content=["json"],
 result_serializer="json",
 timezone="Asia/Shanghai",
 enable_utc=True,
)

@celery.task
def send_email_task(to: str, subject: str, body: str):
 """发送邮件任务"""
 print(f"发送邮件到 {to}: {subject}")
 return {"status": "sent", "to": to}

@celery.task
def process_large_file(file_path: str):
 """处理大文件"""
 import time
 time.sleep(10) # 模拟耗时操作
 return {"status": "completed", "file": file_path}
```

### FastAPI 集成

```python
# main.py
from fastapi import FastAPI
from celery.result import AsyncResult
from celery_app import celery, send_email_task, process_large_file

app = FastAPI()

@app.post("/send-email-async")
async def send_email_async(to: str, subject: str, body: str):
 """异步发送邮件"""
 task = send_email_task.delay(to, subject, body)
 return {"task_id": task.id, "status": "pending"}

@app.get("/task-status/{task_id}")
async def get_task_status(task_id: str):
 """获取任务状态"""
 result = AsyncResult(task_id, app=celery)
 
 return {
 "task_id": task_id,
 "status": result.status,
 "result": result.result if result.ready() else None
 }

@app.post("/process-file")
async def process_file(file_path: str):
 """处理大文件"""
 task = process_large_file.delay(file_path)
 return {"task_id": task.id, "status": "processing"}
```


## 定时任务

### Celery Beat

```python
# celery_app.py
from celery import Celery
from celery.schedules import crontab

celery = Celery("worker", broker="redis://localhost:6379/0")

@celery.task
def cleanup_expired_sessions():
 """清理过期会话"""
 print("清理过期会话...")

# 定时任务配置
celery.conf.beat_schedule = {
 "cleanup-every-hour": {
 "task": "celery_app.cleanup_expired_sessions",
 "schedule": crontab(minute=0), # 每小时执行
 },
 "cleanup-every-day": {
 "task": "celery_app.cleanup_expired_sessions",
 "schedule": crontab(hour=0, minute=0), # 每天凌晨执行
 },
}
```

## 完整示例

```python
from fastapi import FastAPI, BackgroundTasks, HTTPException
from pydantic import BaseModel, EmailStr
from typing import Optional
import logging
import asyncio

app = FastAPI(title="后台任务示例")
logger = logging.getLogger(__name__)

# 模型
class Notification(BaseModel):
 email: EmailStr
 title: str
 content: str

class TaskResult(BaseModel):
 task_id: str
 status: str
 result: Optional[str] = None

# 任务存储
tasks: dict[str, dict] = {}

# 后台任务函数
def send_notification(email: str, title: str, content: str):
 """发送通知"""
 import time
 time.sleep(2) # 模拟耗时操作
 logger.info(f"通知已发送到 {email}: {title}")

def generate_report(report_id: str):
 """生成报告"""
 import time
 tasks[report_id]["status"] = "processing"
 time.sleep(5) # 模拟耗时操作
 tasks[report_id]["status"] = "completed"
 tasks[report_id]["result"] = "报告内容..."

# 路由
@app.post("/notifications", status_code=202)
async def create_notification(
 notification: Notification,
 background_tasks: BackgroundTasks
):
 """
 发送通知
 
 返回 202 Accepted，表示请求已接受但尚未完成
 """
 background_tasks.add_task(
 send_notification,
 email=notification.email,
 title=notification.title,
 content=notification.content
 )
 
 return {
 "message": "通知已加入发送队列",
 "email": notification.email
 }

@app.post("/reports", response_model=TaskResult, status_code=202)
async def create_report(background_tasks: BackgroundTasks):
 """
 生成报告
 
 返回任务 ID，可通过 /reports/{task_id} 查询状态
 """
 import uuid
 task_id = str(uuid.uuid4())
 
 tasks[task_id] = {"status": "pending", "result": None}
 
 background_tasks.add_task(generate_report, task_id)
 
 return TaskResult(task_id=task_id, status="pending")

@app.get("/reports/{task_id}", response_model=TaskResult)
async def get_report_status(task_id: str):
 """获取报告生成状态"""
 if task_id not in tasks:
 raise HTTPException(status_code=404, detail="任务不存在")
 
 task = tasks[task_id]
 return TaskResult(
 task_id=task_id,
 status=task["status"],
 result=task.get("result")
 )
```

## 小结

后台任务要点：

1. **BackgroundTasks**：简单的后台任务
2. **任务顺序**：按添加顺序执行
3. **Celery**：复杂的分布式任务队列
4. **定时任务**：使用 Celery Beat
5. **状态查询**：返回任务 ID 供查询

> **提示**：简单任务使用 BackgroundTasks，复杂任务使用 Celery。
