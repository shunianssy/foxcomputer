# 3.3 响应状态码

HTTP 状态码是 API 响应的重要组成部分，它告诉客户端请求的处理结果。FastAPI 提供了灵活的状态码设置方式。

## HTTP 状态码分类

| 类别 | 范围 | 说明 |
|------|------|------|
| 1xx | 100-199 | 信息响应 |
| 2xx | 200-299 | 成功响应 |
| 3xx | 300-399 | 重定向 |
| 4xx | 400-499 | 客户端错误 |
| 5xx | 500-599 | 服务器错误 |

### 常用状态码

| 状态码 | 名称 | 说明 |
|--------|------|------|
| 200 | OK | 请求成功 |
| 201 | Created | 资源创建成功 |
| 204 | No Content | 成功但无返回内容 |
| 400 | Bad Request | 请求参数错误 |
| 401 | Unauthorized | 未认证 |
| 403 | Forbidden | 无权限 |
| 404 | Not Found | 资源不存在 |
| 422 | Unprocessable Entity | 验证失败 |
| 500 | Internal Server Error | 服务器错误 |

## 默认状态码

```python
from fastapi import FastAPI

app = FastAPI()

@app.get("/items/")
async def read_items():
    """
    默认返回 200 OK
    """
    return {"items": []}

@app.post("/items/")
async def create_item():
    """
    默认返回 200 OK
    """
    return {"id": 1, "name": "Item"}
```

## 设置状态码

### 使用 status_code 参数

```python
from fastapi import FastAPI, status

app = FastAPI()

@app.post("/items/", status_code=201)
async def create_item():
    """
    返回 201 Created
    """
    return {"id": 1, "name": "Item"}

@app.delete("/items/{item_id}", status_code=204)
async def delete_item(item_id: int):
    """
    返回 204 No Content
    注意：204 不应该有响应体
    """
    return None
```

### 使用 status 模块

```python
from fastapi import FastAPI, status

app = FastAPI()

@app.post("/items/", status_code=status.HTTP_201_CREATED)
```

## 动态状态码

### 使用 Response 对象

```python
from fastapi import FastAPI, Response, status

app = FastAPI()

@app.post("/items/")
async def create_item(response: Response):
    """
    动态设置状态码
    """
    # 模拟创建逻辑
    success = True
    
    if success:
        response.status_code = status.HTTP_201_CREATED
        return {"id": 1, "message": "创建成功"}
    else:
        response.status_code = status.HTTP_400_BAD_REQUEST
        return {"message": "创建失败"}
```

### 条件状态码

```python
from fastapi import FastAPI, Response, status
from fastapi.responses import JSONResponse

app = FastAPI()

@app.get("/items/{item_id}")
async def read_item(item_id: int):
    """
    根据条件返回不同状态码
    """
    if item_id < 0:
        return JSONResponse(
            status_code=status.HTTP_400_BAD_REQUEST,
            content={"message": "无效的 ID"}
        )
    
    if item_id > 100:
        return JSONResponse(
            status_code=status.HTTP_404_NOT_FOUND,
            content={"message": "商品不存在"}
        )
    
    return {"id": item_id, "name": "Item"}
```

## HTTPException

### 基本用法

```python
from fastapi import FastAPI, HTTPException, status

app = FastAPI()

@app.get("/items/{item_id}")
async def read_item(item_id: int):
    """
    使用 HTTPException 抛出错误
    """
    if item_id > 100:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="商品不存在"
        )
    
    return {"id": item_id, "name": "Item"}
```

### 自定义错误信息

```python
@app.get("/items/{item_id}")
async def read_item(item_id: int):
    if item_id < 0:
        raise HTTPException(
            status_code=400,
            detail={
                "code": "INVALID_ID",
                "message": "商品 ID 不能为负数",
                "item_id": item_id
            }
        )
    
    return {"id": item_id}
```

### 自定义响应头

```python
@app.get("/items/{item_id}")
async def read_item(item_id: int):
    if item_id > 100:
        raise HTTPException(
            status_code=404,
            detail="商品不存在",
            headers={"X-Error": "Item not found"}
        )
    
    return {"id": item_id}
```

## 自定义异常处理器

### 基本异常处理器

```python
from fastapi import FastAPI, Request, HTTPException
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError

app = FastAPI()

@app.exception_handler(HTTPException)
async def http_exception_handler(request: Request, exc: HTTPException):
    """自定义 HTTP 异常处理器"""
    return JSONResponse(
        status_code=exc.status_code,
        content={
            "code": exc.status_code,
            "message": exc.detail,
            "path": request.url.path
        }
    )
```

### 验证错误处理器

```python
from fastapi.exceptions import RequestValidationError

@app.exception_handler(RequestValidationError)
async def validation_exception_handler(request: Request, exc: RequestValidationError):
    """自定义验证错误处理器"""
    errors = []
    for error in exc.errors():
        errors.append({
            "field": ".".join(str(loc) for loc in error["loc"]),
            "message": error["msg"],
            "type": error["type"]
        })
    
    return JSONResponse(
        status_code=422,
        content={
            "code": 422,
            "message": "数据验证失败",
            "errors": errors
        }
    )
```

### 自定义异常

```python
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

app = FastAPI()

class ItemNotFoundException(Exception):
    """商品不存在异常"""
    def __init__(self, item_id: int):
        self.item_id = item_id

class InsufficientStockException(Exception):
    """库存不足异常"""
    def __init__(self, item_id: int, requested: int, available: int):
        self.item_id = item_id
        self.requested = requested
        self.available = available

@app.exception_handler(ItemNotFoundException)
async def item_not_found_handler(request: Request, exc: ItemNotFoundException):
    """商品不存在异常处理器"""
    return JSONResponse(
        status_code=404,
        content={
            "code": "ITEM_NOT_FOUND",
            "message": f"商品 ID {exc.item_id} 不存在"
        }
    )

@app.exception_handler(InsufficientStockException)
async def insufficient_stock_handler(request: Request, exc: InsufficientStockException):
    """库存不足异常处理器"""
    return JSONResponse(
        status_code=400,
        content={
            "code": "INSUFFICIENT_STOCK",
            "message": f"库存不足，请求 {exc.requested}，可用 {exc.available}"
        }
    )

@app.get("/items/{item_id}")
async def read_item(item_id: int):
    if item_id > 100:
        raise ItemNotFoundException(item_id)
    return {"id": item_id, "name": "Item"}

@app.post("/items/{item_id}/purchase")
async def purchase_item(item_id: int, quantity: int):
    available = 5
    if quantity > available:
        raise InsufficientStockException(item_id, quantity, available)
    return {"message": "购买成功"}
```

## 完整示例

```python
from fastapi import FastAPI, HTTPException, status, Request
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError
from pydantic import BaseModel
from typing import Optional
from datetime import datetime

app = FastAPI()

# 模型定义
class ErrorResponse(BaseModel):
    code: int
    message: str
    detail: Optional[dict] = None
    timestamp: str = Field(default_factory=lambda: datetime.now().isoformat())

# 自定义异常
class BusinessException(Exception):
    """业务异常基类"""
    def __init__(self, code: int, message: str, detail: dict = None):
        self.code = code
        self.message = message
        self.detail = detail

class NotFoundException(BusinessException):
    """资源不存在异常"""
    def __init__(self, resource: str, resource_id: int):
        super().__init__(
            code=404,
            message=f"{resource}不存在",
            detail={"resource": resource, "id": resource_id}
        )

class ConflictException(BusinessException):
    """资源冲突异常"""
    def __init__(self, message: str, detail: dict = None):
        super().__init__(code=409, message=message, detail=detail)

# 异常处理器
@app.exception_handler(BusinessException)
async def business_exception_handler(request: Request, exc: BusinessException):
    """业务异常处理器"""
    return JSONResponse(
        status_code=exc.code,
        content=ErrorResponse(
            code=exc.code,
            message=exc.message,
            detail=exc.detail
        ).model_dump()
    )

@app.exception_handler(RequestValidationError)
async def validation_exception_handler(request: Request, exc: RequestValidationError):
    """验证错误处理器"""
    return JSONResponse(
        status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
        content=ErrorResponse(
            code=422,
            message="请求参数验证失败",
            detail={"errors": exc.errors()}
        ).model_dump()
    )

# 模拟数据库
items_db = {
    1: {"id": 1, "name": "iPhone", "price": 5999},
    2: {"id": 2, "name": "MacBook", "price": 9999},
}

# 路由
@app.get("/items/{item_id}", status_code=status.HTTP_200_OK)
async def read_item(item_id: int):
    """获取商品"""
    if item_id not in items_db:
        raise NotFoundException("商品", item_id)
    return items_db[item_id]

@app.post("/items/", status_code=status.HTTP_201_CREATED)
async def create_item(item: dict):
    """创建商品"""
    item_id = item.get("id")
    if item_id in items_db:
        raise ConflictException(
            message="商品已存在",
            detail={"id": item_id}
        )
    items_db[item_id] = item
    return item

@app.delete("/items/{item_id}", status_code=status.HTTP_204_NO_CONTENT)
async def delete_item(item_id: int):
    """删除商品"""
    if item_id not in items_db:
        raise NotFoundException("商品", item_id)
    del items_db[item_id]
    return None
```

## 小结

状态码使用要点：

1. **默认状态码**：GET/POST 默认 200，需要手动设置 201
2. **status 模块**：使用 `status.HTTP_XXX` 常量更清晰
3. **HTTPException**：用于抛出 HTTP 错误
4. **自定义异常**：创建业务异常类统一处理
5. **异常处理器**：使用 `@app.exception_handler` 自定义响应格式

> 💡 **提示**：合理使用 HTTP 状态码可以让 API 更加规范，客户端可以根据状态码判断请求结果。
