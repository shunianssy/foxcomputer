# 4.4 数据库迁移

数据库迁移是管理数据库结构变更的重要工具。本章介绍如何使用 Alembic 进行数据库迁移。

## 安装 Alembic

```bash
pip install alembic
```

## 初始化 Alembic

```bash
# 在项目根目录执行
alembic init alembic
```

生成的目录结构：

```
project/
├── alembic/
│   ├── versions/        # 迁移文件目录
│   ├── env.py          # Alembic 配置
│   ├── script.py.mako  # 迁移模板
│   └── README
├── alembic.ini         # Alembic 配置文件
├── app/
│   ├── database.py
│   └── models.py
└── requirements.txt
```

## 配置 Alembic

### 修改 alembic.ini

```ini
# alembic.ini

[alembic]
# 迁移脚本目录
script_location = alembic

# 数据库连接 URL
# 方式一：直接配置
sqlalchemy.url = sqlite:///./app.db

# 方式二：从环境变量读取（推荐）
# sqlalchemy.url = driver://user:pass@localhost/dbname

[post_write_hooks]
# 迁移后自动格式化
hooks = black
black.type = console_scripts
black.entrypoint = black
black.options = -q

[loggers]
keys = root,sqlalchemy,alembic

[handlers]
keys = console

[formatters]
keys = generic
```

### 修改 env.py

```python
# alembic/env.py
from logging.config import fileConfig
from sqlalchemy import engine_from_config
from sqlalchemy import pool
from alembic import context
import sys
import os

# 添加项目路径
sys.path.insert(0, os.path.dirname(os.path.dirname(__file__)))

# 导入模型和数据库配置
from app.database import Base
from app.models import *  # 导入所有模型

# Alembic 配置对象
config = context.config

# 日志配置
if config.config_file_name is not None:
    fileConfig(config.config_file_name)

# 模型元数据
target_metadata = Base.metadata

# 从环境变量获取数据库 URL
# from dotenv import load_dotenv
# load_dotenv()
# db_url = os.getenv("DATABASE_URL")
# config.set_main_option("sqlalchemy.url", db_url)

def run_migrations_offline() -> None:
    """
    离线模式运行迁移
    
    不需要连接数据库，只生成 SQL 脚本
    """
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    """
    在线模式运行迁移
    
    需要连接数据库
    """
    connectable = engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection,
            target_metadata=target_metadata
        )

        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
```

## 创建迁移

### 自动生成迁移

```bash
# 根据模型变更自动生成迁移
alembic revision --autogenerate -m "Initial migration"

# 添加商品表
alembic revision --autogenerate -m "Add items table"

# 添加用户表
alembic revision --autogenerate -m "Add users table"
```

### 手动创建迁移

```bash
# 创建空白迁移文件
alembic revision -m "Add custom field"
```

生成的迁移文件：

```python
# alembic/versions/xxx_add_items_table.py
"""Add items table

Revision ID: xxx
Revises: 
Create Date: 2024-01-01 00:00:00.000000
"""
from alembic import op
import sqlalchemy as sa

# 迁移版本信息
revision = 'xxx'
down_revision = None
branch_labels = None
depends_on = None

def upgrade() -> None:
    """
    升级操作
    """
    # 创建表
    op.create_table(
        'items',
        sa.Column('id', sa.Integer(), nullable=False),
        sa.Column('name', sa.String(length=100), nullable=False),
        sa.Column('description', sa.String(length=500), nullable=True),
        sa.Column('price', sa.Float(), nullable=False),
        sa.Column('quantity', sa.Integer(), nullable=True),
        sa.Column('created_at', sa.DateTime(), nullable=True),
        sa.PrimaryKeyConstraint('id')
    )
    
    # 创建索引
    op.create_index(op.f('ix_items_name'), 'items', ['name'], unique=False)
    op.create_index(op.f('ix_items_id'), 'items', ['id'], unique=False)


def downgrade() -> None:
    """
    降级操作
    """
    # 删除索引
    op.drop_index(op.f('ix_items_id'), table_name='items')
    op.drop_index(op.f('ix_items_name'), table_name='items')
    
    # 删除表
    op.drop_table('items')
```

## 执行迁移

### 升级到最新版本

```bash
# 升级到最新版本
alembic upgrade head

# 升级到指定版本
alembic upgrade +1  # 升级一个版本
alembic upgrade xxx  # 升级到指定版本
```

### 降级版本

```bash
# 降级一个版本
alembic downgrade -1

# 降级到指定版本
alembic downgrade xxx

# 降级所有版本
alembic downgrade base
```

### 查看迁移状态

```bash
# 查看当前版本
alembic current

# 查看迁移历史
alembic history

# 查看详细信息
alembic history --verbose
```

## 常用操作

### 添加列

```python
def upgrade() -> None:
    # 添加列
    op.add_column('items', sa.Column('is_active', sa.Boolean(), default=True))
    
    # 添加带默认值的列
    op.add_column(
        'items',
        sa.Column(
            'status',
            sa.String(20),
            nullable=False,
            server_default='active'
        )
    )

def downgrade() -> None:
    # 删除列
    op.drop_column('items', 'is_active')
    op.drop_column('items', 'status')
```

### 修改列

```python
def upgrade() -> None:
    # 修改列类型
    op.alter_column(
        'items',
        'name',
        existing_type=sa.String(50),
        type_=sa.String(100),
        existing_nullable=False
    )
    
    # 修改列名
    op.alter_column('items', 'name', new_column_name='item_name')
    
    # 修改列约束
    op.alter_column(
        'items',
        'price',
        existing_type=sa.Float(),
        nullable=False
    )

def downgrade() -> None:
    op.alter_column(
        'items',
        'item_name',
        new_column_name='name'
    )
```

### 创建外键

```python
def upgrade() -> None:
    # 添加外键列
    op.add_column(
        'orders',
        sa.Column('user_id', sa.Integer(), nullable=False)
    )
    
    # 创建外键约束
    op.create_foreign_key(
        'fk_orders_user_id',
        'orders',
        'users',
        ['user_id'],
        ['id']
    )

def downgrade() -> None:
    op.drop_constraint('fk_orders_user_id', 'orders', type_='foreignkey')
    op.drop_column('orders', 'user_id')
```

### 创建索引

```python
def upgrade() -> None:
    # 创建普通索引
    op.create_index('ix_items_name', 'items', ['name'])
    
    # 创建唯一索引
    op.create_index('uq_items_name', 'items', ['name'], unique=True)
    
    # 创建复合索引
    op.create_index('ix_items_name_price', 'items', ['name', 'price'])

def downgrade() -> None:
    op.drop_index('ix_items_name_price', table_name='items')
    op.drop_index('uq_items_name', table_name='items')
    op.drop_index('ix_items_name', table_name='items')
```

### 数据迁移

```python
from sqlalchemy import table, column
from sqlalchemy import String, Integer

def upgrade() -> None:
    # 创建临时表对象
    items = table(
        'items',
        column('name', String),
        column('status', String)
    )
    
    # 批量更新数据
    op.execute(
        items.update()
        .where(items.c.name == 'old_name')
        .values(status='inactive')
    )
    
    # 批量插入数据
    op.execute(
        items.insert().values([
            {'name': 'Item 1', 'status': 'active'},
            {'name': 'Item 2', 'status': 'active'},
        ])
    )

def downgrade() -> None:
    # 回滚数据
    items = table('items', column('name', String))
    op.execute(
        items.delete().where(items.c.name.in_(['Item 1', 'Item 2']))
    )
```

## 完整示例

```python
# alembic/versions/001_initial.py
"""Initial migration

Revision ID: 001
Revises: 
Create Date: 2024-01-01 00:00:00.000000
"""
from alembic import op
import sqlalchemy as sa

revision = '001'
down_revision = None
branch_labels = None
depends_on = None

def upgrade() -> None:
    # 创建用户表
    op.create_table(
        'users',
        sa.Column('id', sa.Integer(), nullable=False),
        sa.Column('username', sa.String(50), nullable=False),
        sa.Column('email', sa.String(100), nullable=False),
        sa.Column('hashed_password', sa.String(255), nullable=False),
        sa.Column('is_active', sa.Boolean(), default=True),
        sa.Column('created_at', sa.DateTime(), server_default=sa.func.now()),
        sa.PrimaryKeyConstraint('id'),
        sa.UniqueConstraint('username'),
        sa.UniqueConstraint('email')
    )
    op.create_index('ix_users_id', 'users', ['id'])
    
    # 创建商品表
    op.create_table(
        'items',
        sa.Column('id', sa.Integer(), nullable=False),
        sa.Column('name', sa.String(100), nullable=False),
        sa.Column('description', sa.String(500)),
        sa.Column('price', sa.Float(), nullable=False),
        sa.Column('quantity', sa.Integer(), default=0),
        sa.Column('is_active', sa.Boolean(), default=True),
        sa.Column('created_at', sa.DateTime(), server_default=sa.func.now()),
        sa.Column('updated_at', sa.DateTime(), onupdate=sa.func.now()),
        sa.PrimaryKeyConstraint('id')
    )
    op.create_index('ix_items_id', 'items', ['id'])
    op.create_index('ix_items_name', 'items', ['name'])
    
    # 创建订单表
    op.create_table(
        'orders',
        sa.Column('id', sa.Integer(), nullable=False),
        sa.Column('user_id', sa.Integer(), nullable=False),
        sa.Column('total', sa.Float(), default=0),
        sa.Column('status', sa.String(20), default='pending'),
        sa.Column('created_at', sa.DateTime(), server_default=sa.func.now()),
        sa.ForeignKeyConstraint(['user_id'], ['users.id']),
        sa.PrimaryKeyConstraint('id')
    )
    op.create_index('ix_orders_id', 'orders', ['id'])
    op.create_index('ix_orders_user_id', 'orders', ['user_id'])


def downgrade() -> None:
    op.drop_index('ix_orders_user_id', table_name='orders')
    op.drop_index('ix_orders_id', table_name='orders')
    op.drop_table('orders')
    
    op.drop_index('ix_items_name', table_name='items')
    op.drop_index('ix_items_id', table_name='items')
    op.drop_table('items')
    
    op.drop_index('ix_users_id', table_name='users')
    op.drop_table('users')
```

## 小结

数据库迁移要点：

1. **初始化**：使用 `alembic init` 初始化迁移目录
2. **配置**：修改 `alembic.ini` 和 `env.py`
3. **生成迁移**：使用 `--autogenerate` 自动生成
4. **执行迁移**：使用 `upgrade` 和 `downgrade` 命令
5. **版本控制**：使用 `stamp` 标记数据库版本

> 💡 **提示**：在生产环境执行迁移前，务必先备份数据库，并在测试环境验证迁移脚本。
