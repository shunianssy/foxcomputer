# 3.2 Pydantic 模型

Pydantic 是 FastAPI 的核心依赖，用于数据验证和设置管理。本章将详细介绍 Pydantic 的使用方法。

## 基本模型

### 定义模型

```python
from pydantic import BaseModel

class User(BaseModel):
    """用户模型"""
    id: int
    name: str
    email: str
    age: int | None = None  # 可选字段
```

### 创建实例

```python
# 从字典创建
user = User(id=1, name="张三", email="zhangsan@example.com")

# 从字典解包创建
data = {"id": 1, "name": "张三", "email": "zhangsan@example.com"}
user = User(**data)

# 从 JSON 字符串创建
import json
json_data = '{"id": 1, "name": "张三", "email": "zhangsan@example.com"}'
user = User.model_validate_json(json_data)
```

### 模型方法

```python
# 转换为字典
user_dict = user.model_dump()
# {"id": 1, "name": "张三", "email": "zhangsan@example.com", "age": None}

# 转换为 JSON
user_json = user.model_dump_json()
# '{"id": 1, "name": "张三", "email": "zhangsan@example.com", "age": null}'

# 排除未设置的字段
user = User(id=1, name="张三", email="zhangsan@example.com")
user_dict = user.model_dump(exclude_unset=True)
# {"id": 1, "name": "张三", "email": "zhangsan@example.com"}

# 排除 None 值
user_dict = user.model_dump(exclude_none=True)
# {"id": 1, "name": "张三", "email": "zhangsan@example.com"}
```

## 字段验证

### 使用 Field

```python
from pydantic import BaseModel, Field
from typing import Annotated

class Item(BaseModel):
    # 字符串验证
    name: Annotated[str, Field(
        min_length=1,
        max_length=100,
        pattern=r"^[a-zA-Z0-9_]+$",  # 正则表达式
        description="商品名称",
        examples=["iPhone15"]
    )]
    
    # 数值验证
    price: Annotated[float, Field(
        gt=0,        # 大于 0
        le=100000,   # 小于等于 100000
        description="商品价格"
    )]
    
    quantity: Annotated[int, Field(
        ge=0,        # 大于等于 0
        default=0,
        description="库存数量"
    )]
    
    # 列表验证
    tags: Annotated[list[str], Field(
        min_length=1,   # 至少 1 个元素
        max_length=10,  # 最多 10 个元素
        default_factory=list
    )]
```

### 自定义验证器

```python
from pydantic import BaseModel, field_validator, model_validator
import re

class User(BaseModel):
    username: str
    password: str
    confirm_password: str
    
    @field_validator("username")
    @classmethod
    def validate_username(cls, v: str) -> str:
        """验证用户名"""
        if not re.match(r"^[a-zA-Z0-9_]{3,20}$", v):
            raise ValueError("用户名必须是 3-20 个字母、数字或下划线")
        return v
    
    @field_validator("password")
    @classmethod
    def validate_password(cls, v: str) -> str:
        """验证密码"""
        if len(v) < 8:
            raise ValueError("密码至少 8 个字符")
        if not re.search(r"[A-Z]", v):
            raise ValueError("密码必须包含大写字母")
        if not re.search(r"[a-z]", v):
            raise ValueError("密码必须包含小写字母")
        if not re.search(r"\d", v):
            raise ValueError("密码必须包含数字")
        return v
    
    @model_validator(mode="after")
    def validate_passwords_match(self):
        """验证两次密码一致"""
        if self.password != self.confirm_password:
            raise ValueError("两次密码不一致")
        return self
```

### 验证前后处理

```python
from pydantic import BaseModel, field_validator

class Item(BaseModel):
    name: str
    price: float
    
    @field_validator("name", mode="before")
    @classmethod
    def strip_name(cls, v):
        """验证前处理：去除首尾空格"""
        if isinstance(v, str):
            return v.strip()
        return v
    
    @field_validator("price", mode="after")
    @classmethod
    def round_price(cls, v: float) -> float:
        """验证后处理：四舍五入到两位小数"""
        return round(v, 2)
```

## 嵌套模型

```python
from pydantic import BaseModel
from typing import List, Optional

class Address(BaseModel):
    """地址模型"""

    }
)
```

## 模型配置

```python
from pydantic import BaseModel, ConfigDict

class Item(BaseModel):
    # 配置选项
    model_config = ConfigDict(
        # 允许从字段别名读取
        populate_by_name=True,
        
        # 验证赋值
        validate_assignment=True,
        
        # 允许额外字段
        extra="forbid",  # "ignore" 忽略, "allow" 允许, "forbid" 禁止
        
        # 字段别名生成器
        alias_generator=lambda x: x.replace("_", "-"),
        
        # JSON 编码
        json_encoders={
            datetime: lambda v: v.isoformat()
        }
    )
    
    name: str
    price: float
```

### 常用配置选项

| 选项 | 说明 |
|------|------|
| `populate_by_name` | 允许通过字段名或别名填充 |
| `validate_assignment` | 赋值时验证 |
| `extra` | 额外字段处理方式 |
| `alias_generator` | 别名生成器 |
| `str_strip_whitespace` | 自动去除字符串首尾空格 |
| `str_min_length` | 字符串最小长度 |
| `str_max_length` | 字符串最大长度 |

## 动态模型

### 使用 TypeAdapter

```python
from pydantic import TypeAdapter

# 验证简单类型
adapter = TypeAdapter(list[int])
result = adapter.validate_python([1, 2, 3])  # [1, 2, 3]

# 验证复杂类型
adapter = TypeAdapter(dict[str, int])
result = adapter.validate_python({"a": 1, "b": 2})
```

### 创建动态模型

```python
from pydantic import create_model

# 动态创建模型
DynamicModel = create_model(
    "DynamicModel",
    name=(str, ...),
    age=(int, 18),
    __config__=ConfigDict(extra="forbid")
)

instance = DynamicModel(name="张三")
```

## 模型序列化

### JSON 序列化

```python
from pydantic import BaseModel
from datetime import datetime

class Item(BaseModel):
    name: str
    created_at: datetime
    
    model_config = ConfigDict(
        json_encoders={
            datetime: lambda v: v.strftime("%Y-%m-%d %H:%M:%S")
        }
    )

item = Item(name="iPhone", created_at=datetime.now())

# 自定义序列化
json_str = item.model_dump_json()
```

### 自定义序列化器

```python
from pydantic import BaseModel, field_serializer

class Item(BaseModel):
    name: str
    price: float
    
    @field_serializer("price")
    def serialize_price(self, value: float) -> str:
        """自定义价格序列化"""
        return f"¥{value:.2f}"

item = Item(name="iPhone", price=5999.0)
print(item.model_dump())  # {"name": "iPhone", "price": "¥5999.00"}
```

## 实用示例

```python
from pydantic import BaseModel, Field, field_validator, model_validator, ConfigDict
from typing import Annotated, Optional, List
from datetime import datetime
from enum import Enum
import re

class UserRole(str, Enum):
    """用户角色枚举"""
    ADMIN = "admin"
    USER = "user"
    GUEST = "guest"

class Address(BaseModel):
    """地址模型"""
    province: Annotated[str, Field(min_length=2, description="省份")]
    city: Annotated[str, Field(min_length=2, description="城市")]
    detail: Annotated[str, Field(min_length=5, description="详细地址")]
    zip_code: Annotated[str, Field(pattern=r"^\d{6}$", description="邮编")]

class UserBase(BaseModel):
    """用户基础模型"""
    model_config = ConfigDict(
        str_strip_whitespace=True,
        validate_assignment=True
    )
    
    username: Annotated[str, Field(
        min_length=3,
        max_length=20,
        pattern=r"^[a-zA-Z0-9_]+$",
        description="用户名"
    )]
    email: Annotated[str, Field(description="邮箱")]
    role: UserRole = UserRole.USER

class UserCreate(UserBase):
    """创建用户模型"""
    password: Annotated[str, Field(min_length=8, description="密码")]
    confirm_password: str
    addresses: List[Address] = []
    
    @field_validator("email")
    @classmethod
    def validate_email(cls, v: str) -> str:
        if not re.match(r"^[\w\.-]+@[\w\.-]+\.\w+$", v):
            raise ValueError("邮箱格式不正确")
        return v
    
    @field_validator("password")
    @classmethod
    def validate_password(cls, v: str) -> str:
        if not re.search(r"[A-Z]", v):
            raise ValueError("密码必须包含大写字母")
        if not re.search(r"[a-z]", v):
            raise ValueError("密码必须包含小写字母")
        if not re.search(r"\d", v):
            raise ValueError("密码必须包含数字")
        return v
    
    @model_validator(mode="after")
    def validate_passwords_match(self):
        if self.password != self.confirm_password:
            raise ValueError("两次密码不一致")
        return self

class User(UserBase):
    """用户响应模型"""
    id: int
    addresses: List[Address] = []
    created_at: datetime
    is_active: bool = True
    
    @field_serializer("created_at")
    def serialize_datetime(self, dt: datetime) -> str:
        return dt.strftime("%Y-%m-%d %H:%M:%S")
```

## 小结

Pydantic 模型要点：

1. **类型验证**：自动进行类型检查和转换
2. **字段验证**：使用 `Field` 进行详细验证
3. **自定义验证器**：使用 `@field_validator` 和 `@model_validator`
4. **嵌套模型**：支持复杂的数据结构
5. **模型配置**：使用 `model_config` 配置行为

> 💡 **提示**：Pydantic 是 FastAPI 的核心，熟练掌握 Pydantic 可以大大提高开发效率和代码质量。
