# 6.3 WebSocket

WebSocket 提供了全双工通信能力，适合实时应用。FastAPI 原生支持 WebSocket。

## 基本用法

### 简单 WebSocket

```python
from fastapi import FastAPI, WebSocket

app = FastAPI()

@app.websocket("/ws")
async def websocket_endpoint(websocket: WebSocket):
 """
 WebSocket 端点
 
 客户端连接后可以双向通信
 """
 await websocket.accept()
 
 while True:
 # 接收消息
 data = await websocket.receive_text()
 
 # 发送消息
 await websocket.send_text(f"收到: {data}")
```

### 客户端示例

```javascript
// JavaScript 客户端
const ws = new WebSocket("ws://localhost:8000/ws");

ws.onopen = function() {
 console.log("连接已建立");
 ws.send("Hello, Server!");
};

ws.onmessage = function(event) {
 console.log("收到消息:", event.data);
};

ws.onclose = function() {
 console.log("连接已关闭");
};
```

## 连接管理

### 连接管理器

```python
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from typing import List

app = FastAPI()

class ConnectionManager:
 """WebSocket 连接管理器"""
 
 def __init__(self):
 self.active_connections: List[WebSocket] = []
 
 async def connect(self, websocket: WebSocket):
 """接受新连接"""
 await websocket.accept()
 self.active_connections.append(websocket)
 
 def disconnect(self, websocket: WebSocket):
 """断开连接"""
 self.active_connections.remove(websocket)
 
 async def send_personal_message(self, message: str, websocket: WebSocket):
 """发送个人消息"""
 await websocket.send_text(message)
 
 async def broadcast(self, message: str):
 """广播消息"""
 for connection in self.active_connections:
 await connection.send_text(message)

manager = ConnectionManager()

@app.websocket("/ws/{client_id}")
async def websocket_endpoint(websocket: WebSocket, client_id: int):
 """带客户端 ID 的 WebSocket"""
 await manager.connect(websocket)
 
 try:
 while True:
 data = await websocket.receive_text()
 
 # 发送个人消息
 await manager.send_personal_message(
 f"你发送了: {data}",
 websocket
 )
 
 # 广播给其他客户端
 await manager.broadcast(
 f"客户端 #{client_id} 说: {data}"
 )
 
 except WebSocketDisconnect:
 manager.disconnect(websocket)
 await manager.broadcast(f"客户端 #{client_id} 离开了聊天")
```

## 聊天室示例

```python
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from pydantic import BaseModel
from typing import Dict, List
from datetime import datetime
import json

app = FastAPI()

class Message(BaseModel):
 """消息模型"""
 username: str
 content: str
 timestamp: str = None
 
 def __init__(self, **data):
 super().__init__(**data)
 if not self.timestamp:
 self.timestamp = datetime.now().isoformat()

class ChatRoom:
 """聊天室"""
 
 def __init__(self, name: str):
 self.name = name
 self.connections: Dict[str, WebSocket] = {}
 
 async def connect(self, username: str, websocket: WebSocket):
 """用户加入"""
 await websocket.accept()
 self.connections[username] = websocket
 
 # 广播加入消息
 await self.broadcast({
 "type": "join",
 "username": username,
 "message": f"{username} 加入了聊天室"
 })
 
 def disconnect(self, username: str):
 """用户离开"""
 if username in self.connections:
 del self.connections[username]
 
 async def send_to_user(self, username: str, message: dict):
 """发送给特定用户"""
 if username in self.connections:
 await self.connections[username].send_json(message)
 
 async def broadcast(self, message: dict, exclude: str = None):
 """广播消息"""
 for username, connection in self.connections.items():
 if username != exclude:
 await connection.send_json(message)

class ChatManager:
 """聊天管理器"""
 
 def __init__(self):
 self.rooms: Dict[str, ChatRoom] = {}
 
 def get_room(self, room_name: str) -> ChatRoom:
 """获取或创建聊天室"""
 if room_name not in self.rooms:
 self.rooms[room_name] = ChatRoom(room_name)
 return self.rooms[room_name]

chat_manager = ChatManager()

@app.websocket("/chat/{room_name}/{username}")
async def chat_endpoint(
 websocket: WebSocket,
 room_name: str,
 username: str
):
 """聊天室 WebSocket"""
 room = chat_manager.get_room(room_name)
 
 await room.connect(username, websocket)
 
 try:
 while True:
 # 接收消息
 data = await websocket.receive_text()
 
 # 创建消息
 message = Message(username=username, content=data)
 
 # 广播消息
 await room.broadcast({
 "type": "message",
 "username": message.username,
 "content": message.content,
 "timestamp": message.timestamp
 })
 
 except WebSocketDisconnect:
 room.disconnect(username)
 await room.broadcast({
 "type": "leave",
 "username": username,
 "message": f"{username} 离开了聊天室"
 })
```

## 认证 WebSocket

```python
from fastapi import FastAPI, WebSocket, WebSocketDisconnect, Query, HTTPException
from jose import jwt, JWTError

app = FastAPI()

SECRET_KEY = "your-secret-key"

async def get_user_from_token(token: str):
 """从令牌获取用户"""
 try:
 payload = jwt.decode(token, SECRET_KEY, algorithms=["HS256"])
 return payload.get("sub")
 except JWTError:
 return None

@app.websocket("/ws-auth")
async def websocket_auth(
 websocket: WebSocket,
 token: str = Query(...)
):
 """需要认证的 WebSocket"""
 
 # 验证令牌
 username = await get_user_from_token(token)
 if not username:
 await websocket.close(code=4001, reason="认证失败")
 return
 
 await websocket.accept()
 
 try:
 while True:
 data = await websocket.receive_text()
 await websocket.send_text(f"[{username}] {data}")
 
 except WebSocketDisconnect:
 print(f"{username} 断开连接")
```

## 心跳检测

```python
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
import asyncio

app = FastAPI()

@app.websocket("/ws-heartbeat")
async def websocket_with_heartbeat(websocket: WebSocket):
 """带心跳检测的 WebSocket"""
 await websocket.accept()
 
 # 心跳任务
 async def heartbeat():
 while True:
 try:
 await websocket.send_json({"type": "ping"})
 await asyncio.sleep(30) # 每 30 秒发送心跳
 except:
 break
 
 # 启动心跳任务
 heartbeat_task = asyncio.create_task(heartbeat())
 
 try:
 while True:
 data = await websocket.receive_json()
 
 if data.get("type") == "pong":
 # 收到心跳响应
 continue
 
 # 处理其他消息
 await websocket.send_json({
 "type": "message",
 "content": f"收到: {data}"
 })
 
 except WebSocketDisconnect:
 heartbeat_task.cancel()
 print("客户端断开连接")
```

## 完整示例

```python
from fastapi import FastAPI, WebSocket, WebSocketDisconnect, Query
from pydantic import BaseModel
from typing import Dict, List, Optional
from datetime import datetime
import json
import asyncio

app = FastAPI(title="WebSocket 示例")

# ============ 模型 ============
class ChatMessage(BaseModel):
 type: str # message, join, leave, system
 username: str
 content: Optional[str] = None
 timestamp: str = None
 
 model_config = {"json_schema_extra": {
 "examples": [{
 "type": "message",
 "username": "张三",
 "content": "大家好！",
 "timestamp": "2024-01-01T12:00:00"
 }]
 }}

# ============ 连接管理 ============
class ConnectionManager:
 def __init__(self):
 self.active_connections: Dict[str, WebSocket] = {}
 
 async def connect(self, user_id: str, websocket: WebSocket):
 await websocket.accept()
 self.active_connections[user_id] = websocket
 
 def disconnect(self, user_id: str):
 if user_id in self.active_connections:
 del self.active_connections[user_id]
 
 async def send_to(self, user_id: str, message: dict):
 if user_id in self.active_connections:
 await self.active_connections[user_id].send_json(message)
 
 async def broadcast(self, message: dict, exclude: str = None):
 for user_id, connection in self.active_connections.items():
 if user_id != exclude:
 try:
 await connection.send_json(message)
 except:
 pass
 
 def get_online_users(self) -> List[str]:
 return list(self.active_connections.keys())

manager = ConnectionManager()

# ============ 路由 ============
@app.websocket("/ws/{user_id}")
async def websocket_endpoint(
 websocket: WebSocket,
 user_id: str,
 token: str = Query(...)
):
 """
 WebSocket 端点
 
 - user_id: 用户标识
 - token: 认证令牌
 """
 # 简单验证
 if not token:
 await websocket.close(code=4001, reason="缺少认证令牌")
 return
 
 await manager.connect(user_id, websocket)
 
 # 广播用户加入
 await manager.broadcast({
 "type": "join",
 "username": user_id,
 "content": f"{user_id} 加入了聊天",
 "timestamp": datetime.now().isoformat()
 })
 
 # 发送在线用户列表
 await manager.send_to(user_id, {
 "type": "system",
 "content": "欢迎加入聊天室",
 "online_users": manager.get_online_users(),
 "timestamp": datetime.now().isoformat()
 })
 
 try:
 while True:
 # 接收消息
 data = await websocket.receive_text()
 
 # 广播消息
 message = {
 "type": "message",
 "username": user_id,
 "content": data,
 "timestamp": datetime.now().isoformat()
 }
 await manager.broadcast(message)
 
 except WebSocketDisconnect:
 manager.disconnect(user_id)
 
 # 广播用户离开
 await manager.broadcast({
 "type": "leave",
 "username": user_id,
 "content": f"{user_id} 离开了聊天",
 "timestamp": datetime.now().isoformat()
 })

# HTTP 接口获取在线用户
@app.get("/online-users")
async def get_online_users():
 """获取在线用户列表"""
 return {"users": manager.get_online_users()}
```

## 小结

WebSocket 要点：

1. **基本用法**：使用 `@app.websocket()` 装饰器
2. **消息类型**：支持文本、JSON、二进制
3. **连接管理**：管理多个客户端连接
4. **广播消息**：向所有连接发送消息
5. **认证**：通过查询参数或首条消息验证

> **提示**：WebSocket 适合实时聊天、通知推送、实时数据等场景。
