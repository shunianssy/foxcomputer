# 4.2 异步数据库

FastAPI 原生支持异步编程，配合异步数据库驱动可以显著提高性能。本章介绍如何使用异步数据库操作。

## 安装依赖

```bash
# 异步数据库驱动
pip install aiosqlite # SQLite
pip install asyncpg # PostgreSQL
pip install aiomysql # MySQL

# SQLAlchemy 异步支持
pip install sqlalchemy[asyncio]
```

## 异步数据库配置

### SQLite 异步配置

```python
# database.py
from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession
from sqlalchemy.orm import sessionmaker
from sqlalchemy.ext.declarative import declarative_base

# 异步数据库 URL
SQLALCHEMY_DATABASE_URL = "sqlite+aiosqlite:///./app.db"

# 创建异步引擎
engine = create_async_engine(
 SQLALCHEMY_DATABASE_URL,
 echo=True
)

# 创建异步会话工厂
AsyncSessionLocal = sessionmaker(
 engine,
 class_=AsyncSession,
 expire_on_commit=False
)

Base = declarative_base()
```

### PostgreSQL 异步配置

```python
# PostgreSQL 异步 URL
SQLALCHEMY_DATABASE_URL = "postgresql+asyncpg://user:password@localhost/dbname"
```

## 异步模型定义

```python
# models.py
from sqlalchemy import Column, Integer, String, Float, DateTime
from datetime import datetime
from database import Base

class Item(Base):
 __tablename__ = "items"
 
 id = Column(Integer, primary_key=True, index=True)
 name = Column(String(100), nullable=False, index=True)
 description = Column(String(500), nullable=True)
 price = Column(Float, nullable=False)
 quantity = Column(Integer, default=0)
 created_at = Column(DateTime, default=datetime.utcnow)
```

## 异步数据库会话

```python
# main.py
from fastapi import FastAPI, Depends
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select

from database import AsyncSessionLocal, engine, Base
from models import Item as ItemModel

app = FastAPI()

# 创建表
@app.on_event("startup")
async def startup():
 async with engine.begin() as conn:
 await conn.run_sync(Base.metadata.create_all)

# 异步会话依赖
async def get_db():
 async with AsyncSessionLocal() as session:
 try:
 yield session
 finally:
 await session.close()
```

## 异步 CRUD 操作

```python
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

@app.get("/items/")
async def read_items(
 skip: int = 0,
 limit: int = 10,
 db: AsyncSession = Depends(get_db)
):
 """获取商品列表"""
 stmt = select(ItemModel).offset(skip).limit(limit)
 result = await db.execute(stmt)
 items = result.scalars().all()
 return items

@app.get("/items/{item_id}")
async def read_item(item_id: int, db: AsyncSession = Depends(get_db)):
 """获取单个商品"""
 stmt = select(ItemModel).where(ItemModel.id == item_id)
 result = await db.execute(stmt)
 item = result.scalar_one_or_none()
 
 if not item:
 raise HTTPException(status_code=404, detail="商品不存在")
 return item

@app.post("/items/", status_code=201)

 db_item = ItemModel(**item.model_dump())

 await db.refresh(db_item)
 return db_item

@app.put("/items/{item_id}")

 db: AsyncSession = Depends(get_db)
):
 """更新商品"""
 stmt = select(ItemModel).where(ItemModel.id == item_id)
 result = await db.execute(stmt)
 db_item = result.scalar_one_or_none()

 raise HTTPException(status_code=404, detail="商品不存在")
 
 update_data = item.model_dump(exclude_unset=True)
 for key, value in update_data.items():
 setattr(db_item, key, value)
 
 await db.commit()
 await db.refresh(db_item)
 return db_item

@app.delete("/items/{item_id}")
async def delete_item(item_id: int, db: AsyncSession = Depends(get_db)):
 """删除商品"""
 stmt = select(ItemModel).where(ItemModel.id == item_id)
 result = await db.execute(stmt)
 db_item = result.scalar_one_or_none()
 
 if not db_item:
 raise HTTPException(status_code=404, detail="商品不存在")
 
 await db.delete(db_item)
 await db.commit()
 return {"message": "删除成功"}
```

## 小结

异步数据库要点：

1. **异步驱动**：使用 `aiosqlite`、`asyncpg`、`aiomysql` 等
2. **异步引擎**：使用 `create_async_engine` 创建
3. **异步会话**：使用 `AsyncSession` 管理会话
4. **异步查询**：使用 `select()` 和 `await db.execute()`
5. **异步提交**：使用 `await db.commit()`

> **提示**：异步数据库操作可以显著提高高并发场景下的性能。
