# 4.1 SQLAlchemy 集成

SQLAlchemy 是 Python 最流行的 ORM 框架，与 FastAPI 配合使用可以轻松实现数据库操作。本章介绍如何在 FastAPI 中集成 SQLAlchemy。

## 安装依赖

```bash
pip install sqlalchemy
# 数据库驱动（根据需要选择）
pip install psycopg2-binary  # PostgreSQL
pip install pymysql          # MySQL
pip install aiosqlite        # SQLite 异步
```

## 数据库配置

### 基本配置

```python
# database.py
from sqlalchemy import create_engine
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker

# 数据库连接 URL
# SQLite
SQLALCHEMY_DATABASE_URL = "sqlite:///./app.db"

# PostgreSQL
# SQLALCHEMY_DATABASE_URL = "postgresql://user:password@localhost/dbname"

# MySQL
# SQLALCHEMY_DATABASE_URL = "mysql+pymysql://user:password@localhost/dbname"

# 创建引擎
engine = create_engine(
    SQLALCHEMY_DATABASE_URL,
    connect_args={"check_same_thread": False}  # SQLite 需要
)

# 创建会话工厂
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

# 创建基类
Base = declarative_base()
```

### 使用环境变量

```python
# database.py
import os
```

## 定义模型

```python
# models.py
from sqlalchemy import Column, Integer, String, Float, DateTime, ForeignKey
from sqlalchemy.orm import relationship
from datetime import datetime
from database import Base

class Item(Base):
    """商品模型"""
    __tablename__ = "items"
    
    id = Column(Integer, primary_key=True, index=True)
    name = Column(String(100), nullable=False, index=True)
    description = Column(String(500), nullable=True)
    price = Column(Float, nullable=False)
    quantity = Column(Integer, default=0)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    
    # 关联关系
    orders = relationship("OrderItem", back_populates="item")

class User(Base):
    """用户模型"""
    __tablename__ = "users"
    
    id = Column(Integer, primary_key=True, index=True)
    username = Column(String(50), unique=True, index=True, nullable=False)
    email = Column(String(100), unique=True, index=True, nullable=False)
    hashed_password = Column(String(255), nullable=False)
    is_active = Column(Integer, default=1)
    created_at = Column(DateTime, default=datetime.utcnow)
    
    # 关联关系
    orders = relationship("Order", back_populates="user")

class Order(Base):
    """订单模型"""
    __tablename__ = "orders"
    
    id = Column(Integer, primary_key=True, index=True)
    user_id = Column(Integer, ForeignKey("users.id"), nullable=False)
    total = Column(Float, default=0)
    status = Column(String(20), default="pending")
    created_at = Column(DateTime, default=datetime.utcnow)
    
    # 关联关系
    user = relationship("User", back_populates="orders")
    items = relationship("OrderItem", back_populates="order")

class OrderItem(Base):
    """订单商品关联模型"""
    __tablename__ = "order_items"
    
    id = Column(Integer, primary_key=True, index=True)
    order_id = Column(Integer, ForeignKey("orders.id"), nullable=False)
    item_id = Column(Integer, ForeignKey("items.id"), nullable=False)
    quantity = Column(Integer, default=1)
    price = Column(Float, nullable=False)
    
    # 关联关系
    order = relationship("Order", back_populates="items")
    item = relationship("Item", back_populates="orders")
```

## Pydantic 模型

```python
# schemas.py
from pydantic import BaseModel, Field
from typing import Optional, List
from datetime import datetime

# ============ 商品模型 ============
class ItemBase(BaseModel):
    name: str = Field(..., min_length=1, max_length=100)
    description: Optional[str] = Field(None, max_length=500)
    price: float = Field(..., gt=0)
    quantity: int = Field(default=0, ge=0)

class ItemCreate(ItemBase):
    pass

class ItemUpdate(BaseModel):
    name: Optional[str] = Field(None, min_length=1, max_length=100)
    description: Optional[str] = None
    price: Optional[float] = Field(None, gt=0)
    quantity: Optional[int] = Field(None, ge=0)

class Item(ItemBase):
    id: int
    created_at: datetime
    updated_at: Optional[datetime] = None
    
    model_config = {"from_attributes": True}

# ============ 用户模型 ============
class UserBase(BaseModel):
    username: str = Field(..., min_length=3, max_length=50)
    email: str

class UserCreate(UserBase):
    password: str = Field(..., min_length=8)

class User(UserBase):
    id: int
    is_active: bool
    created_at: datetime
    
    model_config = {"from_attributes": True}

# ============ 订单模型 ============
class OrderItemBase(BaseModel):
    item_id: int
    quantity: int = Field(default=1, gt=0)

class OrderCreate(BaseModel):
    items: List[OrderItemBase]

class OrderItemResponse(BaseModel):
    item_id: int
    item_name: str
    quantity: int
    price: float
    
    model_config = {"from_attributes": True}

class Order(BaseModel):
    id: int
    user_id: int
    total: float
    status: str
    created_at: datetime
    items: List[OrderItemResponse]
    
    model_config = {"from_attributes": True}
```

## 数据库会话

### 依赖注入方式

```python
# main.py
from fastapi import FastAPI, Depends, HTTPException
from sqlalchemy.orm import Session
from typing import List

from database import SessionLocal, engine, Base
from models import Item as ItemModel
from schemas import Item, ItemCreate, ItemUpdate

# 创建表
Base.metadata.create_all(bind=engine)

app = FastAPI()

# 数据库会话依赖
def get_db():
    """
    获取数据库会话
    
    使用依赖注入确保会话正确关闭
    """
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

# ============ 路由 ============
@app.get("/items/", response_model=List[Item])
async def read_items(
    skip: int = 0,
    limit: int = 10,
    db: Session = Depends(get_db)
):
    """获取商品列表"""
    items = db.query(ItemModel).offset(skip).limit(limit).all()
    return items

@app.get("/items/{item_id}", response_model=Item)
async def read_item(item_id: int, db: Session = Depends(get_db)):
    """获取单个商品"""
    item = db.query(ItemModel).filter(ItemModel.id == item_id).first()
    if not item:
        raise HTTPException(status_code=404, detail="商品不存在")
    return item

@app.post("/items/", response_model=Item, status_code=201)
async def create_item(item: ItemCreate, db: Session = Depends(get_db)):
    """创建商品"""
    db_item = ItemModel(**item.model_dump())
    db.add(db_item)
    db.commit()
    db.refresh(db_item)
    return db_item

@app.put("/items/{item_id}", response_model=Item)
async def update_item(
    item_id: int,
    item: ItemUpdate,
    db: Session = Depends(get_db)
):
    """更新商品"""
    db_item = db.query(ItemModel).filter(ItemModel.id == item_id).first()
    if not db_item:
        raise HTTPException(status_code=404, detail="商品不存在")
    
    # 只更新提供的字段
    update_data = item.model_dump(exclude_unset=True)
    for key, value in update_data.items():
        setattr(db_item, key, value)
    
    db.commit()
    db.refresh(db_item)
    return db_item

@app.delete("/items/{item_id}")
async def delete_item(item_id: int, db: Session = Depends(get_db)):
    """删除商品"""
    db_item = db.query(ItemModel).filter(ItemModel.id == item_id).first()
    if not db_item:
        raise HTTPException(status_code=404, detail="商品不存在")
    
    db.delete(db_item)
    db.commit()
    return {"message": "删除成功"}
```

## CRUD 操作封装

```python
# crud.py
from sqlalchemy.orm import Session
from typing import List, Optional
from models import Item as ItemModel
from schemas import ItemCreate, ItemUpdate

class ItemCRUD:
    """商品 CRUD 操作"""
    
    @staticmethod
    def get(db: Session, item_id: int) -> Optional[ItemModel]:
        """获取单个商品"""
        return db.query(ItemModel).filter(ItemModel.id == item_id).first()
    
    @staticmethod
    def get_list(
        db: Session,
        skip: int = 0,
        limit: int = 10,
        name: str = None
    ) -> List[ItemModel]:
        """获取商品列表"""
        query = db.query(ItemModel)
        if name:
            query = query.filter(ItemModel.name.ilike(f"%{name}%"))
        return query.offset(skip).limit(limit).all()
    
    @staticmethod
    def create(db: Session, item: ItemCreate) -> ItemModel:
        """创建商品"""
        db_item = ItemModel(**item.model_dump())
        db.add(db_item)
        db.commit()
        db.refresh(db_item)
        return db_item
    
    @staticmethod
    def update(
        db: Session,
        item_id: int,
        item: ItemUpdate
    ) -> Optional[ItemModel]:
        """更新商品"""
        db_item = db.query(ItemModel).filter(ItemModel.id == item_id).first()
        if not db_item:
            return None
        
        update_data = item.model_dump(exclude_unset=True)
        for key, value in update_data.items():
            setattr(db_item, key, value)
        
        db.commit()
        db.refresh(db_item)
        return db_item
    
    @staticmethod
    def delete(db: Session, item_id: int) -> bool:
        """删除商品"""
        db_item = db.query(ItemModel).filter(ItemModel.id == item_id).first()
        if not db_item:
            return False
        
        db.delete(db_item)
        db.commit()
        return True

# 使用 CRUD
@app.get("/items/{item_id}", response_model=Item)
async def read_item(item_id: int, db: Session = Depends(get_db)):
    item = ItemCRUD.get(db, item_id)
    if not item:
        raise HTTPException(status_code=404, detail="商品不存在")
    return item
```

## 数据库迁移

使用 Alembic 进行数据库迁移：

```bash
# 安装 Alembic
pip install alembic

# 初始化
alembic init alembic

# 配置 alembic.ini
# sqlalchemy.url = sqlite:///./app.db

# 自动生成迁移
alembic revision --autogenerate -m "Initial migration"

# 执行迁移
alembic upgrade head
```

## 小结

SQLAlchemy 集成要点：

1. **数据库配置**：使用 `create_engine` 创建引擎
2. **模型定义**：继承 `Base` 类定义 ORM 模型
3. **会话管理**：使用依赖注入管理数据库会话
4. **CRUD 封装**：封装常用数据库操作
5. **数据迁移**：使用 Alembic 管理数据库变更

> 💡 **提示**：使用依赖注入管理数据库会话可以确保会话正确关闭，避免资源泄漏。
