# 3.4 错误处理

良好的错误处理是 API 设计的重要组成部分。本章将介绍 FastAPI 中的错误处理机制和最佳实践。

## HTTPException

### 基本用法

```python
from fastapi import FastAPI, HTTPException

app = FastAPI()

@app.get("/items/{item_id}")
async def read_item(item_id: int):
 """
 使用 HTTPException 抛出错误
 
 FastAPI 会自动捕获并转换为 HTTP 响应
 """
 if item_id > 100:
 raise HTTPException(
 status_code=404,
 detail="商品不存在"
 )
 return {"id": item_id}
```

### 自定义错误详情

```python
@app.get("/items/{item_id}")
async def read_item(item_id: int):
 if item_id < 0:
 raise HTTPException(
 status_code=400,
 detail={
 "code": "INVALID_ID",
 "message": "商品 ID 不能为负数",
 "input": item_id
 }
 )
 return {"id": item_id}
```

### 添加响应头

```python
@app.get("/items/{item_id}")
async def read_item(item_id: int):
 if item_id > 100:
 raise HTTPException(
 status_code=404,
 detail="商品不存在",
 headers={
 "X-Error-Code": "ITEM_NOT_FOUND",
 "X-Request-ID": "12345"
 }
 )
 return {"id": item_id}
```

## 自定义异常

### 定义异常类

```python
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

app = FastAPI()

class AppException(Exception):
 """应用异常基类"""
 def __init__(self, code: int, message: str, detail: dict = None):
 self.code = code
 self.message = message
 self.detail = detail or {}

class NotFoundException(AppException):
 """资源不存在异常"""
 def __init__(self, resource: str, resource_id: int):
 super().__init__(
 code=404,
 message=f"{resource}不存在",
 detail={"resource": resource, "id": resource_id}
 )

class ValidationException(AppException):
 """验证异常"""
 def __init__(self, field: str, message: str):
 super().__init__(
 code=400,
 message="数据验证失败",
 detail={"field": field, "reason": message}
 )

class AuthenticationException(AppException):
 """认证异常"""
 def __init__(self, message: str = "认证失败"):
 super().__init__(code=401, message=message)

class AuthorizationException(AppException):
 """授权异常"""
 def __init__(self, message: str = "无权限访问"):
 super().__init__(code=403, message=message)
```

### 注册异常处理器

```python
@app.exception_handler(AppException)
async def app_exception_handler(request: Request, exc: AppException):
 """应用异常统一处理器"""
 return JSONResponse(
 status_code=exc.code,
 content={
 "success": False,
 "code": exc.code,
 "message": exc.message,
 "detail": exc.detail,
 "path": str(request.url.path)
 }
 )

# 使用示例
@app.get("/items/{item_id}")
async def read_item(item_id: int):
 if item_id > 100:
 raise NotFoundException("商品", item_id)
 return {"id": item_id}
```

## 验证错误处理

### 默认验证错误

FastAPI 默认返回 422 错误：

```json
{
 "detail": [
 {
 "loc": ["body", "name"],
 "msg": "field required",
 "type": "value_error.missing"
 }
 ]
}
```

### 自定义验证错误格式

```python
from fastapi import FastAPI, Request
from fastapi.exceptions import RequestValidationError
from fastapi.responses import JSONResponse

app = FastAPI()

@app.exception_handler(RequestValidationError)
async def validation_exception_handler(request: Request, exc: RequestValidationError):
 """
 自定义验证错误响应格式
 """
 errors = []
 for error in exc.errors():
 field = ".".join(str(loc) for loc in error["loc"])
 errors.append({
 "field": field,
 "message": error["msg"],
 "type": error["type"]
 })
 
 return JSONResponse(
 status_code=422,
 content={
 "success": False,
 "code": 422,
 "message": "请求参数验证失败",
 "errors": errors
 }
 )
```

## 全局异常处理

### 捕获所有异常

```python
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
import logging

app = FastAPI()
logger = logging.getLogger(__name__)

@app.exception_handler(Exception)
async def global_exception_handler(request: Request, exc: Exception):
 """
 全局异常处理器
 
 捕获所有未处理的异常
 """
 # 记录错误日志
 logger.error(f"Unhandled exception: {exc}", exc_info=True)
 
 return JSONResponse(
 status_code=500,
 content={
 "success": False,
 "code": 500,
 "message": "服务器内部错误",
 "detail": str(exc) if app.debug else None
 }
 )
```

### 捕获特定异常

```python
from sqlalchemy.exc import IntegrityError
from redis.exceptions import RedisError

@app.exception_handler(IntegrityError)
async def integrity_error_handler(request: Request, exc: IntegrityError):
 """数据库完整性错误"""
 return JSONResponse(
 status_code=409,
 content={
 "success": False,
 "code": 409,
 "message": "数据冲突，可能已存在相同记录"
 }
 )

@app.exception_handler(RedisError)
async def redis_error_handler(request: Request, exc: RedisError):
 """Redis 错误"""
 logger.error(f"Redis error: {exc}")
 return JSONResponse(
 status_code=503,
 content={
 "success": False,
 "code": 503,
 "message": "服务暂时不可用，请稍后重试"
 }
 )
```

## 错误响应模型

### 定义统一响应格式

```python
from pydantic import BaseModel
from typing import Generic, TypeVar, Optional
from datetime import datetime

T = TypeVar("T")

class ResponseModel(BaseModel, Generic[T]):
 """统一响应模型"""
 success: bool
 code: int
 message: str
 data: Optional[T] = None
 timestamp: str = Field(default_factory=lambda: datetime.now().isoformat())

class ErrorDetail(BaseModel):
 """错误详情"""
 field: Optional[str] = None
 reason: Optional[str] = None
 value: Optional[str] = None

class ErrorResponse(BaseModel):
 """错误响应模型"""
 success: bool = False
 code: int
 message: str
 errors: Optional[list[ErrorDetail]] = None
 timestamp: str = Field(default_factory=lambda: datetime.now().isoformat())
```

### 使用响应模型

```python
from fastapi import FastAPI, HTTPException
from fastapi.responses import JSONResponse

app = FastAPI()

@app.get("/items/{item_id}", response_model=ResponseModel[dict])
async def read_item(item_id: int):
 if item_id > 100:
 raise HTTPException(
 status_code=404,
 detail=ErrorResponse(
 code=404,
 message="商品不存在",
 errors=[ErrorDetail(field="item_id", reason="商品ID不存在")]
 ).model_dump()
 )
 
 return ResponseModel(
 success=True,
 code=200,
 message="获取成功",
 data={"id": item_id, "name": "商品"}
 )
```

## 错误日志记录

### 配置日志

```python
import logging
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

# 配置日志
logging.basicConfig(
 level=logging.INFO,
 format="%(asctime)s - %(name)s - %(levelname)s - %(message)s",
 handlers=[
 logging.FileHandler("app.log"),
 logging.StreamHandler()
 ]
)

logger = logging.getLogger(__name__)
app = FastAPI()

@app.exception_handler(Exception)
async def log_exception_handler(request: Request, exc: Exception):
 """记录异常日志"""
 logger.error(
 f"Exception occurred: {exc}",
 exc_info=True,
 extra={
 "request_url": str(request.url),
 "request_method": request.method,
 "request_headers": dict(request.headers)
 }
 )
 
 return JSONResponse(
 status_code=500,
 content={
 "success": False,
 "code": 500,
 "message": "服务器内部错误"
 }
 )
```

## 完整示例

```python
from fastapi import FastAPI, Request, HTTPException, status
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError
from pydantic import BaseModel, Field
from typing import Optional, List
from datetime import datetime
import logging

# 配置日志
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

app = FastAPI()

# ============ 响应模型 ============
class ErrorDetail(BaseModel):
 """错误详情"""
 field: Optional[str] = None
 message: str
 value: Optional[str] = None

class ErrorResponse(BaseModel):
 """错误响应"""
 success: bool = False
 code: int
 message: str
 errors: Optional[List[ErrorDetail]] = None
 timestamp: str = Field(default_factory=lambda: datetime.now().isoformat())
 path: Optional[str] = None

# ============ 自定义异常 ============
class AppException(Exception):
 """应用异常基类"""
 def __init__(
 self,
 code: int,
 message: str,
 errors: List[ErrorDetail] = None
 ):
 self.code = code
 self.message = message
 self.errors = errors or []

class NotFoundException(AppException):
 """资源不存在"""
 def __init__(self, resource: str, resource_id: int):
 super().__init__(
 code=404,
 message=f"{resource}不存在",
 errors=[ErrorDetail(field="id", message=f"{resource}ID {resource_id} 不存在")]
 )

class ConflictException(AppException):
 """资源冲突"""
 def __init__(self, message: str, field: str = None):
 errors = [ErrorDetail(field=field, message=message)] if field else []
 super().__init__(code=409, message=message, errors=errors)

class ValidationException(AppException):
 """验证失败"""
 def __init__(self, errors: List[ErrorDetail]):
 super().__init__(code=400, message="数据验证失败", errors=errors)

# ============ 异常处理器 ============
@app.exception_handler(AppException)
async def app_exception_handler(request: Request, exc: AppException):
 """应用异常处理器"""
 logger.warning(f"App exception: {exc.message}")
 
 return JSONResponse(
 status_code=exc.code,
 content=ErrorResponse(
 code=exc.code,
 message=exc.message,
 errors=exc.errors,
 path=str(request.url.path)
 ).model_dump()
 )

@app.exception_handler(RequestValidationError)
async def validation_exception_handler(request: Request, exc: RequestValidationError):
 """验证错误处理器"""
 errors = [
 ErrorDetail(
 field=".".join(str(loc) for loc in error["loc"]),
 message=error["msg"],
 value=str(error.get("input", ""))
 )
 for error in exc.errors()
 ]
 
 return JSONResponse(
 status_code=422,
 content=ErrorResponse(
 code=422,
 message="请求参数验证失败",
 errors=errors,
 path=str(request.url.path)
 ).model_dump()
 )

@app.exception_handler(HTTPException)
async def http_exception_handler(request: Request, exc: HTTPException):
 """HTTP 异常处理器"""
 return JSONResponse(
 status_code=exc.status_code,
 content=ErrorResponse(
 code=exc.status_code,
 message=str(exc.detail),
 path=str(request.url.path)
 ).model_dump()
 )

@app.exception_handler(Exception)
async def global_exception_handler(request: Request, exc: Exception):
 """全局异常处理器"""
 logger.error(f"Unhandled exception: {exc}", exc_info=True)
 
 return JSONResponse(
 status_code=500,
 content=ErrorResponse(
 code=500,
 message="服务器内部错误",
 path=str(request.url.path)
 ).model_dump()
 )

# ============ 业务模型 ============
class Item(BaseModel):
 id: int
 name: str = Field(..., min_length=1, max_length=100)
 price: float = Field(..., gt=0)

# 模拟数据库
items_db: dict[int, dict] = {
 1: {"id": 1, "name": "iPhone", "price": 5999},
 2: {"id": 2, "name": "MacBook", "price": 9999},
}

# ============ 路由 ============
@app.get("/items/{item_id}")
async def read_item(item_id: int):
 """获取商品"""
 if item_id not in items_db:
 raise NotFoundException("商品", item_id)
 return items_db[item_id]

@app.post("/items/", status_code=status.HTTP_201_CREATED)
async def create_item(item: Item):
 """创建商品"""
 if item.id in items_db:
 raise ConflictException(f"商品ID {item.id} 已存在", field="id")
 items_db[item.id] = item.model_dump()
 return item

@app.delete("/items/{item_id}", status_code=status.HTTP_204_NO_CONTENT)
async def delete_item(item_id: int):
 """删除商品"""
 if item_id not in items_db:
 raise NotFoundException("商品", item_id)
 del items_db[item_id]
 return None
```

## 小结

错误处理要点：

1. **HTTPException**：用于抛出 HTTP 错误
2. **自定义异常**：创建业务异常类统一管理
3. **异常处理器**：使用 `@app.exception_handler` 统一处理
4. **验证错误**：自定义 `RequestValidationError` 处理器
5. **日志记录**：记录错误日志便于排查问题

> **提示**：统一的错误响应格式可以让客户端更容易处理错误，提高 API 的可用性。
