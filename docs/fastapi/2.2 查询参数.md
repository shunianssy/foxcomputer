# 2.2 查询参数

查询参数是 URL 中 `?` 后面的部分，用于传递额外信息。FastAPI 提供了强大的查询参数处理能力。

## 基本查询参数

```python
from fastapi import FastAPI

app = FastAPI()

@app.get("/items/")
async def read_items(skip: int, limit: int):
 """
 访问 /items/?skip=0&limit=10
 
 skip 和 limit 就是查询参数
 """
 return {"skip": skip, "limit": limit}
```

## 默认值

```python
@app.get("/items/")
async def read_items(skip: int = 0, limit: int = 10):
 """
 带默认值的查询参数
 
 访问 /items/ 时，skip=0, limit=10
 访问 /items/?skip=20 时，skip=20, limit=10
 """
 return {"skip": skip, "limit": limit}
```

## 可选查询参数

```python
from typing import Optional

@app.get("/items/")
async def read_items(skip: int = 0, limit: int = 10, q: Optional[str] = None):
 """
 可选查询参数
 
 q 是可选的，不传时为 None
 """
 results = {"skip": skip, "limit": limit}
 if q:
 results["q"] = q
 return results
```

使用 Python 3.10+ 的新语法：

```python
@app.get("/items/")
async def read_items(skip: int = 0, limit: int = 10, q: str | None = None):
 """Python 3.10+ 的类型注解"""
 return {"skip": skip, "limit": limit, "q": q}
```

## 必需查询参数

```python
@app.get("/items/")
async def read_items(skip: int, limit: int):
 """
 没有默认值的参数是必需的
 
 访问 /items/ 会返回 422 错误
 必须访问 /items/?skip=0&limit=10
 """
 return {"skip": skip, "limit": limit}
```

## 查询参数类型转换

FastAPI 会自动进行类型转换：

```python
@app.get("/items/{item_id}")
async def read_item(item_id: int, name: str, price: float, is_active: bool):
 """
 自动类型转换：
 - item_id: int
 - name: str
 - price: float
 - is_active: bool
 
 访问 /items/42?name=test&price=99.9&is_active=true
 """
 return {
 "item_id": item_id,
 "name": name,
 "price": price,
 "is_active": is_active
 }
```

布尔值转换规则：

| 输入值 | 转换结果 |
|--------|----------|
| `true`, `True`, `1`, `on`, `yes` | `True` |
| `false`, `False`, `0`, `off`, `no` | `False` |

## 多个查询参数

```python
@app.get("/users/")
async def read_users(
 name: str | None = None,
 email: str | None = None,
 age: int | None = None,
 is_active: bool = True
):
 """
 多个查询参数
 
 访问 /users/?name=张三&age=25&is_active=true
 """
 filters = {"is_active": is_active}
 if name:
 filters["name"] = name
 if email:
 filters["email"] = email
 if age:
 filters["age"] = age
 return filters
```

## 查询参数验证

使用 `Query` 进行参数验证：

```python
from fastapi import FastAPI, Query

app = FastAPI()

@app.get("/items/")
async def read_items(
 q: str | None = Query(
 default=None,
 min_length=3, # 最小长度
 max_length=50, # 最大长度
 pattern="^[a-zA-Z]+$", # 正则表达式
 title="搜索关键词",
 description="用于搜索商品的关键词"
 )
):
 """
 验证查询参数
 
 q 必须是 3-50 个字符，且只能包含字母
 """
 return {"q": q}
```

### 数值验证

```python
@app.get("/items/")
async def read_items(
 skip: int = Query(default=0, ge=0), # 大于等于 0
 limit: int = Query(default=10, le=100), # 小于等于 100
 price: float = Query(default=0.0, gt=0, le=10000) # 大于 0，小于等于 10000
):
 """
 数值验证：
 - ge: greater than or equal (>=)
 - gt: greater than (>)
 - le: less than or equal (<=)
 - lt: less than (<)
 """
 return {"skip": skip, "limit": limit, "price": price}
```

### 必需参数验证

```python
@app.get("/items/")
async def read_items(
 q: str = Query(min_length=3) # 没有默认值，参数必需
):
 """必需的查询参数，带验证"""
 return {"q": q}

# 或者使用 ...
from fastapi import Query

@app.get("/items/")
async def read_items(q: str = Query(default=..., min_length=3)):
 """使用 ... 表示必需参数"""
 return {"q": q}
```

## 查询参数列表

接收多个相同名称的参数：

```python
@app.get("/items/")
async def read_items(q: list[str] | None = Query(default=None)):
 """
 接收多个值
 
 访问 /items/?q=foo&q=bar
 q = ["foo", "bar"]
 """
 return {"q": q}

# 带默认值
@app.get("/items/")
async def read_items(q: list[str] = Query(default=["foo", "bar"])):
 """带默认值的列表参数"""
 return {"q": q}
```

## 查询参数别名

```python
@app.get("/items/")
async def read_items(
 item_query: str = Query(default="", alias="item-query")
):
 """
 使用别名
 
 访问 /items/?item-query=test
 参数名是 item-query，但在函数中使用 item_query
 """
 return {"item_query": item_query}
```

## 废弃参数

```python
@app.get("/items/")
async def read_items(
 q: str | None = Query(
 default=None,
 deprecated=True, # 标记为废弃
 title="已废弃",
 description="请使用 search 参数代替"
 ),
 search: str | None = Query(default=None)
):
 """
 废弃的参数会在文档中显示删除线
 """
 return {"q": q, "search": search}
```

## 查询参数模型

使用 Pydantic 模型组织多个查询参数：

```python
from fastapi import FastAPI, Query
from pydantic import BaseModel, Field

app = FastAPI()

class ItemFilter(BaseModel):
 """商品筛选模型"""
 name: str | None = Field(default=None, description="商品名称")
 min_price: float | None = Field(default=None, ge=0, description="最低价格")
 max_price: float | None = Field(default=None, description="最高价格")
 category: str | None = Field(default=None, description="商品分类")

@app.get("/items/")
async def read_items(filter: ItemFilter = Query()):
 """
 使用模型组织查询参数
 
 访问 /items/?name=手机&min_price=1000&max_price=5000
 """
 return filter.model_dump()
```

## 完整示例

```python
from fastapi import FastAPI, Query, HTTPException
from typing import Annotated

app = FastAPI()

# 模拟数据库
items_db = [
 {"id": 1, "name": "iPhone", "price": 5999, "category": "phone"},
 {"id": 2, "name": "MacBook", "price": 9999, "category": "computer"},
 {"id": 3, "name": "iPad", "price": 3999, "category": "tablet"},
 {"id": 4, "name": "AirPods", "price": 1299, "category": "accessory"},
]

@app.get("/items/")
async def list_items(
 skip: Annotated[int, Query(ge=0, description="跳过的记录数")] = 0,
 limit: Annotated[int, Query(ge=1, le=100, description="返回的记录数")] = 10,
 name: Annotated[str | None, Query(min_length=1, description="商品名称（模糊搜索）")] = None,
 min_price: Annotated[float | None, Query(ge=0, description="最低价格")] = None,
 max_price: Annotated[float | None, Query(ge=0, description="最高价格")] = None,
 category: Annotated[str | None, Query(description="商品分类")] = None,
):
 """
 商品列表接口
 
 支持分页和多种筛选条件
 """
 results = items_db
 
 # 名称筛选
 if name:
 results = [item for item in results if name.lower() in item["name"].lower()]
 
 # 价格筛选
 if min_price is not None:
 results = [item for item in results if item["price"] >= min_price]
 if max_price is not None:
 results = [item for item in results if item["price"] <= max_price]
 
 # 分类筛选
 if category:
 results = [item for item in results if item["category"] == category]
 
 # 分页
 total = len(results)
 results = results[skip:skip + limit]
 
 return {
 "total": total,
 "skip": skip,
 "limit": limit,
 "items": results
 }
```

## 小结

查询参数要点：

1. **基本用法**：函数参数自动识别为查询参数
2. **默认值**：有默认值的参数是可选的
3. **类型验证**：FastAPI 自动进行类型转换和验证
4. **Query 验证**：使用 `Query()` 进行更详细的验证
5. **列表参数**：支持接收多个相同名称的参数
6. **别名**：使用 `alias` 支持特殊字符的参数名

> **提示**：合理使用查询参数验证，可以在 API 入口处拦截无效请求，减少不必要的处理。
