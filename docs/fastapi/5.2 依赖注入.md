# 5.2 依赖注入

依赖注入（Dependency Injection）是 FastAPI 的核心特性之一，它提供了一种优雅的方式来管理组件之间的依赖关系。

## 基本概念

依赖注入是一种设计模式，它允许我们将依赖项（如数据库连接、配置、服务等）注入到函数中，而不是在函数内部创建它们。

```python
from fastapi import FastAPI, Depends

app = FastAPI()

# 定义依赖
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

# 使用依赖
@app.get("/items/")
async def read_items(db = Depends(get_db)):
    return db.query(Item).all()
```

## 依赖类型

### 函数依赖

```python
from fastapi import FastAPI, Depends

app = FastAPI()

def common_parameters(q: str = None, skip: int = 0, limit: int = 10):
    """通用查询参数"""
    return {"q": q, "skip": skip, "limit": limit}

@app.get("/items/")
async def read_items(commons: dict = Depends(common_parameters)):
    return commons

@app.get("/users/")
async def read_users(commons: dict = Depends(common_parameters)):
    return commons
```

### 类依赖

```python
from fastapi import FastAPI, Depends

app = FastAPI()

class CommonParams:
    """通用参数类"""
    def __init__(self, q: str = None, skip: int = 0, limit: int = 10):
        self.q = q
        self.skip = skip
        self.limit = limit

@app.get("/items/")
async def read_items(commons: CommonParams = Depends(CommonParams)):
    return {
        "q": commons.q,
        "skip": commons.skip,
        "limit": commons.limit
    }
```

### 生成器依赖

```python
from fastapi import FastAPI, Depends

app = FastAPI()

def get_db():
    """使用 yield 的依赖"""
    db = SessionLocal()
    try:
        yield db
    finally:
        # 清理资源
        db.close()

@app.get("/items/")
async def read_items(db = Depends(get_db)):
    return db.query(Item).all()
```

## 依赖嵌套

依赖可以嵌套使用：

```python
from fastapi import FastAPI, Depends, HTTPException, Header

app = FastAPI()

async def verify_token(token: str = Header(...)):
    """验证令牌"""
    if token != "valid-token":
        raise HTTPException(status_code=401, detail="Invalid token")
    return token

async def get_current_user(token: str = Depends(verify_token)):
    """获取当前用户"""
    # 根据 token 获取用户
    return {"id": 1, "username": "admin"}

@app.get("/users/me")
async def read_users_me(user: dict = Depends(get_current_user)):
    """需要认证的路由"""
    return user
```

## 全局依赖

### 应用级依赖

```python
from fastapi import FastAPI, Depends

app = FastAPI(dependencies=[Depends(verify_token)])

# 所有路由都需要验证令牌
@app.get("/items/")
async def read_items():
    return []

@app.get("/users/")
async def read_users():
    return []
```

### 路由级依赖

```python
from fastapi import FastAPI, APIRouter, Depends

app = FastAPI()
router = APIRouter(dependencies=[Depends(verify_token)])

@router.get("/items/")
async def read_items():
    return []

app.include_router(router, prefix="/api")
```

## 可调用实例

使用 `__call__` 方法创建可调用的依赖：

```python
from fastapi import FastAPI, Depends, Request

class Pagination:
    """分页依赖"""
    def __init__(self, default_skip: int = 0, default_limit: int = 10):
        self.default_skip = default_skip
        self.default_limit = default_limit
    
    def __call__(self, skip: int = 0, limit: int = 10):
        return {
            "skip": skip or self.default_skip,
            "limit": min(limit or self.default_limit, 100)
        }

pagination = Pagination(default_skip=0, default_limit=20)

app = FastAPI()

@app.get("/items/")
async def read_items(page: dict = Depends(pagination)):
    return page
```

## 缓存依赖

使用 `use_cache` 控制依赖缓存：

```python
from fastapi import FastAPI, Depends

app = FastAPI()

def expensive_dependency():
    """耗时依赖"""
    print("执行依赖")
    return {"value": "expensive"}

@app.get("/cached/")
async def cached(dep: dict = Depends(expensive_dependency, use_cache=True)):
    """缓存依赖结果，同一请求只执行一次"""
    return dep

@app.get("/no-cache/")
async def no_cache(dep: dict = Depends(expensive_dependency, use_cache=False)):
    """不缓存，每次调用都执行"""
    return dep
```

## 依赖覆盖

在测试中覆盖依赖：

```python
from fastapi import FastAPI, Depends
from fastapi.testclient import TestClient

app = FastAPI()

def get_db():
    return "production_db"

@app.get("/items/")
async def read_items(db = Depends(get_db)):
    return {"db": db}

# 测试时覆盖依赖
def override_get_db():
    return "test_db"

app.dependency_overrides[get_db] = override_get_db

client = TestClient(app)

def test_read_items():
    response = client.get("/items/")
    assert response.json() == {"db": "test_db"}
```

## 实用示例

### 数据库会话

```python
from fastapi import FastAPI, Depends
from sqlalchemy.orm import Session

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

@app.get("/items/")
async def read_items(db: Session = Depends(get_db)):
    return db.query(Item).all()
```

### 用户认证

```python
from fastapi import FastAPI, Depends, HTTPException
from fastapi.security import OAuth2PasswordBearer

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="token")

async def get_current_user(token: str = Depends(oauth2_scheme)):
    user = verify_token(token)
    if not user:
        raise HTTPException(status_code=401, detail="Invalid token")
    return user

@app.get("/users/me")
async def read_users_me(user = Depends(get_current_user)):
    return user
```

### 权限检查

```python
from fastapi import FastAPI, Depends, HTTPException
from enum import Enum

class Role(str, Enum):
    ADMIN = "admin"
    USER = "user"
    GUEST = "guest"

def require_role(required_role: Role):
    """角色检查依赖工厂"""
    async def role_checker(user = Depends(get_current_user)):
        if user.role != required_role and user.role != Role.ADMIN:
            raise HTTPException(status_code=403, detail="权限不足")
        return user
    return role_checker

@app.get("/admin/")
async def admin_only(user = Depends(require_role(Role.ADMIN))):
    return {"message": "Admin access"}
```

## 小结

依赖注入要点：

1. **函数依赖**：使用 `yield` 管理资源
2. **类依赖**：封装复杂逻辑
3. **依赖嵌套**：组合多个依赖
4. **全局依赖**：应用或路由级别
5. **依赖覆盖**：测试时替换依赖

> 💡 **提示**：依赖注入是 FastAPI 的核心特性，合理使用可以大大提高代码的可维护性和可测试性。
