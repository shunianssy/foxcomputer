# 3.1 响应模型

响应模型定义了 API 返回数据的格式，FastAPI 使用 `response_model` 来声明响应结构，实现自动验证、过滤和文档生成。

## 基本用法

```python
from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()

class Item(BaseModel):
    """商品响应模型"""
    id: int
    name: str
    price: float

@app.get("/items/{item_id}", response_model=Item)
async def read_item(item_id: int):
    """
    使用 response_model 定义响应格式
    
    返回的数据会自动转换为 Item 格式
    """
    return {"id": item_id, "name": "iPhone", "price": 5999.0}
```

## 响应模型的作用

### 数据过滤

```python
from pydantic import BaseModel

class User(BaseModel):
    """用户完整模型"""
    id: int
    username: str
    email: str
    password: str  # 敏感信息
    is_active: bool

class UserResponse(BaseModel):
    """用户响应模型（不包含密码）"""
    id: int
    username: str
    email: str
    is_active: bool

@app.get("/users/{user_id}", response_model=UserResponse)
async def read_user(user_id: int):
    """
    响应模型会自动过滤不需要的字段
    
    即使返回了 password，也会被过滤掉
    """
    return {
        "id": user_id,
        "username": "zhangsan",
        "email": "zhangsan@example.com",
        "password": "secret123",  # 会被过滤
        "is_active": True
    }
```

### 数据转换

```python
@app.get("/items/{item_id}", response_model=Item)
async def read_item(item_id: int):
    """
    返回的数据会自动转换为模型定义的类型
    """
    # 字符串价格会被转换为 float
    return {"id": item_id, "name": "iPhone", "price": "5999"}
```

## 响应模型配置

### 排除未设置的字段

```python
from fastapi import FastAPI
from pydantic import BaseModel
from typing import Optional

app = FastAPI()

class Item(BaseModel):
    name: str
    description: Optional[str] = None
    price: float
    tax: Optional[float] = None

@app.post("/items/", response_model=Item, response_model_exclude_unset=True)
async def create_item(item: Item):
    """
    response_model_exclude_unset=True
    
    只返回实际设置的字段，不返回默认值字段
    """
    return item

# 请求: {"name": "iPhone", "price": 5999}
# 响应: {"name": "iPhone", "price": 5999}
# description 和 tax 不会出现在响应中
```

### 排除 None 值

```python
@app.get("/items/{item_id}", response_model=Item, response_model_exclude_none=True)
async def read_item(item_id: int):
    """
    response_model_exclude_none=True
    
    不返回值为 None 的字段
    """
    return {
        "id": item_id,
        "name": "iPhone",
        "description": None,  # 会被排除
        "price": 5999
    }
```

### 排除特定字段

```python
@app.get("/users/{user_id}", response_model=User, response_model_exclude={"password"})
async def read_user(user_id: int):
    """
    response_model_exclude 排除特定字段
    """
    return {"id": user_id, "username": "zhangsan", "password": "secret"}

@app.get("/users/{user_id}", response_model=User, response_model_include={"id", "username"})
async def read_user_simple(user_id: int):
    """
    response_model_include 只包含特定字段
    """
    return {"id": user_id, "username": "zhangsan", "email": "test@example.com"}
```

## 多种响应模型

### 不同状态码的响应

```python
from fastapi import FastAPI, HTTPException
from fastapi.responses import JSONResponse
from pydantic import BaseModel

app = FastAPI()

class Item(BaseModel):
    id: int
    name: str

class Error(BaseModel):
    code: int
    message: str

@app.get(
    "/items/{item_id}",
    response_model=Item,
    responses={
        200: {"model": Item, "description": "成功"},
        404: {"model": Error, "description": "商品不存在"},
        400: {"model": Error, "description": "无效的 ID"}
    }
)
async def read_item(item_id: int):
    """
    定义多种响应模型
    
    文档会显示所有可能的响应格式
    """
    if item_id < 0:
        return JSONResponse(
            status_code=400,
            content={"code": 400, "message": "无效的 ID"}
        )
    if item_id > 100:
        return JSONResponse(
            status_code=404,
            content={"code": 404, "message": "商品不存在"}
        )
    return {"id": item_id, "name": "示例商品"}
```

### 动态响应模型

```python
from typing import Union

class User(BaseModel):
    username: str

class Admin(BaseModel):
    username: str
    permissions: List[str]

@app.get("/users/{user_id}", response_model=Union[User, Admin])
async def read_user(user_id: int, is_admin: bool = False):
    """
    动态选择响应模型
    """
    if is_admin:
        return {
            "username": "admin",
            "permissions": ["read", "write", "delete"]
        }
    return {"username": "user"}
```

## 响应模型继承

```python
class BaseItem(BaseModel):
    """基础商品模型"""
    name: str
    price: float

class CreateItem(BaseItem):
    """创建商品模型"""
    pass

class Item(BaseItem):
    """商品响应模型"""
    id: int
    created_at: str

@app.post("/items/", response_model=Item, status_code=201)
async def create_item(item: CreateItem):
    """
    使用继承简化模型定义
    """
    return {
        "id": 1,
        "name": item.name,
        "price": item.price,
        "created_at": "2024-01-01"
    }
```

## 自定义响应

### 直接返回 Response

```python
from fastapi import FastAPI, Response
from fastapi.responses import JSONResponse, PlainTextResponse, HTMLResponse

app = FastAPI()

@app.get("/json/")
async def get_json():
    """返回 JSON 响应"""
    return JSONResponse(
        content={"message": "Hello"},
        status_code=200,
        headers={"X-Custom-Header": "value"}
    )

@app.get("/text/")
async def get_text():
    """返回纯文本响应"""
    return PlainTextResponse(content="Hello, World!")

@app.get("/html/")
async def get_html():
    """返回 HTML 响应"""
    return HTMLResponse(content="<h1>Hello, World!</h1>")
```

### 自定义响应类

```python
from fastapi import FastAPI
from fastapi.responses import Response
import orjson

app = FastAPI()

class ORJSONResponse(Response):
    """使用 orjson 加速 JSON 序列化"""
    media_type = "application/json"
    
    def render(self, content) -> bytes:
        return orjson.dumps(content)

@app.get("/items/", response_class=ORJSONResponse)
async def read_items():
    """使用自定义响应类"""
    return {"message": "Hello"}
```

## 响应头

```python
from fastapi import FastAPI, Response

app = FastAPI()

@app.get("/items/")
async def read_items(response: Response):
    """
    设置响应头
    """
    response.headers["X-Custom-Header"] = "custom-value"
    response.headers["Cache-Control"] = "no-cache"
    return {"message": "Hello"}

@app.get("/items-with-headers/")
async def read_items_with_headers():
    """
    使用 Response 对象设置响应头
    """
    return JSONResponse(
        content={"message": "Hello"},
        headers={
            "X-Custom-Header": "value",
            "X-Another-Header": "another-value"
        }
    )
```

## 完整示例

```python
from fastapi import FastAPI, HTTPException, status
from pydantic import BaseModel, Field
from typing import List, Optional
from datetime import datetime

app = FastAPI()

# 模型定义
class ItemBase(BaseModel):
    """商品基础模型"""
    name: str = Field(..., min_length=1, max_length=100)
    price: float = Field(..., gt=0)
    description: Optional[str] = Field(None, max_length=500)

class ItemCreate(ItemBase):
    """创建商品请求模型"""
    pass

class Item(ItemBase):
    """商品响应模型"""
    id: int
    created_at: datetime
    updated_at: Optional[datetime] = None

class ItemList(BaseModel):
    """商品列表响应模型"""
    total: int
    items: List[Item]
    skip: int
    limit: int

class ErrorResponse(BaseModel):
    """错误响应模型"""
    code: int
    message: str
    detail: Optional[str] = None

# 模拟数据库
items_db: dict[int, dict] = {}
item_counter = 0

@app.post(
    "/items/",
    response_model=Item,
    status_code=status.HTTP_201_CREATED,
    responses={
        201: {"model": Item, "description": "创建成功"},
        400: {"model": ErrorResponse, "description": "请求错误"}
    }
)
async def create_item(item: ItemCreate):
    """创建商品"""
    global item_counter
    item_counter += 1
    
    now = datetime.now()
    items_db[item_counter] = {
        "id": item_counter,
        "name": item.name,
        "price": item.price,
        "description": item.description,
        "created_at": now,
        "updated_at": None
    }
    
    return items_db[item_counter]

@app.get(
    "/items/",
    response_model=ItemList,
    responses={
        200: {"model": ItemList, "description": "获取成功"}
    }
)
async def list_items(skip: int = 0, limit: int = 10):
    """获取商品列表"""
    items = list(items_db.values())[skip:skip + limit]
    return {
        "total": len(items_db),
        "items": items,
        "skip": skip,
        "limit": limit
    }

@app.get(
    "/items/{item_id}",
    response_model=Item,
    responses={
        200: {"model": Item, "description": "获取成功"},
        404: {"model": ErrorResponse, "description": "商品不存在"}
    }
)
async def read_item(item_id: int):
    """获取单个商品"""
    if item_id not in items_db:
        raise HTTPException(
            status_code=404,
            detail={"code": 404, "message": "商品不存在", "detail": f"ID {item_id} 不存在"}
        )
    return items_db[item_id]
```

## 小结

响应模型要点：

1. **自动过滤**：只返回模型定义的字段
2. **类型转换**：自动转换为模型定义的类型
3. **文档生成**：自动生成响应文档
4. **多种配置**：支持排除未设置、排除 None、排除特定字段
5. **自定义响应**：可以返回各种类型的响应

> 💡 **提示**：合理使用响应模型可以保护敏感数据、规范 API 输出格式，是构建安全 API 的重要手段。
