# 2.1 路径操作与路由

FastAPI 的路由系统非常灵活且强大。本章将详细介绍如何定义和管理路由。

## 路径操作装饰器

FastAPI 使用装饰器来定义路由，装饰器名称对应 HTTP 方法：

```python
from fastapi import FastAPI

app = FastAPI()

# 常用的 HTTP 方法
@app.get("/") # 获取资源
@app.post("/") # 创建资源
@app.put("/") # 完整更新资源
@app.patch("/") # 部分更新资源
@app.delete("/") # 删除资源
@app.options("/") # 获取支持的 HTTP 方法
@app.head("/") # 只获取响应头
@app.trace("/") # 路径追踪（调试用）
```

## 路径参数

### 基本路径参数

```python
@app.get("/items/{item_id}")
async def read_item(item_id: int):
 """
 路径参数会自动转换为指定类型
 
 访问 /items/42 时，item_id = 42
 """
 return {"item_id": item_id}
```

### 路径参数类型

FastAPI 支持多种路径参数类型：

```python
# 整数类型
@app.get("/items/{item_id:int}")
async def read_item(item_id: int):
 return {"item_id": item_id}

# 浮点数类型
@app.get("/price/{price:float}")
async def read_price(price: float):
 return {"price": price}

# 字符串类型（默认）
@app.get("/users/{username}")
async def read_user(username: str):
 return {"username": username}

# 路径类型（包含斜杠）
@app.get("/files/{file_path:path}")
async def read_file(file_path: str):
 """
 path 类型可以匹配包含斜杠的路径
 
 访问 /files/home/user/document.txt
 file_path = "home/user/document.txt"
 """
 return {"file_path": file_path}
```

### 枚举路径参数

使用 Enum 限制路径参数的值：

```python
from enum import Enum
from fastapi import FastAPI

app = FastAPI()

class ModelName(str, Enum):
 """模型名称枚举"""
 alexnet = "alexnet"
 resnet = "resnet"
 lenet = "lenet"

@app.get("/models/{model_name}")
async def get_model(model_name: ModelName):
 """
 model_name 只能是 alexnet、resnet 或 lenet
 
 如果传入其他值，会返回 422 错误
 """
 if model_name == ModelName.alexnet:
 return {"model": model_name, "message": "Deep Learning FTW!"}
 if model_name == ModelName.resnet:
 return {"model": model_name, "message": "LeCNN all the images"}
 return {"model": model_name, "message": "Have some residuals"}
```

### 多个路径参数

```python
@app.get("/users/{user_id}/items/{item_id}")
async def read_user_item(user_id: int, item_id: int):
 """
 可以定义多个路径参数
 
 访问 /users/1/items/42
 user_id = 1, item_id = 42
 """
 return {"user_id": user_id, "item_id": item_id}
```

## 路由顺序

路由的顺序很重要，FastAPI 按定义顺序匹配：

```python
# 错误示例
@app.get("/users/me")
async def read_user_me():
 """这个路由永远不会被访问到！"""
 return {"user_id": "current user"}

@app.get("/users/{user_id}")
async def read_user(user_id: str):
 """这个会先匹配到 /users/me"""
 return {"user_id": user_id}

# 正确示例
@app.get("/users/me")
async def read_user_me():
 """固定路径要放在前面"""
 return {"user_id": "current user"}

@app.get("/users/{user_id}")
async def read_user(user_id: str):
 """动态路径放在后面"""
 return {"user_id": user_id}
```

## 路由前缀

使用 `APIRouter` 添加路由前缀：

```python
from fastapi import FastAPI, APIRouter

app = FastAPI()

# 创建路由器，设置前缀
router = APIRouter(prefix="/api/v1")

@router.get("/items")
async def read_items():
 """完整路径: /api/v1/items"""
 return [{"id": 1}, {"id": 2}]

@router.get("/users")
async def read_users():
 """完整路径: /api/v1/users"""
 return [{"id": 1, "name": "张三"}]

# 注册路由器
app.include_router(router)
```

## 路由标签

使用标签组织路由：

```python
from fastapi import FastAPI

app = FastAPI()

@app.get("/items/", tags=["items"])
async def get_items():
 """获取商品列表"""
 return []

@app.post("/items/", tags=["items"])
async def create_item():
 """创建商品"""
 return {}

@app.get("/users/", tags=["users"])
async def get_users():
 """获取用户列表"""
 return []
```

## 路由分组

使用 `APIRouter` 进行路由分组：

```python
from fastapi import FastAPI, APIRouter

app = FastAPI()

# 商品路由
items_router = APIRouter(
 prefix="/items",
 tags=["items"],
 responses={404: {"description": "Not found"}}
)

@items_router.get("/")
async def get_items():
 return []

@items_router.get("/{item_id}")
async def get_item(item_id: int):
 return {"id": item_id}

@items_router.post("/")
async def create_item():
 return {"created": True}

# 用户路由
users_router = APIRouter(
 prefix="/users",
 tags=["users"]
)

@users_router.get("/")
async def get_users():
 return []

# 注册所有路由
app.include_router(items_router)
app.include_router(users_router)
```

## 路由依赖

为路由添加依赖：

```python
from fastapi import FastAPI, Depends, HTTPException

app = FastAPI()

# 定义依赖函数
async def verify_token(token: str):
 """验证令牌"""
 if token != "secret":
 raise HTTPException(status_code=401, detail="Invalid token")
 return token

# 在路由中使用依赖
@app.get("/protected", dependencies=[Depends(verify_token)])
async def protected_route():
 """需要验证的路由"""
 return {"message": "Access granted"}

# 获取依赖返回值
@app.get("/token-info")
async def token_info(token: str = Depends(verify_token)):
 """获取令牌信息"""
 return {"token": token}
```

## 路由装饰器参数

完整的路由装饰器参数：

```python
@app.get(
 "/items/{item_id}",
 response_model=Item, # 响应模型
 status_code=200, # 默认状态码
 tags=["items"], # 标签
 summary="获取商品", # 简短描述
 description="根据 ID 获取商品详情", # 详细描述
 response_description="商品信息", # 响应描述
 deprecated=False, # 是否废弃
 operation_id="get_item_by_id", # 操作 ID
 response_model_exclude_unset=True, # 排除未设置的字段
 response_model_exclude_none=True, # 排除 None 值
 include_in_schema=True, # 是否包含在文档中
 name="get_item" # 路由名称
)
async def read_item(item_id: int):
 return {"id": item_id}
```

## 路由回调

FastAPI 支持 OpenAPI 的 callbacks：

```python
from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()

class CallbackItem(BaseModel):
 id: int
 name: str

@app.post(
 "/items/",
 callbacks=[
 # 定义回调路由
 ]
)
async def create_item(item: dict):
 """
 创建商品后会回调通知
 """
 return item
```

## 实用示例

### RESTful 风格路由

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from typing import List, Optional

app = FastAPI()

class Item(BaseModel):
 id: int
 name: str
 price: float
 description: Optional[str] = None

# 模拟数据库
items_db: dict[int, Item] = {}

@app.get("/items", response_model=List[Item], tags=["items"])
async def list_items(skip: int = 0, limit: int = 10):
 """获取商品列表"""
 return list(items_db.values())[skip:skip + limit]

@app.post("/items", response_model=Item, status_code=201, tags=["items"])
async def create_item(item: Item):
 """创建商品"""
 if item.id in items_db:
 raise HTTPException(status_code=400, detail="Item already exists")
 items_db[item.id] = item
 return item

@app.get("/items/{item_id}", response_model=Item, tags=["items"])
async def read_item(item_id: int):
 """获取单个商品"""
 if item_id not in items_db:
 raise HTTPException(status_code=404, detail="Item not found")
 return items_db[item_id]

@app.put("/items/{item_id}", response_model=Item, tags=["items"])
async def update_item(item_id: int, item: Item):
 """完整更新商品"""
 if item_id not in items_db:
 raise HTTPException(status_code=404, detail="Item not found")
 items_db[item_id] = item
 return item

@app.patch("/items/{item_id}", response_model=Item, tags=["items"])
async def partial_update_item(item_id: int, item: dict):
 """部分更新商品"""
 if item_id not in items_db:
 raise HTTPException(status_code=404, detail="Item not found")
 stored_item = items_db[item_id]
 update_data = stored_item.model_dump()
 update_data.update(item)
 items_db[item_id] = Item(**update_data)
 return items_db[item_id]

@app.delete("/items/{item_id}", tags=["items"])
async def delete_item(item_id: int):
 """删除商品"""
 if item_id not in items_db:
 raise HTTPException(status_code=404, detail="Item not found")
 del items_db[item_id]
 return {"message": "Item deleted successfully"}
```

## 小结

FastAPI 路由系统要点：

1. **装饰器定义**：使用 `@app.get()` 等装饰器定义路由
2. **路径参数**：使用 `{param}` 语法定义，支持类型转换
3. **路由顺序**：固定路径要在动态路径之前定义
4. **路由分组**：使用 `APIRouter` 组织路由
5. **标签分类**：使用 tags 组织 API 文档

> **提示**：合理组织路由结构，使用路由前缀和标签，可以让 API 更加清晰易维护。
